1//! The repeaters a node has heard directly, and how it answers a
2//! [Peer Repeaters Request](../../docs/protocol/src/mac-commands.md).
3//!
4//! The list is a neighborhood: repeaters heard off their own transmitters.
5//! Two sources feed it, because neither is enough on its own:
6//!
7//! - **Identities.** A repeater that advertises tells its neighbors its name,
8//!   its position, and the regions it forwards for. None of that can be
9//!   recovered from a hint. Only an identity heard directly counts—one that
10//!   arrived forwarded says nothing about whether its owner is in range.
11//! - **Transmitter observations** ([`umsh_mac::TransmitterObservations`]).
12//!   A repeater prepends its router hint to the trace route of every frame
13//!   it forwards, so every traced frame off the air proves which repeater
14//!   just transmitted it and how well it was heard—including repeaters that
15//!   never send this node anything of their own.
16//!
17//! A [`RouterHint`] is the first two bytes of a public key and a [`NodeHint`]
18//! the first three, so the observation's key is a prefix of the identity's.
19//! That is what lets the two merge: an identity claims the observation whose
20//! hint it starts with, and an observation nothing claims becomes an entry
21//! naming a hop by its router hint and reporting only what was heard.
22
23use alloc::string::String;
24use alloc::vec::Vec;

32/// How many identity-bearing repeaters the table remembers.
33///
34/// Sized to the MAC's observation table: the two merge into one listing, and
35/// a listing longer than the 1-byte Total the response reports would have to
36/// be truncated anyway.
37pub const MAX_PEER_REPEATERS: usize = 16;
38
39/// Region codes kept per peer, matching the identity option's own cap.
40pub const MAX_PEER_REGIONS: usize = 10;
41
42/// What one repeater's identity told this node about it.
43#[derive(Clone, Debug, PartialEq, Eq)]
44pub struct PeerRepeaterRecord {
45    /// The whole node hint, since an identity carries the whole key.
46    pub hint: NodeHint,
47    pub name: Option<String>,
48    pub location: Option<NodeLocation>,
49    /// The codes the peer flood-forwards for, derived from the region
50    /// strings its identity carried.
51    pub regions: Vec<[u8; 2]>,
52    /// How well the identity was heard, when the radio measured it.
53    pub rssi_snr: Option<(i16, Snr)>,
54    /// When the identity arrived, on the monotonic clock.
55    pub last_identity_ms: u64,
56}
57
58impl PeerRepeaterRecord {
59    /// The router hint an observation would be keyed by—the first two
60    /// bytes of the same public key.
61    pub fn router_hint(&self) -> RouterHint {
62        RouterHint([self.hint.0[0], self.hint.0[1]])
63    }
64}

66/// The repeaters whose identities this node has heard directly.
67///
68/// RAM-only: a listing describes a neighborhood as it is now, and a table
69/// restored from flash would name repeaters that may have moved or gone.
70#[derive(Clone, Debug, Default)]
71pub struct PeerRepeaterTable {
72    records: Vec<PeerRepeaterRecord>,
73    /// Bumped on every mutation, so a paging cursor can tell that the list
74    /// it was walking is no longer the list it started on.
75    generation: u16,
76}
77
78impl PeerRepeaterTable {
79    pub fn new() -> Self {
80        Self::default()
81    }
82
83    /// Whether an identity describes a repeater, and so belongs here.
84    ///
85    /// The capability is the claim that matters—a node may forward while
86    /// presenting itself as something else—but a node whose whole role is
87    /// repeating counts even if it advertises no capability bitmap.
88    pub fn is_repeater(identity: &NodeIdentityPayload) -> bool {
89        identity.capabilities.contains(NodeCapabilities::REPEATER)
90            || identity.role == NodeRole::Repeater
91    }
92
93    /// Record what an identity heard directly said, if it came from a
94    /// repeater.
95    ///
96    /// The caller vouches that the identity came off its owner's own
97    /// transmitter; `rssi_snr` is that reception's measurement, when the
98    /// radio made one. Returns whether the table changed. A repeat identity
99    /// replaces the record rather than merging with it: the newest
100    /// advertisement is the node's own account of itself.
101    pub fn observe_identity(
102        &mut self,
103        from: &PublicKey,
104        identity: &NodeIdentityPayload,
105        rssi_snr: Option<(i16, Snr)>,
106        now_ms: u64,
107    ) -> bool {
108        if !Self::is_repeater(identity) {
109            return false;
110        }
111        let record = PeerRepeaterRecord {
112            hint: from.hint(),
113            name: identity.name.clone(),
114            location: identity.location,
115            regions: region_codes(identity),
116            rssi_snr,
117            last_identity_ms: now_ms,
118        };
119        self.generation = self.generation.wrapping_add(1);
120        if let Some(existing) = self
121            .records
122            .iter_mut()
123            .find(|entry| entry.hint == record.hint)
124        {
125            *existing = record;
126            return true;
127        }
128        if self.records.len() < MAX_PEER_REPEATERS {
129            self.records.push(record);
130            return true;
131        }
132        // Full: the identity heard longest ago is the one whose absence says
133        // the least.
134        let Some(oldest) = self
135            .records
136            .iter_mut()
137            .min_by_key(|entry| entry.last_identity_ms)
138        else {
139            return false;
140        };
141        *oldest = record;
142        true
143    }
144
145    pub fn iter(&self) -> impl Iterator<Item = &PeerRepeaterRecord> {
146        self.records.iter()
147    }
148
149    pub fn len(&self) -> usize {
150        self.records.len()
151    }
152
153    pub fn is_empty(&self) -> bool {
154        self.records.is_empty()
155    }
156
157    /// The value a paging cursor carries, so a follow-up request that
158    /// resumes into a changed list is recognized as stale.
159    pub fn generation(&self) -> u16 {
160        self.generation
161    }
162}

164/// Derive the forwarding codes from an identity's region strings.
165///
166/// The identity carries strings and an entry carries codes: the entry format
167/// is tighter on space, and the string form is available from the peer's own
168/// identity when one is wanted.
169fn region_codes(identity: &NodeIdentityPayload) -> Vec<[u8; 2]> {
170    let Some(regions) = identity.supported_regions.as_ref() else {
171        return Vec::new();
172    };
173    regions
174        .iter()
175        .filter_map(|region| region.parse::<umsh_core::RegionCode>().ok())
176        .map(|code| code.to_bytes())
177        .take(MAX_PEER_REGIONS)
178        .collect()
179}

181/// One row of a merged listing: everything known about one peer repeater,
182/// from either source or both.
183#[derive(Clone, Debug, PartialEq, Eq)]
184pub struct MergedPeerRepeater {
185    /// Three bytes when an identity supplied the whole node hint, two when
186    /// only an observation named this hop.
187    pub hint: Vec<u8>,
188    pub name: Option<String>,
189    pub location: Option<NodeLocation>,
190    pub regions: Vec<[u8; 2]>,
191    /// The most recent measured reception, from whichever source heard the
192    /// peer last.
193    pub rssi_dbm: Option<i16>,
194    pub snr: Option<Snr>,
195    /// Minutes since this peer was last heard from, by either source.
196    pub last_heard_min: Option<u16>,
197}
198
199/// Merge the identity table with the MAC's transmitter observations.
200///
201/// An identity claims the observation whose router hint it starts with; each
202/// remaining observation becomes a two-byte-hint entry. Both sources are
203/// direct receptions, so whichever heard the peer more recently supplies the
204/// signal figures.
205///
206/// The listing is ordered most recently heard first. The first page is the
207/// one a requester is surest to read, and the neighbors on the air right
208/// now are the ones it is asking about.
209pub fn merge<'a>(
210    identities: &PeerRepeaterTable,
211    observations: impl IntoIterator<Item = &'a umsh_mac::TransmitterObservation>,
212    now_ms: u64,
213) -> Vec<MergedPeerRepeater> {
214    let observations: Vec<&umsh_mac::TransmitterObservation> = observations.into_iter().collect();
215    // Each entry alongside the monotonic time it was last heard, which orders
216    // the list more finely than the wire's whole minutes can.
217    let mut merged: Vec<(u64, MergedPeerRepeater)> = Vec::new();
218    let mut claimed: Vec<RouterHint> = Vec::new();
219
220    for record in identities.iter() {
221        let router_hint = record.router_hint();
222        let observation = observations
223            .iter()
224            .find(|entry| entry.hint == router_hint)
225            .copied();
226        if observation.is_some() {
227            claimed.push(router_hint);
228        }
229        // The newer reception's figures, falling back to the older one's
230        // when the newer had no radio in it to measure.
231        let rssi_snr = match observation {
232            Some(entry) if entry.last_seen_ms >= record.last_identity_ms => {
233                entry.rssi_snr.or(record.rssi_snr)
234            }
235            Some(entry) => record.rssi_snr.or(entry.rssi_snr),
236            None => record.rssi_snr,
237        };
238        let last_heard_ms = observation
239            .map(|entry| entry.last_seen_ms.max(record.last_identity_ms))
240            .unwrap_or(record.last_identity_ms);
241        merged.push((
242            last_heard_ms,
243            MergedPeerRepeater {
244                hint: Vec::from(&record.hint.0[..]),
245                name: record.name.clone(),
246                location: record.location,
247                regions: record.regions.clone(),
248                rssi_dbm: rssi_snr.map(|(rssi, _)| rssi),
249                snr: rssi_snr.map(|(_, snr)| snr),
250                last_heard_min: minutes_since(Some(last_heard_ms), now_ms),
251            },
252        ));
253    }
254
255    for observation in observations {
256        if claimed.contains(&observation.hint) {
257            continue;
258        }
259        merged.push((
260            observation.last_seen_ms,
261            MergedPeerRepeater {
262                hint: Vec::from(&observation.hint.0[..]),
263                name: None,
264                location: None,
265                regions: Vec::new(),
266                rssi_dbm: observation.rssi_snr.map(|(rssi, _)| rssi),
267                snr: observation.rssi_snr.map(|(_, snr)| snr),
268                last_heard_min: minutes_since(Some(observation.last_seen_ms), now_ms),
269            },
270        ));
271    }
272
273    // Stable, so peers heard in the same instant keep identity-first order.
274    merged.sort_by_key(|(last_heard_ms, _)| core::cmp::Reverse(*last_heard_ms));
275    merged.into_iter().map(|(_, entry)| entry).collect()
276}

278/// Whole minutes between `then_ms` and now, saturating at the two octets the
279/// wire form allows—about 45 days, past which "longer ago than that" is the
280/// only useful answer anyway.
281fn minutes_since(then_ms: Option<u64>, now_ms: u64) -> Option<u16> {
282    let then_ms = then_ms?;
283    let elapsed_min = now_ms.saturating_sub(then_ms) / 60_000;
284    Some(u16::try_from(elapsed_min).unwrap_or(u16::MAX))
285}

368    /// A router hint is a node hint's first two bytes, which is the whole

438    /// Whoever was heard last is listed first, whichever source heard them.
439    #[test]
440    fn the_listing_is_ordered_most_recently_heard_first() {
441        let mut table = PeerRepeaterTable::new();
442        // An identity heard long ago whose owner was heard forwarding since,
443        // one heard recently, and one heard long ago and never since.
444        table.observe_identity(
445            &key(0xAA),
446            &repeater_identity("Refreshed", &[]),
447            None,
448            10_000,
449        );
450        table.observe_identity(&key(0xBB), &repeater_identity("Recent", &[]), None, 200_000);
451        table.observe_identity(&key(0xCC), &repeater_identity("Stale", &[]), None, 20_000);
452        let refreshed = key(0xAA).hint();
453        let observations = [
454            observation(RouterHint([refreshed.0[0], refreshed.0[1]]), 300_000),
455            // Never introduced, heard forwarding between the other two.
456            observation(RouterHint([0x11, 0x22]), 100_000),

460        let order: Vec<&[u8]> = merged.iter().map(|entry| &entry.hint[..]).collect();
461        assert_eq!(
462            order,
463            [

470        assert_eq!(merged[0].last_heard_min, Some(1));
