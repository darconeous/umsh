1//! Built-in responder for the [Identity Request](crate::mac_command) MAC command.
2//!
3//! When enabled on a [`LocalNode`](crate::LocalNode), the node answers a
4//! matching Identity Request with a **targeted authenticated unicast** carrying
5//! its own [`NodeIdentityPayload`], echoing any request `NONCE` into identity
6//! option 5. Responses are never signed and never fall back to broadcast: the
7//! pairwise MIC, frame counter, and echoed nonce already make the reply
8//! authentic and fresh, and a request whose source cannot be resolved to a key
9//! is simply dropped by the MAC before the responder ever runs.
10//!
11//! The application supplies a [`NodeIdentityProfile`] (its role, capabilities,
12//! and descriptive fields) and, optionally, a respond **policy** — a
13//! registerable discriminator that inspects the [`IdentityRequestContext`] and
14//! decides whether, and how, to answer (e.g. "known peers only", "only on a
15//! given channel", "repeater → always"). No signing key is required.
16

29/// The narrowest and widest random hold applied to a reply to a
30/// broadcast/multicast solicitation, per the Identity Request
31/// flood-management rules. Every node the solicitation selected is answering
32/// the same frame, so the hold spreads the replies across the window instead
33/// of piling them onto one another.
34///
35/// A solicitation narrowed to a single node is exempt: there is no crowd to
36/// spread, and holding the one reply would only make the answer late. See
37/// [`IdentityRequestFilters::hint_names_one_node`].
38pub(crate) const RESPONSE_MIN_DELAY_MS: u16 = 500;
39pub(crate) const RESPONSE_MAX_DELAY_MS: u16 = 30_000;
40
41/// How long one answered solicitation suppresses further replies to it.
42///
43/// A solicitation reaches a node once per path it travels, and the copies do
44/// not arrive together: a repeater carrying the second copy holds it for its
45/// own contention delay first. The window must also outlive the reply's own
46/// random hold, since a duplicate landing while the first reply still sits in
47/// the transmit queue is the case that queues a second one — so it is derived
48/// from that hold rather than chosen independently.
49///
50/// An immediate reply is covered by the same window with room to spare, and
51/// keeping one window for both cases costs nothing: it suppresses repeats of a
52/// solicitation, and a requester wanting a fresh answer inside it asks with a
53/// fresh nonce.
54const SOLICITATION_SUPPRESSION_MS: u64 = RESPONSE_MAX_DELAY_MS as u64 * 2;
55
56/// How many recently answered solicitations are remembered at once.
57///
58/// One entry per distinct solicitation, so this is how many separate
59/// questions a node can be holding answers to at once. Overflow evicts the
60/// oldest, which can then be answered twice — the pre-existing behavior
61/// rather than a new failure.
62const ANSWERED_CAPACITY: usize = 8;
63
64/// One solicitation this node has already answered.
65struct AnsweredSolicitation {
66    /// The requester, by hint rather than full key: enough to tell two
67    /// askers apart, and a collision costs one suppressed reply rather than
68    /// anything worse.
69    requester: NodeHint,
70    /// The request's `NONCE`, when it carried one. This is what makes the
71    /// entry name a *solicitation* and not merely a peer — a requester that
72    /// wants another answer inside the window asks with a fresh nonce, which
73    /// is what the nonce is for. A request that carries none cannot be told
74    /// apart from a repeat of itself, so within the window it is treated as
75    /// one.
76    nonce: Option<u32>,
77    answered_at_ms: u64,
78}
79
80/// This node's own identity, used to answer Identity Requests.
81///
82/// Holds descriptive fields only — **no signing key**. Config-like fields
83/// (`role`, `capabilities`, `name`, `supported_regions`) are typically set once
84/// at bring-up; live fields (`location`, `altitude_m`) can be refreshed at any
85/// time via [`LocalNode::update_identity_profile`](crate::LocalNode::update_identity_profile),
86/// e.g. from a GPS task.
87#[derive(Clone, Debug)]
88pub struct NodeIdentityProfile {
89    /// This node's public key. Its [hint](PublicKey::hint) is matched against a
90    /// request's `FILTER_NODE_HINT`; the key itself reaches the requester via
91    /// the reply's MAC source address, not the identity payload.
92    pub public_key: PublicKey,
93    pub role: NodeRole,
94    pub capabilities: NodeCapabilities,
95    pub name: Option<String>,
96    pub location: Option<NodeLocation>,
97    pub altitude_m: Option<i32>,
98    pub supported_regions: Option<Vec<u8>>,
99    /// Where identity option 3 comes from: the current Unix time, or
100    /// `None` on a node that does not know what time it is.
101    ///
102    /// A source rather than a value, because option 3 dates the *payload*
103    /// and every payload is built fresh — a stored number would be the
104    /// time some earlier payload was built. It lives here so that the one
105    /// canonical builder stamps every framing identically; a node with no
106    /// clock keeps the default and simply omits the option.
107    pub clock: fn() -> Option<u32>,
108}
109
110impl NodeIdentityProfile {
111    /// Create a minimal profile (role + capabilities), no descriptive options.
112    pub fn new(public_key: PublicKey, role: NodeRole, capabilities: NodeCapabilities) -> Self {
113        Self {
114            public_key,
115            role,
116            capabilities,
117            name: None,
118            location: None,
119            altitude_m: None,
120            supported_regions: None,
121            clock: || None,
122        }
123    }
124
125    /// Set the display name (builder style).
126    pub fn with_name(mut self, name: impl Into<String>) -> Self {
127        self.name = Some(name.into());
128        self
129    }
130
131    /// Set the geographic location (builder style).
132    pub fn with_location(mut self, location: NodeLocation) -> Self {
133        self.location = Some(location);
134        self
135    }
136
137    /// This node's hint, derived from its public key.
138    pub fn hint(&self) -> NodeHint {
139        self.public_key.hint()
140    }
141
142    /// Build the (unsigned) identity payload this node advertises,
143    /// stamping `nonce` into option 5.
144    ///
145    /// The one canonical builder, deliberately shared by both framings of
146    /// a node identity: the Identity Request reply, which carries a
147    /// request nonce and is authenticated by the enclosing unicast, and
148    /// the standalone signed blob, which carries no nonce and is
149    /// authenticated by a detached signature. They are different objects
150    /// on the wire and the same statement about the node; the difference
151    /// must not be able to drift into the contents.
152    pub fn to_payload(&self, nonce: Option<u32>) -> NodeIdentityPayload {
153        NodeIdentityPayload {
154            role: self.role,
155            capabilities: self.capabilities,
156            name: self.name.clone(),
157            location: self.location,
158            altitude_m: self.altitude_m,
159            // Option 3 dates this payload, so it is read now rather than
160            // carried: the freshness marker exists to stop a captured
161            // identity being presented indefinitely, and a stamp copied
162            // from an earlier build would be exactly that capture.
163            timestamp: (self.clock)(),
164            supported_regions: self.supported_regions.clone(),
165            nonce,
166            signature: None,
167        }
168    }
169}

171/// Reception context for an incoming Identity Request, handed to the respond
172/// policy so it can decide whether — and how — to answer.
173///
174/// The request has already passed the filter gate (its `FILTER_*` options
175/// select this node) and its source has already been resolved to a key, so the
176/// policy only governs the "do I want to answer *this sender*?" decision.
177pub struct IdentityRequestContext<'a> {
178    /// Resolved sender key. Always present: an unresolvable source is dropped
179    /// before the policy runs.
180    pub from_key: PublicKey,
181    /// Sender hint, when the frame carried one.
182    pub from_hint: Option<NodeHint>,
183    /// Whether the request frame was authenticated (pairwise or channel MIC).
184    pub source_authenticated: bool,
185    /// Whether the request carried the sender's full 32-byte key.
186    pub has_full_source: bool,
187    /// The channel the request arrived on, if any (`None` for plain
188    /// broadcast/unicast).
189    pub channel: Option<ChannelId>,
190    /// Coarse packet family (Unicast / Broadcast / Multicast / BlindUnicast).
191    pub family: PacketFamily,
192    /// The request's filter/option block, for policies that inspect it further.
193    pub filters: IdentityRequestFilters<'a>,
194    /// Received signal strength of the request, if measured.
195    pub rssi: Option<i16>,
196    /// Signal-to-noise ratio of the request, if measured.
197    pub snr: Option<Snr>,
198    /// The request's accumulated trace route, as packed option bytes.
199    ///
200    /// Repeaters prepend their hint when forwarding, so this reads
201    /// front-to-back as the path *back* to the requester and needs no
202    /// reversal. It is the only route home a broadcast solicitation offers:
203    /// broadcast reception registers no route with the MAC.
204    pub trace_route: &'a [u8],
205}
206
207/// A respond policy's verdict for one Identity Request.
208#[derive(Clone, Copy, Debug, PartialEq, Eq)]
209pub enum RespondDecision {
210    /// Do not answer this request.
211    Ignore,
212    /// Answer with an authenticated unicast identity response.
213    Respond {
214        /// Include our full 32-byte key in the reply's source address, so a
215        /// requester that only had our hint can authenticate the reply without
216        /// a prior key exchange. Set `false` only when the requester already
217        /// holds our key.
218        full_source: bool,
219    },
220}
221
222/// A registerable respond policy: given the request context, decide the verdict.
223pub type RespondPolicy = dyn FnMut(&IdentityRequestContext<'_>) -> RespondDecision;
224
225/// The default respond policy: answer every request that reached the policy.
226///
227/// Includes our full source key unless the request was authenticated to us
228/// (an authenticated pairwise request implies the sender already holds our
229/// key). The blind-unicast-over-channel case, where a channel-authenticated
230/// sender may still lack our key, is a known edge a custom policy can override.
231pub fn default_respond_policy(ctx: &IdentityRequestContext<'_>) -> RespondDecision {
232    RespondDecision::Respond {
233        full_source: !ctx.source_authenticated,
234    }
235}
236
237/// A respond policy that answers nothing.
238///
239/// Lets a node keep a live profile while declining to be discovered. The
240/// profile is what unsolicited advertisements are built from, so silencing
241/// the responder this way — rather than by uninstalling it — is what keeps
242/// the two behaviors independent.
243pub fn never_respond_policy(_ctx: &IdentityRequestContext<'_>) -> RespondDecision {
244    RespondDecision::Ignore
245}

247/// Installed responder state: the profile, the active policy, and the record
248/// of what has already been answered.
249pub(crate) struct IdentityResponder {
250    pub(crate) profile: NodeIdentityProfile,
251    pub(crate) policy: Box<RespondPolicy>,
252    /// Insertion-ordered, so the oldest entry is always at the front and
253    /// expiry prunes from that end without a scan.
254    answered: Deque<AnsweredSolicitation, ANSWERED_CAPACITY>,
255}
256
257impl IdentityResponder {
258    pub(crate) fn new(profile: NodeIdentityProfile, policy: Box<RespondPolicy>) -> Self {
259        Self {
260            profile,
261            policy,
262            answered: Deque::new(),
263        }
264    }
265}

267/// A resolved plan to answer one Identity Request, produced synchronously while
268/// the node state is borrowed and executed later by the async pump.
269pub(crate) struct IdentityResponsePlan {
270    /// Destination (the requester).
271    pub(crate) to: PublicKey,
272    /// Whether the reply should carry our full source key.
273    pub(crate) full_source: bool,
274    /// Whether the reply must be held for a random delay before transmit.
275    ///
276    /// Set for a broadcast/multicast solicitation that more than one node may
277    /// satisfy, where every selected node answers the same frame and undelayed
278    /// replies would collide on the channel. A request narrowed to a single
279    /// node has no crowd to spread and is answered immediately.
280    pub(crate) delayed: bool,
281    /// Whether the reply must carry no flood hop count field at all.
282    /// Set for a solicitation that no `FILTER_NODE_HINT` narrowed: such a
283    /// request was confined to the requester's neighbours on the way in, and
284    /// its reply stays there too rather than being flooded back.
285    pub(crate) no_flood: bool,
286    /// Routers to steer the reply back through, in send order; empty when the
287    /// request arrived with no trace to follow.
288    ///
289    /// A steered solicitation is the case this exists for: the requester is
290    /// several hops away, the reply carries no flood budget, and broadcast
291    /// reception left the MAC with no cached route to them. The trace the
292    /// request accumulated on the way in is the path home.
293    pub(crate) route: Vec<RouterHint>,
294    /// The framed reply payload: `PayloadType::NodeIdentity` + encoded identity.
295    pub(crate) framed: Vec<u8>,
296}
297
298impl IdentityResponder {
299    /// Evaluate an incoming request against the profile and policy.
300    ///
301    /// Returns `Some(plan)` when the node should answer: the request's filters
302    /// select this node, the solicitation has not already been answered,
303    /// **and** the policy returns `Respond`. Returns `None` otherwise (not
304    /// selected, already answered, policy said `Ignore`, or the reply could
305    /// not be encoded).
306    pub(crate) fn evaluate(
307        &mut self,
308        ctx: &IdentityRequestContext<'_>,
309        now_ms: u64,
310    ) -> Option<IdentityResponsePlan> {
311        // Filter gate: does this request target a node like us?
312        let our_hint = self.profile.hint();
313        if !ctx
314            .filters
315            .selects(self.profile.role, self.profile.capabilities, &our_hint)
316            .unwrap_or(false)
317        {
318            return None;
319        }
320
321        // Freshness gate: one reply per solicitation, however many copies of
322        // it arrive. A plain broadcast request carries no frame counter, so
323        // nothing below the node layer can recognize a second copy of one;
324        // the echoed nonce is the only thing that names the question.
325        let nonce = ctx.filters.nonce().ok().flatten();
326        let requester = ctx.from_key.hint();
327        self.expire_answered(now_ms);
328        if self.already_answered(&requester, nonce) {
329            return None;
330        }
331
332        // Policy gate: do we want to answer this particular sender?
333        let full_source = match (self.policy)(ctx) {
334            RespondDecision::Ignore => return None,
335            RespondDecision::Respond { full_source } => full_source,
336        };
337
338        // Build the framed reply, echoing the request nonce into option 5.
339        let payload = self.profile.to_payload(nonce);
340        let mut buf = [0u8; 192];
341        buf[0] = PayloadType::NodeIdentity as u8;
342        let len = 1 + payload.encode(&mut buf[1..]).ok()?;
343        let solicitation = matches!(
344            ctx.family,
345            crate::PacketFamily::Broadcast | crate::PacketFamily::Multicast
346        );
347        // Recorded only now that there is a reply to send: a request the
348        // policy declined or the encoder could not frame was never answered,
349        // and must not suppress a later attempt that would succeed.
350        self.record_answered(requester, nonce, now_ms);
351        // Repeaters prepend as they forward, so the accumulated trace already
352        // reads as the path back and is copied verbatim rather than reversed.
353        let route = RouteHops::new(ctx.trace_route).collect::<Vec<_>>();
354        Some(IdentityResponsePlan {
355            to: ctx.from_key,
356            full_source,
357            delayed: solicitation && !ctx.filters.hint_names_one_node(),
358            no_flood: solicitation && !ctx.filters.hint_filtered(),
359            route,
360            framed: Vec::from(&buf[..len]),
361        })
362    }
363
364    fn already_answered(&self, requester: &NodeHint, nonce: Option<u32>) -> bool {
365        self.answered
366            .iter()
367            .any(|entry| entry.requester == *requester && entry.nonce == nonce)
368    }
369
370    /// Drop every entry older than the suppression window.
371    ///
372    /// A clock that has gone backwards leaves an entry looking younger than
373    /// it is, never older, so suppression can only be held slightly too long
374    /// — never released early, which is the direction that would let the
375    /// duplicate replies back.
376    fn expire_answered(&mut self, now_ms: u64) {
377        while let Some(entry) = self.answered.front() {
378            if now_ms.saturating_sub(entry.answered_at_ms) < SOLICITATION_SUPPRESSION_MS {
379                return;
380            }
381            let _ = self.answered.pop_front();
382        }
383    }
384
385    fn record_answered(&mut self, requester: NodeHint, nonce: Option<u32>, now_ms: u64) {
386        if self.answered.is_full() {
387            let _ = self.answered.pop_front();
388        }
389        let _ = self.answered.push_back(AnsweredSolicitation {
390            requester,
391            nonce,
392            answered_at_ms: now_ms,
393        });
394    }
395}

397#[cfg(test)]
398mod tests {
399    use super::*;
400    use crate::location::NodeLocation;
401
402    /// A profile's position reaches the wire, and every payload built
403    /// from it is dated when it was built rather than carrying a stamp
404    /// from some earlier one.
405    #[test]
406    fn a_profile_position_reaches_the_wire_dated_at_build_time() {
407        let mut profile = NodeIdentityProfile::new(
408            PublicKey([9; 32]),
409            NodeRole::Tracker,
410            NodeCapabilities::empty(),
411        );
412        let here = NodeLocation::from_e7(377_749_000, -1_224_194_000, 5);
413        profile.location = Some(here);
414        profile.altitude_m = Some(-17);
415        profile.clock = || Some(1_785_000_000);
416
417        let payload = profile.to_payload(None);
418        let mut buf = [0u8; 192];
419        let len = payload.encode(&mut buf).expect("encode");
420        let decoded = NodeIdentityPayload::from_bytes(&buf[..len]).expect("decode");
421
422        assert_eq!(decoded.location, Some(here));
423        assert_eq!(decoded.altitude_m, Some(-17));
424        assert_eq!(decoded.timestamp, Some(1_785_000_000));
425    }
426
427    /// A node that does not know what time it is omits option 3 rather
428    /// than inventing a date for the payload.
429    #[test]
430    fn a_clockless_node_omits_the_timestamp() {
431        let profile = NodeIdentityProfile::new(
432            PublicKey([9; 32]),
433            NodeRole::Tracker,
434            NodeCapabilities::empty(),
435        );
436        assert_eq!(profile.to_payload(None).timestamp, None);
437    }
