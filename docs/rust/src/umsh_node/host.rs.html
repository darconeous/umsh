25/// Multi-identity orchestration layer for the node API.
26///
27/// `Host` owns the shared MAC run loop and routes inbound events to the correct
28/// [`LocalNode`], including ephemeral PFS identities that belong to a long-term node.
29/// Applications typically:
30///
31/// 1. construct a `Host` from a shared [`MacHandle`](umsh_mac::MacHandle)
32/// 2. register one or more [`LocalNode`]s with [`add_node`](Self::add_node)
33/// 3. attach callbacks to nodes / peers / wrappers
34/// 4. drive progress with [`run`](Self::run) or [`pump_once`](Self::pump_once)
35///
36/// `run()` is the preferred long-lived driver. `pump_once()` exists for callers that need to
37/// multiplex UMSH progress with other async work using `select!`.
38///
39/// # Callback re-entrancy
40///
41/// Receive and control callbacks (`on_receive`, `on_text`, `on_ack_received`, …) are invoked
42/// synchronously from inside [`pump_once`](Self::pump_once) while the coordinator borrow is
43/// held. A callback therefore MUST NOT call back into the MAC—it cannot `await` a `send`,
44/// and a blocking or re-entrant borrow of the coordinator would deadlock. Keep callbacks
45/// short and side-effect-free with respect to the MAC: record what you need (e.g. push to a
46/// queue or set a flag) and perform any follow-up sends after `pump_once` / `run` returns
47/// control to your own task.
48pub struct Host<M: MacBackend> {
49    mac: M,
50    dispatcher: Rc<RefCell<EventDispatcher>>,
51    nodes: Vec<(LocalIdentityId, LocalNode<M>)>,
52    pfs_control_options: SendOptions,
53}
54

83    /// Create and register a [`LocalNode`] for an already-registered local identity.
84    ///
85    /// The returned handle is cheap to clone and becomes the application-facing entry point
86    /// for sending traffic and attaching callbacks for that identity.
87    pub fn add_node(&mut self, identity_id: LocalIdentityId) -> LocalNode<M> {
88        let membership = Rc::new(RefCell::new(NodeMembership::new()));
89        let state = Rc::new(RefCell::new(LocalNodeState::new()));
90        let node = LocalNode::new(
91            identity_id,
92            self.mac.clone(),
93            self.dispatcher.clone(),
94            membership,
95            state,
96        );
97        self.nodes.push((identity_id, node.clone()));
98        node
99    }

101    /// Look up a previously added node by identity id.
102    pub fn node(&self, identity_id: LocalIdentityId) -> Option<LocalNode<M>> {
103        self.nodes
104            .iter()
105            .find(|(id, _)| *id == identity_id)
106            .map(|(_, node)| node.clone())
107    }

110    fn route_node(&self, identity_id: LocalIdentityId) -> Option<LocalNode<M>> {
111        if let Some(node) = self.node(identity_id) {
112            return Some(node);
113        }
114        self.nodes
115            .iter()
116            .find(|(_, node)| node.owns_ephemeral_identity(identity_id))
117            .map(|(_, node)| node.clone())

120    /// Drive the shared MAC until one wake cycle completes.
121    ///
122    /// This is a single wake-driven step, not a non-blocking poll. It waits until the MAC has
123    /// meaningful work to do (radio activity or a protocol deadline), dispatches any resulting
124    /// callbacks, services PFS command handling, and then returns.
125    ///
126    /// Use this when you need to multiplex UMSH progress with other async sources using
127    /// `select!`. If UMSH owns the task, prefer [`run`](Self::run).
128    pub async fn pump_once(&mut self) -> Result<(), HostError<M::RunError>> {
129        let now_ms = self.mac.now_ms().await;
130        let pending_pfs = Rc::new(RefCell::new(Vec::<(
131            LocalIdentityId,
132            umsh_core::PublicKey,
133            Option<umsh_core::ChannelId>,
134            OwnedMacCommand,
135        )>::new()));
136        let pending_pfs_ref = pending_pfs.clone();
137        let pending_identity = Rc::new(RefCell::new(
138            Vec::<(LocalNode<M>, IdentityResponsePlan)>::new(),
139        ));
140        let pending_identity_ref = pending_identity.clone();
141        let pending_peer_repeaters = Rc::new(RefCell::new(Vec::<(
142            LocalNode<M>,
143            umsh_core::PublicKey,
144            Option<umsh_core::ChannelId>,
145            Vec<u8>,
146        )>::new()));
147        let pending_peer_repeaters_ref = pending_peer_repeaters.clone();
148        let pending_mobility = Rc::new(RefCell::new(Vec::<(umsh_core::PublicKey, bool)>::new()));
149        let pending_mobility_ref = pending_mobility.clone();
150        let dispatcher = self.dispatcher.clone();
151        let nodes = self.nodes.clone();
152        self.mac
153            .next_event(move |identity_id, event| {
154                dispatcher
155                    .borrow_mut()
156                    .dispatch_ticket_state(identity_id, &event);
157                let Some(node) = route_node(&nodes, identity_id) else {
158                    return;
159                };
160                match event {
161                    umsh_mac::MacEventRef::Received(packet) => {
162                        let _ = node.dispatch_received_packet(&packet);
163                        if packet.packet_type() == umsh_core::PacketType::Broadcast
164                            && packet.payload().is_empty()
165                        {
166                            if let Some(from_hint) = packet.from_hint() {
167                                node.dispatch_beacon(from_hint, packet.from_key());
168                            }
169                        } else if let Some(from) = packet.from_key() {
170                            dispatch_payload_callbacks(
171                                &node,
172                                &packet,
173                                from,
174                                &pending_pfs_ref,
175                                &pending_identity_ref,
176                                &pending_peer_repeaters_ref,
177                                &pending_mobility_ref,
178                                now_ms,
179                            );
180                        }
181                    }
182                    umsh_mac::MacEventRef::AckReceived { peer, receipt, .. } => {
183                        node.dispatch_ack_received(
184                            peer,
185                            crate::SendToken::new(identity_id, receipt),
186                        );
187                    }
188                    umsh_mac::MacEventRef::AckTimeout { peer, receipt } => {
189                        node.dispatch_ack_timeout(
190                            peer,
191                            crate::SendToken::new(identity_id, receipt),
192                        );
193                    }
194                    umsh_mac::MacEventRef::Transmitted { wire_bytes, .. } => {
195                        node.dispatch_transmitted(wire_bytes);
196                    }
197                    umsh_mac::MacEventRef::Forwarded { .. } => {}
198                    // Ticket resolution (failed + finished) is handled by
199                    // dispatch_ticket_state above; there is no peer to
200                    // notify since the frame never aired.
201                    umsh_mac::MacEventRef::TxAbandoned { .. } => {}
202                }
203            })
204            .await
205            .map_err(HostError::Mac)?;
206
207        let queued: Vec<(
208            LocalIdentityId,
209            umsh_core::PublicKey,
210            Option<umsh_core::ChannelId>,
211            OwnedMacCommand,
212        )> = pending_pfs.borrow_mut().drain(..).collect();
213        // Forward secrecy needs the software crypto back end; without it, a
214        // peer's request goes unanswered.
215        #[cfg(feature = "software-crypto")]
216        for (identity_id, from, channel, command) in queued {
217            self.handle_pfs_command(identity_id, from, channel, command)
218                .await;
219        }
220        #[cfg(not(feature = "software-crypto"))]
221        drop(queued);
222
223        let identity_replies: Vec<(LocalNode<M>, IdentityResponsePlan)> =
224            pending_identity.borrow_mut().drain(..).collect();
225        for (node, plan) in identity_replies {
226            node.send_identity_response(plan).await;
227        }
228
229        let peer_repeater_requests: Vec<(
230            LocalNode<M>,
231            umsh_core::PublicKey,
232            Option<umsh_core::ChannelId>,
233            Vec<u8>,
234        )> = pending_peer_repeaters.borrow_mut().drain(..).collect();
235        for (node, from, channel, request) in peer_repeater_requests {
236            node.answer_peer_repeaters_request(from, channel, &request)
237                .await;
238        }
239
240        let mobility: Vec<(umsh_core::PublicKey, bool)> =
241            pending_mobility.borrow_mut().drain(..).collect();
242        for (peer, mobile) in mobility {
243            self.mac.set_peer_mobile(&peer, mobile).await;
244        }
245
246        self.service_protocol_timeouts().await;
247
248        Ok(())
249    }

251    /// Service node-layer deadlines independently of MAC/radio wake events.
252    ///
253    /// A quiet radio is not a MAC wake source, so applications that multiplex
254    /// [`pump_once`](Self::pump_once) with other work must also call this from
255    /// their own timer. In particular, ping and PFS request timeouts must fire
256    /// even when the remote peer sends no response at all.
257    pub async fn service_protocol_timeouts(&mut self) {
258        let now_ms = self.mac.now_ms().await;
259        service_node_timeouts(&self.nodes, now_ms).await;
260    }
261
262    /// Detach an owned handle that services the same node-layer deadlines as
263    /// [`service_protocol_timeouts`](Self::service_protocol_timeouts).
264    ///
265    /// Use this when the host pump runs as its own long-lived future (so the
266    /// `Host` is exclusively borrowed) and a sibling timer task must still
267    /// fire ping/PFS timeouts—e.g. while the pump is parked awaiting a slow
268    /// physical transmit. The handle snapshots the current node set; nodes
269    /// added afterwards are not covered by it.
270    pub fn protocol_timeout_servicer(&self) -> ProtocolTimeoutServicer<M>
271    where
272        M: Clone,
273    {
274        ProtocolTimeoutServicer {
275            mac: self.mac.clone(),
276            nodes: self.nodes.clone(),
277        }
278    }
279
280    /// Run the shared MAC/Host loop forever.
281    ///
282    /// This is the preferred long-lived driver for node-based applications. It keeps the
283    /// runtime wake policy inside the MAC/Host stack rather than requiring callers to write
284    /// poll/sleep loops themselves.
285    pub async fn run(&mut self) -> Result<(), HostError<M::RunError>> {
286        loop {
287            self.pump_once().await?;
288        }
289    }

291    #[cfg(feature = "software-crypto")]
292    async fn handle_pfs_command(
293        &mut self,
294        identity_id: LocalIdentityId,
295        from: umsh_core::PublicKey,
296        channel: Option<umsh_core::ChannelId>,
297        command: OwnedMacCommand,
298    ) {
299        let Some(node) = self.route_node(identity_id) else {
300            return;
301        };
302
303        match node
304            .handle_pfs_command(&from, channel, &command, &self.pfs_control_options)
305            .await
306        {
307            Ok(Some(PfsLifecycle::Established(peer))) => node.dispatch_pfs_established(peer),
308            Ok(Some(PfsLifecycle::Ended(peer))) => node.dispatch_pfs_ended(peer),
309            Ok(None) => {}
310            // Previously the error was dropped here, so a failed negotiation
311            // (e.g. no ephemeral identity slot) left both sides silently stuck.
312            // Surface it so applications can report the failure.
313            Err(err) => node.dispatch_pfs_failed(from, err.pfs_failure()),
314        }
315    }
316}

318/// Owned deadline-service handle detached from a [`Host`]; see
319/// [`Host::protocol_timeout_servicer`].
320pub struct ProtocolTimeoutServicer<M: MacBackend> {
321    mac: M,
322    nodes: Vec<(LocalIdentityId, LocalNode<M>)>,
323}
324
325impl<M: MacBackend> ProtocolTimeoutServicer<M> {
326    /// Fire any elapsed ping and PFS deadlines on the snapshot's nodes.
327    pub async fn service(&self) {
328        let now_ms = self.mac.now_ms().await;
329        service_node_timeouts(&self.nodes, now_ms).await;
330    }
331}
332
333async fn service_node_timeouts<M: MacBackend>(
334    nodes: &[(LocalIdentityId, LocalNode<M>)],
335    now_ms: u64,
336) {
337    #[cfg(feature = "software-crypto")]
338    for (_, node) in nodes {
339        if let Ok(expired) = node.expire_pfs_sessions().await {
340            for peer in expired {
341                node.dispatch_pfs_ended(peer);
342            }
343        }
344        // Abandon sent PFS requests that were never answered, so the
345        // requester reports a timeout instead of sitting in `Requested`.
346        for peer in node.expire_pfs_requests(now_ms) {
347            node.dispatch_pfs_failed(peer, crate::node::PfsFailure::Timeout);
348        }
349    }
350
351    for (_, node) in nodes {
352        node.expire_pings(now_ms);
353    }
354}

356fn route_node<M: MacBackend>(
357    nodes: &[(LocalIdentityId, LocalNode<M>)],
358    identity_id: LocalIdentityId,
359) -> Option<LocalNode<M>> {
360    let node = nodes
361        .iter()
362        .find(|(id, _)| *id == identity_id)
363        .map(|(_, node)| node.clone());
364    // An ephemeral identity answers for the node that made it.
365    #[cfg(feature = "software-crypto")]
366    let node = node.or_else(|| {
367        nodes
368            .iter()
369            .find(|(_, node)| node.owns_ephemeral_identity(identity_id))
370            .map(|(_, node)| node.clone())
371    });
372    node
373}

375/// Whether an identity payload's claims are its sender's own: a signature
376/// that verifies, or none on a frame whose source authenticated.
377#[cfg(feature = "software-crypto")]
378pub fn identity_is_trustworthy(
379    packet: &ReceivedPacketRef<'_>,
380    from: &umsh_core::PublicKey,
381    identity: &NodeIdentityPayload,
382) -> bool {
383    match identity.signature_state(from, packet.payload()) {
384        crate::IdentitySignature::Valid => true,
385        crate::IdentitySignature::Unsigned => packet.source_authenticated(),
386        crate::IdentitySignature::Invalid => false,
387    }
388}
389
390/// Without the crypto to check a signature, only an unsigned payload on a
391/// frame whose source authenticated can be trusted.
392#[cfg(not(feature = "software-crypto"))]
393pub fn identity_is_trustworthy(
394    packet: &ReceivedPacketRef<'_>,
395    _from: &umsh_core::PublicKey,
396    identity: &NodeIdentityPayload,
397) -> bool {
398    identity.signature.is_none() && packet.source_authenticated()
399}

401fn dispatch_payload_callbacks<M: MacBackend>(
402    node: &LocalNode<M>,
403    packet: &ReceivedPacketRef<'_>,
404    from: umsh_core::PublicKey,
405    pending_pfs: &Rc<
406        RefCell<
407            Vec<(
408                LocalIdentityId,
409                umsh_core::PublicKey,
410                Option<umsh_core::ChannelId>,
411                OwnedMacCommand,
412            )>,
413        >,
414    >,
415    pending_identity: &Rc<RefCell<Vec<(LocalNode<M>, IdentityResponsePlan)>>>,
416    pending_peer_repeaters: &Rc<
417        RefCell<
418            Vec<(
419                LocalNode<M>,
420                umsh_core::PublicKey,
421                Option<umsh_core::ChannelId>,
422                Vec<u8>,
423            )>,
424        >,
425    >,
426    pending_mobility: &Rc<RefCell<Vec<(umsh_core::PublicKey, bool)>>>,
427    now_ms: u64,
428) {
429    if packet.payload_type() == PayloadType::NodeIdentity {
430        if let Ok(identity) = NodeIdentityPayload::from_bytes(packet.payload()) {
431            // Whether the node moves decides whether a route to it keeps a
432            // flood hop of slack. That holds however the advertisement got
433            // here, so long as the claim is the node's own.
434            if identity_is_trustworthy(packet, &from, &identity) {
435                pending_mobility.borrow_mut().push((
436                    from,
437                    identity
438                        .capabilities
439                        .contains(crate::NodeCapabilities::MOBILE),
440                ));
441            }
442            // Only an advertisement heard from its owner directly places
443            // the owner in this node's repeater neighborhood: one that
444            // arrived forwarded says nothing about whether the owner is a
445            // neighbor, one replayed from a device's queue was heard
446            // minutes ago, and a copy of what this stack's own antenna
447            // just sent came from nobody else. A host attached over a
448            // point-to-point link is a neighbor with no radio between it
449            // and this node, so its advertisement counts and carries no
450            // measurement. Recorded before the callback so an observer
451            // that asks for the peer-repeater listing from inside it sees
452            // this identity in it.
453            let heard_directly = packet.hop_count() == Some(1)
454                && packet.rx().buffered_age_s() == 0
455                && packet.rx().origin() != umsh_hal::RxOrigin::LocalTx;
456            if heard_directly {
457                node.observe_peer_identity(
458                    from,
459                    &identity,
460                    packet.rssi().zip(packet.snr()),
461                    now_ms,
462                );
463            }
464            node.dispatch_node_discovered(from, identity.name.as_deref());
465        }
466        return;
467    }
468
469    if packet.payload_type() == PayloadType::MacCommand {
470        if let Ok(command) = mac_command::parse(packet.payload()) {
471            // A MAC command is addressed to one node. A command that arrives by
472            // multicast or broadcast is ignored unless its own definition gives
473            // rules for that carriage (mac-commands.md), and today the Identity
474            // Request is the only one that does. Anything else is dropped here,
475            // before observers see it.
476            let addressed_to_one_node = matches!(
477                packet.packet_family(),
478                umsh_mac::PacketFamily::Unicast | umsh_mac::PacketFamily::BlindUnicast
479            );
480            if !addressed_to_one_node
481                && !matches!(command, mac_command::MacCommand::IdentityRequest { .. })
482            {
483                return;
484            }
485            // The carriage this request arrived on, which is the carriage its
486            // response owes it back. `Some` only for a blind unicast: a plain
487            // unicast is answered in kind, and the identity responder decides
488            // for itself how to answer a multicast or broadcast solicitation.
489            let reply_channel =
490                matches!(packet.packet_family(), umsh_mac::PacketFamily::BlindUnicast)
491                    .then(|| packet.channel().map(|c| c.id()))
492                    .flatten();
493            // Match borrowed variants before converting to owned.
494            if let mac_command::MacCommand::EchoResponse { data } = command {
495                node.match_pong(
496                    from,
497                    data,
498                    packet,
499                    packet.received_at_ms().unwrap_or(now_ms),
500                );
501            }
502            // Identity Request: let the built-in responder (if enabled) build a
503            // reply plan now, while the reception context is live; the async
504            // pump sends it after this synchronous dispatch returns.
505            if let mac_command::MacCommand::IdentityRequest { options } = command {
506                if let Some(plan) = node.evaluate_identity_request(
507                    packet,
508                    from,
509                    options,
510                    packet.received_at_ms().unwrap_or(now_ms),
511                ) {
512                    pending_identity.borrow_mut().push((node.clone(), plan));
513                }
514            }
515            // Peer Repeaters Request: the answer needs the MAC's transmitter
516            // observations, which only an async borrow reaches, so it is
517            // built by the pump after this synchronous dispatch returns.
518            if let mac_command::MacCommand::PeerRepeatersRequest { options } = command
519                && node.peer_repeaters_responder_enabled()
520            {
521                pending_peer_repeaters.borrow_mut().push((
522                    node.clone(),
523                    from,
524                    reply_channel,
525                    Vec::from(options),
526                ));
527            }
528            let owned = OwnedMacCommand::from(command);
529            node.dispatch_mac_command(from, &owned);
530            if matches!(
531                owned,
532                OwnedMacCommand::PfsSessionRequest { .. }
533                    | OwnedMacCommand::PfsSessionResponse { .. }
534                    | OwnedMacCommand::EndPfsSession
535            ) {
536                pending_pfs
537                    .borrow_mut()
538                    .push((node.identity_id(), from, reply_channel, owned));
539            }
540        }
541    }
542}

555    /// An advertisement from `from` carrying `payload`, its source
556    /// authenticated or not.
557    fn advertisement(
558        from: PublicKey,
559        payload: &[u8],
560        source_authenticated: bool,
561    ) -> ReceivedPacketRef<'static> {
562        let mut bytes = Vec::from([PayloadType::NodeIdentity as u8]);
563        bytes.extend_from_slice(payload);
564        let wire: &'static [u8] = alloc::boxed::Box::leak(bytes.into_boxed_slice());
565        let header = PacketHeader {
566            fcf: Fcf::new(PacketType::Broadcast, true, false),
567            options_range: 0..0,
568            flood_hops: None,
569            dst: None,
570            channel: None,
571            source: umsh_core::SourceAddrRef::Hint(from.hint()),
572            sec_info: None,
573            body_range: 0..wire.len(),
574            mic_range: wire.len()..wire.len(),
575            total_len: wire.len(),
576        };
577        ReceivedPacketRef::new(
578            wire,
579            wire,
580            header,
581            ParsedOptions::default(),
582            Some(from),
583            Some(from.hint()),
584            source_authenticated,
585            None,
586            umsh_mac::RxMetadata::default(),
587        )
588    }

590    fn trusted(packet: &ReceivedPacketRef<'_>, from: &PublicKey) -> bool {
591        let identity = NodeIdentityPayload::from_bytes(packet.payload()).unwrap();
592        identity_is_trustworthy(packet, from, &identity)
593    }
