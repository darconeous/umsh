21/// How long a sent PFS request waits for a response before it is abandoned and
22/// reported as a [`PfsFailure::Timeout`](crate::PfsFailure::Timeout). A
23/// `Requested` session holds no MAC resources (the ephemeral identity is only
24/// registered once a response arrives), so timing one out just drops the entry.
25#[cfg(feature = "software-crypto")]
26const PFS_REQUEST_TIMEOUT_MS: u64 = 30_000;
27
28#[cfg(feature = "software-crypto")]
29/// Lifecycle state for a PFS session.
30#[derive(Clone, Copy, Debug, PartialEq, Eq)]
31pub(crate) enum PfsState {
32    /// Session requested but not fully activated.
33    Requested,
34    /// Session active and mapped to an ephemeral MAC identity.
35    Active,
36}

38#[cfg(feature = "software-crypto")]
39/// One tracked PFS session.
40pub(crate) struct PfsSession {
41    /// Peer long-term identity.
42    pub peer_long_term: PublicKey,
43    /// Local ephemeral MAC identity.
44    pub local_ephemeral_id: LocalIdentityId,
45    /// Peer ephemeral public key.
46    pub peer_ephemeral: PublicKey,
47    /// Absolute expiry time in milliseconds.
48    ///
49    /// While the session is still [`Requested`](PfsState::Requested), this is a
50    /// provisional value computed at request time and replaced when the session
51    /// becomes active.
52    pub expires_ms: u64,
53    /// Session lifecycle state.
54    pub state: PfsState,
55    /// Deadline by which a [`Requested`](PfsState::Requested) session must
56    /// receive its response, after which it is abandoned. Unused once `Active`.
57    pub request_deadline_ms: u64,
58    pending_local: Option<SoftwareIdentity>,
59}

89#[cfg(feature = "software-crypto")]
90impl PfsSessionManager {
91    /// Create an empty session manager.
92    pub(crate) fn new() -> Self {
93        Self {
94            sessions: Vec::new(),
95            max_sessions: DEFAULT_MAX_PFS_SESSIONS,
96        }
97    }

99    /// Borrow the tracked sessions.
100    pub(crate) fn sessions(&self) -> &[PfsSession] {
101        &self.sessions
102    }
103
104    /// Return the active ephemeral route for `peer`, if any.
105    pub(crate) fn active_route(
106        &self,
107        peer: &PublicKey,
108        now_ms: u64,
109    ) -> Option<(LocalIdentityId, PublicKey)> {
110        self.sessions
111            .iter()
112            .find(|session| {
113                session.peer_long_term == *peer
114                    && session.state == PfsState::Active
115                    && session.expires_ms > now_ms
116            })
117            .map(|session| (session.local_ephemeral_id, session.peer_ephemeral))
118    }
119
120    /// Initiate a new PFS session request.
121    pub(crate) async fn request_session<M: MacBackend>(
122        &mut self,
123        mac: &M,
124        parent: LocalIdentityId,
125        peer: &PublicKey,
126        duration_minutes: u16,
127        options: &SendOptions,
128    ) -> Result<Option<SendReceipt>, NodeError<M>> {
129        self.remove_existing(mac, peer).await?;
130        self.ensure_capacity()?;
131        let mut secret = [0u8; 32];
132        mac.fill_random(&mut secret).await;
133        let local_ephemeral = SoftwareIdentity::from_secret_bytes(&secret);
134        secret.zeroize();
135        let now_ms = mac.now_ms().await;
136        let receipt = send_pfs_command(
137            mac,
138            parent,
139            peer,
140            &MacCommand::PfsSessionRequest {
141                ephemeral_key: *local_ephemeral.public_key(),
142                duration_minutes,
143            },
144            options,
145        )
146        .await?;
147        self.sessions.push(PfsSession {
148            peer_long_term: *peer,
149            local_ephemeral_id: LocalIdentityId(u8::MAX),
150            peer_ephemeral: PublicKey([0u8; 32]),
151            expires_ms: now_ms.saturating_add(u64::from(duration_minutes) * 60_000),
152            state: PfsState::Requested,
153            request_deadline_ms: now_ms.saturating_add(PFS_REQUEST_TIMEOUT_MS),
154            pending_local: Some(local_ephemeral),
155        });
156        Ok(receipt)
157    }
158
159    /// Accept an inbound PFS request.
160    pub(crate) async fn accept_request<M: MacBackend>(
161        &mut self,
162        mac: &M,
163        parent: LocalIdentityId,
164        peer_long_term: PublicKey,
165        peer_ephemeral: PublicKey,
166        duration_minutes: u16,
167        options: &SendOptions,
168    ) -> Result<(), NodeError<M>> {
169        self.remove_existing(mac, &peer_long_term).await?;
170        self.ensure_capacity()?;
171        let mut secret = [0u8; 32];
172        mac.fill_random(&mut secret).await;
173        let local_ephemeral = SoftwareIdentity::from_secret_bytes(&secret);
174        secret.zeroize();
175        let active = activate_identity(
176            mac,
177            parent,
178            local_ephemeral,
179            peer_ephemeral,
180            duration_minutes,
181        )
182        .await?;
183        send_pfs_command(
184            mac,
185            parent,
186            &peer_long_term,
187            &MacCommand::PfsSessionResponse {
188                ephemeral_key: active.peer_local_public,
189                duration_minutes,
190            },
191            options,
192        )
193        .await?;
194        self.sessions.push(PfsSession {
195            peer_long_term,
196            local_ephemeral_id: active.local_ephemeral_id,
197            peer_ephemeral,
198            expires_ms: active.expires_ms,
199            state: PfsState::Active,
200            request_deadline_ms: u64::MAX,
201            pending_local: None,
202        });
203        Ok(())
204    }

206    /// Accept an inbound PFS response for an existing request.
207    pub(crate) async fn accept_response<M: MacBackend>(
208        &mut self,
209        mac: &M,
210        parent: LocalIdentityId,
211        peer_long_term: PublicKey,
212        peer_ephemeral: PublicKey,
213        duration_minutes: u16,
214    ) -> Result<bool, NodeError<M>> {
215        let Some(index) = self
216            .sessions
217            .iter()
218            .position(|session| session.peer_long_term == peer_long_term)
219        else {
220            return Err(NodeError::PfsSessionMissing);
221        };
222        let mut session = self.sessions.swap_remove(index);
223        let Some(local_ephemeral) = session.pending_local.take() else {
224            self.sessions.push(session);
225            return Err(NodeError::PfsSessionMissing);
226        };
227        let active = activate_identity(
228            mac,
229            parent,
230            local_ephemeral,
231            peer_ephemeral,
232            duration_minutes,
233        )
234        .await?;
235        session.local_ephemeral_id = active.local_ephemeral_id;
236        session.peer_ephemeral = peer_ephemeral;
237        session.expires_ms = active.expires_ms;
238        session.state = PfsState::Active;
239        self.sessions.push(session);
240        Ok(true)
241    }
242
243    /// End one session and optionally notify the peer.
244    pub(crate) async fn end_session<M: MacBackend>(
245        &mut self,
246        mac: &M,
247        parent: LocalIdentityId,
248        peer: &PublicKey,
249        notify_peer: bool,
250        options: &SendOptions,
251    ) -> Result<bool, NodeError<M>> {
252        let Some(index) = self
253            .sessions
254            .iter()
255            .position(|session| session.peer_long_term == *peer)
256        else {
257            return Err(NodeError::PfsSessionMissing);
258        };
259        let session = self.sessions.swap_remove(index);
260        if session.state == PfsState::Active {
261            let _ = mac.remove_ephemeral(session.local_ephemeral_id).await;
262        }
263        if notify_peer {
264            send_pfs_command(mac, parent, peer, &MacCommand::EndPfsSession, options).await?;
265        }
266        Ok(true)
267    }

269    /// Expire any sessions whose deadlines have passed.
270    pub(crate) async fn expire_sessions<M: MacBackend>(
271        &mut self,
272        mac: &M,
273        now_ms: u64,
274    ) -> Result<Vec<PublicKey>, NodeError<M>> {
275        let mut expired = Vec::new();
276        let mut index = 0usize;
277        while index < self.sessions.len() {
278            if self.sessions[index].expires_ms > now_ms {
279                index += 1;
280                continue;
281            }
282            let session = self.sessions.swap_remove(index);
283            if session.state == PfsState::Active {
284                let _ = mac.remove_ephemeral(session.local_ephemeral_id).await;
285            }
286            expired.push(session.peer_long_term);
287        }
288        Ok(expired)
289    }
290
291    /// Drop any `Requested` sessions whose response deadline has passed and
292    /// return their peers. These hold no MAC resources (no ephemeral identity
293    /// is registered until a response arrives), so no `mac` cleanup is needed.
294    pub(crate) fn expire_requests(&mut self, now_ms: u64) -> Vec<PublicKey> {
295        let mut timed_out = Vec::new();
296        let mut index = 0usize;
297        while index < self.sessions.len() {
298            let session = &self.sessions[index];
299            if session.state == PfsState::Requested && session.request_deadline_ms <= now_ms {
300                timed_out.push(self.sessions.swap_remove(index).peer_long_term);
301            } else {
302                index += 1;
303            }
304        }
305        timed_out
306    }
307
308    async fn remove_existing<M: MacBackend>(
309        &mut self,
310        mac: &M,
311        peer: &PublicKey,
312    ) -> Result<(), NodeError<M>> {
313        if let Some(index) = self
314            .sessions
315            .iter()
316            .position(|session| session.peer_long_term == *peer)
317        {
318            let session = self.sessions.swap_remove(index);
319            if session.state == PfsState::Active {
320                let _ = mac.remove_ephemeral(session.local_ephemeral_id).await;
321            }
322        }
323        Ok(())
324    }
325
326    fn ensure_capacity<M: MacBackend>(&self) -> Result<(), NodeError<M>> {
327        if self.sessions.len() >= self.max_sessions {
328            Err(NodeError::PfsSessionTableFull)
329        } else {
330            Ok(())
331        }
332    }

342#[cfg(feature = "software-crypto")]
343async fn activate_identity<M: MacBackend>(
344    mac: &M,
345    parent: LocalIdentityId,
346    local_ephemeral: SoftwareIdentity,
347    peer_ephemeral: PublicKey,
348    duration_minutes: u16,
349) -> Result<ActivatedPfsIdentity, NodeError<M>> {
350    let _peer_id = mac.add_peer(peer_ephemeral).await?;
351    let peer_local_public = *local_ephemeral.public_key();
352    let local_ephemeral_id = mac.register_ephemeral(parent, local_ephemeral).await?;
353    let now_ms = mac.now_ms().await;
354    Ok(ActivatedPfsIdentity {
355        local_ephemeral_id,
356        peer_local_public,
357        expires_ms: now_ms.saturating_add(u64::from(duration_minutes) * 60_000),
358    })
359}

361#[cfg(feature = "software-crypto")]
362async fn send_pfs_command<M: MacBackend>(
363    mac: &M,
364    from: LocalIdentityId,
365    peer: &PublicKey,
366    command: &MacCommand<'_>,
367    options: &SendOptions,
368) -> Result<Option<SendReceipt>, NodeError<M>> {
369    let mut body = [0u8; 80];
370    let len = mac_command::encode(command, &mut body)?;
371    let mut payload = Vec::with_capacity(len + 1);
372    payload.push(PayloadType::MacCommand as u8);
373    payload.extend_from_slice(&body[..len]);
374    mac.send_unicast(from, peer, &payload, options)
375        .await
376        .map_err(Into::into)
377}
