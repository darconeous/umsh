1//! The device's answer to Node Management Requests.
2//!
3//! This is the seam between the mesh and the ULCP session. The node's
4//! receive path taps arriving Node Management Request payloads,
5//! [`admit`] decides whether the sender is entitled to be heard at all,
6//! and [`responder_loop`] runs each admitted exchange: the
7//! [`DeviceEngine`] reads the envelope, the session—reached through the
8//! driver's event loop, since it belongs to that task—serves the frame
9//! inside, and the engine wraps the answer, retaining it where executing
10//! the request again would be wrong.
11//!
12//! Two things are deliberately not here. The session's property surface
13//! is unchanged, because an administrative exchange runs through the same
14//! dispatch as the local link (`Session::handle_admin_frame`). And no
15//! part of the exchange holds a borrow across an await: the request
16//! crosses to the driver as bytes and the response comes back as bytes.
17

40/// What one secure unicast frame spends before its application payload
41/// begins, itemized against the packet format:
42///
43/// ```text
44///   1  FCF
45///   3  destination node hint
46///   3  source node hint
47///   5  SECINFO: SCF and the frame counter
48///  32  a source route of up to 15 router hints, with its option header
49///   3  region code
50///   1  trace route
51///   1  end-of-options marker
52///   8  MIC
53///   1  payload type
54///  --
55///  58
56/// ```
57///
58/// Taken as 75 to leave room for an option this list does not yet know
59/// about. The asymmetry is deliberate: over-reserving costs a slightly
60/// smaller fragment and one more exchange, while under-reserving costs a
61/// response that cannot be sent at all—which an administrator cannot
62/// tell from a lost packet.
63///
64/// A full 32-octet source key is not reserved for. A response goes to an
65/// administrator, which is by definition a registered peer, so the hint
66/// form always suffices.
67const FRAME_RESERVE: usize = 75;
68
69/// The largest Node Management payload this device produces.
70pub const ADMIN_PAYLOAD_MAX: usize = umsh_radio_loraphy::MAX_PAYLOAD - FRAME_RESERVE;
71
72/// The largest reply the session hands back for one exchange: an
73/// [`AdminFrame`] is all that crosses back from the driver.
74const ADMIN_REPLY_MAX: usize = FRAME_IN_MAX;
75
76/// The octets every retained reply shares. A read too large for one
77/// payload is retained whole and every continuation is cut from it, so
78/// this is room for two administrators to be partway through the largest
79/// read at once, or for one to be while writes are retained around it.
80const ADMIN_POOL: usize = 2 * ADMIN_REPLY_MAX;
81
82/// A payload that cannot hold an envelope plus a frame is not a budget,
83/// it is a bug in the reserve above.
84const _: () = assert!(
85    ADMIN_PAYLOAD_MAX > umsh_node_mgmt::envelope::OVERHEAD_MAX + 32,
86    "FRAME_RESERVE leaves no room for a Node Management exchange"
87);
88
89/// An administrator cannot ask what a device's budget is, so it assumes
90/// the smallest one a device is allowed to have. A device below that
91/// would silently drop requests it is required to answer.
92const _: () = assert!(
93    ADMIN_PAYLOAD_MAX >= umsh_node_mgmt::PAYLOAD_MAX,
94    "this radio's payload is smaller than an administrator assumes"
95);

97/// One admitted request on its way to the responder.
98struct Request {
99    from: [u8; 32],
100    /// The channel the request arrived on, when it came as a blind unicast.
101    /// The response follows it back rather than naming both parties in the
102    /// clear.
103    channel: Option<ChannelId>,
104    /// The request's `ack_mic`, which the response carries back so the
105    /// administrator's MAC can take the response as the acknowledgment
106    /// should the standalone one be lost.
107    ack_mic: Option<[u8; 4]>,
108    payload: heapless::Vec<u8, ADMIN_PAYLOAD_MAX>,
109}
110
111/// Admitted requests, from the node's receive callback to the responder
112/// task.
113///
114/// One slot. An administrator may not have more than one exchange
115/// outstanding, and the responder serves one at a time, so a second
116/// arrival while one is in flight is either a different administrator or
117/// a retransmission—and both are better served by being asked again
118/// than by being queued behind an exchange that may take several radio
119/// round trips.
120static REQUESTS: Channel<NodeMutex, Request, 1> = Channel::new();
121
122/// The mirrored `PROP_DEV_ADMINS`.
123///
124/// Mirrored rather than read from the session because authorization
125/// happens in the node's receive callback, which cannot borrow the
126/// session and cannot await. An empty list disables node management
127/// entirely, which is the post-reset default.
128static ADMINS: BlockingMutex<
129    CriticalSectionRawMutex,
130    RefCell<heapless::Vec<[u8; 32], MAX_DEV_ADMINS>>,
131> = BlockingMutex::new(RefCell::new(heapless::Vec::new()));
132
133/// Requests dropped before reaching the engine, by reason.
134static UNAUTHORIZED: AtomicU32 = AtomicU32::new(0);
135static NOT_UNICAST: AtomicU32 = AtomicU32::new(0);
136static OVERSIZE: AtomicU32 = AtomicU32::new(0);
137static BUSY: AtomicU32 = AtomicU32::new(0);
138static RESPONSES_DROPPED: AtomicU32 = AtomicU32::new(0);
139
140/// What the binding refused, for a device that has to explain itself
141/// without having answered anybody.
142#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
143pub struct AdminCounters {
144    /// Requests from a node that is not a listed administrator, or whose
145    /// source the MAC could not authenticate.
146    pub unauthorized: u32,
147    /// Requests that arrived by multicast or broadcast.
148    pub not_unicast: u32,
149    /// Requests larger than this device's payload ceiling.
150    pub oversize: u32,
151    /// Requests that arrived while another exchange was in flight.
152    pub busy: u32,
153    /// Response payloads, which a device never solicits.
154    pub responses_dropped: u32,
155}
156
157/// What the binding has refused since boot.
158pub fn counters() -> AdminCounters {
159    AdminCounters {
160        unauthorized: UNAUTHORIZED.load(Ordering::Relaxed),
161        not_unicast: NOT_UNICAST.load(Ordering::Relaxed),
162        oversize: OVERSIZE.load(Ordering::Relaxed),
163        busy: BUSY.load(Ordering::Relaxed),
164        responses_dropped: RESPONSES_DROPPED.load(Ordering::Relaxed),
165    }
166}

168/// Republish the administrator list from a device-domain snapshot. Called
169/// by the device-domain sync loop on every snapshot, so a key added over
170/// either binding takes effect without a reboot.
171pub fn publish_dev_domain(snapshot: &DevDomainSnapshot) {
172    ADMINS.lock(|cell| {
173        let mut admins = cell.borrow_mut();
174        admins.clear();
175        for key in snapshot.admins.iter() {
176            let _ = admins.push(*key);
177        }
178    });
179}
180
181fn is_admin(key: &[u8; 32]) -> bool {
182    ADMINS.lock(|cell| cell.borrow().iter().any(|listed| listed == key))
183}

185/// Decide whether an arriving packet is a Node Management Request this
186/// device will act on, and queue it if so.
187///
188/// Registered as a receive handler at bring-up; returns `false` always,
189/// so the packet still reaches every other observer. Everything refused
190/// is refused silently—an unlisted sender learns nothing about whether
191/// the device is manageable, not even that it declined to say.
192pub fn admit(packet: &ReceivedPacketRef<'_>) {
193    match packet.payload_type() {
194        PayloadType::NodeManagementRequest => {}
195        // A device never solicits anything, so a response addressed to it
196        // is either misdirected or an attempt to confuse it.
197        PayloadType::NodeManagementResponse => {
198            RESPONSES_DROPPED.fetch_add(1, Ordering::Relaxed);
199            return;
200        }
201        _ => return,
202    }
203    if !matches!(
204        packet.packet_family(),
205        PacketFamily::Unicast | PacketFamily::BlindUnicast
206    ) {
207        NOT_UNICAST.fetch_add(1, Ordering::Relaxed);
208        return;
209    }
210    // Both conditions matter and neither implies the other: an
211    // unauthenticated source could claim any key, and an authenticated
212    // one that is not listed is simply not an administrator here.
213    let Some(from) = packet.from_key().filter(|_| packet.source_authenticated()) else {
214        UNAUTHORIZED.fetch_add(1, Ordering::Relaxed);
215        return;
216    };
217    if !is_admin(&from.0) {
218        UNAUTHORIZED.fetch_add(1, Ordering::Relaxed);
219        return;
220    }
221    let mut payload = heapless::Vec::new();
222    if payload.extend_from_slice(packet.payload()).is_err() {
223        // Larger than this device can answer within. Dropped rather than
224        // truncated: a truncated envelope would be answered
225        // STATUS_PARSE_ERROR, which is a worse explanation than silence.
226        OVERSIZE.fetch_add(1, Ordering::Relaxed);
227        return;
228    }
229    if REQUESTS
230        .try_send(Request {
231            from: from.0,
232            channel: packet.channel().map(|c| c.id()),
233            ack_mic: packet.ack_mic(),
234            payload,
235        })
236        .is_err()
237    {
238        BUSY.fetch_add(1, Ordering::Relaxed);
239    }
240}

242/// Serve admitted Node Management exchanges forever.
243///
244/// `nonce` is drawn once per boot from the platform's cryptographic RNG.
245/// It is what stops a cursor issued before a reboot from being honored
246/// after one, when the engine's serials have started over.
247pub async fn responder_loop<CS: CounterStore + 'static, M: RawMutex + 'static>(
248    node: DeviceNode<CS>,
249    input: &'static InputChannel<M>,
250    nonce: u16,
251) {
252    let mut engine: DeviceEngine<ADMIN_PAYLOAD_MAX, ADMIN_POOL> = DeviceEngine::new(nonce);
253    let mut out = [0u8; ADMIN_PAYLOAD_MAX];
254    loop {
255        let request = REQUESTS.receive().await;
256        let now_ms = Instant::now().as_millis();
257        let mut completion = None;
258        let len = match engine.begin(&request.from, &request.payload, now_ms, &mut out) {
259            Ingress::Drop(reason) => {
260                debug_log(format_args!("admin: dropped {reason:?}"));
261                continue;
262            }
263            Ingress::Respond { len } => Some(len),
264            Ingress::Dispatch(dispatch) => {
265                let reply = serve(input, &dispatch).await;
266                completion = reply.completion;
267                match engine.complete(&reply.frame, &mut out) {
268                    Ok(len) => len,
269                    Err(error) => {
270                        if let Some(completion) = completion {
271                            completion.complete(false);
272                        }
273                        // Nothing goes out. The administrator retransmits,
274                        // and gets here again—which is the honest
275                        // outcome for a reply this device cannot carry.
276                        debug_log(format_args!("admin: reply REFUSED {error:?}"));
277                        continue;
278                    }
279                }
280            }
281        };
282        // A reset-class command is answered by no payload at all; its
283        // delivery was confirmed by the MAC acknowledgment of the request.
284        let Some(len) = len else { continue };
285        let sent = respond(
286            &node,
287            &request.from,
288            request.channel,
289            request.ack_mic,
290            &out[..len],
291            completion,
292        )
293        .await;
294        if let Some(completion) = completion {
295            if !completion.complete(sent) {
296                engine.fail_retained(&request.from, [request.payload[0], request.payload[1]]);
297            }
298        }
299    }
300}

302/// Hand one frame to the session and wait for its answer.
303///
304/// The session belongs to the driver's task, so the exchange crosses the
305/// event loop. It always answers—an empty reply is the answer for a
306/// reset—so this never has to time out.
307async fn serve<M: RawMutex + 'static>(
308    input: &'static InputChannel<M>,
309    dispatch: &Dispatch<'_>,
310) -> AdminReply {
311    let mut frame = AdminFrame::new();
312    if frame.extend_from_slice(dispatch.frame).is_err() {
313        // The frame came out of a payload smaller than this buffer.
314        debug_assert!(
315            false,
316            "admin request frame exceeds the driver's frame buffer"
317        );
318        return AdminReply {
319            frame: AdminFrame::new(),
320            completion: None,
321        };
322    }
323    // A read is built whole and continued from the engine's copy; anything
324    // else—a write sequence above all—is measured against one payload,
325    // because there is no continuing it: the binding's rule is that an
326    // entry whose reply would not fit is not executed at all. The engine
327    // has already said which.
328    input
329        .send(InEvent::Admin {
330            frame,
331            reply_budget: dispatch.budget,
332        })
333        .await;
334    ADMIN_REPLY.receive().await
335}

337/// Send one response payload back to the administrator.
338///
339/// No acknowledgment is requested. The administrator's token retry is the
340/// reliability layer for this binding, and it covers a lost response and
341/// a lost request alike—an ack would only tell the device something it
342/// has no use for. The response does carry the request's `ack_mic`, so it
343/// acknowledges the request itself if the standalone ack goes missing.
344async fn respond<CS: CounterStore + 'static>(
345    node: &DeviceNode<CS>,
346    to: &[u8; 32],
347    channel: Option<ChannelId>,
348    ack_mic: Option<[u8; 4]>,
349    payload: &[u8],
350    completion: Option<ReplyCompletion>,
351) -> bool {
352    let mut wire = heapless::Vec::<u8, { ADMIN_PAYLOAD_MAX + 1 }>::new();
353    if wire
354        .push(PayloadType::NodeManagementResponse as u8)
355        .is_err()
356        || wire.extend_from_slice(payload).is_err()
357    {
358        debug_assert!(false, "admin response exceeds ADMIN_PAYLOAD_MAX");
359        return false;
360    }
361    if let Some(completion) = completion {
362        if !completion.is_pending() {
363            return false;
364        }
365        let Ok(ticket) = node
366            .send_response_tracked(&PublicKey(*to), channel, &wire, ack_mic)
367            .await
368        else {
369            return false;
370        };
371        while completion.is_pending() && !ticket.is_finished() {
372            Timer::after(Duration::from_millis(5)).await;
373        }
374        let sent = ticket.was_transmitted() && completion.is_pending();
375        if let Some(token) = ticket.token() {
376            node.cancel_send(token).await;
377        }
378        return sent;
379    }
380    // Built in the call rather than held across this function's awaits:
381    // this task's future is a static, and every octet it keeps is one
382    // fewer for the stack on the boards whose stack is what statics leave.
383    let options = SendOptions {
384        ack_mic,
385        ..SendOptions::default()
386    };
387    if node
388        .send_response(&PublicKey(*to), channel, &wire, &options)
389        .await
390        .is_err()
391    {
392        debug_log(format_args!(
393            "admin: response to {:02x}{:02x}.. FAILED",
394            to[0], to[1]
395        ));
396        return false;
397    }
398    true
