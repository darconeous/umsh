1//! The board-agnostic ULCP session driver (Phase 5, increment C).
2//!
3//! This is the extraction of the nRF firmware's `device_task` select loop, its
4//! `apply_effect` radio-effect dispatcher, and the `Emitter` frame stager—
5//! the one copy of the session-driving logic shared by every device
6//! firmware (T-Echo, T-1000E, Heltec V3). The board personalities that used
7//! to be `cfg(feature = "t1000e")` forks inside the loop are expressed as
8//! [`DeviceEnv`] hooks with no-op defaults, so a new board supplies exactly the
9//! couplings it has and nothing else.
10//!
11//! The split of responsibilities:
12//!
13//! - **This module** owns the protocol loop: transport arbitration, frame
14//!   handling, radio RX/TX-completion processing, every deferred `Effect`
15//!   arm (save/clear/wipe/provision/PIN/RSSI/battery/drain), asynchronous
16//!   property publication ([`DeviceEnv::battery_event`]), and the
17//!   device-domain mirror.
18//! - **The board** owns the edges: transport tasks feeding [`InEvent`]s and
19//!   draining [`TransportChannels`], the radio runner + mux serving the
20//!   session's virtual [`Channels`] bundle, and an [`DeviceEnv`] implementation
21//!   wiring persistence, entropy, pairing, and indicators to its hardware.

49/// Derive a device identity's public key and its persisted record from a
50/// raw Ed25519 secret.
51///
52/// Shared by first-boot generation and `Effect::ProvisionIdentity` so the
53/// two can never disagree about the derivation or the record layout.
54///
55/// The caller supplies `secret` and owns the question this function
56/// cannot answer: it **MUST** come from a cryptographic RNG with real
57/// entropy behind it. On the nRF boards that is the hardware TRNG with
58/// bias correction enabled; on Espressif it is `EspCryptoRng`, which
59/// refuses to exist unless the RF noise source is live.
60pub fn device_identity_record(secret: &[u8; 32]) -> ([u8; 32], [u8; proto::IDENTITY_PAYLOAD_LEN]) {
61    let public_key = SoftwareIdentity::from_secret_bytes(secret).public_key().0;
62    (public_key, proto::encode_identity(secret, &public_key))
63}
64
65/// Read the device identity a boot mount of the identity journal
66/// returned, checking the stored public key against its secret.
67///
68/// A record whose halves disagree is [`proto::StoredIdentity::
69/// Unparseable`]: there is no telling which half is damaged, and the
70/// record must survive for whoever can tell.
71pub fn stored_device_identity(payload: Option<&[u8]>) -> proto::StoredIdentity {
72    match proto::classify_identity(payload) {
73        proto::StoredIdentity::Keys { secret, public }
74            if device_identity_record(&secret).0 != public =>
75        {
76            proto::StoredIdentity::Unparseable
77        }
78        stored => stored,
79    }
80}

253/// Frames held for the device's own interface until it reads them.
254pub const DEVICE_LINK_DEPTH: usize = 8;

256/// The return path to the device's own settings interface: the answers
257/// to its [`InEvent::Device`] requests, and everything the device
258/// publishes.
259///
260/// An answer always gets through: the driver waits for room. A
261/// publication never waits, because the interface's task may itself be
262/// waiting on the driver. One that finds the link full is dropped, and
263/// the interface hears [`DeviceFrame::Overrun`] in its place, after the
264/// frames that were queued ahead of it.
265pub struct DeviceLink<M: RawMutex> {
266    frames: Channel<M, FrameBuf, DEVICE_LINK_DEPTH>,
267    overrun: AtomicBool,
268}
269
270/// What the device's own interface reads from its [`DeviceLink`].
271pub enum DeviceFrame {
272    /// An answer or a publication, as the session emitted it.
273    Frame(FrameBuf),
274    /// Publications were dropped while the link was full, so anything the
275    /// interface holds may be out of date.
276    Overrun,
277}
278
279impl<M: RawMutex> DeviceLink<M> {
280    pub const fn new() -> Self {
281        Self {
282            frames: Channel::new(),
283            overrun: AtomicBool::new(false),
284        }
285    }
286
287    /// The next frame for the interface, if there is one.
288    pub fn try_receive(&self) -> Option<DeviceFrame> {
289        if let Ok(frame) = self.frames.try_receive() {
290            return Some(DeviceFrame::Frame(frame));
291        }
292        self.overrun
293            .swap(false, Ordering::AcqRel)
294            .then_some(DeviceFrame::Overrun)
295    }
296
297    /// Wait until [`DeviceLink::try_receive`] has something.
298    ///
299    /// An overrun is only ever recorded against a full link, so waiting
300    /// for a frame is waiting for it too.
301    pub async fn ready(&self) {
302        self.frames.ready_to_receive().await
303    }
304
305    fn publish(&self, frame: &[u8]) {
306        let Ok(copy) = FrameBuf::from_slice(frame) else {
307            debug_assert!(false, "a publication exceeds FrameBuf");
308            return;
309        };
310        if self.frames.try_send(copy).is_err() {
311            self.overrun.store(true, Ordering::Release);
312        }
313    }
314
315    async fn answer(&self, frame: &[u8]) {
316        match FrameBuf::from_slice(frame) {
317            Ok(copy) => self.frames.send(copy).await,
318            Err(()) => debug_assert!(false, "an answer exceeds FrameBuf"),
319        }
320    }
321}

323impl<M: RawMutex> Default for DeviceLink<M> {
324    fn default() -> Self {
325        Self::new()
326    }
327}

1007/// Collects frames emitted synchronously by the session, then flushes
1008/// them to the active transport's output queue asynchronously. The
1009/// session emits at most one frame per call; two slots give headroom.
1010struct Emitter {
1011    finish_ble_wipe: bool,
1012    completion: Option<crate::reply_completion::ReplyCompletion>,
1013    /// The request being served changes properties, so its answers are
1014    /// told to the listeners that did not ask.
1015    mirror: bool,
1016    bufs: [[u8; FRAME_OUT_MAX]; 2],
1017    lens: [usize; 2],
1018    count: usize,
1019}
1020
1021impl Emitter {
1022    const fn new() -> Self {
1023        Self {
1024            finish_ble_wipe: false,
1025            completion: None,
1026            mirror: false,
1027            bufs: [[0; FRAME_OUT_MAX]; 2],
1028            lens: [0; 2],
1029            count: 0,
1030        }
1031    }
1032
1033    /// Copy one raw ULCP frame into the next slot.
1034    ///
1035    /// The session is expected to emit at most `bufs.len()` frames per call
1036    /// and every frame is expected to fit `FRAME_OUT_MAX`. Both invariants are
1037    /// asserted in debug builds so a future session change that violates
1038    /// them is caught rather than silently dropping a response.
1039    fn push(&mut self, frame: &[u8]) {
1040        if self.count >= self.bufs.len() {
1041            debug_assert!(
1042                false,
1043                "Emitter overflow: session emitted more frames per call than staging slots"
1044            );
1045            return;
1046        }
1047        if frame.len() <= FRAME_OUT_MAX {
1048            self.bufs[self.count][..frame.len()].copy_from_slice(frame);
1049            self.lens[self.count] = frame.len();
1050            self.count += 1;
1051        } else {
1052            debug_assert!(false, "Emitter: ULCP frame exceeds FRAME_OUT_MAX");
1053        }
1054    }
1055
1056    /// Hand all staged frames to whoever is being answered.
1057    async fn flush<M: RawMutex>(&mut self, sink: &mut ReplySink<'_, M>) {
1058        for index in 0..self.count {
1059            let frame = &self.bufs[index][..self.lens[index]];
1060            let to_host = HostFrame {
1061                completion: self.completion,
1062                finish_ble_wipe: self.finish_ble_wipe && index + 1 == self.count,
1063            };
1064            match sink {
1065                ReplySink::Transport {
1066                    destination,
1067                    out,
1068                    device,
1069                } => {
1070                    if let Some(link) = device {
1071                        if is_publication(frame) {
1072                            link.publish(frame);
1073                        } else if self.mirror {
1074                            let mut mirrors = Mirrors::new(frame);
1075                            while let Some(mirror) = mirrors.next() {
1076                                link.publish(mirror);
1077                            }
1078                        }
1079                    }
1080                    to_host.send(*destination, out, frame).await;
1081                }
1082                ReplySink::Admin {
1083                    reply,
1084                    destination,
1085                    out,
1086                    device,
1087                } => {
1088                    // An exchange is one request and one response. A
1089                    // second frame would mean the session emitted
1090                    // something unsolicited, which this binding does not
1091                    // carry, so keep the first and account for the rest.
1092                    if reply.is_empty() {
1093                        let _ = reply.extend_from_slice(frame);
1094                        if self.mirror {
1095                            let mut mirrors = Mirrors::new(frame);
1096                            while let Some(mirror) = mirrors.next() {
1097                                if let Some(link) = device {
1098                                    link.publish(mirror);
1099                                }
1100                                HostFrame::default().send(*destination, out, mirror).await;
1101                            }
1102                        }
1103                    } else {
1104                        debug_assert!(false, "admin exchange emitted more than one frame");
1105                    }
1106                }
1107                ReplySink::Device {
1108                    link,
1109                    destination,
1110                    out,
1111                } => {
1112                    if is_publication(frame) {
1113                        // Everyone listening hears a publication, the host
1114                        // included, whoever's request it came out of.
1115                        link.publish(frame);
1116                        HostFrame::default().send(*destination, out, frame).await;
1117                    } else {
1118                        if self.mirror {
1119                            let mut mirrors = Mirrors::new(frame);
1120                            while let Some(mirror) = mirrors.next() {
1121                                HostFrame::default().send(*destination, out, mirror).await;
1122                            }
1123                        }
1124                        // Queued is delivered: the interface is in process.
1125                        link.answer(frame).await;
1126                        if let Some(completion) = self.completion {
1127                            completion.complete(true);
1128                        }
1129                    }
1130                }
1131            }
1132        }
1133        self.count = 0;
1134        self.finish_ble_wipe = false;
1135        self.completion = None;
1136    }
1137}

1139/// Whether `frame` is one everyone listening to the device receives,
1140/// rather than an answer for whoever asked.
1141fn is_publication(frame: &[u8]) -> bool {
1142    umsh_ulcp::Frame::parse(frame).is_ok_and(|parsed| parsed.is_publication())
1143}

1145/// How one frame goes out to the attached host.
1146#[derive(Default)]
1147struct HostFrame {
1148    completion: Option<crate::reply_completion::ReplyCompletion>,
1149    /// The final frame of a bond wipe, if it goes out over BLE.
1150    finish_ble_wipe: bool,
1151}
1152
1153impl HostFrame {
1154    async fn send<M: RawMutex>(
1155        self,
1156        destination: Option<(Transport, u32)>,
1157        out: &TransportChannels<M>,
1158        frame: &[u8],
1159    ) {
1160        // Nobody attached: the frame has nowhere to go.
1161        let Some((transport, generation)) = destination else {
1162            if let Some(completion) = self.completion {
1163                completion.complete(false);
1164            }
1165            return;
1166        };
1167        let mut copy: FrameBuf = heapless::Vec::new();
1168        if copy.extend_from_slice(frame).is_err() {
1169            // FRAME_OUT_MAX == FrameBuf capacity, so this cannot happen;
1170            // assert in debug rather than silently drop.
1171            debug_assert!(false, "Emitter frame copy exceeded FrameBuf capacity");
1172            return;
1173        }
1174        let send = out.for_transport(transport).send(OutFrame {
1175            completion: self.completion,
1176            finish_ble_wipe: self.finish_ble_wipe && transport == Transport::Ble,
1177            generation,
1178            frame: copy,
1179        });
1180        if let Some(completion) = self.completion {
1181            if embassy_time::with_deadline(completion.deadline(), send)
1182                .await
1183                .is_err()
1184            {
1185                completion.complete(false);
1186            }
1187        } else {
1188            send.await;
1189        }
1190    }
1191}

1193/// Where the frames a session emits while serving one command are
1194/// delivered.
1195///
1196/// The command paths do not know which they are feeding—that is the
1197/// point. A deferred property read makes the same `respond_*` call
1198/// whether the value is going to an attached host over USB or back to an
1199/// administrator across the mesh.
1200enum ReplySink<'a, M: RawMutex> {
1201    /// The attached host's output queue, or nowhere when none is
1202    /// attached. What the device publishes reaches its own interface
1203    /// too, on a board that has one.
1204    Transport {
1205        destination: Option<(Transport, u32)>,
1206        out: &'a TransportChannels<M>,
1207        device: Option<&'a DeviceLink<M>>,
1208    },
1209    /// The response frame of a Node Management exchange, staged for the
1210    /// responder that will envelope and transmit it. Empty when the
1211    /// command produced no response, which is what a reset does. What
1212    /// the administrator changed reaches the attached host and the
1213    /// device's own interface as publications.
1214    Admin {
1215        reply: &'a mut AdminFrame,
1216        destination: Option<(Transport, u32)>,
1217        out: &'a TransportChannels<M>,
1218        device: Option<&'a DeviceLink<M>>,
1219    },
1220    /// The device's own interface. Answers go back to it alone, and what
1221    /// the device publishes while serving it reaches the attached host
1222    /// too.
1223    Device {
1224        link: &'a DeviceLink<M>,
1225        destination: Option<(Transport, u32)>,
1226        out: &'a TransportChannels<M>,
1227    },
1228}

1230impl<M: RawMutex + 'static> ReplySink<'static, M> {
1231    /// Frames for the attached host: everything the session emits outside
1232    /// an exchange, and the answers to the host's own frames.
1233    fn host<const RX: usize, const TX: usize>(
1234        arbitration: &SessionArbitration,
1235        rt: &DeviceRuntime<M, RX, TX>,
1236    ) -> Self {
1237        Self::Transport {
1238            destination: arbitration.destination(),
1239            out: rt.out,
1240            device: rt.device,
1241        }
1242    }

1246async fn apply_effect<A, S, const TXQ: usize, M, const RX: usize, const TX: usize, E>(
1247    session: &Session<A, S, TXQ>,
1248    effect: Option<Effect>,
1249    rt: &DeviceRuntime<M, RX, TX>,
1250    env: &mut E,
1251) where
1252    A: AesProvider,

1254    M: RawMutex,

1261            // sets these converters accept, so None here is
1262            // unreachable; bail out defensively rather than panic.

1282            // builds.
1283            #[cfg(feature = "device-node")]

1285                crate::device_node::set_tx_power_dbm(settings.tx_power_dbm);
1286                crate::device_node::set_radio_enabled(settings.enabled);
1287            }
1288        }
1289        Some(Effect::StartTransmit) => {

1292                env.trace(format_args!(
1293                    "radio tx staging=FAILED len={}",
1294                    session.tx_data().len()
1295                ));
1296                return;
1297            }
1298            let power_dbm = match session.tx_power() {
1299                TxPower::Default => None,
1300                TxPower::Max => Some(i32::from(session.max_tx_power_dbm())),
1301                TxPower::Dbm(dbm) => Some(i32::from(dbm)),
1302            };
1303            // Mark the load for the board's battery level estimator (the
1304            // radio runner transmits within milliseconds of this).
1305            env.note_transmit_load();

1307                CadPolicy::Skip
1308            } else {
1309                CadPolicy::Gate
1310            };

1312                .tx
1313                .send(TxRequest {
1314                    data,
1315                    power_dbm,
1316                    cad,
1317                })
1318                .await;
1319        }
1320        Some(Effect::DeviceNameChanged) => {
1321            env.publish_device_name(session.device_name()).await;
1322        }
1323        Some(Effect::ApplyAlert(state)) => {
1324            env.set_alert(state);
1325        }
1326        Some(Effect::ApplyTime { epoch }) => {
1327            env.apply_time(epoch).await;
1328        }
1329        // Deferred effects needing `&mut Session` + the emitter are
1330        // handled inline in the run loop rather than here.
1331        Some(Effect::SampleRssi { .. })
1332        | Some(Effect::SignIdentity { .. })
1333        | Some(Effect::SampleBattery { .. })
1334        | Some(Effect::SampleBatteryGroup { .. })
1335        | Some(Effect::SampleIlluminance { .. })
1336        | Some(Effect::SampleTemperatures { .. })
1337        | Some(Effect::ReadTemperatureNames { .. })
1338        | Some(Effect::ReadTime { .. })
1339        | Some(Effect::SampleGnss { .. })
1340        | Some(Effect::SetPairingPin { .. })
1341        | Some(Effect::ClearBleBonds { .. })
1342        | Some(Effect::SetBlePairing { .. })
1343        | Some(Effect::SetWifiScanning { .. })
1344        | Some(Effect::SelectWifiNetwork { .. })
1345        | Some(Effect::InsertWifiNetwork { .. })
1346        | Some(Effect::RemoveWifiNetwork { .. })
1347        | Some(Effect::ReadNetworkTable { .. })
1348        | Some(Effect::I2cTransfer { .. })
1349        | Some(Effect::I2cScan { .. })
1350        | Some(Effect::DrainQueue)
1351        | Some(Effect::SaveSnapshot { .. })
1352        | Some(Effect::ClearSaved { .. })
1353        | Some(Effect::ProvisionIdentity { .. })
1354        | Some(Effect::FactoryReset)
1355        | Some(Effect::Reboot)
1356        | Some(Effect::Dfu { .. })
1357        | Some(Effect::Announce { .. })
1358        | None => {}
1359    }

1362/// Keep the saved snapshot in step with a change the device made on its
1363/// own—a switch the operator flipped at the board, a network picked from
1364/// its menu—so the change survives the next reboot.
1365///
1366/// Only a [`SavedStatus::Current`] snapshot is kept in step. A device
1367/// with nothing saved gets nothing saved: manufacturing a snapshot from
1368/// a button press would persist every other live-only value with it. A
1369/// device that rejected its newest generation at boot (`Fallback` or
1370/// `Unreadable`) must not write one either: the write would become the
1371/// newest generation, bury the rejected record under defaults or stale
1372/// values, and clear the fault `PROP_SAVED` exists to report. The
1373/// change still applies live in both cases; only a host's `CMD_SAVE`
1374/// writes over a fault.
1375async fn persist_device_change<A, S, const TXQ: usize, E>(
1376    session: &mut Session<A, S, TXQ>,
1377    snapshot_buf: &mut [u8],
1378    env: &mut E,
1379) where
1380    A: AesProvider,
1381    S: Sha256Provider,
1382    E: DeviceEnv,
1383{
1384    if session.saved_status() == SavedStatus::Current
1385        && let Some(len) = session.encode_snapshot(snapshot_buf)
1386        && env.persist_snapshot(&snapshot_buf[..len]).await.is_ok()
1387    {
1388        session.note_snapshot_saved();
1389    }
1390}

1392/// Mirror the session's device-domain node tables to the device node
1393/// when their generation moved (device-node plan increment 3).
1394/// `synced_version` is the caller's cache of the last published
1395/// generation. Cheap when nothing changed—one u32 compare—so the
1396/// loop runs it after every session interaction.
1397/// Generate, persist, and install a fresh device identity.
1398///
1399/// The counterpart to first-boot generation, for the one runtime path
1400/// that can leave the session without one: `CMD_CLEAR` followed by the
1401/// `CMD_RST` that completes a factory reset. A device identity is not a
1402/// commissioning step, so there is no state in which the operator has to
1403/// supply one.
1404///
1405/// A failure to draw entropy or to persist leaves the session
1406/// identityless, which is a worse outcome than either but not one this
1407/// layer can repair: it is reported and the next boot regenerates.
1408async fn regenerate_device_identity<A, S, const TXQ: usize, E>(
1409    session: &mut Session<A, S, TXQ>,
1410    env: &mut E,
1411) where
1412    A: AesProvider,
1413    S: Sha256Provider,
1414    E: DeviceEnv,
1415{
1416    let mut secret = [0u8; 32];
1417    if env.fill_secret(&mut secret).is_err() {
1418        env.trace(format_args!("device identity regenerate: entropy FAILED"));
1419        return;
1420    }
1421    let (public_key, payload) = device_identity_record(&secret);
1422    match env.persist_identity(&payload).await {
1423        Ok(()) => {
1424            session.set_boot_identity(public_key);
1425            env.trace(format_args!(
1426                "device identity regenerated after clear+reset"
1427            ));
1428        }
1429        Err(()) => env.trace(format_args!(
1430            "device identity regenerate: persist FAILED—none in effect"
1431        )),
1432    }
1433}

1526/// One ULCP frame to serve, and what its binding needs to know.
1527enum Exchange<'a> {
1528    /// A frame from the attached host. The reply is bounded by the
1529    /// transport frame, which is what the session assumes by default.
1530    Local(&'a [u8]),
1531    /// A frame from a mesh administrator, whose reply must fit
1532    /// `reply_budget` octets.
1533    Admin {
1534        frame: &'a [u8],
1535        reply_budget: usize,
1536    },
1537    /// A frame from the device's own settings interface.
1538    Device(&'a [u8]),
1539}
1540
1541/// Serve one ULCP frame to completion, deferred platform round trips
1542/// included, delivering everything the session emits to `sink`.
1543///
1544/// This is the whole of the driver's command path, and it is deliberately
1545/// one function for both bindings. A multi-property command is served
1546/// entry by entry: each deferred value returns here for its platform
1547/// round trip, and `resume_multi` hands back the next one until the reply
1548/// is emitted. For every other command `resume_multi` answers `None` and
1549/// the loop runs exactly once. None of that changes because the answer is
1550/// going to the mesh instead of to a cable—the only thing that changes
1551/// is where the frames go, which is `sink`'s business.
1552async fn serve_frame<A, S, const TXQ: usize, M, const RX: usize, const TX: usize, E>(
1553    session: &mut Session<A, S, TXQ>,
1554    exchange: Exchange<'_>,
1555    emitter: &mut Emitter,
1556    sink: &mut ReplySink<'_, M>,
1557    snapshot_buf: &mut [u8; SNAPSHOT_MAX],
1558    rt: &DeviceRuntime<M, RX, TX>,
1559    env: &mut E,
1560) -> Option<(umsh_ulcp::DfuMode, crate::reply_completion::ReplyCompletion)>
1561where
1562    A: AesProvider,
1563    S: Sha256Provider,
1564    M: RawMutex,
1565    E: DeviceEnv,
1566{
1567    let mut dfu = None;
1568    let now_ms = Instant::now().as_millis();
1569    let request = match exchange {
1570        Exchange::Local(bytes) | Exchange::Admin { frame: bytes, .. } | Exchange::Device(bytes) => {
1571            bytes
1572        }
1573    };
1574    emitter.mirror = Writes::of(request).any;
1575    let mut pending = match exchange {
1576        Exchange::Local(bytes) => {
1577            session.handle_frame(bytes, now_ms, &mut |frame: &[u8]| emitter.push(frame))
1578        }
1579        Exchange::Admin {
1580            frame,
1581            reply_budget,
1582        } => session.handle_admin_frame(frame, now_ms, reply_budget, &mut |frame: &[u8]| {
1583            emitter.push(frame)
1584        }),
1585        Exchange::Device(bytes) => {
1586            session.handle_device_frame(bytes, now_ms, &mut |frame: &[u8]| emitter.push(frame))
1587        }
1588    };
1589    emitter.flush(sink).await;
1590    while pending.is_some() {
1591        match pending.take() {
1592            Some(Effect::SampleRssi { tid }) => {
1593                // Round-trip to the radio runner for an
1594                // instantaneous RSSI sample, then answer the
1595                // deferred PROP_PHY_RSSI get.
1596                rt.ctl.request_rssi();
1597                let sample = rt.ctl.wait_rssi().await;
1598                session.respond_rssi(tid, sample, &mut |frame: &[u8]| emitter.push(frame));
1599                emitter.flush(sink).await;
1600            }
1601            Some(Effect::SignIdentity { tid }) => {
1602                // The session holds no signing key, so the
1603                // platform builds the canonical node-identity
1604                // payload from the same profile the Identity
1605                // Request responder advertises and signs it
1606                // with the device identity.
1607                let mut blob = [0u8; IDENTITY_BLOB_MAX];
1608                let signed = env.sign_identity(&mut blob).await;
1609                session.respond_identity_blob(
1610                    tid,
1611                    signed.map(|len| &blob[..len]).ok_or(()),
1612                    &mut |frame: &[u8]| emitter.push(frame),
1613                );
1614                emitter.flush(sink).await;
1615            }
1616            Some(Effect::SampleBattery { tid }) => {
1617                // Round-trip to the platform battery
1618                // source for a fresh measurement, then
1619                // answer the deferred PROP_BATTERY get.
1620                let sample = env.sample_battery().await;
1621                session.respond_battery(tid, sample, &mut |frame: &[u8]| emitter.push(frame));
1622                emitter.flush(sink).await;
1623            }
1624            Some(Effect::SampleBatteryGroup { tid, key, fields }) => {
1625                let sample = env.sample_battery_group(fields).await;
1626                session.respond_battery_group(tid, key, sample, &mut |frame: &[u8]| {
1627                    emitter.push(frame)
1628                });
1629                emitter.flush(sink).await;
1630            }
1631            Some(Effect::SampleTemperatures { tid }) => {
1632                let mut out =
1633                    [umsh_ulcp::temperature::UNKNOWN; umsh_ulcp_device::MAX_TEMPERATURE_SENSORS];
1634                let result = env.sample_temperatures(&mut out).await;
1635                let values = result.and_then(|len| out.get(..len).ok_or(Status::NOMEM));
1636                session.respond_temperatures(tid, values, &mut |frame: &[u8]| emitter.push(frame));
1637                emitter.flush(sink).await;
1638            }
1639            Some(Effect::ReadTemperatureNames { tid }) => {
1640                let mut out = [0; umsh_ulcp_device::TEMPERATURE_NAMES_MAX];
1641                let result = env.read_temperature_names(&mut out).await;
1642                let names = result.and_then(|len| out.get(..len).ok_or(Status::NOMEM));
1643                session
1644                    .respond_temperature_names(tid, names, &mut |frame: &[u8]| emitter.push(frame));
1645                emitter.flush(sink).await;
1646            }
1647            Some(Effect::SampleIlluminance { tid }) => {
1648                let millilux = env.sample_illuminance().await;
1649                session.respond_illuminance(tid, millilux, &mut |frame: &[u8]| emitter.push(frame));
1650                emitter.flush(sink).await;
1651            }
1652            Some(Effect::ReadTime { tid }) => {
1653                let epoch = env.read_time().await;
1654                session.respond_time(tid, epoch, &mut |frame: &[u8]| emitter.push(frame));
1655                emitter.flush(sink).await;
1656            }
1657            Some(Effect::SampleGnss { tid, key }) => {
1658                let sample = env.sample_gnss().await;
1659                session.respond_gnss(tid, key, sample, &mut |frame: &[u8]| emitter.push(frame));
1660                emitter.flush(sink).await;
1661            }
1662            Some(Effect::DrainQueue) => {
1663                // Deliver the covered frames one per
1664                // step, flushing between steps so the
1665                // two-slot emitter never overflows and
1666                // the transport applies backpressure.
1667                loop {
1668                    let more = session
1669                        .drain_step(Instant::now().as_millis(), &mut |frame: &[u8]| {
1670                            emitter.push(frame)
1671                        });
1672                    emitter.flush(sink).await;
1673                    if !more {
1674                        break;
1675                    }
1676                }
1677                env.queue_emptied();
1678            }
1679            Some(Effect::SaveSnapshot { tid }) => {
1680                let result = match session.encode_snapshot(snapshot_buf) {
1681                    Some(len) => env.persist_snapshot(&snapshot_buf[..len]).await,
1682                    None => Err(()),
1683                };
1684                session.respond_save(tid, result, &mut |frame: &[u8]| emitter.push(frame));
1685                emitter.flush(sink).await;
1686            }
1687            Some(Effect::ClearSaved { tid }) => {
1688                // CMD_CLEAR covers all persisted
1689                // provisioning: the snapshot and the
1690                // independently persisted device
1691                // identity. Each journal's tombstone is
1692                // individually atomic; an interruption
1693                // between them reports failure and the
1694                // host's retry completes the erase.
1695                let mut result = match env.clear_snapshot().await {
1696                    Ok(()) => env.clear_identity().await,
1697                    Err(()) => Err(()),
1698                };
1699                // The lists the board's own store keeps are
1700                // provisioning too.
1701                if result.is_ok() && session.holds_lists() {
1702                    result = env.persist_lists(None).await;
1703                }
1704                // With the identity durably gone, its
1705                // counter boundaries are dead weight;
1706                // drop them with it. (Kept if the
1707                // identity clear failed—the identity
1708                // then survives the reboot and still
1709                // needs its TX boundary.)
1710                if result.is_ok() {
1711                    env.clear_counters().await;
1712                }
1713                session.respond_clear(tid, result, &mut |frame: &[u8]| emitter.push(frame));
1714                emitter.flush(sink).await;
1715            }
1716            Some(Effect::ProvisionIdentity { tid }) => {
1717                // Build the keypair (drawing a fresh
1718                // secret from the platform RNG for
1719                // on-device generation), persist it, and
1720                // only then report the public key.
1721                let result = match session.identity_request() {
1722                    Some(source) => {
1723                        let secret = match source {
1724                            IdentitySource::Install(secret) => Ok(secret),
1725                            IdentitySource::Generate => {
1726                                let mut secret = [0u8; 32];
1727                                env.fill_secret(&mut secret).map(|()| secret)
1728                            }
1729                        };
1730                        match secret {
1731                            Ok(secret) => {
1732                                let (public_key, payload) = device_identity_record(&secret);
1733                                env.persist_identity(&payload).await.map(|()| public_key)
1734                            }
1735                            Err(()) => Err(()),
1736                        }
1737                    }
1738                    None => Err(()),
1739                };
1740                session.respond_identity(tid, result, &mut |frame: &[u8]| emitter.push(frame));
1741                emitter.flush(sink).await;
1742            }
1743            Some(Effect::SetPairingPin { tid, pin }) => {
1744                let applied = env.apply_pairing_pin(pin).await;
1745                session.respond_pin_set(
1746                    tid,
1747                    applied.then_some(()).ok_or(()),
1748                    &mut |frame: &[u8]| emitter.push(frame),
1749                );
1750                emitter.flush(sink).await;
1751            }
1752            Some(Effect::ClearBleBonds { tid }) => {
1753                // Commit revocation first, then mark the response for the BLE
1754                // sender's controller-completion fence. Queue insertion does
1755                // not authorize disconnecting before transmission completes.
1756                env.trace(format_args!(
1757                    "PROP_BLE_BOND_COUNT <- 0: forgetting every host"
1758                ));
1759                let reply_over_ble = matches!(
1760                    sink,
1761                    ReplySink::Transport {
1762                        destination: Some((Transport::Ble, _)),
1763                        ..
1764                    }
1765                );
1766                let cleared = env.clear_ble_bonds(reply_over_ble).await;
1767                emitter.finish_ble_wipe = cleared && reply_over_ble;
1768                session.respond_ble_bond_count(
1769                    tid,
1770                    cleared.then_some(()).ok_or(()),
1771                    &mut |frame: &[u8]| emitter.push(frame),
1772                );
1773                emitter.flush(sink).await;
1774            }
1775            Some(Effect::SetBlePairing { tid, open }) => {
1776                let applied = env.set_ble_pairing(open).await;
1777                env.trace(format_args!(
1778                    "PROP_BLE_PAIRING <- {open}: {}",
1779                    if applied { "applied" } else { "refused" }
1780                ));
1781                session.respond_ble_pairing(
1782                    tid,
1783                    applied.then_some(open).ok_or(()),
1784                    &mut |frame: &[u8]| emitter.push(frame),
1785                );
1786                emitter.flush(sink).await;
1787            }
1788            Some(Effect::SetWifiScanning { tid, scanning }) => {
1789                let result = env.set_wifi_scanning(scanning).await;
1790                session.respond_wifi_scanning(tid, result, &mut |frame: &[u8]| emitter.push(frame));
1791                emitter.flush(sink).await;
1792            }
1793            Some(Effect::SelectWifiNetwork { tid }) => {
1794                let result = env.select_wifi_network(session.pending_network()).await;
1795                session.respond_wifi_network(tid, result, &mut |frame: &[u8]| emitter.push(frame));
1796                emitter.flush(sink).await;
1797            }
1798            Some(Effect::InsertWifiNetwork { tid }) => {
1799                // The one place a Wi-Fi credential is handled, and
1800                // deliberately untraced.
1801                let result = env
1802                    .store_wifi_network(session.pending_network_entry())
1803                    .await;
1804                session.respond_wifi_network_stored(tid, result, &mut |frame: &[u8]| {
1805                    emitter.push(frame)
1806                });
1807                emitter.flush(sink).await;
1808            }
1809            Some(Effect::RemoveWifiNetwork { tid }) => {
1810                let result = env.forget_wifi_network(session.pending_network()).await;
1811                session.respond_wifi_network_forgotten(tid, result, &mut |frame: &[u8]| {
1812                    emitter.push(frame)
1813                });
1814                emitter.flush(sink).await;
1815            }
1816            Some(Effect::ReadNetworkTable { tid, key }) => {
1817                let mut table = [0u8; NETWORK_TABLE_MAX];
1818                let read = env.read_network_table(key, &mut table).await;
1819                let items = read.map(|len| &table[..len]);
1820                session.respond_network_table(tid, key, items, &mut |frame: &[u8]| {
1821                    emitter.push(frame)
1822                });
1823                emitter.flush(sink).await;
1824            }
1825            Some(Effect::I2cTransfer { tid }) => {
1826                // The result is staged in a static rather than on this
1827                // frame: `device_task` is shared by every image and the
1828                // smaller ESP32 boards have no stack headroom for a
1829                // 255-octet local.
1830                let mut out = I2C_RESULT.lock().await;
1831                let result = env.i2c_transfer(session.i2c_request(), &mut out[..]).await;
1832                env.trace(format_args!("CMD_I2C_TRANSFER: {result:?}"));
1833                let data = result.map(|len| &out[..len]);
1834                session.respond_i2c(tid, data, &mut |frame: &[u8]| emitter.push(frame));
1835                emitter.flush(sink).await;
1836            }
1837            Some(Effect::I2cScan { tid }) => {
1838                let mut out = I2C_RESULT.lock().await;
1839                let result = env.i2c_scan(session.i2c_scan_request(), &mut out[..]).await;
1840                env.trace(format_args!("CMD_I2C_SCAN: {result:?}"));
1841                let data = result.map(|len| &out[..len]);
1842                session.respond_i2c(tid, data, &mut |frame: &[u8]| emitter.push(frame));
1843                emitter.flush(sink).await;
1844            }
1845            Some(Effect::Announce { tid, request }) => {
1846                // The node behind the session is what announces, so a
1847                // build without one has nothing to do this with. Where
1848                // there is a node, the answer is whether the trigger
1849                // queue took the request—it is short by design, and a
1850                // full one means an announcement is already pending.
1851                #[cfg(feature = "device-node")]
1852                let result = {
1853                    use crate::device_node::{BeaconTrigger, request_beacon};
1854                    if request_beacon(BeaconTrigger::Host(request)) {
1855                        Ok(())
1856                    } else {
1857                        Err(Status::BUSY)
1858                    }
1859                };
1860                #[cfg(not(feature = "device-node"))]
1861                let result = {
1862                    let _ = request;
1863                    Err(Status::UNIMPLEMENTED)
1864                };
1865                env.trace(format_args!("CMD_ANNOUNCE: queued={}", result.is_ok()));
1866                session.respond_announce(tid, result, &mut |frame: &[u8]| emitter.push(frame));
1867                emitter.flush(sink).await;
1868            }
1869            Some(Effect::Dfu { tid, mode }) => {
1870                let prepared = env.prepare_dfu(mode).await;
1871                if let Ok(mode) = prepared {
1872                    let seconds = if matches!(sink, ReplySink::Admin { .. }) {
1873                        60
1874                    } else {
1875                        10
1876                    };
1877                    let completion = crate::reply_completion::ReplyCompletion::begin(
1878                        Instant::now() + embassy_time::Duration::from_secs(seconds),
1879                    );
1880                    emitter.completion = Some(completion);
1881                    dfu = Some((mode, completion));
1882                }
1883                session.respond_dfu(tid, prepared.map(|_| ()), &mut |frame: &[u8]| {
1884                    emitter.push(frame)
1885                });
1886                emitter.flush(sink).await;
1887            }
1888            Some(Effect::FactoryReset) => {
1889                // Hand off to the platform, which erases every
1890                // persistent journal and reboots. This never
1891                // returns; no acknowledgement is sent because the
1892                // reset drops the link. Any frames the session
1893                // already staged were flushed above.
1894                env.trace(format_args!("CMD_FACTORY_RESET: wiping all state + reboot"));
1895                env.factory_reset().await
1896            }
1897            Some(Effect::Reboot) => {
1898                // Same shape as the factory reset above and none of the
1899                // erasing: the platform restarts and the device comes
1900                // back as itself. Never returns, and answers nothing—
1901                // the reboot drops the link.
1902                env.trace(format_args!("CMD_REBOOT: restarting"));
1903                env.reboot().await
1904            }
1905            other => apply_effect(session, other, rt, env).await,
1906        }
1907        pending = session.resume_multi(Instant::now().as_millis(), &mut |frame: &[u8]| {
1908            emitter.push(frame)
1909        });
1910        emitter.flush(sink).await;
1911    }
1912    // Whatever this exchange was, the next frame is served on its own
1913    // terms. Restoring the local binding here rather than at the next
1914    // frame's arrival keeps every other emitting path—a publication, a
1915    // transmit completion, an alert deadline—reading the binding it
1916    // expects.
1917    session.end_admin_exchange();
1918    // A device identity always exists. `CMD_CLEAR` erases the stored one
1919    // without touching live state, and the `CMD_RST` that completes a
1920    // factory reset is where the live copy catches up—leaving the device
1921    // with none, which is the one state the invariant forbids. Regenerate
1922    // here, exactly as first boot would, so the only way to reach an
1923    // identityless device is to physically remove it from existence.
1924    //
1925    // The running device node keeps the *previous* key in its MAC until
1926    // the next boot (identity is fixed at bring-up), and stops
1927    // originating traffic because the dev-domain sync gate compares keys
1928    // rather than counting them.
1929    if session.dev_key().is_none() {
1930        regenerate_device_identity(session, env).await;
1931    }
1932    if session.queued_frame_count() == 0 {
1933        env.queue_emptied();
1934    }
1935    emitter.mirror = false;
1936    dfu
1937}

1939async fn finish_dfu(
1940    env: &mut impl DeviceEnv,
1941    dfu: Option<(umsh_ulcp::DfuMode, crate::reply_completion::ReplyCompletion)>,
1942) {
1943    if let Some((mode, completion)) = dfu {

1945            env.enter_dfu(mode);
1946        }
1947        env.trace(format_args!("DFU canceled: response transmission failed"));
1948    }

1951/// Drive the ULCP session forever: restore persisted state, then
1952/// select over host frames, radio receptions, and transmit completions,
1953/// executing every session effect through the board's [`DeviceEnv`].
1954///
1955/// The caller constructs the [`Session`] with its board profile
1956/// (`SessionConfig`) and boot status, mounts its journals, and hands the
1957/// stored payloads in; the driver owns everything after that.
1958pub async fn run<A, S, const TXQ: usize, M, const RX: usize, const TX: usize, E>(
1959    mut session: Session<A, S, TXQ>,
1960    boot_snapshot: Option<&[u8]>,
1961    boot_identity: Option<[u8; 32]>,
1962    rt: DeviceRuntime<M, RX, TX>,
1963    env: E,
1964) -> !
1965where
1966    A: AesProvider,
1967    S: Sha256Provider,
1968    M: RawMutex,
1969    E: DeviceEnv,
1970{
1971    let mut snapshot_buf = [0u8; SNAPSHOT_MAX];
1972    run_with_storage(
1973        &mut session,
1974        &mut snapshot_buf,
1975        boot_snapshot,
1976        boot_identity,
1977        rt,
1978        env,
1979    )
1980    .await
1981}
1982
1983/// Run with explicitly placed session and snapshot storage. No allocation occurs here.
1984pub async fn run_with_storage<A, S, const TXQ: usize, M, const RX: usize, const TX: usize, E>(
1985    session: &mut Session<A, S, TXQ>,
1986    snapshot_buf: &mut [u8; SNAPSHOT_MAX],
1987    boot_snapshot: Option<&[u8]>,
1988    boot_identity: Option<[u8; 32]>,
1989    rt: DeviceRuntime<M, RX, TX>,
1990    mut env: E,
1991) -> !
1992where
1993    A: AesProvider,
1994    S: Sha256Provider,
1995    M: RawMutex,
1996    E: DeviceEnv,
1997{
1998    let mut emitter = Emitter::new();
1999    let mut arbitration = SessionArbitration::new(rt.session_gen.load(Ordering::Acquire));
2000    // Last device-domain generation mirrored to the device node.
2001    // Force the first publication even on a bare device with no identity or
2002    // snapshot, then use generation changes for subsequent publications.
2003    let mut dev_domain_synced: u32 = session.dev_domain_version().wrapping_sub(1);
2004    // Shared staging buffer for the durable-write effect arms
2005    // (save/wipe). Held across their persist awaits, so as a
2006    // loop-lifetime local it costs one future slot instead of one
2007    // per arm.
2008
2009    // A board with its own interface hears what the device publishes
2010    // from the start, attached host or not.
2011    if rt.device.is_some() {
2012        session.open_device_interface();
2013    }
2014
2015    // The device identity is persisted independently of snapshots;
2016    // its post-reset value is whatever the identity journal holds.
2017    if let Some(public_key) = boot_identity {
2018        session.set_boot_identity(public_key);
2019    }
2020
2021    // Restore a stored snapshot before processing any host command:
2022    // the saved configuration is applied, the PHY re-enabled if it
2023    // was enabled when saved, and detached operation begins
2024    // immediately.
2025    //
2026    // A payload that does not decode is not the end of it. The journal
2027    // is multi-record and newest-generation-wins, so an older readable
2028    // generation usually sits behind the rejected one—and for an
2029    // unattended repeater, falling back to it is the only outcome that
2030    // keeps the device forwarding. Walk back a bounded number of
2031    // generations: a systematically undecodable payload is a firmware
2032    // bug rather than corruption, and re-walking the whole journal on
2033    // every boot would just be slower about it.
2034    if let Some(payload) = boot_snapshot {
2035        let mut generation = 0usize;
2036        let mut restored = session.restore_at_boot(payload);
2037        while let Err(error) = restored {
2038            env.trace(format_args!(
2039                "proto-store boot-restore generation=-{generation} REJECTED error={error:?}"
2040            ));
2041            session.note_snapshot_rejected();
2042            generation += 1;
2043            if generation > SNAPSHOT_FALLBACK_LIMIT {
2044                env.trace(format_args!(
2045                    "proto-store boot-restore fallback=EXHAUSTED limit={SNAPSHOT_FALLBACK_LIMIT}"
2046                ));
2047                break;
2048            }
2049            let Some(len) = env.older_snapshot(snapshot_buf).await else {
2050                env.trace(format_args!("proto-store boot-restore fallback=NONE"));
2051                break;
2052            };
2053            restored = session.restore_at_boot(&snapshot_buf[..len]);
2054        }
2055        match restored {
2056            Ok(effect) => {
2057                if generation > 0 {
2058                    env.trace(format_args!(
2059                        "proto-store boot-restore=FALLBACK generation=-{generation}"
2060                    ));
2061                    env.report_snapshot_rejected(true);
2062                } else {
2063                    env.trace(format_args!("proto-store boot-restore=ok"));
2064                }
2065                apply_effect(&session, Some(effect), &rt, &mut env).await;
2066            }
2067            Err(_) => {
2068                env.trace(format_args!("proto-store boot-restore=BARE"));
2069                env.report_snapshot_rejected(false);
2070            }
2071        }
2072    }
2073
2074    // A bare device has no restore effect. Publish its authoritative default
2075    // too, so the local radio switch has a value before the first host write.
2076    #[cfg(feature = "device-node")]
2077    crate::device_node::set_radio_enabled(session.settings().enabled);
2078
2079    // Publish the device domain once before any host interaction, on every
2080    // boot path rather than only after a successful restore.
2081    //
2082    // Two things depend on it. Detached multicast processing needs the
2083    // restored tables without waiting for an attach—the original reason.
2084    // And anything that waits for the domain to be published before acting
2085    // needs that publication to happen on a device that has never been
2086    // configured, where the answer is "the post-reset defaults" rather than
2087    // silence: the boot-time GNSS clock read waits on exactly this, and on
2088    // a bare device it would otherwise wait for a host that may never come.
2089    env.publish_device_name(session.device_name()).await;
2090    sync_dev_domain(&session, &mut dev_domain_synced, &mut env);
2091    env.boot_settings_ready();
2092
2093    loop {
2094        // Resolve the next event in its own statement so the select's
2095        // futures—one of which mutably borrows `env`—are dropped
2096        // before the arms below use `env` again. A `match select4(..)`
2097        // scrutinee would hold them for the whole match.
2098        let event = {
2099            // Only wait for a TX completion while one is outstanding,
2100            // so a spurious tx_done can never be consumed early.
2101            let tx_done = async {
2102                if session.has_pending_tx() {
2103                    rt.radio.tx_done.wait().await
2104                } else {
2105                    core::future::pending().await
2106                }
2107            };
2108            // The locate alert's deadline. Enforced here rather than by
2109            // each board so that "a device MUST bound how long it will
2110            // remain in ALERT_LOCATE" holds for every board that
2111            // advertises CAP_ALERT, including ones whose UX layer has no
2112            // timer of its own. Idle (never completes) while no alert is
2113            // running. It borrows `session` immutably, alongside
2114            // `tx_done`—only `publish_event` touches `env`.
2115            let alert_deadline = async {
2116                match session.alert_deadline_ms() {
2117                    Some(deadline) => Timer::at(Instant::from_millis(deadline)).await,
2118                    None => core::future::pending().await,
2119                }
2120            };
2121            select4(
2122                rt.input.receive(),
2123                rt.radio.rx.receive(),
2124                tx_done,
2125                select(env.publish_event(), alert_deadline),
2126            )
2127            .await
2128        };
2129
2130        match event {
2131            Either4::First(InEvent::Attached(transport)) => {
2132                // Fresh session state for the new host session; the
2133                // device domain (PHY configuration and enable state,
2134                // device name, duty accounting) is deliberately
2135                // untouched and nothing is emitted (full-protocol
2136                // attach semantics).
2137                arbitration.attach(transport);
2138                rt.session_gen
2139                    .store(arbitration.generation(), Ordering::Release);
2140                env.set_advertising_allowed(arbitration.advertising_allowed());
2141                // Both transports meet their provisioning-security
2142                // binding here: the wired transport by physical
2143                // possession, BLE because the ULCP GATT service
2144                // refuses any access outside an encrypted LESC-bonded
2145                // link.
2146                session.attach(true);
2147            }
2148            Either4::First(InEvent::Detached(transport)) => {
2149                // Only the active transport's detach ends the
2150                // session; a displaced transport's stale detach
2151                // must not clear the successor's session state.
2152                if arbitration.detach(transport) {
2153                    env.set_advertising_allowed(true);
2154                    session.detach();
2155                }
2156            }
2157            Either4::First(InEvent::CancelAlert) => {
2158                // Whoever found the radio silenced it. Publishing the
2159                // transition is not conditional on a host being
2160                // attached—`cancel_alert` handles that—and a press
2161                // with no alert running is simply nothing.
2162                let effect = session.cancel_alert(&mut |frame: &[u8]| emitter.push(frame));
2163                emitter.flush(&mut ReplySink::host(&arbitration, &rt)).await;
2164                apply_effect(&session, effect, &rt, &mut env).await;
2165            }
2166            Either4::First(InEvent::Toggle(setting)) => {
2167                // PHY enable needs the same radio effect as a host write;
2168                // other switches reach the platform through the domain mirror.
2169                let mut effect = None;
2170                let flipped = match setting {
2171                    Setting::Radio => {
2172                        let (enabled, radio_effect) =
2173                            session.toggle_radio(&mut |frame: &[u8]| emitter.push(frame));
2174                        effect = Some(radio_effect);
2175                        Some(enabled)
2176                    }
2177                    Setting::Wifi => session.toggle_wifi(&mut |frame: &[u8]| emitter.push(frame)),
2178                    Setting::Bluetooth => {
2179                        session.toggle_ble(&mut |frame: &[u8]| emitter.push(frame))
2180                    }
2181                    Setting::MotionWake => {
2182                        session.toggle_display_motion_wake(&mut |frame: &[u8]| emitter.push(frame))
2183                    }
2184                    Setting::Gnss => session.toggle_gnss(&mut |frame: &[u8]| emitter.push(frame)),
2185                    Setting::ShareLocation => {
2186                        session.toggle_gnss_ident_update(&mut |frame: &[u8]| emitter.push(frame))
2187                    }
2188                    Setting::Forwarding => {
2189                        session.toggle_repeater(&mut |frame: &[u8]| emitter.push(frame))
2190                    }
2191                };
2192                if let Some(enabled) = flipped {
2193                    emitter.flush(&mut ReplySink::host(&arbitration, &rt)).await;
2194                    if setting == Setting::Gnss {
2195                        env.gnss_switched(enabled);
2196                    }
2197                    apply_effect(&session, effect, &rt, &mut env).await;
2198                    // Keep an existing snapshot in step, so a switch the
2199                    // operator flipped is still flipped after a reboot.
2200                    persist_device_change(session, snapshot_buf, &mut env).await;
2201                    crate::log::debug_log(format_args!(
2202                        "ulcp: {:?} {} at the device",
2203                        setting,
2204                        if enabled { "ON" } else { "off" }
2205                    ));
2206                }
2207            }
2208            #[cfg(feature = "wifi")]
2209            Either4::First(InEvent::SelectWifiNetwork(network)) => {
2210                let result = session
2211                    .choose_wifi_network(network.as_bytes(), &mut |frame| emitter.push(frame));
2212                emitter.flush(&mut ReplySink::host(&arbitration, &rt)).await;
2213                if result.is_ok() {
2214                    persist_device_change(session, snapshot_buf, &mut env).await;
2215                }
2216                env.wifi_selection_result(result);
2217            }
2218            Either4::First(InEvent::ForceBluetoothOn) => {
2219                if session
2220                    .force_ble_on(&mut |frame: &[u8]| emitter.push(frame))
2221                    .is_some()
2222                {
2223                    emitter.flush(&mut ReplySink::host(&arbitration, &rt)).await;
2224                    // Persisted like a toggle: the gesture is the device's
2225                    // own control for the property, and a radio rescued by
2226                    // it should stay rescued across the next reboot.
2227                    persist_device_change(session, snapshot_buf, &mut env).await;
2228                    crate::log::debug_log(format_args!(
2229                        "ulcp: Bluetooth forced ON by the power-on gesture"
2230                    ));
2231                }
2232            }
2233            Either4::First(InEvent::Frame(transport, frame_bytes)) => {
2234                if arbitration.accepts_frame(transport) {
2235                    let mut sink = ReplySink::host(&arbitration, &rt);
2236                    let dfu = serve_frame(
2237                        session,
2238                        Exchange::Local(&frame_bytes),
2239                        &mut emitter,
2240                        &mut sink,
2241                        snapshot_buf,
2242                        &rt,
2243                        &mut env,
2244                    )
2245                    .await;
2246                    finish_dfu(&mut env, dfu).await;
2247                }
2248            }
2249            Either4::First(InEvent::Admin {
2250                frame,
2251                reply_budget,
2252            }) => {
2253                // The responder is blocked on the reply channel, so this
2254                // arm must always answer. An empty reply is the answer
2255                // for a reset-class command, and for anything the session
2256                // declined to respond to at all.
2257                let mut reply = AdminFrame::new();
2258                let dfu = {
2259                    let mut sink = ReplySink::Admin {
2260                        reply: &mut reply,
2261                        destination: arbitration.destination(),
2262                        out: rt.out,
2263                        device: rt.device,
2264                    };
2265                    serve_frame(
2266                        session,
2267                        Exchange::Admin {
2268                            frame: &frame,
2269                            reply_budget,
2270                        },
2271                        &mut emitter,
2272                        &mut sink,
2273                        snapshot_buf,
2274                        &rt,
2275                        &mut env,
2276                    )
2277                    .await
2278                };
2279                ADMIN_REPLY
2280                    .send(AdminReply {
2281                        frame: reply,
2282                        completion: dfu.map(|(_, completion)| completion),
2283                    })
2284                    .await;
2285                finish_dfu(&mut env, dfu).await;
2286            }
2287            Either4::First(InEvent::Device(frame)) => {
2288                // A board without a link has no interface to send this,
2289                // so a request here has nowhere to be answered.
2290                if let Some(link) = rt.device {
2291                    let mut sink = ReplySink::Device {
2292                        link,
2293                        destination: arbitration.destination(),
2294                        out: rt.out,
2295                    };
2296                    let dfu = serve_frame(
2297                        session,
2298                        Exchange::Device(&frame),
2299                        &mut emitter,
2300                        &mut sink,
2301                        snapshot_buf,
2302                        &rt,
2303                        &mut env,
2304                    )
2305                    .await;
2306                    // A change made at the device keeps an existing
2307                    // snapshot in step, as a switch flipped there does.
2308                    if Writes::with(&frame, |key| session.snapshots(key)).saved {
2309                        persist_device_change(session, snapshot_buf, &mut env).await;
2310                    }
2311                    finish_dfu(&mut env, dfu).await;
2312                } else {
2313                    debug_assert!(false, "a device request on a board with no device link");
2314                }
2315            }
2316            Either4::Second(RxFrame { data, info }) => {
2317                // While detached this may stage a delegated MAC
2318                // acknowledgement (Effect::StartTransmit).
2319                let rx_info = match info.origin {
2320                    RxOrigin::Air => {
2321                        RadioRxInfo::measured(info.rssi, info.snr.as_centibels(), info.lqi)
2322                    }
2323                    // Backhaul frames travel host-to-node, so the session
2324                    // never sees one arriving.
2325                    RxOrigin::LocalTx | RxOrigin::Backhaul => RadioRxInfo::self_transmitted(),
2326                    // Heard on a bearer, where nothing measures a signal.
2327                    RxOrigin::Bearer(_) => RadioRxInfo::default(),
2328                };
2329                let effect = session.on_radio_rx(
2330                    &data,
2331                    &rx_info,
2332                    Instant::now().as_millis(),
2333                    &mut |frame: &[u8]| emitter.push(frame),
2334                );
2335                if let Some(notice) = session.take_queued_notice() {
2336                    env.frame_queued(notice);
2337                }
2338                emitter.flush(&mut ReplySink::host(&arbitration, &rt)).await;
2339                apply_effect(&session, effect, &rt, &mut env).await;
2340            }
2341            Either4::Third(result) => {
2342                let now_ms = Instant::now().as_millis();
2343                let outcome = match result {
2344                    Ok(()) => TxOutcome::Sent,
2345                    Err(umsh_hal::TxError::CadTimeout) => TxOutcome::ChannelBusy,
2346                    Err(umsh_hal::TxError::Io(_)) => TxOutcome::Failed,
2347                };
2348                let effect =
2349                    session.on_tx_result(outcome, now_ms, &mut |frame: &[u8]| emitter.push(frame));
2350                emitter.flush(&mut ReplySink::host(&arbitration, &rt)).await;
2351                apply_effect(&session, effect, &rt, &mut env).await;
2352            }
2353            Either4::Fourth(Either::First(event)) => {
2354                // The board decided this is worth announcing; publish it
2355                // unsolicited. Dropped silently while no host is
2356                // attached, and no effect can result—a publication is
2357                // not an operation.
2358                let emit = &mut |frame: &[u8]| emitter.push(frame);
2359                match event {
2360                    PublishEvent::Battery(sample) => {
2361                        session.publish_battery(sample, emit);
2362                    }
2363                    PublishEvent::Time(epoch) => {
2364                        session.publish_time(epoch, emit);
2365                    }
2366                    PublishEvent::Gnss(key, snapshot) => {
2367                        session.publish_gnss(key, &snapshot, emit);
2368                    }
2369                    PublishEvent::BleBondCount(count) => {
2370                        session.set_ble_bond_count(count, emit);
2371                    }
2372                    PublishEvent::BleLink(state) => {
2373                        session.set_ble_link(state, emit);
2374                    }
2375                    PublishEvent::BlePairing(open) => {
2376                        session.set_ble_pairing(open, emit);
2377                    }
2378                    PublishEvent::WifiScanning(scanning) => {
2379                        // The clear goes first, so every insert that
2380                        // follows lands in a table the host knows to be
2381                        // empty (spec §PROP_WIFI_SCANNING).
2382                        if scanning {
2383                            session.clear_scan_results(emit);
2384                        }
2385                        session.set_wifi_scanning(scanning, emit);
2386                    }
2387                    PublishEvent::WifiScanResult(item) => {
2388                        if let Ok(result) = umsh_ulcp::wifi::ScanResult::decode(&item) {
2389                            session.publish_scan_result(&result, emit);
2390                        }
2391                    }
2392                    PublishEvent::WifiLink(link) => {
2393                        session.set_wifi_link(link, emit);
2394                    }
2395                    PublishEvent::WifiRssi(dbm) => {
2396                        session.set_wifi_rssi(dbm);
2397                    }
2398                    PublishEvent::WifiMac(mac) => {
2399                        session.set_wifi_mac(mac);
2400                    }
2401                    #[cfg(feature = "wifi")]
2402                    PublishEvent::NetworkTableChanged(key, value) => {
2403                        session.publish_network_table(key, &value, emit);
2404                    }
2405                    PublishEvent::IpState(key, state) => {
2406                        session.set_ip_state(key, state, emit);
2407                    }
2408                    #[cfg(feature = "bridge-client")]
2409                    PublishEvent::BridgeLink(link) => session.set_bridge_link(link, emit),
2410                    PublishEvent::LanBearerState(state) => {
2411                        session.set_lan_bearer_state(state, emit)
2412                    }
2413                    PublishEvent::IdentityFix(location, altitude_m) => {
2414                        // The session clamps, compares, and bumps the
2415                        // device-domain version if the advertised
2416                        // position actually moved; the sync at the foot
2417                        // of this loop carries it to the identity.
2418                        session.absorb_ident_fix(&location, altitude_m);
2419                    }
2420                }
2421                emitter.flush(&mut ReplySink::host(&arbitration, &rt)).await;
2422            }
2423            Either4::Fourth(Either::Second(())) => {
2424                // The alert outlived its deadline: stop the indication
2425                // and publish the transition the host did not command.
2426                let effect = session
2427                    .poll_alert(Instant::now().as_millis(), &mut |frame: &[u8]| {
2428                        emitter.push(frame)
2429                    });
2430                emitter.flush(&mut ReplySink::host(&arbitration, &rt)).await;
2431                apply_effect(&session, effect, &rt, &mut env).await;
2432            }
2433        }
2434        // Any of the arms may have moved the device-domain tables
2435        // (property mutation, CMD_RST, CMD_RESTORE); one u32
2436        // compare when they did not.
2437        sync_dev_domain(&session, &mut dev_domain_synced, &mut env);
2438        // The board's own store keeps the device lists as they change.
2439        // One it could not take is offered again after the next event.
2440        if let Some(len) = session.lists_to_keep(snapshot_buf)
2441            && env.persist_lists(Some(&snapshot_buf[..len])).await.is_ok()
2442        {
2443            session.note_lists_kept();
2444        }
2445        // PROP_MAC_BACKHAUL is session-scoped but its routing lives in
2446        // the mux, which nothing inside the session can reach. Mirroring
2447        // it here rather than pushing it from each write is what keeps
2448        // the two agreeing across every path that moves it—a host's
2449        // write, an attach that discards session state, a CMD_RST off
2450        // the mesh. The store is a relaxed atomic, so doing it every
2451        // iteration costs nothing.
2452        crate::radio_mux::MUX_MODE.set_backhaul(session.backhaul());
2453        // Session state was discarded under a host that did not ask for
2454        // it; tell it, so it can re-establish whatever it was holding
2455        // there. Drained here rather than at the discard because a
2456        // CMD_RST off the mesh discards while the admin binding is
2457        // still held: by now `serve_frame` has released it and the sink
2458        // below is the host's own transport.
2459        if session.take_session_reset_notice(&mut |frame: &[u8]| emitter.push(frame)) {
2460            emitter.flush(&mut ReplySink::host(&arbitration, &rt)).await;
2461        }
2462    }
