1//! The board-agnostic ULCP session driver (Phase 5, increment C).
2//!
3//! This is the extraction of the nRF firmware's `device_task` select loop, its
4//! `apply_effect` radio-effect dispatcher, and the `Emitter` frame stager —
5//! the one copy of the session-driving logic shared by every device
6//! firmware (T-Echo, T-1000E, Heltec V3). The board personalities that used
7//! to be `cfg(feature = "t1000e")` forks inside the loop are expressed as
8//! [`DeviceEnv`] hooks with no-op defaults, so a new board supplies exactly the
9//! couplings it has and nothing else.
10//!
11//! The split of responsibilities:
12//!
13//! - **This module** owns the protocol loop: transport arbitration, frame
14//!   handling, radio RX/TX-completion processing, every deferred `Effect`
15//!   arm (save/clear/wipe/provision/PIN/RSSI/battery/drain), asynchronous
16//!   property publication ([`DeviceEnv::battery_event`]), and the
17//!   device-domain mirror.
18//! - **The board** owns the edges: transport tasks feeding [`InEvent`]s and
19//!   draining [`TransportChannels`], the radio runner + mux serving the
20//!   session's virtual [`Channels`] bundle, and an [`DeviceEnv`] implementation
21//!   wiring persistence, entropy, pairing, and indicators to its hardware.
22

45/// Derive a device identity's public key and its persisted record from a
46/// raw Ed25519 secret.
47///
48/// Shared by first-boot generation and `Effect::ProvisionIdentity` so the
49/// two can never disagree about the derivation or the record layout.
50///
51/// The caller supplies `secret` and owns the question this function
52/// cannot answer: it **MUST** come from a cryptographic RNG with real
53/// entropy behind it. On the nRF boards that is the hardware TRNG with
54/// bias correction enabled; on Espressif it is `EspCryptoRng`, which
55/// refuses to exist unless the RF noise source is live.
56pub fn device_identity_record(secret: &[u8; 32]) -> ([u8; 32], [u8; proto::IDENTITY_PAYLOAD_LEN]) {
57    let public_key = SoftwareIdentity::from_secret_bytes(secret).public_key().0;
58    (public_key, proto::encode_identity(secret, &public_key))
59}

61/// How many older snapshot generations boot will try after the newest
62/// one is rejected.
63///
64/// Bounded deliberately. Corruption is expected to affect one record, so
65/// a handful of generations covers it; a payload that is *systematically*
66/// undecodable is a firmware bug, and walking the whole journal for it on
67/// every boot would only delay booting bare and reporting the fact.
68pub const SNAPSHOT_FALLBACK_LIMIT: usize = 4;

103/// One raw ULCP frame in a transport output queue, stamped with the

111/// The per-transport outbound frame queues, drained by the board's
112/// transport output tasks. `wired` is the physical-possession transport

114/// `transport_policy::Transport` names.
115pub struct TransportChannels<M: RawMutex> {
116    pub wired: Channel<M, OutFrame, 4>,
117    pub ble: Channel<M, OutFrame, 4>,

120impl<M: RawMutex> TransportChannels<M> {
121    pub const fn new() -> Self {
122        Self {
123            wired: Channel::new(),
124            ble: Channel::new(),
125        }
126    }

128    fn for_transport(&self, transport: Transport) -> &Channel<M, OutFrame, 4> {

136/// The session's device-domain tables, mirrored to the board's device
137/// node whenever their generation moves (device-node plan increment 3).
138pub struct DevDomainSnapshot {
139    pub channel_keys: heapless::Vec<[u8; 32], MAX_CHANNEL_KEYS>,
140    pub peers: heapless::Vec<[u8; 32], MAX_DEV_PEERS>,
141    /// The session's live `PROP_DEV_KEY`.
142    ///
143    /// The device node compares this against the key its MAC was built
144    /// with rather than merely checking that *some* identity exists. The
145    /// two can disagree — a newly installed `PROP_DEV_PRIVATE_KEY` takes
146    /// effect at the next boot, and `CMD_CLEAR` + `CMD_RST` erases the
147    /// stored one — and in every such case the running node is no longer

160    /// MAC's region filter and hand the same bytes to the Supported
161    /// Regions identity option without reshaping either.

163    /// `PROP_MAC_REPEATER_DEFAULT_REGION`: the code the node inserts

166    /// `PROP_MAC_REPEATER_MIN_RSSI`: minimum RSSI in dBm to
167    /// flood-forward, or `None` for no threshold.
168    pub repeater_min_rssi: Option<i16>,
169    /// `PROP_MAC_REPEATER_MIN_SNR`: minimum SNR in whole dB to
170    /// flood-forward, or `None` for no threshold.
171    pub repeater_min_snr: Option<i8>,
172    /// `PROP_IDENT_ROLE`: the advertised `ROLE` byte, or `None` to
173    /// derive it from the forwarding state.
174    pub ident_role: Option<u8>,
175    /// `PROP_IDENT_MOBILE`: whether to advertise the `MOB` capability
176    /// bit.
177    pub ident_mobile: bool,
178    /// `PROP_DEV_DISCOVERABLE`: whether the device identity answers
179    /// Identity Requests.
180    pub discoverable: bool,
181    /// `PROP_ADVERT_INTERVAL`: seconds between unsolicited
182    /// advertisements, 0 for none.
183    pub advert_interval_s: u32,
184    /// `PROP_BEACON_INTERVAL`: seconds between unsolicited beacons, 0 for
185    /// none.
186    pub beacon_interval_s: u32,
187    /// `PROP_STARTUP_BEACON`: whether one beacon goes out at bring-up.
188    pub startup_beacon: bool,
189    /// `PROP_TZ_OFFSET`: minutes east of UTC, for whatever renders a
190    /// local time.
191    pub tz_offset_min: i16,
192    /// `PROP_GNSS_ENABLED`: whether the receiver should be powered.
193    /// Always false on a board without `CAP_GNSS`.
194    ///
195    /// Carried here rather than as an effect of its own so that a host

199    pub gnss_enabled: bool,
200    /// `PROP_GNSS_IDENT_UPDATE`: whether fixes refresh the advertised

202    pub gnss_ident_update: bool,

211/// A device-initiated property publication, yielded by
212/// [`DeviceEnv::publish_event`].
213///
214/// The driver has exactly one select arm for everything the board pushes
215/// unasked, because a hook per property would need one `&mut self` borrow
216/// per arm and no two of those can coexist. One arm, one enum, and the
217/// board decides internally which of its sources woke it.
218pub enum PublishEvent {
219    /// An unsolicited `PROP_BATTERY`.
220    Battery(umsh_ulcp::battery::BatteryStatus),
221    /// An unsolicited `PROP_TIME`; `None` is a clock that has gone back
222    /// to not knowing what time it is.
223    Time(Option<u32>),
224    /// An unsolicited positioning property, named by its key, encoded
225    /// from the accompanying snapshot.
226    Gnss(u32, umsh_ulcp::gnss::GnssSnapshot),
227}

229/// Board couplings of the session driver. Everything the loop needs from
230/// the platform, expressed as one trait so the driver itself stays free
231/// of HAL types and `cfg` board forks. Hooks a board doesn't have keep
232/// their no-op defaults (e.g. only the T-1000E implements the attention
233/// indicator and transmit-load hooks today).
234// Single-executor embedded consumers; `Send` futures are irrelevant here,
235// same as the embassy ecosystem's own async traits.
236#[allow(async_fn_in_trait)]
237pub trait DeviceEnv {
238    /// Durably persist the encoded protocol snapshot (CMD_SAVE / host wipe).
239    async fn persist_snapshot(&mut self, bytes: &[u8]) -> Result<(), ()>;
240    /// Tombstone the snapshot journal (CMD_CLEAR).
241    async fn clear_snapshot(&mut self) -> Result<(), ()>;
242    /// Copy the newest committed snapshot generation strictly older than
243    /// the one last handed to the driver into `out`, returning its
244    /// length.
245    ///
246    /// Called only after a payload is rejected, so the cost is paid on a
247    /// boot that is already going wrong. Implementations re-scan the
248    /// journal rather than retaining a runner-up, keeping the mount
249    /// path's "never buffers a second copy" discipline. The default
250    /// refuses, which makes rejection terminal for boards whose journal
251    /// cannot walk back.
252    async fn older_snapshot(&mut self, out: &mut [u8]) -> Option<usize> {
253        let _ = out;
254        None
255    }
256    /// A stored snapshot was rejected at boot. Boards with an indicator
257    /// surface it locally: the host-visible report reaches nobody on an
258    /// unattended repeater, which is exactly the deployment this
259    /// matters for.
260    fn report_snapshot_rejected(&mut self, fell_back: bool) {
261        let _ = fell_back;
262    }
263    /// Durably persist the encoded device identity.
264    async fn persist_identity(&mut self, bytes: &[u8]) -> Result<(), ()>;
265    /// Tombstone the identity journal (CMD_CLEAR).
266    async fn clear_identity(&mut self) -> Result<(), ()>;
267    /// Drop persisted frame-counter boundaries after a successful
268    /// identity clear. Boards without a device node keep the default.
269    async fn clear_counters(&mut self) {}
270    /// Fill `secret` from the platform's cryptographic RNG. Fails closed:
271    /// an error refuses identity generation rather than degrading.
272    fn fill_secret(&mut self, secret: &mut [u8; 32]) -> Result<(), ()>;
273    /// One fresh battery measurement (`Effect::SampleBattery`). Only
274    /// emitted when the board's `SessionConfig::battery` advertises
275    /// fields, so the default refuses.
276    async fn sample_battery(&mut self) -> Result<umsh_ulcp::battery::BatteryStatus, ()> {
277        Err(())
278    }
279    /// One fresh ambient light measurement in millilux
280    /// (`Effect::SampleIlluminance`). Only emitted on a board whose
281    /// `SessionConfig::illuminance` is set, so the default reports nothing.
282    ///
283    /// `None` is a legitimate answer — the sensor exists but could not be
284    /// read — and reaches the host as the empty value rather than an error.
285    async fn sample_illuminance(&mut self) -> Option<u32> {
286        None
287    }
288    /// Wait for a battery measurement the board considers worth
289    /// announcing, for publication as an unsolicited `PROP_BATTERY`
290    /// (`Session::publish_battery`).
291    ///
292    /// The board owns the whole policy: the sampling cadence, the
293    /// charge-state edges, and which changes matter. It is the only layer
294    /// that sees every sample, so filtering there keeps the session free
295    /// of cached readings and keeps this hook's contract simple — every
296    /// value it yields is published.
297    ///
298    /// Cancellation-safe: the driver drops and re-creates this future on
299    /// every other loop iteration, so an implementation must not lose an
300    /// update it was cancelled on (an `embassy_sync::watch::Watch`
301    /// receiver behaves correctly here; a bare `Signal` does not).
302    ///
303    /// The default never completes, so boards without battery push add
304    /// nothing to the select.
305    async fn battery_event(&mut self) -> umsh_ulcp::battery::BatteryStatus {
306        core::future::pending().await
307    }
308    /// Wait for anything the board wants to publish unasked, across every
309    /// property it pushes.
310    ///
311    /// This is the driver's single select arm for device-initiated
312    /// publication. The default delegates to
313    /// [`battery_event`](Self::battery_event), so a board that pushes only
314    /// battery measurements implements that and nothing else. A board that
315    /// also pushes time or position overrides this instead and selects
316    /// over its own sources — which it can do without fighting the
317    /// borrow checker, since those are its own fields rather than three
318    /// `&mut self` calls.
319    ///
320    /// Cancellation-safe on the same terms as
321    /// [`battery_event`](Self::battery_event).
322    async fn publish_event(&mut self) -> PublishEvent {
323        PublishEvent::Battery(self.battery_event().await)
324    }
325    /// Read the platform wall clock (`Effect::ReadTime`): Unix seconds,
326    /// or `None` when the device does not know what time it is.
327    ///
328    /// Not knowing is the honest answer for a board that has never had a
329    /// fix and was never told, and it is what stops a display from
330    /// showing a clock. The default is exactly that, so a board without
331    /// `CAP_TIME` never has to implement it.
332    async fn read_time(&mut self) -> Option<u32> {
333        None
334    }
335    /// Apply a `PROP_TIME` write (`Effect::ApplyTime`): set the wall
336    /// clock, or return it to not knowing.
337    ///
338    /// A manual set outranks every receiver-derived one, so this applies
339    /// regardless of `PROP_GNSS_TIME_TRUST`.
340    async fn apply_time(&mut self, epoch: Option<u32>) {
341        let _ = epoch;
342    }
343    /// Sample the receiver's current view of position and constellation
344    /// (`Effect::SampleGnss`). Only emitted on a board whose
345    /// `SessionConfig::gnss` advertises the capability, so the default
346    /// refuses.
347    async fn sample_gnss(&mut self) -> Result<umsh_ulcp::gnss::GnssSnapshot, ()> {
348        Err(())
349    }
350    /// Build and sign the device identity's node-identity blob into
351    /// `out` (`Effect::SignIdentity`), returning its length.
352    ///
353    /// The board owns both halves the session does not: the signing key,
354    /// and the advertised profile the Identity Request responder uses.
355    /// Boards without a device node keep the default, which refuses.
356    async fn sign_identity(&mut self, out: &mut [u8]) -> Option<usize> {
357        let _ = out;
358        None
359    }
360    /// Apply a `PROP_BLE_PAIRING_PIN` write against the bond journal and
361    /// the live BLE stack; `true` when it took effect.
362    async fn apply_pairing_pin(&mut self, pin: Option<u32>) -> bool;
363    /// `CMD_FACTORY_RESET`: erase EVERY piece of persistent state the
364    /// platform owns — saved snapshot, device identity, frame-counter
365    /// boundaries, BLE bonds, pairing PIN, and any other journal — then
366    /// reboot. Never returns: the reset discards in-RAM state and the
367    /// board comes back factory-fresh. There is no separate "clear bonds"
368    /// hook because a reboot reloads bonds from the (now-erased) journal,
369    /// so the live BLE stack never has to be touched.
370    async fn factory_reset(&mut self) -> !;
371    /// Publish the transport-arbitration advertising policy (a wired
372    /// attach suppresses BLE advertising). Diagnostic builds may
373    /// deliberately ignore `allowed`.
374    fn set_advertising_allowed(&mut self, allowed: bool);
375    /// Publish the session's device name to the board's consumers
376    /// (advertising data, device node, UI).
377    async fn publish_device_name(&mut self, name: &str);
378    /// Deliver a device-domain mirror to the board's device node.
379    fn publish_dev_domain(&mut self, snapshot: DevDomainSnapshot);
380    /// Start or stop the board's locate indication (`PROP_ALERT`).
381    ///
382    /// Carries the authoritative state and is called for every
383    /// transition — host write, local cancellation, and deadline — so an
384    /// implementation can treat it as idempotent and needs no notion of
385    /// *why* the alert ended. `AlertState::Locate` must override a local
386    /// silence setting without clearing it (spec §PROP_ALERT); boards
387    /// without `CAP_ALERT` never see this and keep the default.
388    fn set_alert(&mut self, state: umsh_ulcp::alert::AlertState) {
389        let _ = state;
390    }
391    /// The receiver switch was flipped at the device, and is now
392    /// `enabled`.
393    ///
394    /// Only for the local gesture: a host write already knows what it
395    /// asked for, and a board that indicated one would announce the
396    /// phone's own settings screen back at it. Carries the resulting
397    /// state rather than the fact of a press, because "on" and "off"
398    /// are what the operator needs told apart.
399    fn gnss_switched(&mut self, enabled: bool) {
400        let _ = enabled;
401    }
402    /// A covered frame was queued for an attached-or-future host
403    /// (T-1000E: request the attention LED).
404    fn request_attention(&mut self) {}
405    /// The host-facing queue drained to empty (T-1000E: clear it).
406    fn clear_attention(&mut self) {}
407    /// A transmit is about to start; boards with a battery-level
408    /// estimator mark the load spike.
409    fn note_transmit_load(&mut self) {}
410    /// Diagnostic trace line (routed to the board's debug channel; the
411    /// default discards).
412    fn trace(&mut self, args: core::fmt::Arguments<'_>) {
413        let _ = args;
414    }
415}

421    pub input: &'static InputChannel<M>,

435/// Collects frames emitted synchronously by the session, then flushes
436/// them to the active transport's output queue asynchronously. The
437/// session emits at most one frame per call; two slots give headroom.
438struct Emitter {
439    bufs: [[u8; FRAME_OUT_MAX]; 2],
440    lens: [usize; 2],
441    count: usize,
442}
443
444impl Emitter {
445    const fn new() -> Self {
446        Self {
447            bufs: [[0; FRAME_OUT_MAX]; 2],
448            lens: [0; 2],
449            count: 0,
450        }
451    }
452
453    /// Copy one raw ULCP frame into the next slot.
454    ///
455    /// The session is expected to emit at most `bufs.len()` frames per call
456    /// and every frame is expected to fit `FRAME_OUT_MAX`. Both invariants are
457    /// asserted in debug builds so a future session change that violates
458    /// them is caught rather than silently dropping a response.
459    fn push(&mut self, frame: &[u8]) {
460        if self.count >= self.bufs.len() {
461            debug_assert!(
462                false,
463                "Emitter overflow: session emitted more frames per call than staging slots"
464            );
465            return;
466        }
467        if frame.len() <= FRAME_OUT_MAX {
468            self.bufs[self.count][..frame.len()].copy_from_slice(frame);
469            self.lens[self.count] = frame.len();
470            self.count += 1;
471        } else {
472            debug_assert!(false, "Emitter: ULCP frame exceeds FRAME_OUT_MAX");
473        }
474    }
475
476    /// Queue all staged frames for the active transport output task.
477    async fn flush<M: RawMutex>(
478        &mut self,
479        destination: Option<(Transport, u32)>,
480        out: &TransportChannels<M>,
481    ) {
482        if let Some((transport, generation)) = destination {
483            for index in 0..self.count {
484                let mut frame: FrameBuf = heapless::Vec::new();
485                if frame
486                    .extend_from_slice(&self.bufs[index][..self.lens[index]])
487                    .is_err()
488                {
489                    // FRAME_OUT_MAX == FrameBuf capacity, so this cannot
490                    // happen; assert in debug rather than silently drop.
491                    debug_assert!(false, "Emitter frame copy exceeded FrameBuf capacity");
492                    continue;
493                }
494                out.for_transport(transport)
495                    .send(OutFrame { generation, frame })
496                    .await;
497            }
498        }
499        self.count = 0;
500    }
501}

503/// Execute a radio side effect requested by the session.
504async fn apply_effect<A, S, const TXQ: usize, M, const RX: usize, const TX: usize, E>(
505    session: &Session<A, S, TXQ>,
506    effect: Option<Effect>,
507    rt: &DeviceRuntime<M, RX, TX>,
508    env: &mut E,
509) where
510    A: AesProvider,
511    S: Sha256Provider,
512    M: RawMutex,
513    E: DeviceEnv,
514{
515    match effect {
516        Some(Effect::ApplyRadio(settings)) => {
517            env.publish_device_name(session.device_name()).await;
518            // The session validates values against the same discrete
519            // sets these converters accept, so None here is
520            // unreachable; bail out defensively rather than panic.
521            let (Some(sf), Some(bw), Some(cr)) = (
522                spreading_factor_from_u8(settings.sf),
523                bandwidth_from_hz(settings.bw_hz),
524                coding_rate_from_denom(settings.cr_denom),
525            ) else {
526                return;
527            };
528            rt.ctl.apply(DeviceSettings {
529                enabled: settings.enabled,
530                freq_hz: settings.freq_khz.saturating_mul(1_000),
531                sf,
532                bw,
533                cr,
534                power_dbm: i32::from(settings.tx_power_dbm),
535            });
536            // Published for anything that wants to show what the radio is
537            // actually set to — a board's stats page, in particular —
538            // without having to hold the session to ask. The statics live
539            // with the device node, which not every driver consumer
540            // builds.
541            #[cfg(feature = "device-node")]
542            crate::device_node::set_tx_power_dbm(settings.tx_power_dbm);
543        }
544        Some(Effect::StartTransmit) => {
545            let mut data: heapless::Vec<u8, MAX_PAYLOAD> = heapless::Vec::new();
546            if data.extend_from_slice(session.tx_data()).is_err() {
547                env.trace(format_args!(
548                    "radio tx staging=FAILED len={}",
549                    session.tx_data().len()
550                ));
551                return;
552            }
553            let power_dbm = match session.tx_power() {
554                TxPower::Default => None,
555                TxPower::Max => Some(i32::from(session.max_tx_power_dbm())),
556                TxPower::Dbm(dbm) => Some(i32::from(dbm)),
557            };
558            // Mark the load for the board's battery level estimator (the
559            // radio runner transmits within milliseconds of this).
560            env.note_transmit_load();
561            let cad = if session.tx_nocca() {
562                CadPolicy::Skip
563            } else {
564                CadPolicy::Gate
565            };
566            rt.radio
567                .tx
568                .send(TxRequest {
569                    data,
570                    power_dbm,
571                    cad,
572                })
573                .await;
574        }
575        Some(Effect::DeviceNameChanged) => {
576            env.publish_device_name(session.device_name()).await;
577        }
578        Some(Effect::ApplyAlert(state)) => {
579            env.set_alert(state);
580        }
581        Some(Effect::ApplyTime { epoch }) => {
582            env.apply_time(epoch).await;
583        }
584        Some(Effect::ApplyBackhaul { enabled }) => {
585            crate::radio_mux::MUX_MODE.set_backhaul(enabled);
586        }
587        // Deferred effects needing `&mut Session` + the emitter are
588        // handled inline in the run loop rather than here.
589        Some(Effect::SampleRssi { .. })
590        | Some(Effect::SignIdentity { .. })
591        | Some(Effect::SampleBattery { .. })
592        | Some(Effect::SampleIlluminance { .. })
593        | Some(Effect::ReadTime { .. })
594        | Some(Effect::SampleGnss { .. })
595        | Some(Effect::SetPairingPin { .. })
596        | Some(Effect::DrainQueue)
597        | Some(Effect::SaveSnapshot { .. })
598        | Some(Effect::ClearSaved { .. })
599        | Some(Effect::ProvisionIdentity { .. })
600        | Some(Effect::FactoryReset)
601        | None => {}
602    }
603}

605/// Mirror the session's device-domain node tables to the device node
606/// when their generation moved (device-node plan increment 3).
607/// `synced_version` is the caller's cache of the last published
608/// generation. Cheap when nothing changed — one u32 compare — so the
609/// loop runs it after every session interaction.
610/// Generate, persist, and install a fresh device identity.
611///
612/// The counterpart to first-boot generation, for the one runtime path
613/// that can leave the session without one: `CMD_CLEAR` followed by the
614/// `CMD_RST` that completes a factory reset. A device identity is not a
615/// commissioning step, so there is no state in which the operator has to
616/// supply one.
617///
618/// A failure to draw entropy or to persist leaves the session
619/// identityless, which is a worse outcome than either but not one this
620/// layer can repair: it is reported and the next boot regenerates.
621async fn regenerate_device_identity<A, S, const TXQ: usize, E>(
622    session: &mut Session<A, S, TXQ>,
623    env: &mut E,
624) where
625    A: AesProvider,
626    S: Sha256Provider,
627    E: DeviceEnv,
628{
629    let mut secret = [0u8; 32];
630    if env.fill_secret(&mut secret).is_err() {
631        env.trace(format_args!("device identity regenerate: entropy FAILED"));
632        return;
633    }
634    let (public_key, payload) = device_identity_record(&secret);
635    match env.persist_identity(&payload).await {
636        Ok(()) => {
637            session.set_boot_identity(public_key);
638            env.trace(format_args!(
639                "device identity regenerated after clear+reset"
640            ));
641        }
642        Err(()) => env.trace(format_args!(
643            "device identity regenerate: persist FAILED — none in effect"
644        )),
645    }
646}
647
648fn sync_dev_domain<A, S, const TXQ: usize, E>(
649    session: &Session<A, S, TXQ>,
650    synced_version: &mut u32,
651    env: &mut E,
652) where
653    A: AesProvider,
654    S: Sha256Provider,
655    E: DeviceEnv,
656{
657    if session.dev_domain_version() == *synced_version {
658        return;
659    }
660    *synced_version = session.dev_domain_version();
661    let mut snapshot = DevDomainSnapshot {
662        channel_keys: heapless::Vec::new(),
663        peers: heapless::Vec::new(),
664        dev_key: session.dev_key().copied(),
665        repeater_enabled: session.repeater_enabled(),
666        repeater_regions: heapless::Vec::from_slice(session.repeater_regions()).unwrap_or_default(),
667        repeater_default_region: session.repeater_default_region(),
668        repeater_min_rssi: session.repeater_min_rssi(),
669        repeater_min_snr: session.repeater_min_snr(),
670        ident_role: session.ident_role(),
671        ident_mobile: session.ident_mobile(),
672        discoverable: session.dev_discoverable(),
673        advert_interval_s: session.advert_interval_s(),
674        beacon_interval_s: session.beacon_interval_s(),
675        startup_beacon: session.startup_beacon(),
676        tz_offset_min: session.tz_offset_min(),
677        gnss_enabled: session.gnss_enabled(),
678        gnss_ident_update: session.gnss_ident_update(),
679        gnss_ident_precision: session.gnss_ident_precision(),
680        gnss_time_trust: session.gnss_time_trust(),
681    };
682    for key in session.dev_channel_keys() {
683        let _ = snapshot.channel_keys.push(key);
684    }
685    for public_key in session.dev_peers() {
686        let _ = snapshot.peers.push(public_key);
687    }
688    env.publish_dev_domain(snapshot);
689}
690
691/// Drive the ULCP session forever: restore persisted state, then
692/// select over host frames, radio receptions, and transmit completions,
693/// executing every session effect through the board's [`DeviceEnv`].
694///
695/// The caller constructs the [`Session`] with its board profile
696/// (`SessionConfig`) and boot status, mounts its journals, and hands the
697/// stored payloads in; the driver owns everything after that.
698pub async fn run<A, S, const TXQ: usize, M, const RX: usize, const TX: usize, E>(
699    mut session: Session<A, S, TXQ>,
700    boot_snapshot: Option<&[u8]>,
701    boot_identity: Option<[u8; 32]>,
702    rt: DeviceRuntime<M, RX, TX>,
703    mut env: E,
704) -> !
705where
706    A: AesProvider,
707    S: Sha256Provider,
708    M: RawMutex,
709    E: DeviceEnv,
710{
711    let mut emitter = Emitter::new();
712    let mut arbitration = SessionArbitration::new(rt.session_gen.load(Ordering::Acquire));
713    // Last device-domain generation mirrored to the device node.
714    // Matches the session's initial value; the first mutation (or a
715    // boot restore) publishes the first snapshot.
716    let mut dev_domain_synced: u32 = session.dev_domain_version();
717    // Shared staging buffer for the durable-write effect arms
718    // (save/wipe). Held across their persist awaits, so as a
719    // loop-lifetime local it costs one future slot instead of one
720    // per arm.
721    let mut snapshot_buf = [0u8; SNAPSHOT_MAX];
722
723    // The device identity is persisted independently of snapshots;
724    // its post-reset value is whatever the identity journal holds.
725    if let Some(public_key) = boot_identity {
726        session.set_boot_identity(public_key);
727    }
728
729    // Restore a stored snapshot before processing any host command:
730    // the saved configuration is applied, the PHY re-enabled if it
731    // was enabled when saved, and detached operation begins
732    // immediately.
733    //
734    // A payload that does not decode is not the end of it. The journal
735    // is multi-record and newest-generation-wins, so an older readable
736    // generation usually sits behind the rejected one — and for an
737    // unattended repeater, falling back to it is the only outcome that
738    // keeps the device forwarding. Walk back a bounded number of
739    // generations: a systematically undecodable payload is a firmware
740    // bug rather than corruption, and re-walking the whole journal on
741    // every boot would just be slower about it.
742    if let Some(payload) = boot_snapshot {
743        let mut generation = 0usize;
744        let mut restored = session.restore_at_boot(payload);
745        while let Err(error) = restored {
746            env.trace(format_args!(
747                "proto-store boot-restore generation=-{generation} REJECTED error={error:?}"
748            ));
749            session.note_snapshot_rejected();
750            generation += 1;
751            if generation > SNAPSHOT_FALLBACK_LIMIT {
752                env.trace(format_args!(
753                    "proto-store boot-restore fallback=EXHAUSTED limit={SNAPSHOT_FALLBACK_LIMIT}"
754                ));
755                break;
756            }
757            let Some(len) = env.older_snapshot(&mut snapshot_buf).await else {
758                env.trace(format_args!("proto-store boot-restore fallback=NONE"));
759                break;
760            };
761            restored = session.restore_at_boot(&snapshot_buf[..len]);
762        }
763        match restored {
764            Ok(effect) => {
765                if generation > 0 {
766                    env.trace(format_args!(
767                        "proto-store boot-restore=FALLBACK generation=-{generation}"
768                    ));
769                    env.report_snapshot_rejected(true);
770                } else {
771                    env.trace(format_args!("proto-store boot-restore=ok"));
772                }
773                apply_effect(&session, Some(effect), &rt, &mut env).await;
774            }
775            Err(_) => {
776                env.trace(format_args!("proto-store boot-restore=BARE"));
777                env.report_snapshot_rejected(false);
778            }
779        }
780    }
781
782    // Publish the device domain once before any host interaction, on every
783    // boot path rather than only after a successful restore.
784    //
785    // Two things depend on it. Detached multicast processing needs the
786    // restored tables without waiting for an attach — the original reason.
787    // And anything that waits for the domain to be published before acting
788    // needs that publication to happen on a device that has never been
789    // configured, where the answer is "the post-reset defaults" rather than
790    // silence: the boot-time GNSS clock read waits on exactly this, and on
791    // a bare device it would otherwise wait for a host that may never come.
792    sync_dev_domain(&session, &mut dev_domain_synced, &mut env);
793
794    loop {
795        // Resolve the next event in its own statement so the select's
796        // futures — one of which mutably borrows `env` — are dropped
797        // before the arms below use `env` again. A `match select4(..)`
798        // scrutinee would hold them for the whole match.
799        let event = {
800            // Only wait for a TX completion while one is outstanding,
801            // so a spurious tx_done can never be consumed early.
802            let tx_done = async {
803                if session.has_pending_tx() {
804                    rt.radio.tx_done.wait().await
805                } else {
806                    core::future::pending().await
807                }
808            };
809            // The locate alert's deadline. Enforced here rather than by
810            // each board so that "a device MUST bound how long it will
811            // remain in ALERT_LOCATE" holds for every board that
812            // advertises CAP_ALERT, including ones whose UX layer has no
813            // timer of its own. Idle (never completes) while no alert is
814            // running. It borrows `session` immutably, alongside
815            // `tx_done` — only `publish_event` touches `env`.
816            let alert_deadline = async {
817                match session.alert_deadline_ms() {
818                    Some(deadline) => Timer::at(Instant::from_millis(deadline)).await,
819                    None => core::future::pending().await,
820                }
821            };
822            select4(
823                rt.input.receive(),
824                rt.radio.rx.receive(),
825                tx_done,
826                select(env.publish_event(), alert_deadline),
827            )
828            .await
829        };
830
831        match event {
832            Either4::First(InEvent::Attached(transport)) => {
833                // Fresh session state for the new host session; the
834                // device domain (PHY configuration and enable state,
835                // device name, duty accounting) is deliberately
836                // untouched and nothing is emitted (full-protocol
837                // attach semantics).
838                arbitration.attach(transport);
839                rt.session_gen
840                    .store(arbitration.generation(), Ordering::Release);
841                env.set_advertising_allowed(arbitration.advertising_allowed());
842                // Both transports meet their provisioning-security
843                // binding here: the wired transport by physical
844                // possession, BLE because the ULCP GATT service
845                // refuses any access outside an encrypted LESC-bonded
846                // link.
847                session.attach(true);
848                // PROP_MAC_BACKHAUL is session-scoped but its routing
849                // lives in the mux, which the session-state reset above
850                // cannot reach. A displaced session attaches without any
851                // detach in between, so this is the reset that keeps a
852                // predecessor's backhaul mode from leaving the new
853                // session deaf to the air while its property reads 0.
854                crate::radio_mux::MUX_MODE.set_backhaul(false);
855            }
856            Either4::First(InEvent::Detached(transport)) => {
857                // Only the active transport's detach ends the
858                // session; a displaced transport's stale detach
859                // must not clear the successor's session state.
860                if arbitration.detach(transport) {
861                    env.set_advertising_allowed(true);
862                    session.detach();
863                    // The session that enabled backhaul is gone; without
864                    // this, detached operation would keep the mux routing
865                    // for a host that no longer exists and the device
866                    // would stop queueing what it hears off the air.
867                    crate::radio_mux::MUX_MODE.set_backhaul(false);
868                }
869            }
870            Either4::First(InEvent::CancelAlert) => {
871                // Whoever found the radio silenced it. Publishing the
872                // transition is not conditional on a host being
873                // attached — `cancel_alert` handles that — and a press
874                // with no alert running is simply nothing.
875                let effect = session.cancel_alert(&mut |frame: &[u8]| emitter.push(frame));
876                emitter.flush(arbitration.destination(), rt.out).await;
877                apply_effect(&session, effect, &rt, &mut env).await;
878            }
879            Either4::First(InEvent::ToggleGnss) => {
880                // The switch itself reaches the receiver through the
881                // device-domain mirror at the bottom of this loop, like
882                // every other write to it.
883                if let Some(enabled) = session.toggle_gnss(&mut |frame: &[u8]| emitter.push(frame))
884                {
885                    emitter.flush(arbitration.destination(), rt.out).await;
886                    env.gnss_switched(enabled);
887                    // Keep an existing snapshot in step, so a switch the
888                    // operator flipped is still flipped after a reboot.
889                    // A device with nothing saved gets nothing saved:
890                    // manufacturing a snapshot from a button press would
891                    // persist every other live-only value with it.
892                    if session.saved_status() != SavedStatus::None
893                        && let Some(len) = session.encode_snapshot(&mut snapshot_buf)
894                        && env.persist_snapshot(&snapshot_buf[..len]).await.is_ok()
895                    {
896                        session.note_snapshot_saved();
897                    }
898                    crate::log::debug_log(format_args!(
899                        "ulcp: gnss {} at the device",
900                        if enabled { "ON" } else { "off" }
901                    ));
902                }
903            }
904            Either4::First(InEvent::Frame(transport, frame_bytes)) => {
905                if arbitration.accepts_frame(transport) {
906                    let now_ms = Instant::now().as_millis();
907                    let effect =
908                        session.handle_frame(&frame_bytes, now_ms, &mut |frame: &[u8]| {
909                            emitter.push(frame)
910                        });
911                    emitter.flush(arbitration.destination(), rt.out).await;
912                    match effect {
913                        Some(Effect::SampleRssi { tid }) => {
914                            // Round-trip to the radio runner for an
915                            // instantaneous RSSI sample, then answer the
916                            // deferred PROP_PHY_RSSI get.
917                            rt.ctl.request_rssi();
918                            let sample = rt.ctl.wait_rssi().await;
919                            session
920                                .respond_rssi(tid, sample, &mut |frame: &[u8]| emitter.push(frame));
921                            emitter.flush(arbitration.destination(), rt.out).await;
922                        }
923                        Some(Effect::SignIdentity { tid }) => {
924                            // The session holds no signing key, so the
925                            // platform builds the canonical node-identity
926                            // payload from the same profile the Identity
927                            // Request responder advertises and signs it
928                            // with the device identity.
929                            let mut blob = [0u8; IDENTITY_BLOB_MAX];
930                            let signed = env.sign_identity(&mut blob).await;
931                            session.respond_identity_blob(
932                                tid,
933                                signed.map(|len| &blob[..len]).ok_or(()),
934                                &mut |frame: &[u8]| emitter.push(frame),
935                            );
936                            emitter.flush(arbitration.destination(), rt.out).await;
937                        }
938                        Some(Effect::SampleBattery { tid }) => {
939                            // Round-trip to the platform battery
940                            // source for a fresh measurement, then
941                            // answer the deferred PROP_BATTERY get.
942                            let sample = env.sample_battery().await;
943                            session.respond_battery(tid, sample, &mut |frame: &[u8]| {
944                                emitter.push(frame)
945                            });
946                            emitter.flush(arbitration.destination(), rt.out).await;
947                        }
948                        Some(Effect::SampleIlluminance { tid }) => {
949                            let millilux = env.sample_illuminance().await;
950                            session.respond_illuminance(tid, millilux, &mut |frame: &[u8]| {
951                                emitter.push(frame)
952                            });
953                            emitter.flush(arbitration.destination(), rt.out).await;
954                        }
955                        Some(Effect::ReadTime { tid }) => {
956                            let epoch = env.read_time().await;
957                            session
958                                .respond_time(tid, epoch, &mut |frame: &[u8]| emitter.push(frame));
959                            emitter.flush(arbitration.destination(), rt.out).await;
960                        }
961                        Some(Effect::SampleGnss { tid, key }) => {
962                            let sample = env.sample_gnss().await;
963                            session.respond_gnss(tid, key, sample, &mut |frame: &[u8]| {
964                                emitter.push(frame)
965                            });
966                            emitter.flush(arbitration.destination(), rt.out).await;
967                        }
968                        Some(Effect::DrainQueue) => {
969                            // Deliver the covered frames one per
970                            // step, flushing between steps so the
971                            // two-slot emitter never overflows and
972                            // the transport applies backpressure.
973                            loop {
974                                let more = session.drain_step(
975                                    Instant::now().as_millis(),
976                                    &mut |frame: &[u8]| emitter.push(frame),
977                                );
978                                emitter.flush(arbitration.destination(), rt.out).await;
979                                if !more {
980                                    break;
981                                }
982                            }
983                            env.clear_attention();
984                        }
985                        Some(Effect::SaveSnapshot { tid }) => {
986                            let result = match session.encode_snapshot(&mut snapshot_buf) {
987                                Some(len) => env.persist_snapshot(&snapshot_buf[..len]).await,
988                                None => Err(()),
989                            };
990                            session
991                                .respond_save(tid, result, &mut |frame: &[u8]| emitter.push(frame));
992                            emitter.flush(arbitration.destination(), rt.out).await;
993                        }
994                        Some(Effect::ClearSaved { tid }) => {
995                            // CMD_CLEAR covers all persisted
996                            // provisioning: the snapshot and the
997                            // independently persisted device
998                            // identity. Each journal's tombstone is
999                            // individually atomic; an interruption
1000                            // between them reports failure and the
1001                            // host's retry completes the erase.
1002                            let result = match env.clear_snapshot().await {
1003                                Ok(()) => env.clear_identity().await,
1004                                Err(()) => Err(()),
1005                            };
1006                            // With the identity durably gone, its
1007                            // counter boundaries are dead weight;
1008                            // drop them with it. (Kept if the
1009                            // identity clear failed — the identity
1010                            // then survives the reboot and still
1011                            // needs its TX boundary.)
1012                            if result.is_ok() {
1013                                env.clear_counters().await;
1014                            }
1015                            session.respond_clear(tid, result, &mut |frame: &[u8]| {
1016                                emitter.push(frame)
1017                            });
1018                            emitter.flush(arbitration.destination(), rt.out).await;
1019                        }
1020                        Some(Effect::ProvisionIdentity { tid }) => {
1021                            // Build the keypair (drawing a fresh
1022                            // secret from the platform RNG for
1023                            // on-device generation), persist it, and
1024                            // only then report the public key.
1025                            let result = match session.identity_request() {
1026                                Some(source) => {
1027                                    let secret = match source {
1028                                        IdentitySource::Install(secret) => Ok(secret),
1029                                        IdentitySource::Generate => {
1030                                            let mut secret = [0u8; 32];
1031                                            env.fill_secret(&mut secret).map(|()| secret)
1032                                        }
1033                                    };
1034                                    match secret {
1035                                        Ok(secret) => {
1036                                            let (public_key, payload) =
1037                                                device_identity_record(&secret);
1038                                            env.persist_identity(&payload)
1039                                                .await
1040                                                .map(|()| public_key)
1041                                        }
1042                                        Err(()) => Err(()),
1043                                    }
1044                                }
1045                                None => Err(()),
1046                            };
1047                            session.respond_identity(tid, result, &mut |frame: &[u8]| {
1048                                emitter.push(frame)
1049                            });
1050                            emitter.flush(arbitration.destination(), rt.out).await;
1051                        }
1052                        Some(Effect::SetPairingPin { tid, pin }) => {
1053                            let applied = env.apply_pairing_pin(pin).await;
1054                            session.respond_pin_set(
1055                                tid,
1056                                applied.then_some(()).ok_or(()),
1057                                &mut |frame: &[u8]| emitter.push(frame),
1058                            );
1059                            emitter.flush(arbitration.destination(), rt.out).await;
1060                        }
1061                        Some(Effect::FactoryReset) => {
1062                            // Hand off to the platform, which erases every
1063                            // persistent journal and reboots. This never
1064                            // returns; no acknowledgement is sent because the
1065                            // reset drops the link. Any frames the session
1066                            // already staged were flushed above.
1067                            env.trace(format_args!("CMD_FACTORY_RESET: wiping all state + reboot"));
1068                            env.factory_reset().await
1069                        }
1070                        other => apply_effect(&session, other, &rt, &mut env).await,
1071                    }
1072                    // A device identity always exists. `CMD_CLEAR`
1073                    // erases the stored one without touching live state,
1074                    // and the `CMD_RST` that completes a factory reset
1075                    // is where the live copy catches up — leaving the
1076                    // device with none, which is the one state the
1077                    // invariant forbids. Regenerate here, exactly as
1078                    // first boot would, so the only way to reach an
1079                    // identityless device is to physically remove it
1080                    // from existence.
1081                    //
1082                    // The running device node keeps the *previous* key
1083                    // in its MAC until the next boot (identity is
1084                    // fixed at bring-up), and stops originating traffic
1085                    // because the dev-domain sync gate compares keys
1086                    // rather than counting them.
1087                    if session.dev_key().is_none() {
1088                        regenerate_device_identity(&mut session, &mut env).await;
1089                    }
1090                    if session.queued_frame_count() == 0 {
1091                        env.clear_attention();
1092                    }
1093                }
1094            }
1095            Either4::Second(RxFrame { data, info }) => {
1096                // While detached this may stage a delegated MAC
1097                // acknowledgement (Effect::StartTransmit).
1098                let queued_before = session.queued_frame_count();
1099                let rx_info = match info.origin {
1100                    RxOrigin::Air => {
1101                        RadioRxInfo::measured(info.rssi, info.snr.as_centibels(), info.lqi)
1102                    }
1103                    // Backhaul frames travel host-to-node, so the session
1104                    // never sees one arriving.
1105                    RxOrigin::LocalTx | RxOrigin::Backhaul => RadioRxInfo::self_transmitted(),
1106                };
1107                let effect = session.on_radio_rx(
1108                    &data,
1109                    &rx_info,
1110                    Instant::now().as_millis(),
1111                    &mut |frame: &[u8]| emitter.push(frame),
1112                );
1113                if session.queued_frame_count() > queued_before {
1114                    env.request_attention();
1115                }
1116                emitter.flush(arbitration.destination(), rt.out).await;
1117                apply_effect(&session, effect, &rt, &mut env).await;
1118            }
1119            Either4::Third(result) => {
1120                let now_ms = Instant::now().as_millis();
1121                let outcome = match result {
1122                    Ok(()) => TxOutcome::Sent,
1123                    Err(umsh_hal::TxError::CadTimeout) => TxOutcome::ChannelBusy,
1124                    Err(umsh_hal::TxError::Io(_)) => TxOutcome::Failed,
1125                };
1126                let effect =
1127                    session.on_tx_result(outcome, now_ms, &mut |frame: &[u8]| emitter.push(frame));
1128                emitter.flush(arbitration.destination(), rt.out).await;
1129                apply_effect(&session, effect, &rt, &mut env).await;
1130            }
1131            Either4::Fourth(Either::First(event)) => {
1132                // The board decided this is worth announcing; publish it
1133                // unsolicited. Dropped silently while no host is
1134                // attached, and no effect can result — a publication is
1135                // not an operation.
1136                let emit = &mut |frame: &[u8]| emitter.push(frame);
1137                match event {
1138                    PublishEvent::Battery(sample) => {
1139                        session.publish_battery(sample, emit);
1140                    }
1141                    PublishEvent::Time(epoch) => {
1142                        session.publish_time(epoch, emit);
1143                    }
1144                    PublishEvent::Gnss(key, snapshot) => {
1145                        session.publish_gnss(key, &snapshot, emit);
1146                    }
1147                }
1148                emitter.flush(arbitration.destination(), rt.out).await;
1149            }
1150            Either4::Fourth(Either::Second(())) => {
1151                // The alert outlived its deadline: stop the indication
1152                // and publish the transition the host did not command.
1153                let effect = session
1154                    .poll_alert(Instant::now().as_millis(), &mut |frame: &[u8]| {
1155                        emitter.push(frame)
1156                    });
1157                emitter.flush(arbitration.destination(), rt.out).await;
1158                apply_effect(&session, effect, &rt, &mut env).await;
1159            }
1160        }
1161        // Any of the arms may have moved the device-domain tables
1162        // (property mutation, CMD_RST, CMD_RESTORE); one u32
1163        // compare when they did not.
1164        sync_dev_domain(&session, &mut dev_domain_synced, &mut env);
1165    }
