1//! The board-agnostic ULCP session driver (Phase 5, increment C).
2//!
3//! This is the extraction of the nRF firmware's `device_task` select loop, its
4//! `apply_effect` radio-effect dispatcher, and the `Emitter` frame stager —
5//! the one copy of the session-driving logic shared by every device
6//! firmware (T-Echo, T-1000E, Heltec V3). The board personalities that used
7//! to be `cfg(feature = "t1000e")` forks inside the loop are expressed as
8//! [`DeviceEnv`] hooks with no-op defaults, so a new board supplies exactly the
9//! couplings it has and nothing else.
10//!
11//! The split of responsibilities:
12//!
13//! - **This module** owns the protocol loop: transport arbitration, frame
14//!   handling, radio RX/TX-completion processing, every deferred `Effect`
15//!   arm (save/clear/wipe/provision/PIN/RSSI/battery/drain), asynchronous
16//!   property publication ([`DeviceEnv::battery_event`]), and the
17//!   device-domain mirror.
18//! - **The board** owns the edges: transport tasks feeding [`InEvent`]s and
19//!   draining [`TransportChannels`], the radio runner + mux serving the
20//!   session's virtual [`Channels`] bundle, and an [`DeviceEnv`] implementation
21//!   wiring persistence, entropy, pairing, and indicators to its hardware.
22

44/// Derive a device identity's public key and its persisted record from a
45/// raw Ed25519 secret.
46///
47/// Shared by first-boot generation and `Effect::ProvisionIdentity` so the
48/// two can never disagree about the derivation or the record layout.
49///
50/// The caller supplies `secret` and owns the question this function
51/// cannot answer: it **MUST** come from a cryptographic RNG with real
52/// entropy behind it. On the nRF boards that is the hardware TRNG with
53/// bias correction enabled; on Espressif it is `EspCryptoRng`, which
54/// refuses to exist unless the RF noise source is live.
55pub fn device_identity_record(secret: &[u8; 32]) -> ([u8; 32], [u8; proto::IDENTITY_PAYLOAD_LEN]) {
56    let public_key = SoftwareIdentity::from_secret_bytes(secret).public_key().0;
57    (public_key, proto::encode_identity(secret, &public_key))
58}

60/// How many older snapshot generations boot will try after the newest
61/// one is rejected.
62///
63/// Bounded deliberately. Corruption is expected to affect one record, so
64/// a handful of generations covers it; a payload that is *systematically*
65/// undecodable is a firmware bug, and walking the whole journal for it on
66/// every boot would only delay booting bare and reporting the fact.
67pub const SNAPSHOT_FALLBACK_LIMIT: usize = 4;

98/// One raw ULCP frame in a transport output queue, stamped with the

106/// The per-transport outbound frame queues, drained by the board's
107/// transport output tasks. `wired` is the physical-possession transport

109/// `transport_policy::Transport` names.
110pub struct TransportChannels<M: RawMutex> {
111    pub wired: Channel<M, OutFrame, 4>,
112    pub ble: Channel<M, OutFrame, 4>,

115impl<M: RawMutex> TransportChannels<M> {
116    pub const fn new() -> Self {

118            wired: Channel::new(),

131/// The session's device-domain tables, mirrored to the board's device
132/// node whenever their generation moves (device-node plan increment 3).
133pub struct DevDomainSnapshot {
134    pub channel_keys: heapless::Vec<[u8; 32], MAX_CHANNEL_KEYS>,
135    pub peers: heapless::Vec<[u8; 32], MAX_DEV_PEERS>,
136    /// The session's live `PROP_DEV_KEY`.
137    ///
138    /// The device node compares this against the key its MAC was built
139    /// with rather than merely checking that *some* identity exists. The
140    /// two can disagree — a newly installed `PROP_DEV_PRIVATE_KEY` takes
141    /// effect at the next boot, and `CMD_CLEAR` + `CMD_RST` erases the
142    /// stored one — and in every such case the running node is no longer
143    /// the identity the session describes and must stop originating
144    /// traffic under it.
145    pub dev_key: Option<[u8; 32]>,
146    /// `PROP_MAC_REPEATER_ENABLED`: whether the device node should
147    /// forward overheard routable frames. Advertised as the `REP`
148    /// capability bit — a fact about what the node does, not a choice
149    /// about what it calls itself.
150    pub repeater_enabled: bool,
151    /// `PROP_MAC_REPEATER_REGIONS`: the flood-forwarding region filter,
152    /// as concatenated 2-octet codes. Empty imposes no restriction.
153    ///
154    /// Carried in wire order so the device node can both configure the
155    /// MAC's region filter and hand the same bytes to the Supported
156    /// Regions identity option without reshaping either.
157    pub repeater_regions: heapless::Vec<u8, { MAX_REPEATER_REGIONS * 2 }>,
158    /// `PROP_MAC_REPEATER_DEFAULT_REGION`: the code the node inserts
159    /// into an untagged flood packet, or `None` to never tag.
160    pub repeater_default_region: Option<[u8; 2]>,
161    /// `PROP_MAC_REPEATER_MIN_RSSI`: minimum RSSI in dBm to
162    /// flood-forward, or `None` for no threshold.
163    pub repeater_min_rssi: Option<i16>,
164    /// `PROP_MAC_REPEATER_MIN_SNR`: minimum SNR in whole dB to
165    /// flood-forward, or `None` for no threshold.
166    pub repeater_min_snr: Option<i8>,
167    /// `PROP_IDENT_ROLE`: the advertised `ROLE` byte, or `None` to
168    /// derive it from the forwarding state.
169    pub ident_role: Option<u8>,
170    /// `PROP_IDENT_MOBILE`: whether to advertise the `MOB` capability
171    /// bit.
172    pub ident_mobile: bool,
173    /// `PROP_DEV_DISCOVERABLE`: whether the device identity answers
174    /// Identity Requests.
175    pub discoverable: bool,
176}

178/// Board couplings of the session driver. Everything the loop needs from
179/// the platform, expressed as one trait so the driver itself stays free
180/// of HAL types and `cfg` board forks. Hooks a board doesn't have keep
181/// their no-op defaults (e.g. only the T-1000E implements the attention
182/// indicator and transmit-load hooks today).
183// Single-executor embedded consumers; `Send` futures are irrelevant here,
184// same as the embassy ecosystem's own async traits.
185#[allow(async_fn_in_trait)]
186pub trait DeviceEnv {
187    /// Durably persist the encoded protocol snapshot (CMD_SAVE / host wipe).
188    async fn persist_snapshot(&mut self, bytes: &[u8]) -> Result<(), ()>;
189    /// Tombstone the snapshot journal (CMD_CLEAR).
190    async fn clear_snapshot(&mut self) -> Result<(), ()>;
191    /// Copy the newest committed snapshot generation strictly older than
192    /// the one last handed to the driver into `out`, returning its
193    /// length.
194    ///
195    /// Called only after a payload is rejected, so the cost is paid on a
196    /// boot that is already going wrong. Implementations re-scan the
197    /// journal rather than retaining a runner-up, keeping the mount
198    /// path's "never buffers a second copy" discipline. The default
199    /// refuses, which makes rejection terminal for boards whose journal
200    /// cannot walk back.
201    async fn older_snapshot(&mut self, out: &mut [u8]) -> Option<usize> {
202        let _ = out;
203        None
204    }
205    /// A stored snapshot was rejected at boot. Boards with an indicator
206    /// surface it locally: the host-visible report reaches nobody on an
207    /// unattended repeater, which is exactly the deployment this
208    /// matters for.
209    fn report_snapshot_rejected(&mut self, fell_back: bool) {
210        let _ = fell_back;
211    }
212    /// Durably persist the encoded device identity.
213    async fn persist_identity(&mut self, bytes: &[u8]) -> Result<(), ()>;
214    /// Tombstone the identity journal (CMD_CLEAR).
215    async fn clear_identity(&mut self) -> Result<(), ()>;
216    /// Drop persisted frame-counter boundaries after a successful
217    /// identity clear. Boards without a device node keep the default.
218    async fn clear_counters(&mut self) {}
219    /// Fill `secret` from the platform's cryptographic RNG. Fails closed:
220    /// an error refuses identity generation rather than degrading.
221    fn fill_secret(&mut self, secret: &mut [u8; 32]) -> Result<(), ()>;
222    /// One fresh battery measurement (`Effect::SampleBattery`). Only
223    /// emitted when the board's `SessionConfig::battery` advertises
224    /// fields, so the default refuses.
225    async fn sample_battery(&mut self) -> Result<umsh_ulcp::battery::BatteryStatus, ()> {
226        Err(())
227    }
228    /// Wait for a battery measurement the board considers worth
229    /// announcing, for publication as an unsolicited `PROP_BATTERY`
230    /// (`Session::publish_battery`).
231    ///
232    /// The board owns the whole policy: the sampling cadence, the
233    /// charge-state edges, and which changes matter. It is the only layer
234    /// that sees every sample, so filtering there keeps the session free
235    /// of cached readings and keeps this hook's contract simple — every
236    /// value it yields is published.
237    ///
238    /// Cancellation-safe: the driver drops and re-creates this future on
239    /// every other loop iteration, so an implementation must not lose an
240    /// update it was cancelled on (an `embassy_sync::watch::Watch`
241    /// receiver behaves correctly here; a bare `Signal` does not).
242    ///
243    /// The default never completes, so boards without battery push add
244    /// nothing to the select.
245    async fn battery_event(&mut self) -> umsh_ulcp::battery::BatteryStatus {
246        core::future::pending().await
247    }
248    /// Build and sign the device identity's node-identity blob into
249    /// `out` (`Effect::SignIdentity`), returning its length.
250    ///
251    /// The board owns both halves the session does not: the signing key,
252    /// and the advertised profile the Identity Request responder uses.
253    /// Boards without a device node keep the default, which refuses.
254    async fn sign_identity(&mut self, out: &mut [u8]) -> Option<usize> {
255        let _ = out;
256        None
257    }
258    /// Apply a `PROP_BLE_PAIRING_PIN` write against the bond journal and
259    /// the live BLE stack; `true` when it took effect.
260    async fn apply_pairing_pin(&mut self, pin: Option<u32>) -> bool;
261    /// `CMD_FACTORY_RESET`: erase EVERY piece of persistent state the
262    /// platform owns — saved snapshot, device identity, frame-counter
263    /// boundaries, BLE bonds, pairing PIN, and any other journal — then
264    /// reboot. Never returns: the reset discards in-RAM state and the
265    /// board comes back factory-fresh. There is no separate "clear bonds"
266    /// hook because a reboot reloads bonds from the (now-erased) journal,
267    /// so the live BLE stack never has to be touched.
268    async fn factory_reset(&mut self) -> !;
269    /// Publish the transport-arbitration advertising policy (a wired
270    /// attach suppresses BLE advertising). Diagnostic builds may
271    /// deliberately ignore `allowed`.
272    fn set_advertising_allowed(&mut self, allowed: bool);
273    /// Publish the session's device name to the board's consumers
274    /// (advertising data, device node, UI).
275    async fn publish_device_name(&mut self, name: &str);
276    /// Deliver a device-domain mirror to the board's device node.
277    fn publish_dev_domain(&mut self, snapshot: DevDomainSnapshot);
278    /// Start or stop the board's locate indication (`PROP_ALERT`).
279    ///
280    /// Carries the authoritative state and is called for every
281    /// transition — host write, local cancellation, and deadline — so an
282    /// implementation can treat it as idempotent and needs no notion of
283    /// *why* the alert ended. `AlertState::Locate` must override a local
284    /// silence setting without clearing it (spec §PROP_ALERT); boards
285    /// without `CAP_ALERT` never see this and keep the default.
286    fn set_alert(&mut self, state: umsh_ulcp::alert::AlertState) {
287        let _ = state;
288    }
289    /// A covered frame was queued for an attached-or-future host
290    /// (T-1000E: request the attention LED).
291    fn request_attention(&mut self) {}
292    /// The host-facing queue drained to empty (T-1000E: clear it).
293    fn clear_attention(&mut self) {}
294    /// A transmit is about to start; boards with a battery-level
295    /// estimator mark the load spike.
296    fn note_transmit_load(&mut self) {}
297    /// Diagnostic trace line (routed to the board's debug channel; the
298    /// default discards).
299    fn trace(&mut self, args: core::fmt::Arguments<'_>) {
300        let _ = args;
301    }
302}

304/// The driver's `'static` wiring: the channels and control blocks the
305/// loop shares with the board's transport and radio tasks.
306pub struct DeviceRuntime<M: RawMutex + 'static, const RX: usize, const TX: usize> {
307    /// Inbound frames and connection edges from every transport task.
308    pub input: &'static InputChannel<M>,
309    /// The session's radio endpoint — its private virtual `Channels`
310    /// bundle served by the board's radio mux (never the real radio
311    /// bundle directly).
312    pub radio: &'static Channels<M, RX, TX>,
313    /// Runtime radio settings / RSSI sampling into the radio runner.
314    pub ctl: &'static DeviceControl<M>,
315    /// Outbound frame queues drained by the transport output tasks.
316    pub out: &'static TransportChannels<M>,
317    /// Published session epoch, checked by each transport at framing
318    /// edges (`transport_policy::generation_checked`).
319    pub session_gen: &'static AtomicU32,
320}

322/// Collects frames emitted synchronously by the session, then flushes
323/// them to the active transport's output queue asynchronously. The
324/// session emits at most one frame per call; two slots give headroom.
325struct Emitter {
326    bufs: [[u8; FRAME_OUT_MAX]; 2],
327    lens: [usize; 2],
328    count: usize,
329}
330
331impl Emitter {
332    const fn new() -> Self {
333        Self {
334            bufs: [[0; FRAME_OUT_MAX]; 2],
335            lens: [0; 2],
336            count: 0,
337        }
338    }
339
340    /// Copy one raw ULCP frame into the next slot.
341    ///
342    /// The session is expected to emit at most `bufs.len()` frames per call
343    /// and every frame is expected to fit `FRAME_OUT_MAX`. Both invariants are
344    /// asserted in debug builds so a future session change that violates
345    /// them is caught rather than silently dropping a response.
346    fn push(&mut self, frame: &[u8]) {
347        if self.count >= self.bufs.len() {
348            debug_assert!(
349                false,
350                "Emitter overflow: session emitted more frames per call than staging slots"
351            );
352            return;
353        }
354        if frame.len() <= FRAME_OUT_MAX {
355            self.bufs[self.count][..frame.len()].copy_from_slice(frame);
356            self.lens[self.count] = frame.len();
357            self.count += 1;
358        } else {
359            debug_assert!(false, "Emitter: ULCP frame exceeds FRAME_OUT_MAX");
360        }
361    }
362
363    /// Queue all staged frames for the active transport output task.
364    async fn flush<M: RawMutex>(
365        &mut self,
366        destination: Option<(Transport, u32)>,
367        out: &TransportChannels<M>,
368    ) {
369        if let Some((transport, generation)) = destination {
370            for index in 0..self.count {
371                let mut frame: FrameBuf = heapless::Vec::new();
372                if frame
373                    .extend_from_slice(&self.bufs[index][..self.lens[index]])
374                    .is_err()
375                {
376                    // FRAME_OUT_MAX == FrameBuf capacity, so this cannot
377                    // happen; assert in debug rather than silently drop.
378                    debug_assert!(false, "Emitter frame copy exceeded FrameBuf capacity");
379                    continue;
380                }
381                out.for_transport(transport)
382                    .send(OutFrame { generation, frame })
383                    .await;
384            }
385        }
386        self.count = 0;
387    }
388}

390/// Execute a radio side effect requested by the session.
391async fn apply_effect<A, S, const TXQ: usize, M, const RX: usize, const TX: usize, E>(
392    session: &Session<A, S, TXQ>,
393    effect: Option<Effect>,
394    rt: &DeviceRuntime<M, RX, TX>,
395    env: &mut E,
396) where
397    A: AesProvider,
398    S: Sha256Provider,
399    M: RawMutex,
400    E: DeviceEnv,
401{
402    match effect {
403        Some(Effect::ApplyRadio(settings)) => {
404            env.publish_device_name(session.device_name()).await;
405            // The session validates values against the same discrete
406            // sets these converters accept, so None here is
407            // unreachable; bail out defensively rather than panic.
408            let (Some(sf), Some(bw), Some(cr)) = (
409                spreading_factor_from_u8(settings.sf),
410                bandwidth_from_hz(settings.bw_hz),
411                coding_rate_from_denom(settings.cr_denom),
412            ) else {
413                return;
414            };
415            rt.ctl.apply(DeviceSettings {
416                enabled: settings.enabled,
417                freq_hz: settings.freq_khz.saturating_mul(1_000),
418                sf,
419                bw,
420                cr,
421                power_dbm: i32::from(settings.tx_power_dbm),
422            });
423        }
424        Some(Effect::StartTransmit) => {
425            let mut data: heapless::Vec<u8, MAX_PAYLOAD> = heapless::Vec::new();
426            if data.extend_from_slice(session.tx_data()).is_err() {
427                env.trace(format_args!(
428                    "radio tx staging=FAILED len={}",
429                    session.tx_data().len()
430                ));
431                return;
432            }
433            let power_dbm = match session.tx_power() {
434                TxPower::Default => None,
435                TxPower::Max => Some(i32::from(session.max_tx_power_dbm())),
436                TxPower::Dbm(dbm) => Some(i32::from(dbm)),
437            };
438            // Mark the load for the board's battery level estimator (the
439            // radio runner transmits within milliseconds of this).
440            env.note_transmit_load();
441            let cad = if session.tx_nocca() {
442                CadPolicy::Skip
443            } else {
444                CadPolicy::Gate
445            };
446            rt.radio
447                .tx
448                .send(TxRequest {
449                    data,
450                    power_dbm,
451                    cad,
452                })
453                .await;
454        }
455        Some(Effect::DeviceNameChanged) => {
456            env.publish_device_name(session.device_name()).await;
457        }
458        Some(Effect::ApplyAlert(state)) => {
459            env.set_alert(state);
460        }
461        // Deferred effects needing `&mut Session` + the emitter are
462        // handled inline in the run loop rather than here.
463        Some(Effect::SampleRssi { .. })
464        | Some(Effect::SignIdentity { .. })
465        | Some(Effect::SampleBattery { .. })
466        | Some(Effect::SetPairingPin { .. })
467        | Some(Effect::DrainQueue)
468        | Some(Effect::SaveSnapshot { .. })
469        | Some(Effect::ClearSaved { .. })
470        | Some(Effect::ProvisionIdentity { .. })
471        | Some(Effect::FactoryReset)
472        | None => {}
473    }
474}

476/// Mirror the session's device-domain node tables to the device node
477/// when their generation moved (device-node plan increment 3).
478/// `synced_version` is the caller's cache of the last published
479/// generation. Cheap when nothing changed — one u32 compare — so the
480/// loop runs it after every session interaction.
481/// Generate, persist, and install a fresh device identity.
482///
483/// The counterpart to first-boot generation, for the one runtime path
484/// that can leave the session without one: `CMD_CLEAR` followed by the
485/// `CMD_RST` that completes a factory reset. A device identity is not a
486/// commissioning step, so there is no state in which the operator has to
487/// supply one.
488///
489/// A failure to draw entropy or to persist leaves the session
490/// identityless, which is a worse outcome than either but not one this
491/// layer can repair: it is reported and the next boot regenerates.
492async fn regenerate_device_identity<A, S, const TXQ: usize, E>(
493    session: &mut Session<A, S, TXQ>,
494    env: &mut E,
495) where
496    A: AesProvider,
497    S: Sha256Provider,
498    E: DeviceEnv,
499{
500    let mut secret = [0u8; 32];
501    if env.fill_secret(&mut secret).is_err() {
502        env.trace(format_args!("device identity regenerate: entropy FAILED"));
503        return;
504    }
505    let (public_key, payload) = device_identity_record(&secret);
506    match env.persist_identity(&payload).await {
507        Ok(()) => {
508            session.set_boot_identity(public_key);
509            env.trace(format_args!(
510                "device identity regenerated after clear+reset"
511            ));
512        }
513        Err(()) => env.trace(format_args!(
514            "device identity regenerate: persist FAILED — none in effect"
515        )),
516    }
517}

519fn sync_dev_domain<A, S, const TXQ: usize, E>(
520    session: &Session<A, S, TXQ>,
521    synced_version: &mut u32,
522    env: &mut E,
523) where
524    A: AesProvider,
525    S: Sha256Provider,
526    E: DeviceEnv,
527{
528    if session.dev_domain_version() == *synced_version {
529        return;
530    }
531    *synced_version = session.dev_domain_version();
532    let mut snapshot = DevDomainSnapshot {
533        channel_keys: heapless::Vec::new(),
534        peers: heapless::Vec::new(),
535        dev_key: session.dev_key().copied(),
536        repeater_enabled: session.repeater_enabled(),
537        repeater_regions: heapless::Vec::from_slice(session.repeater_regions()).unwrap_or_default(),
538        repeater_default_region: session.repeater_default_region(),
539        repeater_min_rssi: session.repeater_min_rssi(),
540        repeater_min_snr: session.repeater_min_snr(),
541        ident_role: session.ident_role(),
542        ident_mobile: session.ident_mobile(),
543        discoverable: session.dev_discoverable(),
544    };
545    for key in session.dev_channel_keys() {
546        let _ = snapshot.channel_keys.push(key);
547    }
548    for public_key in session.dev_peers() {
549        let _ = snapshot.peers.push(public_key);
550    }
551    env.publish_dev_domain(snapshot);
552}

554/// Drive the ULCP session forever: restore persisted state, then
555/// select over host frames, radio receptions, and transmit completions,
556/// executing every session effect through the board's [`DeviceEnv`].
557///
558/// The caller constructs the [`Session`] with its board profile
559/// (`SessionConfig`) and boot status, mounts its journals, and hands the
560/// stored payloads in; the driver owns everything after that.
561pub async fn run<A, S, const TXQ: usize, M, const RX: usize, const TX: usize, E>(
562    mut session: Session<A, S, TXQ>,
563    boot_snapshot: Option<&[u8]>,
564    boot_identity: Option<[u8; 32]>,
565    rt: DeviceRuntime<M, RX, TX>,
566    mut env: E,
567) -> !
568where
569    A: AesProvider,
570    S: Sha256Provider,
571    M: RawMutex,
572    E: DeviceEnv,
573{
574    let mut emitter = Emitter::new();
575    let mut arbitration = SessionArbitration::new(rt.session_gen.load(Ordering::Acquire));
576    // Last device-domain generation mirrored to the device node.
577    // Matches the session's initial value; the first mutation (or a
578    // boot restore) publishes the first snapshot.
579    let mut dev_domain_synced: u32 = session.dev_domain_version();
580    // Shared staging buffer for the durable-write effect arms
581    // (save/wipe). Held across their persist awaits, so as a
582    // loop-lifetime local it costs one future slot instead of one
583    // per arm.
584    let mut snapshot_buf = [0u8; SNAPSHOT_MAX];
585
586    // The device identity is persisted independently of snapshots;
587    // its post-reset value is whatever the identity journal holds.
588    if let Some(public_key) = boot_identity {
589        session.set_boot_identity(public_key);
590    }
591
592    // Restore a stored snapshot before processing any host command:
593    // the saved configuration is applied, the PHY re-enabled if it
594    // was enabled when saved, and detached operation begins
595    // immediately.
596    //
597    // A payload that does not decode is not the end of it. The journal
598    // is multi-record and newest-generation-wins, so an older readable
599    // generation usually sits behind the rejected one — and for an
600    // unattended repeater, falling back to it is the only outcome that
601    // keeps the device forwarding. Walk back a bounded number of
602    // generations: a systematically undecodable payload is a firmware
603    // bug rather than corruption, and re-walking the whole journal on
604    // every boot would just be slower about it.
605    if let Some(payload) = boot_snapshot {
606        let mut generation = 0usize;
607        let mut restored = session.restore_at_boot(payload);
608        while let Err(error) = restored {
609            env.trace(format_args!(
610                "proto-store boot-restore generation=-{generation} REJECTED error={error:?}"
611            ));
612            session.note_snapshot_rejected();
613            generation += 1;
614            if generation > SNAPSHOT_FALLBACK_LIMIT {
615                env.trace(format_args!(
616                    "proto-store boot-restore fallback=EXHAUSTED limit={SNAPSHOT_FALLBACK_LIMIT}"
617                ));
618                break;
619            }
620            let Some(len) = env.older_snapshot(&mut snapshot_buf).await else {
621                env.trace(format_args!("proto-store boot-restore fallback=NONE"));
622                break;
623            };
624            restored = session.restore_at_boot(&snapshot_buf[..len]);
625        }
626        match restored {
627            Ok(effect) => {
628                if generation > 0 {
629                    env.trace(format_args!(
630                        "proto-store boot-restore=FALLBACK generation=-{generation}"
631                    ));
632                    env.report_snapshot_rejected(true);
633                } else {
634                    env.trace(format_args!("proto-store boot-restore=ok"));
635                }
636                apply_effect(&session, Some(effect), &rt, &mut env).await;
637                // Replay the restored device-domain tables into the
638                // device node before any host interaction: detached
639                // multicast processing must not wait for an attach.
640                sync_dev_domain(&session, &mut dev_domain_synced, &mut env);
641            }
642            Err(_) => {
643                env.trace(format_args!("proto-store boot-restore=BARE"));
644                env.report_snapshot_rejected(false);
645            }
646        }
647    }
648
649    loop {
650        // Resolve the next event in its own statement so the select's
651        // futures — one of which mutably borrows `env` — are dropped
652        // before the arms below use `env` again. A `match select4(..)`
653        // scrutinee would hold them for the whole match.
654        let event = {
655            // Only wait for a TX completion while one is outstanding,
656            // so a spurious tx_done can never be consumed early.
657            let tx_done = async {
658                if session.has_pending_tx() {
659                    rt.radio.tx_done.wait().await
660                } else {
661                    core::future::pending().await
662                }
663            };
664            // The locate alert's deadline. Enforced here rather than by
665            // each board so that "a device MUST bound how long it will
666            // remain in ALERT_LOCATE" holds for every board that
667            // advertises CAP_ALERT, including ones whose UX layer has no
668            // timer of its own. Idle (never completes) while no alert is
669            // running. It borrows `session` immutably, alongside
670            // `tx_done` — only `battery_event` touches `env`.
671            let alert_deadline = async {
672                match session.alert_deadline_ms() {
673                    Some(deadline) => Timer::at(Instant::from_millis(deadline)).await,
674                    None => core::future::pending().await,
675                }
676            };
677            select4(
678                rt.input.receive(),
679                rt.radio.rx.receive(),
680                tx_done,
681                select(env.battery_event(), alert_deadline),
682            )
683            .await
684        };
685
686        match event {
687            Either4::First(InEvent::Attached(transport)) => {
688                // Fresh session state for the new host session; the
689                // device domain (PHY configuration and enable state,
690                // device name, duty accounting) is deliberately
691                // untouched and nothing is emitted (full-protocol
692                // attach semantics).
693                arbitration.attach(transport);
694                rt.session_gen
695                    .store(arbitration.generation(), Ordering::Release);
696                env.set_advertising_allowed(arbitration.advertising_allowed());
697                // Both transports meet their provisioning-security
698                // binding here: the wired transport by physical
699                // possession, BLE because the ULCP GATT service
700                // refuses any access outside an encrypted LESC-bonded
701                // link.
702                session.attach(true);
703            }
704            Either4::First(InEvent::Detached(transport)) => {
705                // Only the active transport's detach ends the
706                // session; a displaced transport's stale detach
707                // must not clear the successor's session state.
708                if arbitration.detach(transport) {
709                    env.set_advertising_allowed(true);
710                    session.detach();
711                }
712            }
713            Either4::First(InEvent::CancelAlert) => {
714                // Whoever found the radio silenced it. Publishing the
715                // transition is not conditional on a host being
716                // attached — `cancel_alert` handles that — and a press
717                // with no alert running is simply nothing.
718                let effect = session.cancel_alert(&mut |frame: &[u8]| emitter.push(frame));
719                emitter.flush(arbitration.destination(), rt.out).await;
720                apply_effect(&session, effect, &rt, &mut env).await;
721            }
722            Either4::First(InEvent::Frame(transport, frame_bytes)) => {
723                if arbitration.accepts_frame(transport) {
724                    let now_ms = Instant::now().as_millis();
725                    let effect =
726                        session.handle_frame(&frame_bytes, now_ms, &mut |frame: &[u8]| {
727                            emitter.push(frame)
728                        });
729                    emitter.flush(arbitration.destination(), rt.out).await;
730                    match effect {
731                        Some(Effect::SampleRssi { tid }) => {
732                            // Round-trip to the radio runner for an
733                            // instantaneous RSSI sample, then answer the
734                            // deferred PROP_PHY_RSSI get.
735                            rt.ctl.request_rssi();
736                            let sample = rt.ctl.wait_rssi().await;
737                            session
738                                .respond_rssi(tid, sample, &mut |frame: &[u8]| emitter.push(frame));
739                            emitter.flush(arbitration.destination(), rt.out).await;
740                        }
741                        Some(Effect::SignIdentity { tid }) => {
742                            // The session holds no signing key, so the
743                            // platform builds the canonical node-identity
744                            // payload from the same profile the Identity
745                            // Request responder advertises and signs it
746                            // with the device identity.
747                            let mut blob = [0u8; IDENTITY_BLOB_MAX];
748                            let signed = env.sign_identity(&mut blob).await;
749                            session.respond_identity_blob(
750                                tid,
751                                signed.map(|len| &blob[..len]).ok_or(()),
752                                &mut |frame: &[u8]| emitter.push(frame),
753                            );
754                            emitter.flush(arbitration.destination(), rt.out).await;
755                        }
756                        Some(Effect::SampleBattery { tid }) => {
757                            // Round-trip to the platform battery
758                            // source for a fresh measurement, then
759                            // answer the deferred PROP_BATTERY get.
760                            let sample = env.sample_battery().await;
761                            session.respond_battery(tid, sample, &mut |frame: &[u8]| {
762                                emitter.push(frame)
763                            });
764                            emitter.flush(arbitration.destination(), rt.out).await;
765                        }
766                        Some(Effect::DrainQueue) => {
767                            // Deliver the covered frames one per
768                            // step, flushing between steps so the
769                            // two-slot emitter never overflows and
770                            // the transport applies backpressure.
771                            loop {
772                                let more = session.drain_step(
773                                    Instant::now().as_millis(),
774                                    &mut |frame: &[u8]| emitter.push(frame),
775                                );
776                                emitter.flush(arbitration.destination(), rt.out).await;
777                                if !more {
778                                    break;
779                                }
780                            }
781                            env.clear_attention();
782                        }
783                        Some(Effect::SaveSnapshot { tid }) => {
784                            let result = match session.encode_snapshot(&mut snapshot_buf) {
785                                Some(len) => env.persist_snapshot(&snapshot_buf[..len]).await,
786                                None => Err(()),
787                            };
788                            session
789                                .respond_save(tid, result, &mut |frame: &[u8]| emitter.push(frame));
790                            emitter.flush(arbitration.destination(), rt.out).await;
791                        }
792                        Some(Effect::ClearSaved { tid }) => {
793                            // CMD_CLEAR covers all persisted
794                            // provisioning: the snapshot and the
795                            // independently persisted device
796                            // identity. Each journal's tombstone is
797                            // individually atomic; an interruption
798                            // between them reports failure and the
799                            // host's retry completes the erase.
800                            let result = match env.clear_snapshot().await {
801                                Ok(()) => env.clear_identity().await,
802                                Err(()) => Err(()),
803                            };
804                            // With the identity durably gone, its
805                            // counter boundaries are dead weight;
806                            // drop them with it. (Kept if the
807                            // identity clear failed — the identity
808                            // then survives the reboot and still
809                            // needs its TX boundary.)
810                            if result.is_ok() {
811                                env.clear_counters().await;
812                            }
813                            session.respond_clear(tid, result, &mut |frame: &[u8]| {
814                                emitter.push(frame)
815                            });
816                            emitter.flush(arbitration.destination(), rt.out).await;
817                        }
818                        Some(Effect::ProvisionIdentity { tid }) => {
819                            // Build the keypair (drawing a fresh
820                            // secret from the platform RNG for
821                            // on-device generation), persist it, and
822                            // only then report the public key.
823                            let result = match session.identity_request() {
824                                Some(source) => {
825                                    let secret = match source {
826                                        IdentitySource::Install(secret) => Ok(secret),
827                                        IdentitySource::Generate => {
828                                            let mut secret = [0u8; 32];
829                                            env.fill_secret(&mut secret).map(|()| secret)
830                                        }
831                                    };
832                                    match secret {
833                                        Ok(secret) => {
834                                            let (public_key, payload) =
835                                                device_identity_record(&secret);
836                                            env.persist_identity(&payload)
837                                                .await
838                                                .map(|()| public_key)
839                                        }
840                                        Err(()) => Err(()),
841                                    }
842                                }
843                                None => Err(()),
844                            };
845                            session.respond_identity(tid, result, &mut |frame: &[u8]| {
846                                emitter.push(frame)
847                            });
848                            emitter.flush(arbitration.destination(), rt.out).await;
849                        }
850                        Some(Effect::SetPairingPin { tid, pin }) => {
851                            let applied = env.apply_pairing_pin(pin).await;
852                            session.respond_pin_set(
853                                tid,
854                                applied.then_some(()).ok_or(()),
855                                &mut |frame: &[u8]| emitter.push(frame),
856                            );
857                            emitter.flush(arbitration.destination(), rt.out).await;
858                        }
859                        Some(Effect::FactoryReset) => {
860                            // Hand off to the platform, which erases every
861                            // persistent journal and reboots. This never
862                            // returns; no acknowledgement is sent because the
863                            // reset drops the link. Any frames the session
864                            // already staged were flushed above.
865                            env.trace(format_args!("CMD_FACTORY_RESET: wiping all state + reboot"));
866                            env.factory_reset().await
867                        }
868                        other => apply_effect(&session, other, &rt, &mut env).await,
869                    }
870                    // A device identity always exists. `CMD_CLEAR`
871                    // erases the stored one without touching live state,
872                    // and the `CMD_RST` that completes a factory reset
873                    // is where the live copy catches up — leaving the
874                    // device with none, which is the one state the
875                    // invariant forbids. Regenerate here, exactly as
876                    // first boot would, so the only way to reach an
877                    // identityless device is to physically remove it
878                    // from existence.
879                    //
880                    // The running device node keeps the *previous* key
881                    // in its MAC until the next boot (identity is
882                    // fixed at bring-up), and stops originating traffic
883                    // because the dev-domain sync gate compares keys
884                    // rather than counting them.
885                    if session.dev_key().is_none() {
886                        regenerate_device_identity(&mut session, &mut env).await;
887                    }
888                    if session.queued_frame_count() == 0 {
889                        env.clear_attention();
890                    }
891                }
892            }
893            Either4::Second(RxFrame { data, info }) => {
894                // While detached this may stage a delegated MAC
895                // acknowledgement (Effect::StartTransmit).
896                let queued_before = session.queued_frame_count();
897                let effect = session.on_radio_rx(
898                    &data,
899                    info.rssi,
900                    info.snr.as_centibels(),
901                    info.lqi,
902                    Instant::now().as_millis(),
903                    &mut |frame: &[u8]| emitter.push(frame),
904                );
905                if session.queued_frame_count() > queued_before {
906                    env.request_attention();
907                }
908                emitter.flush(arbitration.destination(), rt.out).await;
909                apply_effect(&session, effect, &rt, &mut env).await;
910            }
911            Either4::Third(result) => {
912                let now_ms = Instant::now().as_millis();
913                let outcome = match result {
914                    Ok(()) => TxOutcome::Sent,
915                    Err(umsh_hal::TxError::CadTimeout) => TxOutcome::ChannelBusy,
916                    Err(umsh_hal::TxError::Io(_)) => TxOutcome::Failed,
917                };
918                let effect =
919                    session.on_tx_result(outcome, now_ms, &mut |frame: &[u8]| emitter.push(frame));
920                emitter.flush(arbitration.destination(), rt.out).await;
921                apply_effect(&session, effect, &rt, &mut env).await;
922            }
923            Either4::Fourth(Either::First(sample)) => {
924                // The board decided this measurement is worth
925                // announcing; publish it as an unsolicited PROP_BATTERY.
926                // Dropped silently while no host is attached, and no
927                // effect can result — a publication is not an operation.
928                session.publish_battery(sample, &mut |frame: &[u8]| emitter.push(frame));
929                emitter.flush(arbitration.destination(), rt.out).await;
930            }
931            Either4::Fourth(Either::Second(())) => {
932                // The alert outlived its deadline: stop the indication
933                // and publish the transition the host did not command.
934                let effect = session
935                    .poll_alert(Instant::now().as_millis(), &mut |frame: &[u8]| {
936                        emitter.push(frame)
937                    });
938                emitter.flush(arbitration.destination(), rt.out).await;
939                apply_effect(&session, effect, &rt, &mut env).await;
940            }
941        }
942        // Any of the arms may have moved the device-domain tables
943        // (property mutation, CMD_RST, CMD_RESTORE); one u32
944        // compare when they did not.
945        sync_dev_domain(&session, &mut dev_domain_synced, &mut env);
946    }
947}
