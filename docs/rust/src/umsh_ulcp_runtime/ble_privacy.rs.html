27/// Choose the window duration from the retained bond count when it opens.
28pub const fn pairing_window_ms(bond_count: u8) -> u64 {
29    if bond_count == 0 {
30        5 * 60 * 1000
31    } else {
32        2 * 60 * 1000
33    }
34}

36/// Avoid a tight controller rebuild loop while keeping retries short enough
37/// for background reconnection. A stable run resets consecutive-failure delay.
38#[derive(Default)]
39pub struct RestartBackoff(u64);
40impl RestartBackoff {
41    pub fn after_exit(&mut self, uptime_ms: u64) -> u64 {
42        if uptime_ms >= 30_000 {
43            self.0 = 0;
44        }
45        self.0 = (self.0 * 2).clamp(1_000, 5_000);
46        self.0
47    }
48    pub fn reset(&mut self) {
49        self.0 = 0;
50    }
51}

53pub struct AdvertisementData {
54    /// Both controller interval bounds use the same pairing-policy snapshot
55    /// as the payload, including when a pairing window opens or closes.
56    pub interval_us: u64,
57    /// Absolute boot uptime when this advertisement needs reconfiguration.
58    /// Stack recreation must not extend the startup fast-advertising period.
59    pub refresh_at_ms: Option<u64>,
60    pub advertising: [u8; 31],
61    pub advertising_len: usize,
62    pub scan_response: [u8; 31],
63    pub scan_response_len: usize,
64}

66pub fn utf8_prefix_len(bytes: &[u8], limit: usize) -> usize {
67    let Ok(text) = core::str::from_utf8(bytes) else {
68        return 0;
69    };
70    let mut len = bytes.len().min(limit);
71    while !text.is_char_boundary(len) {
72        len -= 1;
73    }
74    len
75}

77/// Service discovery remains public; the model and name are disclosed only
78/// during pairing.
79pub fn advertisement(
80    service_uuid_le: [u8; 16],
81    name: &[u8],
82    model_id: u16,
83    pairing: bool,
84    uptime_ms: u64,
85) -> AdvertisementData {
86    let startup = uptime_ms < STARTUP_FAST_ADVERTISING_MS;
87    let mut data = AdvertisementData {
88        interval_us: if pairing || startup {
89            FAST_ADVERTISING_INTERVAL_US
90        } else {
91            ADVERTISING_INTERVAL_US
92        },
93        refresh_at_ms: if startup && !pairing {
94            Some(STARTUP_FAST_ADVERTISING_MS)
95        } else {
96            None
97        },
98        advertising: [0; 31],
99        advertising_len: 21,
100        scan_response: [0; 31],
101        scan_response_len: 0,
102    };
103    data.advertising[..3].copy_from_slice(&[2, 1, if pairing { 6 } else { 4 }]);
104    data.advertising[3..5].copy_from_slice(&[17, 7]);
105    data.advertising[5..21].copy_from_slice(&service_uuid_le);
106    if pairing {
107        // Hosts that report only the primary packet title the device by its
108        // model: manufacturer data, company identifier first.
109        data.advertising[21..23].copy_from_slice(&[5, 0xff]);
110        data.advertising[23..25].copy_from_slice(&COMPANY_ID.to_le_bytes());
111        data.advertising[25..27].copy_from_slice(&model_id.to_be_bytes());
112        data.advertising_len = 27;
113        // The name travels only in the scan response. A prefix in the primary
114        // packet would be reported, and cached, in place of the full name.
115        let len = utf8_prefix_len(name, 29);
116        if len != 0 {
117            data.scan_response[..2]
118                .copy_from_slice(&[(len + 1) as u8, if len == name.len() { 9 } else { 8 }]);
119            data.scan_response[2..2 + len].copy_from_slice(&name[..len]);
120            data.scan_response_len = 2 + len;
121        }
122    }
123    data
124}

126pub const fn name_visible(pairing: bool, encrypted: bool, durable_bond: bool) -> bool {
127    pairing || (encrypted && durable_bond)
128}

130/// A deadline survives stack recreation; only an explicit open extends it.
131#[derive(Default, Debug)]
132pub struct PairingDeadline(Option<u64>);
133impl PairingDeadline {
134    pub const fn new() -> Self {
135        Self(None)
136    }
137    pub fn open(&mut self, now_ms: u64, window_ms: u64) {
138        self.0 = Some(now_ms.saturating_add(window_ms));
139    }
140    pub fn close(&mut self) {
141        self.0 = None;
142    }
143    pub fn expired(&self, now_ms: u64) -> bool {
144        self.0.is_some_and(|deadline| now_ms >= deadline)
145    }
146    pub fn at(&self) -> Option<u64> {
147        self.0
148    }
149}

302    #[test]
303    fn deadline_does_not_extend_when_stack_restarts() {
304        let mut deadline = PairingDeadline::new();
305        deadline.open(1000, pairing_window_ms(0));
306        assert_eq!(deadline.at(), Some(301_000));
307        assert!(!deadline.expired(300_999));
308        assert!(deadline.expired(301_000));
309        deadline.close();
310        assert!(!deadline.expired(600_000));
311        for bonds in 1..=4 {
312            deadline.open(600_000, pairing_window_ms(bonds));
313            assert_eq!(deadline.at(), Some(720_000));
314            assert!(!deadline.expired(719_999));
315            assert!(deadline.expired(720_000));
316        }
317        // Clearing all hosts gives the next window the initial setup duration.
318        deadline.open(800_000, pairing_window_ms(0));
319        assert_eq!(deadline.at(), Some(1_100_000));
320    }
321    #[test]
322    fn every_pairing_exit_removes_name_and_retains_service_without_resetting_security() {
323        use crate::ble_security::PairingRuntime;
324        let initial = PairingRuntime {
325            pairing_mode: false,
326            failures: 2,
327            locked_out: false,
328        };
329        for exit in 0..4 {
330            let mut state = initial;
331            let mut deadline = PairingDeadline::new();
332            state.pairing_mode = true;
333            deadline.open(1000, 30000);
334            assert_eq!(
335                advertisement([1; 16], b"radio", MODEL, state.pairing_mode, 30_000).interval_us,
336                20_000
337            );
338            assert_ne!(
339                advertisement([1; 16], b"radio", MODEL, state.pairing_mode, 30_000)
340                    .scan_response_len,
341                0
342            );
343            match exit {
344                0 => {
345                    assert!(deadline.expired(31000));
346                    state.pairing_mode = false;
347                }
348                1 => state.pairing_mode = false, // explicit close
349                2 => state = state.pairing_succeeded(),
350                _ => state = state.bonded_reconnect(),
351            }
352            deadline.close();
353            let data = advertisement([1; 16], b"radio", MODEL, state.pairing_mode, 30_000);
354            assert_eq!(data.interval_us, 546_250);
355            assert_eq!(data.advertising_len, 21);
356            assert_eq!(&data.advertising[..5], &[2, 1, 4, 17, 7]);
357            assert_eq!(&data.advertising[5..21], &[1; 16]);
358            assert_eq!(&data.advertising[21..], &[0; 10]);
359            assert_eq!(data.scan_response_len, 0);
360            assert_eq!(data.scan_response, [0; 31]);
361            assert_eq!(deadline.at(), None);
362            assert_eq!(state.failures, if exit == 2 { 0 } else { 2 });
363            // Reopening advertises the current name and the service again.
364            let reopened = advertisement([1; 16], b"renamed", MODEL, true, 30_000);
365            assert_eq!(reopened.interval_us, 20_000);
366            assert_eq!(&reopened.advertising[3..5], &[17, 7]);
367            assert_eq!(&reopened.advertising[5..21], &[1; 16]);
368            assert_eq!(
369                &reopened.scan_response[..reopened.scan_response_len],
370                b"\x08\x09renamed"
371            );
372        }
373    }
