1//! The bus half of `CMD_I2C_TRANSFER` and `CMD_I2C_SCAN`, over any
2//! `embedded-hal-async` bus.
3//!
4//! [`transfer_with`] and [`scan_with`] are the part that is the same on
5//! every board: turning the staged operation list into HAL operations
6//! that read straight into the result buffer, walking a scan range, and
7//! naming what went wrong in the protocol's terms. [`guarded_transfer`]
8//! and [`guarded_scan`] add what a board with a shared bus needs around
9//! them—a bounded wait for the bus mutex, a deadline on the transaction,
10//! and a [`Reservation`] through which the board's own multi-transaction
11//! procedures keep a host out of a peripheral mid-procedure. A board with
12//! an exclusively owned bus can call the bare functions instead.
13

24/// Perform `request` on `bus`, writing the concatenated read data into
25/// `out` and returning its length.
26///
27/// The operations are handed to the HAL as one transaction, so the wire
28/// semantics—repeated START on a change of direction, no restart between
29/// consecutive operations of the same direction, one STOP—are the HAL's
30/// `transaction` contract. Consecutive reads are coalesced before carving
31/// `out`. Write data is packed into scratch space so consecutive writes
32/// also reach the HAL as one operation, without an extra START/address.
33pub async fn transfer_with<I: I2c>(
34    bus: &mut I,
35    request: TransferRequest<'_>,
36    out: &mut [u8],
37) -> Result<usize, Status> {
38    let mut writes = [0u8; I2C_DATA_MAX];
39    let mut written = 0;
40    for op in request.ops() {
41        if let Op::Write(data) = op.map_err(|error| error.status())? {
42            let end = written + data.len();
43            writes
44                .get_mut(written..end)
45                .ok_or(Status::INVALID_ARGUMENT)?
46                .copy_from_slice(data);
47            written = end;
48        }
49    }
50    let mut write_rest = &writes[..written];
51    let mut operations: heapless::Vec<Operation<'_>, I2C_MAX_OPS> = heapless::Vec::new();
52    let mut rest = out;
53    let mut read_len = 0usize;
54    let mut ops = request.ops().peekable();
55    while let Some(op) = ops.next() {
56        let operation = match op.map_err(|error| error.status())? {
57            Op::Write(data) => {
58                let mut len = data.len();
59                while let Some(Ok(Op::Write(next))) = ops.peek() {
60                    len += next.len();
61                    ops.next();
62                }
63                let (head, tail) = write_rest.split_at(len);
64                write_rest = tail;
65                Operation::Write(head)
66            }
67            Op::Read(mut len) => {
68                while let Some(Ok(Op::Read(next))) = ops.peek() {
69                    len = len.checked_add(*next).ok_or(Status::NOMEM)?;
70                    ops.next();
71                }
72                if rest.len() < len {
73                    return Err(Status::NOMEM);
74                }
75                let (head, tail) = core::mem::take(&mut rest).split_at_mut(len);
76                rest = tail;
77                read_len += len;
78                Operation::Read(head)
79            }
80        };
81        operations
82            .push(operation)
83            .map_err(|_| Status::INVALID_ARGUMENT)?;
84    }
85    bus.transaction(request.addr, &mut operations)
86        .await
87        .map_err(|error| status_for(error.kind()))?;
88    Ok(read_len)
89}

91/// Probe every address in `request`'s range with a one-octet read,
92/// writing the addresses that acknowledged into `out` in ascending order
93/// and returning how many.
94///
95/// An address that does not acknowledge is absent; any other failure
96/// ends the scan, because a bus that has just reported an error is not
97/// one whose silence means anything.
98pub async fn scan_with<I: I2c>(
99    bus: &mut I,
100    request: ScanRequest,
101    out: &mut [u8],
102) -> Result<usize, Status> {
103    let mut found = 0usize;
104    for addr in request.first..=request.last {
105        let mut probe = [0u8; 1];
106        match bus.read(addr, &mut probe).await {
107            Ok(()) => {
108                let slot = out.get_mut(found).ok_or(Status::NOMEM)?;
109                *slot = addr;
110                found += 1;
111            }
112            Err(error) => match error.kind() {
113                ErrorKind::NoAcknowledge(_) => {}
114                other => return Err(status_for(other)),
115            },
116        }
117    }
118    Ok(found)
119}

121/// The status a HAL error kind answers with.
122///
123/// An unacknowledged address is nobody home; an unacknowledged data
124/// octet is somebody home who refused it; everything else is the bus
125/// itself misbehaving, which is what a host sees whether the cause was
126/// arbitration, an overrun, or the board's own deadline.
127pub const fn status_for(kind: ErrorKind) -> Status {
128    match kind {
129        ErrorKind::NoAcknowledge(NoAcknowledgeSource::Address) => Status::NO_DEVICE,
130        ErrorKind::NoAcknowledge(_) => Status::NACK,
131        _ => Status::BUS_ERROR,
132    }
133}

135/// The peripheral one of the board's own multi-transaction procedures
136/// currently holds, if any.
137///
138/// The bus mutex serializes transactions, not procedures: a gauge unseal,
139/// read, and reseal are three transactions, and a host transfer to the
140/// gauge between them would land inside the procedure. A procedure
141/// publishes its peripheral's address here for its duration through
142/// [`hold`](Self::hold), and the guarded functions answer `STATUS_BUSY`
143/// to anything addressed to it, or to a scan whose range covers it. Give each
144/// independently running procedure its own reservation and pass their array
145/// to guarded access. A single reservation must never have overlapping holds.
146pub struct Reservation(AtomicU8);
147
148impl Reservation {
149    const NONE: u8 = 0xFF;
150
151    pub const fn new() -> Self {
152        Self(AtomicU8::new(Self::NONE))
153    }
154
155    /// Publish `addr` as held until the returned guard drops. Every exit
156    /// from the procedure, error paths included, releases it.
157    #[must_use = "the reservation lasts as long as the guard"]
158    pub fn hold(&self, addr: u8) -> Held<'_> {
159        self.0.store(addr, Ordering::Release);
160        Held(self)
161    }
162
163    /// Whether `addr` is held.
164    pub fn holds(&self, addr: u8) -> bool {
165        let held = self.0.load(Ordering::Acquire);
166        held != Self::NONE && held == addr
167    }
168
169    /// Whether a scan of `range` would probe the held address.
170    pub fn covers(&self, range: &ScanRequest) -> bool {
171        let held = self.0.load(Ordering::Acquire);
172        held != Self::NONE && range.covers(held)
173    }
174}
175
176impl Default for Reservation {
177    fn default() -> Self {
178        Self::new()
179    }
180}
181
182/// A [`Reservation`] in force; dropping it releases the peripheral.
183pub struct Held<'a>(&'a Reservation);
184
185impl Drop for Held<'_> {
186    fn drop(&mut self) {
187        self.0.0.store(Reservation::NONE, Ordering::Release);
188    }
189}

191/// Independent board procedures may reserve different devices concurrently.
192pub trait Reservations {
193    fn holds(&self, addr: u8) -> bool;
194    fn covers(&self, range: &ScanRequest) -> bool;
195}
196
197impl Reservations for Reservation {
198    fn holds(&self, addr: u8) -> bool {
199        Reservation::holds(self, addr)
200    }
201    fn covers(&self, range: &ScanRequest) -> bool {
202        Reservation::covers(self, range)
203    }
204}
205
206impl<const N: usize> Reservations for [&Reservation; N] {
207    fn holds(&self, addr: u8) -> bool {
208        self.iter().any(|r| r.holds(addr))
209    }
210    fn covers(&self, range: &ScanRequest) -> bool {
211        self.iter().any(|r| r.covers(range))
212    }
213}

215/// The floor under every transaction deadline: clock stretching and the
216/// controller's own bookkeeping cost more than the octets do on a short
217/// transfer.
218const DEADLINE_FLOOR: Duration = Duration::from_millis(100);
219
220/// How long a transaction moving `octets` on a bus clocked at `speed_khz`
221/// may take before it is declared stuck.
222///
223/// Ten times the nominal wire time on top of the floor: generous enough
224/// that a peripheral stretching the clock is not mistaken for a stuck
225/// bus, and still well under the host's response timeout.
226pub const fn deadline_for(octets: usize, speed_khz: u16) -> Duration {
227    // Nine clocks per octet (eight bits and an acknowledge), in
228    // microseconds at `speed_khz` kilohertz.
229    let speed_khz = if speed_khz == 0 { 1 } else { speed_khz as u64 };
230    let nominal_us = (octets as u64 * 9 * 1000) / speed_khz;
231    Duration::from_micros(DEADLINE_FLOOR.as_micros() + 10 * nominal_us)
232}
233
234/// The deadline for a transfer of `shape`: its data plus one address
235/// octet per operation.
236pub const fn transfer_deadline(shape: TransferShape, speed_khz: u16) -> Duration {
237    deadline_for(shape.data_len + shape.ops + 1, speed_khz)
238}
239
240/// The deadline for a whole scan: an address and a probe octet per
241/// address in the range.
242pub const fn scan_deadline(request: &ScanRequest, speed_khz: u16) -> Duration {
243    deadline_for(request.len() * 2, speed_khz)
244}

246/// Acquire `bus` within `lock_timeout`, refuse the transfer if
247/// `reserved` holds its address, and perform it within `deadline`.
248///
249/// The reservation is checked *after* the bus is acquired. Checked
250/// before, a transfer could pass while the peripheral is free, wait for
251/// the mutex while a procedure begins, and then run inside it.
252/// A wait for the bus that outlasts `lock_timeout` is `STATUS_BUSY`; a
253/// transaction that outlasts `deadline` is `STATUS_BUS_ERROR`, and
254/// dropping the HAL's future is how the transaction is abandoned—a HAL
255/// whose drop does not reset the controller is not one to use here.
256pub async fn guarded_transfer<M: RawMutex, I: I2c>(
257    bus: &Mutex<M, I>,
258    reserved: &impl Reservations,
259    lock_timeout: Duration,
260    deadline: Duration,
261    request: TransferRequest<'_>,
262    out: &mut [u8],
263) -> Result<usize, Status> {
264    let mut bus = with_timeout(lock_timeout, bus.lock())
265        .await
266        .map_err(|_| Status::BUSY)?;
267    if reserved.holds(request.addr) {
268        return Err(Status::BUSY);
269    }
270    with_timeout(deadline, transfer_with(&mut *bus, request, out))
271        .await
272        .map_err(|_| Status::BUS_ERROR)?
273}
274
275/// The scan counterpart of [`guarded_transfer`]: a range that covers a
276/// held address is refused whole rather than silently skipping it, so
277/// the host learns to try again instead of drawing a wrong inventory.
278pub async fn guarded_scan<M: RawMutex, I: I2c>(
279    bus: &Mutex<M, I>,
280    reserved: &impl Reservations,
281    lock_timeout: Duration,
282    deadline: Duration,
283    request: ScanRequest,
284    out: &mut [u8],
285) -> Result<usize, Status> {
286    let mut bus = with_timeout(lock_timeout, bus.lock())
287        .await
288        .map_err(|_| Status::BUSY)?;
289    if reserved.covers(&request) {
290        return Err(Status::BUSY);
291    }
292    with_timeout(deadline, scan_with(&mut *bus, request, out))
293        .await
294        .map_err(|_| Status::BUS_ERROR)?
295}

652    /// The race the reservation exists for, in the order that defeats a
653    /// check-before-lock implementation: the transfer is already parked
654    /// on the bus mutex when the procedure publishes its reservation.
655    #[test]
656    fn a_transfer_parked_on_the_bus_sees_a_reservation_published_meanwhile() {
657        let bus: Mutex<NoopRawMutex, Bus> = Mutex::new(Bus::default());
658        let reserved = Reservation::new();
659        let codec = Reservation::new();
660        let _codec = codec.hold(0x18);
661        let reservations = [&reserved, &codec];
662        let payload = request(0, 0x55, &[Op::Write(&[0x00]), Op::Read(2)]);
663        let transfer = TransferRequest::parse(&payload).unwrap();
664        let mut out = [0u8; 4];
665
666        // (1) Something else holds the bus, while only the unrelated codec is reserved.
667        let holder = block_on(bus.lock());
668        let held = {
669            // (2) The transfer starts and parks on the mutex.
670            let mut fut = pin!(guarded_transfer(
671                &bus,
672                &reservations,
673                NEVER,
674                NEVER,
675                transfer,
676                &mut out
677            ));
678            assert!(poll_once(fut.as_mut()).is_pending(), "waiting for the bus");
679            // (3) The procedure begins, then the bus frees up.
680            let held = reserved.hold(0x55);
681            drop(holder);
682            // (4) The transfer is refused and never touched the bus.
683            assert_eq!(block_on(fut.as_mut()), Err(Status::BUSY));
684            held
685        };
686        assert_eq!(block_on(bus.lock()).transactions, 0);
687        // (5) The procedure ends and the same transfer goes through.
688        drop(held);
689        block_on(bus.lock()).expected.push_back(Expected {
690            addr: 0x55,
691            ops: vec![(false, vec![0x00]), (true, vec![0xAA, 0xBB])],
692            result: Ok(()),
693        });
694        let len = block_on(guarded_transfer(
695            &bus,
696            &reservations,
697            NEVER,
698            NEVER,
699            transfer,
700            &mut out,
701        ))
702        .unwrap();
703        assert_eq!(&out[..len], &[0xAA, 0xBB]);
704    }
