1//! The device node: a full `umsh-mac`/`umsh-node` stack running on the
2//! device itself, alongside the ULCP session.
3//!
4//! The device identity "exists even when no phone is attached" (ULCP spec
5//! §Identities); this module is what makes that true. It is an ordinary
6//! MAC + `Host` pump with the device's constraints baked into
7//! [`DeviceNodePlatform`]:
8//!
9//! - **Radio** is a `LoraphyRadio` over the node's virtual mux bundle
10//!   ([`NODE_CH`], mux client B) behind the shared duty-ledger gate: the
11//!   session and the node share one physical radio through `radio_mux`
12//!   and draw from one combined `PROP_PHY_DUTY_LIMIT` budget. A refused
13//!   transmit is shed via the MAC's CAD-backoff path rather than killing
14//!   the pump.
15//! - **Rng** is a ChaCha20 CSPRNG seeded from the board's hardware TRNG
16//!   at boot ([`NodeRng`]): project policy forbids non-crypto RNGs, and
17//!   under BLE builds the RNG peripheral is not ours to read at runtime.
18//! - **The counter store** is the board's — the `CS` parameter — so TX
19//!   reservation boundaries for the device identity and per-peer RX
20//!   replay boundaries survive power cycles, flushed from inside the MAC
21//!   pump one whole-map record per persist block.
22//!
23//! The node **always exists**: a device identity is generated and
24//! persisted at first boot, so bring-up is unconditional and the only
25//! question is whether the node is *transmitting*. That is configuration
26//! — the PHY enable state and the forwarding switch — plus the
27//! `NODE_ACTIVE` gate, which closes while a factory reset is in flight
28//! (the identity has been erased from storage but the running MAC still
29//! holds it until the reboot that completes the wipe).
30//!
31//! Beacon requests arrive through [`BEACON_TRIGGER`] rather than from any
32//! specific button handler: the trigger is an input, because wake- and
33//! timer-driven advertisement policy (reserved device-domain properties
34//! 80–87) will feed the same path later.
35//!
36//! # Board seam
37//!
38//! Embassy task functions cannot be generic, so the spawnable tasks stay
39//! in each firmware as thin shims around the `*_loop` functions here, and
40//! the board owns the two statics whose types depend on `CS`: the MAC
41//! cell and its counter store. Everything else — every static whose type
42//! is fixed, and every line of logic — is here once.
43

68/// The mutex kind guarding the node's statics.
69///
70/// The nRF images run a single thread-mode executor, where
71/// `ThreadModeRawMutex` is a no-op lock — worth keeping, because these
72/// statics sit in the radio RX/TX path on boards whose BLE controller
73/// (MPSL/SDC) has hard real-time deadlines that a critical section would
74/// intrude on. Boards without that constraint take the portable default.
75///
76/// `embassy_sync` only defines `ThreadModeRawMutex` for bare-metal targets,
77/// so the host build takes the portable default regardless of the feature.
78/// Nothing on the host runs the device node — it builds there for tests and
79/// rustdoc — and the portable lock is the correct choice under a hosted OS
80/// anyway.
81#[cfg(all(feature = "node-thread-mode-mutex", target_os = "none"))]
82pub type NodeMutex = embassy_sync::blocking_mutex::raw::ThreadModeRawMutex;
83#[cfg(not(all(feature = "node-thread-mode-mutex", target_os = "none")))]
84pub type NodeMutex = CriticalSectionRawMutex;

143/// Device-node MAC sized to the session's device-domain tables, which are
144/// the only provisioning source it has: 1 identity (the device identity;
145/// no PFS ephemerals on the device node), `MAX_DEV_PEERS` peers,
146/// `MAX_CHANNEL_KEYS` channels (a smaller MAC table would refuse channels
147/// the property surface accepted), 4 pending ACKs, 4 TX slots (beacons
148/// and future acks — no application traffic), 255-byte frames, 32-entry
149/// dup cache. The per-channel replay maps are the RAM hot spot (~330
150/// bytes per tracked sender): 4 full-key + 2 hint-only senders per
151/// channel keeps the whole table ~2 KiB/channel; extra concurrent senders
152/// on one channel fail closed (dropped, never accepted unchecked).
153pub type DeviceNodeMac<CS> =
154    umsh_mac::Mac<DeviceNodePlatform<CS>, 1, MAX_DEV_PEERS, MAX_CHANNEL_KEYS, 4, 4, 255, 32, 4, 2>;
155pub type DeviceNodeHandle<CS> = MacHandle<
156    'static,
157    DeviceNodePlatform<CS>,
158    1,
159    MAX_DEV_PEERS,
160    MAX_CHANNEL_KEYS,
161    4,
162    4,
163    255,
164    32,
165    4,
166    2,
167>;
168pub type DeviceNodeHost<CS> = Host<DeviceNodeHandle<CS>>;
169pub type DeviceNode<CS> = LocalNode<DeviceNodeHandle<CS>>;

204/// The live device name, pushed by the board.
205///
206/// Pushed rather than pulled: the name lives behind a board-owned async
207/// mutex, and the profile paths that need it are called from places that
208/// cannot await one. The board writes it at boot and on every change,
209/// which is the same edge it already had to handle.
210static DEVICE_NAME: BlockingMutex<
211    CriticalSectionRawMutex,
212    RefCell<heapless::Vec<u8, MAX_DEVICE_NAME_LEN>>,
213> = BlockingMutex::new(RefCell::new(heapless::Vec::new()));

215/// Signalled internally by [`set_device_name`] so the Identity Request
216/// responder's profile name stays current.
217static NAME_CHANGED: Signal<NodeMutex, ()> = Signal::new();
218
219/// Publish the live device name to the node. Call at boot and whenever
220/// the session's device name changes.
221pub fn set_device_name(name: &[u8]) {
222    DEVICE_NAME.lock(|cell| {
223        let mut current = cell.borrow_mut();
224        current.clear();
225        let _ = current.extend_from_slice(&name[..name.len().min(MAX_DEVICE_NAME_LEN)]);
226    });
227    NAME_CHANGED.signal(());
228}

230/// Snapshot the live device name as a spec-capped (≤24-byte) Node Name
231/// identity option value, or `None` when unset/empty.
232fn profile_name() -> Option<alloc::string::String> {
233    DEVICE_NAME.lock(|cell| {
234        let current = cell.borrow();
235        core::str::from_utf8(&current)
236            .ok()
237            .map(|name| truncate_utf8(name, 24))
238            .filter(|name| !name.is_empty())
239            .map(alloc::string::String::from)
240    })
241}

243/// Trim to at most `max` bytes without splitting a UTF-8 sequence.
244fn truncate_utf8(text: &str, max: usize) -> &str {
245    if text.len() <= max {
246        return text;
247    }
248    let mut end = max;
249    while !text.is_char_boundary(end) {
250        end -= 1;
251    }
252    &text[..end]
253}

255/// The advertised (role, capabilities) pair for the device identity.
256///
257/// Role and forwarding are separate dimensions, and this is where they
258/// meet without being conflated. Capabilities are **facts**: `REP` is set
259/// exactly when the node is actually forwarding, `MOB` exactly when the
260/// operator has said the device moves. The role is what the device
261/// *presents itself as*, which is configuration — an explicit
262/// `PROP_IDENT_ROLE` is advertised verbatim, so a mobile repeater and a
263/// fixed tracker are both expressible.
264///
265/// With no role configured the device derives one, which is what makes
266/// the default sensible rather than a lie: a forwarding node is a
267/// repeater, and anything else is a tracker.
268fn advertised_identity(
269    is_repeater: bool,
270    is_mobile: bool,
271    configured_role: Option<u8>,
272) -> (NodeRole, NodeCapabilities) {
273    let role = match configured_role {
274        Some(byte) => NodeRole::from_byte(byte),
275        None if is_repeater => NodeRole::Repeater,
276        None => NodeRole::Tracker,
277    };
278    let mut capabilities = NodeCapabilities::empty();
279    capabilities.set(NodeCapabilities::REPEATER, is_repeater);
280    capabilities.set(NodeCapabilities::MOBILE, is_mobile);
281    (role, capabilities)
282}

284/// The regions the node should advertise as supported, given a snapshot.
285///
286/// A device that is not forwarding makes no claim at all and omits the
287/// option (node-identity.md §Supported Regions); a forwarding device with
288/// an empty list forwards regardless of region, which is likewise not a
289/// claim about any particular region. Only a forwarding device with a
290/// configured list has something to say.
291fn advertised_regions(snapshot: &DevDomainSnapshot) -> Option<alloc::vec::Vec<u8>> {
292    (snapshot.repeater_enabled && !snapshot.repeater_regions.is_empty())
293        .then(|| snapshot.repeater_regions.to_vec())
294}

298/// Reconciles the node's MAC against each [`DevDomainSnapshot`]: joins
299/// newly provisioned channels, removes de-provisioned ones (dropping
300/// their replay state), and registers peers. Peer *removal* is not
301/// propagated — MAC registry entries carry no key material, so a stale
302/// entry is inert, and the registry is rebuilt from the live table at the
303/// next boot.
304pub async fn dev_sync_loop<CS: CounterStore + 'static>(
305    node: DeviceNode<CS>,
306    mac: DeviceNodeHandle<CS>,
307    node_key: [u8; 32],
308) {
309    // Channel keys currently applied to the MAC. Starts empty: the MAC is
310    // built bare at bring-up and every channel arrives through here.
311    let mut applied: heapless::Vec<[u8; 32], MAX_CHANNEL_KEYS> = heapless::Vec::new();
312    loop {
313        let snapshot = DEV_SYNC.wait().await;
314        // The gate is key equality, not mere presence. The session's live
315        // identity and the one this MAC was built around can legitimately
316        // differ — an installed `PROP_DEV_PRIVATE_KEY` takes effect at the
317        // next boot, and `CMD_CLEAR` + `CMD_RST` erases the stored key —
318        // and in every such case originating traffic would be signing as
319        // an identity the device no longer claims.
320        let matches_live = snapshot.dev_key == Some(node_key);
321        NODE_ACTIVE.store(matches_live, Ordering::Relaxed);
322        // Reconcile the forwarding switch, the forwarding policy, and the
323        // advertised role/capabilities. All are idempotent; the profile is
324        // refreshed unconditionally because the role and mobility
325        // properties can move without the forwarding flag.
326        mac.set_repeater_enabled(snapshot.repeater_enabled).await;
327        let regions: heapless::Vec<[u8; 2], 8> = snapshot
328            .repeater_regions
329            .chunks_exact(2)
330            .map(|code| [code[0], code[1]])
331            .collect();
332        let stored = mac
333            .set_repeater_policy(
334                &regions,
335                snapshot.repeater_default_region,
336                snapshot.repeater_min_rssi,
337                snapshot.repeater_min_snr,
338            )
339            .await;
340        if stored != regions.len() {
341            debug_log(format_args!(
342                "node dev-sync: repeater regions TRUNCATED {} of {}",
343                stored,
344                regions.len()
345            ));
346        }
347        let (role, capabilities) = advertised_identity(
348            snapshot.repeater_enabled,
349            snapshot.ident_mobile,
350            snapshot.ident_role,
351        );
352        let supported_regions = advertised_regions(&snapshot);
353        // `PROP_DEV_DISCOVERABLE` gates the Identity Request responder.
354        // Enable rebuilds the whole profile rather than patching it:
355        // disabling drops the profile with the responder, so a later
356        // opt-back-in must not depend on a name change to become
357        // correct.
358        if snapshot.discoverable {
359            let mut profile = NodeIdentityProfile::new(PublicKey(node_key), role, capabilities);
360            profile.name = profile_name();
361            profile.supported_regions = supported_regions;
362            node.enable_identity_responder_default(profile);
363        } else {
364            node.disable_identity_responder();
365        }
366        if NODE_IS_REPEATER.swap(snapshot.repeater_enabled, Ordering::Relaxed)
367            != snapshot.repeater_enabled
368        {
369            debug_log(format_args!(
370                "node dev-sync: repeater {}",
371                if snapshot.repeater_enabled {
372                    "ON"
373                } else {
374                    "off"
375                }
376            ));
377        }
378        let mut index = 0;
379        while index < applied.len() {
380            if snapshot.channel_keys.contains(&applied[index]) {
381                index += 1;
382                continue;
383            }
384            let key = applied.swap_remove(index);
385            let _ = node.leave(&umsh_node::Channel::private(ChannelKey(key), ""));
386            mac.remove_channel(&ChannelKey(key)).await;
387            debug_log(format_args!(
388                "node dev-sync: channel {:02x}{:02x}.. removed",
389                key[0], key[1]
390            ));
391        }
392        for key in snapshot.channel_keys.iter() {
393            if applied.contains(key) {
394                continue;
395            }
396            match node
397                .join(&umsh_node::Channel::private(ChannelKey(*key), ""))
398                .await
399            {
400                Ok(_) => {
401                    let _ = applied.push(*key);
402                    debug_log(format_args!(
403                        "node dev-sync: channel {:02x}{:02x}.. joined",
404                        key[0], key[1]
405                    ));
406                }
407                Err(_) => debug_log(format_args!(
408                    "node dev-sync: channel {:02x}{:02x}.. join FAILED",
409                    key[0], key[1]
410                )),
411            }
412        }
413        // Registration is add-or-refresh; repeats are harmless.
414        for public_key in snapshot.peers.iter() {
415            if node.peer(PublicKey(*public_key)).await.is_err() {
416                debug_log(format_args!(
417                    "node dev-sync: peer {:02x}{:02x}.. register FAILED",
418                    public_key[0], public_key[1]
419                ));
420            }
421        }
422        // Seed persisted RX replay boundaries for the registered peers (a
423        // repeat only refreshes each peer's initial boundary; live replay
424        // windows are untouched). Without this, a peer's replay floor
425        // would restart at zero after every power cycle.
426        if !snapshot.peers.is_empty() {
427            let _ = mac.load_all_persisted_rx_counters().await;
428        }
429        debug_log(format_args!(
430            "node dev-sync: {} channels, {} peers, identity-matches-live={}",
431            snapshot.channel_keys.len(),
432            snapshot.peers.len(),
433            matches_live
434        ));
435    }
436}

440/// Why a beacon was requested. Carried through [`BEACON_TRIGGER`] so the
441/// send path never assumes a button: beacon-at-wake and periodic beacons
442/// (device-domain advertisement policy) are planned triggers.
443#[derive(Clone, Copy)]
444pub enum BeaconTrigger {
445    /// The board's primary-action button slot. Boards without one carry
446    /// the variant unused.
447    Button,
448    /// Emit a solicited advertisement — a broadcast carrying the signed
449    /// node identity payload — echoing `nonce` when set. Currently
450    /// unconstructed: the Identity Request responder that will drive it
451    /// (with a targeted unicast reply) is a follow-up; the generator is
452    /// kept for that pass.
453    Advertise { nonce: Option<u32> },
454}
455
456/// Beacon requests into the node. On a boot that skipped node bring-up
457/// the queue is never drained and requests are dropped at the `try_send`
458/// in [`request_beacon`], leaving the slot inert rather than blocking the
459/// caller.
460pub static BEACON_TRIGGER: Channel<NodeMutex, BeaconTrigger, 2> = Channel::new();
461
462/// Fire-and-forget beacon request. A full queue means a beacon (or
463/// advertisement) is already pending, so dropping the extra request loses
464/// nothing — bursts of Advertisement Requests coalesce here.
465pub fn request_beacon(trigger: BeaconTrigger) {
466    let _ = BEACON_TRIGGER.try_send(trigger);
467}

493/// Drives the device node's MAC pump. Never returns while healthy; an
494/// exit means the MAC hit an unrecoverable radio error, and rebooting
495/// through the panic handler beats silently losing the device identity.
496pub async fn pump_loop<CS: CounterStore + 'static>(mut host: DeviceNodeHost<CS>) -> ! {
497    debug_log(format_args!("node pump: running"));
498    let result = host.run().await;
499    debug_log(format_args!("node pump: EXITED ok={}", result.is_ok()));
500    panic!("device node host exited");
501}

503/// Turns beacon triggers into node sends on the device identity: a plain
504/// beacon for the button slot, a signed solicited advertisement for an
505/// Advertisement Request.
506pub async fn beacon_loop<CS: CounterStore + 'static>(
507    node: DeviceNode<CS>,
508    identity: SoftwareIdentity,
509    hooks: NodeHooks,
510) {
511    use umsh_node::Transport as _;
512    loop {
513        let trigger = BEACON_TRIGGER.receive().await;
514        // A factory-cleared identity leaves the slot inert, exactly like
515        // an unprovisioned one.
516        if !NODE_ACTIVE.load(Ordering::Relaxed) {
517            continue;
518        }
519        match trigger {
520            BeaconTrigger::Button => {
521                // A beacon's whole job is "I am here, and here is a path
522                // back to me". The trace route is what carries the second
523                // half: repeaters prepend their hints as they forward, so a
524                // listener that already knows this node learns a usable
525                // source route from a packet that costs no payload at all.
526                let options = SendOptions::default().with_trace_route();
527                if node.send_all(&[], &options).await.is_ok() {
528                    (hooks.beacon_confirm)();
529                }
530            }
531            BeaconTrigger::Advertise { nonce } => {
532                let accepted = send_advertisement(&node, &identity, nonce).await;
533                debug_log(format_args!(
534                    "node advert: nonce={nonce:?} accepted={accepted}"
535                ));
536            }
537        }
538    }
539}

541/// Build, sign, and broadcast a solicited advertisement: the node
542/// identity payload (role, live device name, echoed nonce) with the
543/// standalone EdDSA signature the spec prefers for broadcasts, typed as a
544/// NodeIdentity payload.
545async fn send_advertisement<CS: CounterStore + 'static>(
546    node: &DeviceNode<CS>,
547    identity: &SoftwareIdentity,
548    nonce: Option<u32>,
549) -> bool {
550    use umsh_crypto::NodeIdentity as _;
551    use umsh_node::Transport as _;
552    // The node's own profile is the canonical statement of what this node
553    // is — kept current by `dev_sync_loop` and `identity_profile_loop` —
554    // so build the payload from it rather than assembling a second,
555    // drifting copy here.
556    let Some(payload) = node.with_identity_profile(|profile| profile.to_payload(nonce)) else {
557        return false;
558    };
559    // Payload-type byte + role/caps + name (≤26) + regions (≤20) + nonce
560    // (6) + 0xFF + 64-byte signature — 192 covers it with headroom.
561    let mut buf = [0u8; 192];
562    buf[0] = umsh_core::PayloadType::NodeIdentity as u8;
563    let Ok(body_len) = payload.encode_for_signing(&mut buf[1..]) else {
564        return false;
565    };
566    let mut len = 1 + body_len;
567    // The signature covers ROLE through the 0xFF terminator — the
568    // payload-type byte stays outside the signed range.
569    let Ok(signature) = identity.sign(&buf[1..len]).await else {
570        return false;
571    };
572    if buf.len() < len + 64 {
573        return false;
574    }
575    buf[len..len + 64].copy_from_slice(&signature);
576    len += 64;
577    // Full source, not a hint: the bundle's detached signature is only
578    // checkable against the sender's public key, and a broadcast carries no
579    // MIC to authenticate it otherwise. A hint-only advertisement is
580    // unverifiable by anyone who does not already hold the key, which is
581    // exactly the audience an advertisement is for. Trace route for the same
582    // reason a beacon carries one.
583    let options = SendOptions::default().with_full_source().with_trace_route();
584    node.send_all(&buf[..len], &options).await.is_ok()
585}
586
587/// Keeps the Identity Request responder's profile name synced to the live
588/// device name. The responder builds replies synchronously and cannot
589/// await, so the current name is pushed in here on each change rather
590/// than read at reply time.
591pub async fn identity_profile_loop<CS: CounterStore + 'static>(node: DeviceNode<CS>) {
592    loop {
593        NAME_CHANGED.wait().await;
594        let name = profile_name();
595        node.update_identity_profile(move |profile| profile.name = name);
596    }
597}
598
599/// A `PROP_IDENT` read in flight.
600///
601/// A request/response signal pair rather than a shared handle: the node
602/// is single-executor `Rc`/`RefCell` state and cannot live in a `static`
603/// at all, and this is the same shape the pairing-PIN round trip already
604/// uses.
605static IDENT_REQUEST: Signal<NodeMutex, ()> = Signal::new();
606static IDENT_RESPONSE: Signal<NodeMutex, Option<IdentityBlob>> = Signal::new();
607
608/// A complete signed node-identity blob: the canonical unsigned encoding
609/// followed by its 64-octet detached signature.
610type IdentityBlob = heapless::Vec<u8, 320>;
611
612/// Build and sign this node's identity blob into `out`, returning its
613/// length.
614///
615/// This is the standalone framing of the same statement the Identity
616/// Request responder makes — same profile, same builder — differing only
617/// in that it carries no request nonce and is authenticated by the
618/// signature rather than by an enclosing unicast.
619pub async fn sign_identity_blob(out: &mut [u8]) -> Option<usize> {
620    if !NODE_UP.load(Ordering::Relaxed) {
621        return None;
622    }
623    IDENT_RESPONSE.reset();
624    IDENT_REQUEST.signal(());
625    let blob = IDENT_RESPONSE.wait().await?;
626    if blob.len() > out.len() {
627        return None;
628    }
629    out[..blob.len()].copy_from_slice(&blob);
630    Some(blob.len())
631}
632
633/// Answers [`IDENT_REQUEST`] with the node's current signed identity.
634pub async fn identity_blob_loop<CS: CounterStore + 'static>(
635    node: DeviceNode<CS>,
636    identity: SoftwareIdentity,
637) {
638    use umsh_crypto::NodeIdentity as _;
639    loop {
640        IDENT_REQUEST.wait().await;
641        let mut blob = IdentityBlob::new();
642        let _ = blob.resize_default(blob.capacity());
643        // The node's own profile is the canonical statement of what this
644        // node is; building from it is what keeps the local-control
645        // framing and the over-the-air framing from drifting apart.
646        let signed = async {
647            let payload = node.with_identity_profile(|profile| profile.to_payload(None))?;
648            let body = payload.encode_for_signing(&mut blob).ok()?;
649            let signature = identity.sign(blob.get(..body)?).await.ok()?;
650            blob.get_mut(body..body + 64)?.copy_from_slice(&signature);
651            Some(body + 64)
652        }
653        .await;
654        IDENT_RESPONSE.signal(signed.map(|len| {
655            blob.truncate(len);
656            blob
657        }));
658    }
659}

661// ─── Bring-up ────────────────────────────────────────────────────────────────
662
663/// Everything bring-up produced, for the board to spawn its task shims
664/// around. Embassy tasks cannot be generic, so the spawning itself stays
665/// board-side.
666pub struct DeviceNodeParts<CS: CounterStore + 'static> {
667    pub host: DeviceNodeHost<CS>,
668    pub node: DeviceNode<CS>,
669    pub mac: DeviceNodeHandle<CS>,
670    /// The public key the MAC was actually built around, for the
671    /// device-domain sync gate.
672    pub node_key: [u8; 32],
673}
674
675/// Construct the MAC around the device identity and wire up the node.
676/// Call at most once. The identity is never absent — boot generates and
677/// persists one when the journal is empty — so there is no
678/// "unprovisioned" path here.
679///
680/// `t_frame_ms` is the worst-case airtime hint for the MAC scheduler.
681/// The caller spawns the loops and must do so promptly: `NODE_UP` is set
682/// here, so a `PROP_IDENT` read arriving between this returning and the
683/// spawns would wait on the signal.
684pub async fn bring_up<CS: CounterStore + 'static>(
685    mac_cell: &'static DeviceNodeMacCell<CS>,
686    identity_secret: &[u8; 32],
687    node_seed: [u8; 32],
688    t_frame_ms: u32,
689    counters: CS,
690    duty: &'static umsh_ulcp_device::DutyLedger,
691    hooks: NodeHooks,
692) -> DeviceNodeParts<CS> {
693    // The Mac is ~37 KiB. `init_with` lets the compiler construct it in
694    // place inside the static cell; building it as a stack local (what
695    // `StaticCell::init` does) transits the stack once per move in the
696    // chain — hardware-diagnosed on the nRF images as boot HardFaults
697    // (INVSTATE jumps to 0) and a smashed allocator when the temporaries
698    // blew through the stack budget. Keep the construction a single
699    // in-place expression.
700    let mac_cell: &'static AsyncRefCell<DeviceNodeMac<CS>> = mac_cell.init_with(|| {
701        AsyncRefCell::new(DeviceNodeMac::new(
702            DutyGatedRadio::with_load_hook(
703                umsh_radio_loraphy::LoraphyRadio::new(&NODE_CH, t_frame_ms),
704                duty,
705                EmbassyClock,
706                hooks.note_external_load,
707            ),
708            CryptoEngine::new(SoftwareAes, SoftwareSha256),
709            EmbassyClock,
710            NodeRng::from_seed(node_seed),
711            counters,
712            RepeaterConfig::default(),
713            OperatingPolicy::default(),
714        ))
715    });
716    debug_log(format_args!("node bring-up: mac cell ready"));
717    let identity = SoftwareIdentity::from_secret_bytes(identity_secret);
718    // Retained for the device-domain sync gate, which compares the
719    // session's live PROP_DEV_KEY against the key this MAC actually holds.
720    let node_key = umsh_crypto::NodeIdentity::public_key(&identity).0;
721    let identity_id = mac_cell
722        .try_borrow_mut()
723        .expect("mac cell is unshared during bring-up")
724        .add_identity(identity)
725        .unwrap_or_else(|_| panic!("device node identity"));
726    // Seed the identity's TX frame counter from the persisted boundary so
727    // secured sends can never reuse counter space from a previous boot.
728    // With nothing persisted the random initial counter stands.
729    match MacHandle::new(mac_cell)
730        .load_persisted_counter(identity_id)
731        .await
732    {
733        Ok(counter) => debug_log(format_args!("node bring-up: tx counter {counter}")),
734        Err(_) => debug_log(format_args!("node bring-up: tx counter load FAILED")),
735    }
736
737    let mut host: DeviceNodeHost<CS> = Host::new(MacHandle::new(mac_cell));
738    let node = host.add_node(identity_id);
739    // Permanent observability tap: every packet the node processes is one
740    // debug line. This is the device-domain acceptance instrument
741    // (multicast on a provisioned device channel shows up here) and it
742    // never consumes the packet. The subscription is leaked because the
743    // node lives for the rest of the boot.
744    core::mem::forget(node.on_receive(|packet| {
745        let channel = packet
746            .channel()
747            .map(|info| u16::from_be_bytes(info.id().0))
748            .unwrap_or(0);
749        debug_log(format_args!(
750            "node rx: {:?} ch={:04x} len={} auth={}",
751            packet.packet_family(),
752            channel,
753            packet.payload().len(),
754            packet.source_authenticated(),
755        ));
756        false
757    }));
758    // Identity Request observability tap. The actual reply is produced by
759    // the built-in responder enabled below (a targeted unicast identity,
760    // echoing any NONCE); this handler only logs, and never consumes.
761    core::mem::forget(node.on_mac_command(|from, command| {
762        if let umsh_node::OwnedMacCommand::IdentityRequest { options } = command {
763            let nonce = umsh_node::mac_command::IdentityRequestFilters::new(options)
764                .nonce()
765                .ok()
766                .flatten();
767            debug_log(format_args!(
768                "node identity-request: from {:02x}{:02x}.. nonce={:?}",
769                from.0[0], from.0[1], nonce
770            ));
771        }
772    }));
773    // Enable the built-in Identity Request responder. Role and
774    // capabilities start derived and are corrected by the first
775    // device-domain sync; the name tracks the live device name via
776    // `identity_profile_loop`. The default policy answers every request
777    // whose filters select us, including our full source key when the
778    // request wasn't authenticated to us. Replies are authenticated
779    // unicast — never signed, never a broadcast fallback: a request whose
780    // source can't be resolved to a key is dropped by the MAC before the
781    // responder runs.
782    {
783        use umsh_crypto::NodeIdentity as _;
784        let public_key = *SoftwareIdentity::from_secret_bytes(identity_secret).public_key();
785        let mut profile =
786            NodeIdentityProfile::new(public_key, NodeRole::Tracker, NodeCapabilities::empty());
787        profile.name = profile_name();
788        node.enable_identity_responder_default(profile);
789    }
790    // Let the node answer a brand-new requester that supplied its full
791    // 32-byte source key: the MAC auto-registers it transiently
792    // (LRU-evictable, never pinned) so the pairwise reply can be sealed.
793    // Repeaters specifically must be able to respond this way.
794    MacHandle::new(mac_cell)
795        .set_auto_register_full_key_peers(true)
796        .await;
797    debug_log(format_args!("node bring-up: host ready"));
798    NODE_UP.store(true, Ordering::Relaxed);
799    DeviceNodeParts {
800        host,
801        node: node.clone(),
802        mac: MacHandle::new(mac_cell),
803        node_key,
804    }
805}
