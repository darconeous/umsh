20/// Signal-to-noise ratio represented in centibels (0.1 dB units).
21///
22/// This uses a slightly finer unit than whole decibels while still staying
23/// compact and integer-friendly. Some common LoRa radios report SNR in
24/// quarter-dB steps. Converting those readings into centibels requires
25/// rounding, introducing at most 0.5 cB (0.05 dB) of error.
26#[derive(Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
27pub struct Snr(i16);
28
29impl Snr {
30    /// Construct an SNR value directly from centibels.
31    pub const fn from_centibels(centibels: i16) -> Self {
32        Self(centibels)
33    }
34
35    /// Construct an SNR value from whole decibels.
36    pub const fn from_decibels(db: i8) -> Self {
37        Self((db as i16) * 10)
38    }
39
40    /// Construct an SNR value from quarter-dB steps, rounding to the nearest
41    /// centibel.
42    pub const fn from_quarter_db_steps(steps: i16) -> Self {
43        let scaled = steps * 25;
44        let rounded = if scaled >= 0 {
45            (scaled + 5) / 10
46        } else {
47            (scaled - 5) / 10
48        };
49        Self(rounded)
50    }
51
52    /// Return the stored value in centibels.
53    pub const fn as_centibels(self) -> i16 {
54        self.0
55    }
56
57    /// Return the stored value in decibels.
58    pub const fn as_decibels(self) -> i16 {
59        self.0 / 10
60    }
61
62    /// Return the stored value in quarter-dB steps, rounding to the nearest
63    /// step—the form a peer-repeater entry reports.
64    ///
65    /// The rounding inverse of [`from_quarter_db_steps`](Self::from_quarter_db_steps):
66    /// every value that came from a quarter-dB step returns that step.
67    pub const fn as_quarter_db_steps(self) -> i16 {
68        let scaled = self.0.saturating_mul(2);
69        if scaled >= 0 {
70            (scaled + 2) / 5
71        } else {
72            (scaled - 2) / 5
73        }
74    }
75}

77impl core::fmt::Display for Snr {
78    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
79        let sign = if self.0 < 0 { "-" } else { "" };
80        let magnitude = self.0.unsigned_abs();
81        write!(f, "{sign}{}.{}dB", magnitude / 10, magnitude % 10)
82    }
83}
84
85impl core::fmt::Debug for Snr {
86    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
87        write!(f, "Snr({self})")
88    }

91/// Which of a node's bearers a frame crossed.
92///
93/// A bearer is a shared broadcast medium beside the node's radio, such as
94/// an IP multicast group on the local network. A node numbers its own
95/// bearers from zero; the number means nothing to any other node.
96#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, PartialOrd, Ord)]
97pub struct BearerId(u8);
98
99impl BearerId {
100    /// How many bearers one node can number.
101    pub const COUNT: usize = 8;
102
103    /// The LAN multicast bearer, which every node that has one numbers
104    /// first.
105    pub const LAN: Self = Self(0);
106
107    /// The bearer numbered `index`, if that is below [`Self::COUNT`].
108    pub const fn new(index: u8) -> Option<Self> {
109        if (index as usize) < Self::COUNT {
110            Some(Self(index))
111        } else {
112            None
113        }
114    }
115
116    /// This bearer's number.
117    pub const fn index(self) -> u8 {
118        self.0
119    }
120}

122/// Where a received frame came from.
123///
124/// Only [`RxOrigin::Air`] carries real measurements. The others arrive
125/// through paths with no radio in them, so their `rssi`, `snr`, and `lqi`
126/// carry no information and must be reported as unmeasured wherever a
127/// receiver would otherwise read them as a link quality.
128#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
129pub enum RxOrigin {
130    /// A reception off the air.
131    #[default]
132    Air,
133    /// A copy of a frame this device just transmitted.
134    LocalTx,
135    /// A frame handed over by an attached host across a point-to-point link.
136    Backhaul,
137    /// A frame received from a shared broadcast bearer, heard by every
138    /// node attached to it.
139    Bearer(BearerId),
140}
141
142impl RxOrigin {
143    /// Whether the accompanying signal fields are measurements.
144    pub const fn is_measured(self) -> bool {
145        matches!(self, Self::Air)
146    }
147
148    /// The bearer this frame arrived on, if it arrived on one.
149    pub const fn bearer(self) -> Option<BearerId> {
150        match self {
151            Self::Bearer(bearer) => Some(bearer),
152            _ => None,
153        }
154    }
155}

158/// queue rather than received live.

218/// Half-duplex radio abstraction used by the MAC coordinator.
219pub trait Radio {
220    type Error;
221
222    /// Transmit a complete raw UMSH frame.
223    async fn transmit(
224        &mut self,
225        data: &[u8],
226        options: TxOptions,
227    ) -> Result<(), TxError<Self::Error>>;
228
229    /// Poll reception of one frame into `buf`.
230    ///
231    /// `Poll::Pending` means no frame is currently available right now. The
232    /// call does not reserve any receive state; a later poll after transmit
233    /// completion can resume probing immediately.
234    fn poll_receive(
235        &mut self,
236        cx: &mut Context<'_>,
237        buf: &mut [u8],
238    ) -> Poll<Result<RxInfo, Self::Error>>;
239
240    /// Return the largest supported raw frame size.
241    fn max_frame_size(&self) -> usize;
242    /// Return the approximate airtime for a maximum-length frame.
243    fn t_frame_ms(&self) -> u32;
244
245    /// The LoRa spreading factor this radio is currently receiving with,
246    /// when it knows.
247    ///
248    /// The spreading factor alone sets the SNR a LoRa receiver can
249    /// demodulate down to, so it is what turns a measured SNR into a
250    /// margin. A radio that is not LoRa, or that does not know how its
251    /// far end is configured, reports `None`.
252    fn spreading_factor(&self) -> Option<u8> {
253        None
254    }
255}

257/// Monotonic millisecond clock.
258pub trait Clock {
259    /// Return milliseconds since the device booted.
260    ///
261    /// Most of the stack only ever takes differences and would be happy
262    /// with any monotonic epoch, but `PROP_UPTIME` reports this value
263    /// directly, so the epoch is part of the contract: an implementation
264    /// backed by a clock that starts somewhere else MUST subtract its own
265    /// origin. A simulated device that emulates a reboot has to restart
266    /// this along with the rest of the hardware.
267    fn now_ms(&self) -> u64;
268
269    /// Poll a delay that completes when the monotonic clock reaches `deadline_ms`.
270    ///
271    /// Returns `Poll::Ready(())` if the deadline has already passed. Otherwise
272    /// the implementation MUST register `cx.waker()` with a platform timer and
273    /// return `Poll::Pending`, so the task is woken when the deadline elapses.
274    ///
275    /// The default implementation has no real timer: it schedules an immediate
276    /// re-poll (via `cx.waker().wake_by_ref()`) and returns `Poll::Pending`, causing
277    /// the caller to busy-poll until the monotonic clock reaches the deadline.
278    /// This is correct but wastes CPU; platform clocks backed by a real timer
279    /// (tokio, embassy, etc.) MUST override this to sleep efficiently. A default
280    /// that returned `Ready` unconditionally would spin just as hard; one that
281    /// returned `Pending` without waking would stall timer-driven work entirely.
282    fn poll_delay_until(&self, cx: &mut Context<'_>, deadline_ms: u64) -> Poll<()> {
283        let _ = deadline_ms;
284        cx.waker().wake_by_ref();
285        Poll::Pending
286    }
287}

289/// Persistent peer directory for the node layer.
290///
291/// Stores and retrieves peer records keyed by the raw 32-byte Ed25519 public
292/// key. `alias` is an optional UTF-8 label (the implementation may silently
293/// truncate values longer than 16 bytes). Implementors MUST treat `store_peer`
294/// as an upsert—calling it twice for the same key overwrites the record.
295pub trait PeerStore {
296    type Error;
297
298    /// Upsert the peer record for `key`. `alias`, if present, is a UTF-8
299    /// display label; passing `None` removes any existing alias.
300    async fn store_peer(&self, key: &[u8; 32], alias: Option<&[u8]>) -> Result<(), Self::Error>;
301
302    /// Remove the peer record for `key`. A no-op if the key is not present.
303    async fn delete_peer(&self, key: &[u8; 32]) -> Result<(), Self::Error>;
304
305    /// Invoke `f` for every persisted peer record.
306    ///
307    /// `alias` is `None` when no alias was stored for that key. Callers that
308    /// cannot handle an error mid-iteration should collect into a local buffer
309    /// first, then process asynchronously.
310    async fn for_each_peer(
311        &self,
312        f: &mut dyn FnMut(&[u8; 32], Option<&[u8]>),
313    ) -> Result<(), Self::Error>;
314}
315

418/// Optional power-control hook for higher layers (e.g. the CLI).
419///
420/// Implementations request a controlled shutdown—the actual sequencing
421/// (display, storage flush, GPIO sense, entering System OFF, etc.) lives
422/// in the firmware that owns those peripherals. This call MUST return
423/// promptly; it's typically a `Signal::signal(())` to a shutdown task.
424pub trait PowerControl {
425    fn request_power_off(&self);
426
427    /// Request a soft reboot. The default implementation is a no-op; targets
428    /// without a wired reboot path can leave it as such. Like
429    /// [`request_power_off`](Self::request_power_off), this MUST return
430    /// promptly—typically a `Signal::signal(())` to a reboot task that
431    /// performs any final flushes before triggering a system reset.
432    fn request_reboot(&self) {}
433}

475#[cfg(test)]
476mod tests {
477    use super::Snr;
478
479    /// The wire form of an SNR in a peer-repeater entry is quarter-dB
480    /// steps, so a value that arrived as one has to leave as the same one
481    ///—otherwise relaying an observation would drift it a step at a
482    /// time.
483    #[test]
484    fn quarter_db_steps_round_trip_through_centibels() {
485        for steps in -128..=127i16 {
486            assert_eq!(
487                Snr::from_quarter_db_steps(steps).as_quarter_db_steps(),
488                steps,
489                "steps {steps}"
490            );
491        }
492    }
493
494    #[test]
495    fn quarter_db_steps_round_to_the_nearest_step() {
496        assert_eq!(Snr::from_decibels(0).as_quarter_db_steps(), 0);
497        assert_eq!(Snr::from_decibels(5).as_quarter_db_steps(), 20);
498        assert_eq!(Snr::from_decibels(-7).as_quarter_db_steps(), -28);
499        // 0.3 dB sits between the first and second step and rounds up.
500        assert_eq!(Snr::from_centibels(3).as_quarter_db_steps(), 1);
501    }
502
503    #[test]
504    fn display_keeps_the_sign_off_the_fraction() {
505        extern crate std;
506        use std::format;
507
508        assert_eq!(format!("{}", Snr::from_centibels(25)), "2.5dB");
509        assert_eq!(format!("{}", Snr::from_centibels(0)), "0.0dB");
510        assert_eq!(format!("{}", Snr::from_centibels(-5)), "-0.5dB");
511        assert_eq!(format!("{}", Snr::from_centibels(-15)), "-1.5dB");
512        assert_eq!(format!("{}", Snr::from_centibels(-100)), "-10.0dB");
513        assert_eq!(format!("{}", Snr::from_centibels(i16::MIN)), "-3276.8dB");
514    }
