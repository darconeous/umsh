20/// Signal-to-noise ratio represented in centibels (0.1 dB units).
21///
22/// This uses a slightly finer unit than whole decibels while still staying
23/// compact and integer-friendly. Some common LoRa radios report SNR in
24/// quarter-dB steps. Converting those readings into centibels requires
25/// rounding, introducing at most 0.5 cB (0.05 dB) of error.
26#[derive(Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
27pub struct Snr(i16);
28
29impl Snr {
30    /// Construct an SNR value directly from centibels.
31    pub const fn from_centibels(centibels: i16) -> Self {
32        Self(centibels)
33    }
34
35    /// Construct an SNR value from whole decibels.
36    pub const fn from_decibels(db: i8) -> Self {
37        Self((db as i16) * 10)
38    }
39
40    /// Construct an SNR value from quarter-dB steps, rounding to the nearest
41    /// centibel.
42    pub const fn from_quarter_db_steps(steps: i16) -> Self {
43        let scaled = steps * 25;
44        let rounded = if scaled >= 0 {
45            (scaled + 5) / 10
46        } else {
47            (scaled - 5) / 10
48        };
49        Self(rounded)
50    }
51
52    /// Return the stored value in centibels.
53    pub const fn as_centibels(self) -> i16 {
54        self.0
55    }
56
57    /// Return the stored value in decibels.
58    pub const fn as_decibels(self) -> i16 {
59        self.0 / 10
60    }
61}
62
63impl core::fmt::Display for Snr {
64    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
65        let db = self.as_decibels();
66        let cdb = self.0 % 10;
67        write!(f, "{}.{:01}dB", db, cdb)
68    }
69}
70
71impl core::fmt::Debug for Snr {
72    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
73        write!(f, "Snr({self})")
74    }
75}

77/// Where a received frame came from.
78///
79/// Only [`RxOrigin::Air`] carries real measurements. The other two arrive
80/// through paths with no radio in them, so their `rssi`, `snr`, and `lqi`
81/// carry no information and must be reported as unmeasured wherever a
82/// receiver would otherwise read them as a link quality.
83#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
84pub enum RxOrigin {
85    /// A reception off the air.
86    #[default]
87    Air,
88    /// A copy of a frame this device just transmitted.
89    LocalTx,
90    /// A frame handed over by an attached host across a point-to-point link.
91    Backhaul,
92}
93
94impl RxOrigin {
95    /// Whether the accompanying signal fields are measurements.
96    pub const fn is_measured(self) -> bool {
97        matches!(self, Self::Air)
98    }
99}

101/// Metadata returned with a received frame.

103pub struct RxInfo {
104    /// Number of bytes written into the receive buffer.
105    pub len: usize,
106    /// Received signal strength in dBm.
107    pub rssi: i16,
108    /// Signal-to-noise ratio in centibels.
109    pub snr: Snr,
110    /// Optional link-quality indicator in a radio-specific normalized scale.
111    pub lqi: Option<NonZeroU8>,
112    /// The path this frame took to reach the receiver.
113    pub origin: RxOrigin,
114}

116/// Channel-activity-detection (CAD) policy applied before a transmit.
117#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
118pub enum CadPolicy {
119    /// Skip CAD entirely and transmit immediately.
120    #[default]
121    Skip,
122    /// Perform CAD once and transmit only if the channel is currently clear.
123    /// Equivalent to a retry budget of zero.
124    Gate,
125    /// Retry CAD until the channel is clear or `timeout_ms` elapses.
126    RetryFor { timeout_ms: u32 },
127}

145/// Half-duplex radio abstraction used by the MAC coordinator.
146pub trait Radio {
147    type Error;
148
149    /// Transmit a complete raw UMSH frame.
150    async fn transmit(
151        &mut self,
152        data: &[u8],
153        options: TxOptions,
154    ) -> Result<(), TxError<Self::Error>>;
155
156    /// Poll reception of one frame into `buf`.
157    ///
158    /// `Poll::Pending` means no frame is currently available right now. The
159    /// call does not reserve any receive state; a later poll after transmit
160    /// completion can resume probing immediately.
161    fn poll_receive(
162        &mut self,
163        cx: &mut Context<'_>,
164        buf: &mut [u8],
165    ) -> Poll<Result<RxInfo, Self::Error>>;
166
167    /// Return the largest supported raw frame size.
168    fn max_frame_size(&self) -> usize;
169    /// Return the approximate airtime for a maximum-length frame.
170    fn t_frame_ms(&self) -> u32;
171}

173/// Monotonic millisecond clock.
174pub trait Clock {
175    /// Return milliseconds since an arbitrary monotonic epoch.
176    fn now_ms(&self) -> u64;
177
178    /// Poll a delay that completes when the monotonic clock reaches `deadline_ms`.
179    ///
180    /// Returns `Poll::Ready(())` if the deadline has already passed. Otherwise
181    /// the implementation MUST register `cx.waker()` with a platform timer and
182    /// return `Poll::Pending`, so the task is woken when the deadline elapses.
183    ///
184    /// The default implementation has no real timer: it schedules an immediate
185    /// re-poll (via `cx.waker().wake_by_ref()`) and returns `Poll::Pending`, causing
186    /// the caller to busy-poll until the monotonic clock reaches the deadline.
187    /// This is correct but wastes CPU; platform clocks backed by a real timer
188    /// (tokio, embassy, etc.) MUST override this to sleep efficiently. A default
189    /// that returned `Ready` unconditionally would spin just as hard; one that
190    /// returned `Pending` without waking would stall timer-driven work entirely.
191    fn poll_delay_until(&self, cx: &mut Context<'_>, deadline_ms: u64) -> Poll<()> {
192        let _ = deadline_ms;
193        cx.waker().wake_by_ref();
194        Poll::Pending
195    }
196}

198/// Persistent peer directory for the node layer.
199///
200/// Stores and retrieves peer records keyed by the raw 32-byte Ed25519 public
201/// key. `alias` is an optional UTF-8 label (the implementation may silently
202/// truncate values longer than 16 bytes). Implementors MUST treat `store_peer`
203/// as an upsert — calling it twice for the same key overwrites the record.
204pub trait PeerStore {
205    type Error;
206
207    /// Upsert the peer record for `key`. `alias`, if present, is a UTF-8
208    /// display label; passing `None` removes any existing alias.
209    async fn store_peer(&self, key: &[u8; 32], alias: Option<&[u8]>) -> Result<(), Self::Error>;
210
211    /// Remove the peer record for `key`. A no-op if the key is not present.
212    async fn delete_peer(&self, key: &[u8; 32]) -> Result<(), Self::Error>;
213
214    /// Invoke `f` for every persisted peer record.
215    ///
216    /// `alias` is `None` when no alias was stored for that key. Callers that
217    /// cannot handle an error mid-iteration should collect into a local buffer
218    /// first, then process asynchronously.
219    async fn for_each_peer(
220        &self,
221        f: &mut dyn FnMut(&[u8; 32], Option<&[u8]>),
222    ) -> Result<(), Self::Error>;
223}
224
225/// No-op peer store — use when peer persistence is not needed.
226pub struct NoPeerStore;
227
228impl PeerStore for NoPeerStore {
229    type Error = core::convert::Infallible;
230
231    async fn store_peer(&self, _: &[u8; 32], _: Option<&[u8]>) -> Result<(), Self::Error> {
232        Ok(())
233    }
234
235    async fn delete_peer(&self, _: &[u8; 32]) -> Result<(), Self::Error> {
236        Ok(())
237    }
238
239    async fn for_each_peer(
240        &self,

242    ) -> Result<(), Self::Error> {
243        Ok(())
244    }

293    type Error;
294
295    /// Load the stored counter for `context`, or `0` if missing.

327/// Optional power-control hook for higher layers (e.g. the CLI).
328///
329/// Implementations request a controlled shutdown — the actual sequencing
330/// (display, storage flush, GPIO sense, entering System OFF, etc.) lives
331/// in the firmware that owns those peripherals. This call MUST return
332/// promptly; it's typically a `Signal::signal(())` to a shutdown task.
333pub trait PowerControl {
334    fn request_power_off(&self);
335
336    /// Request a soft reboot. The default implementation is a no-op; targets
337    /// without a wired reboot path can leave it as such. Like
338    /// [`request_power_off`](Self::request_power_off), this MUST return
339    /// promptly — typically a `Signal::signal(())` to a reboot task that
340    /// performs any final flushes before triggering a system reset.
341    fn request_reboot(&self) {}
342}
343
344/// No-op power control — use when shutdown is not implemented for the target.
345pub struct NoPowerControl;
346
347impl PowerControl for NoPowerControl {
348    fn request_power_off(&self) {}
349}
