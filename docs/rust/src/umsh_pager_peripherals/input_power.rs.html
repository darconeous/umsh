1//! Input/display handoff policy, independent of GPIO and the executor.
2
3#[derive(Clone, Copy, Debug, PartialEq, Eq)]
4pub enum Mode {
5    Interactive,
6    Dark,
7    Shutdown,
8}
9
10#[derive(Clone, Copy, Debug, PartialEq, Eq)]
11pub enum Gesture {
12    CancelAlert,
13    Navigate,
14    Wake,
15    Splash,
16    Ignore,
17}
18
19pub struct InputPower {
20    pub mode: Mode,
21    pub epoch: u32,
22    pub armed: bool,
23    pub display_ready: bool,
24    activity: u32,
25    dark_activity: u32,
26    waking: bool,
27    press: Option<Gesture>,
28}
29
30impl Default for InputPower {
31    fn default() -> Self {
32        Self::new()
33    }
34}
35
36impl InputPower {
37    pub const fn new() -> Self {
38        Self {
39            mode: Mode::Interactive,
40            epoch: 0,
41            armed: false,
42            display_ready: false,
43            activity: 0,
44            dark_activity: 0,
45            waking: false,
46            press: None,
47        }
48    }
49
50    pub fn activity(&self) -> u32 {
51        self.activity
52    }
53
54    pub fn note_activity(&mut self) {
55        self.activity = self.activity.wrapping_add(1);
56    }
57
58    pub fn begin_press(&mut self, disposition: Gesture) {
59        self.press = Some(disposition);
60    }
61
62    pub fn alert_started(&mut self) {
63        if self.press.is_some() {
64            self.press = Some(Gesture::CancelAlert);
65        }
66    }
67
68    pub fn press_disposition(&self) -> Gesture {
69        self.press.unwrap_or(Gesture::Ignore)
70    }
71
72    pub fn release_press(&mut self) {
73        self.press = None;
74    }
75
76    /// A display-off operation may have yielded while an input arrived.
77    pub fn dark(&mut self, before_display_off: u32) -> bool {
78        if self.mode == Mode::Shutdown || self.activity != before_display_off || self.waking {
79            return false;
80        }
81        self.mode = Mode::Dark;
82        self.epoch = self.epoch.wrapping_add(1);
83        self.armed = false;
84        self.display_ready = false;
85        self.dark_activity = self.activity;
86        true
87    }
88
89    pub fn wake(&mut self) -> bool {
90        if self.mode == Mode::Shutdown {
91            return false;
92        }
93        if self.mode == Mode::Dark {
94            self.mode = Mode::Interactive;
95            self.epoch = self.epoch.wrapping_add(1);
96        }
97        self.armed = false;
98        self.waking = true;
99        self.display_ready = false;
100        true
101    }
102
103    pub fn arm(&mut self, epoch: u32) -> bool {
104        if self.mode != Mode::Dark || self.epoch != epoch || self.activity != self.dark_activity {
105            return false;
106        }
107        self.armed = true;
108        true
109    }
110
111    pub fn activity_since_dark(&self) -> bool {
112        self.activity != self.dark_activity
113    }
114
115    pub fn shown(&mut self, epoch: u32) {
116        if self.mode == Mode::Interactive && self.epoch == epoch {
117            self.display_ready = true;
118            self.waking = false;
119        }
120    }
121
122    pub fn shutdown(&mut self) {
123        self.mode = Mode::Shutdown;
124        self.epoch = self.epoch.wrapping_add(1);
125        self.armed = false;
126        self.display_ready = false;
127    }
128
129    /// Latch at gesture start, not after debounce or display rendering.
130    pub fn gesture(&mut self, splash: bool, faded: bool, alert: bool) -> Gesture {
131        self.note_activity();
132        if self.mode == Mode::Shutdown {
133            Gesture::Ignore
134        } else if alert {
135            Gesture::CancelAlert
136        } else if splash {
137            Gesture::Splash
138        } else if self.mode == Mode::Dark || (faded && !self.waking) {
139            self.wake();
140            Gesture::Wake
141        } else {
142            Gesture::Navigate
143        }
144    }
145}

147/// Fixed-space qualification summary. No timers or extra acquisitions.
148pub struct PowerWindow {
149    pub count: u32,
150    pub unlocked: u32,
151    pub min_ma: i32,
152    pub max_ma: i32,
153    sum_ma: i64,
154    first_ms: u64,
155    last_ms: u64,
156}
157impl Default for PowerWindow {
158    fn default() -> Self {
159        Self::new()
160    }
161}
162impl PowerWindow {
163    pub const fn new() -> Self {
164        Self {
165            count: 0,
166            unlocked: 0,
167            min_ma: 0,
168            max_ma: 0,
169            sum_ma: 0,
170            first_ms: 0,
171            last_ms: 0,
172        }
173    }
174    pub fn sample(&mut self, now_ms: u64, current_ma: i32, unlocked: bool) {
175        if self.complete() {
176            return;
177        }
178        if self.count == 0 {
179            self.first_ms = now_ms;
180            self.min_ma = current_ma;
181            self.max_ma = current_ma;
182        }
183        self.last_ms = now_ms;
184        self.count += 1;
185        self.unlocked += u32::from(unlocked);
186        self.min_ma = self.min_ma.min(current_ma);
187        self.max_ma = self.max_ma.max(current_ma);
188        self.sum_ma += i64::from(current_ma);
189    }
190    pub fn span_ms(&self) -> u64 {
191        self.last_ms.saturating_sub(self.first_ms)
192    }
193    pub fn complete(&self) -> bool {
194        self.count >= 300 && self.span_ms() >= 300_000
195    }
196    pub fn mean_ma(&self) -> i64 {
197        self.sum_ma.checked_div(i64::from(self.count)).unwrap_or(0)
198    }
199}

205    #[test]
206    fn alert_preempts_splash_dark_and_a_pending_press() {
207        let mut p = InputPower::new();
208        assert!(p.dark(p.activity()));
209        assert_eq!(p.gesture(true, true, true), Gesture::CancelAlert);
210        p.wake();
211        p.shown(p.epoch);
212        p.begin_press(Gesture::Navigate);
213        p.alert_started();
214        // Even if the alert ends before debounce completes, never select.
215        assert_eq!(p.press_disposition(), Gesture::CancelAlert);
216        p.release_press();
217        assert_eq!(p.press_disposition(), Gesture::Ignore);
218        assert_eq!(p.gesture(false, false, false), Gesture::Navigate);
219        p.shutdown();
220        assert_eq!(p.gesture(true, true, true), Gesture::Ignore);
221    }
222
223    #[test]
224    fn power_window_requires_five_minutes_and_keeps_signed_current() {
225        let mut w = PowerWindow::new();
226        for i in 0..300 {
227            w.sample(30_000 + i * 1000, -100, true);
228        }
229        assert!(!w.complete());
230        w.sample(330_000, -200, false);
231        assert!(w.complete());
232        assert_eq!(w.count, 301);
233        assert_eq!(w.unlocked, 300);
234        assert_eq!((w.min_ma, w.max_ma), (-200, -100));
235        assert_eq!(w.mean_ma(), -100);
236        w.sample(331_000, 700, false);
237        assert_eq!(w.count, 301);
238    }

264        let stale = p.epoch;

266        assert!(!p.arm(stale));

271        p.shown(p.epoch);
