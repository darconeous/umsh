1//! Quadrature policy and TCA8418 key events. Quadrature `transition` is
2//! called from the GPIO interrupt handler on BOTH edges of BOTH inputs.
3use embedded_hal_async::i2c::I2c;

5/// Full-step decoder: a detent is a complete cycle back to both inputs high.
6/// Partial reversals cancel; a two-bit jump invalidates the current cycle.
7pub struct Quadrature {
8    previous: u8,
9    quarters: i8,
10    synchronized: bool,
11}
12impl Quadrature {
13    pub const fn new(ab: u8) -> Self {
14        Self {
15            previous: ab & 3,
16            quarters: 0,
17            synchronized: ab & 3 == 3,
18        }
19    }
20    #[inline(always)]
21    pub fn transition(&mut self, ab: u8) -> i8 {
22        let ab = ab & 3;
23        let old = self.previous;
24        self.previous = ab;
25        if old ^ ab == 3 {
26            self.quarters = 0;
27            self.synchronized = false;
28        } else if self.synchronized {
29            self.quarters += match (old, ab) {
30                (3, 1) | (1, 0) | (0, 2) | (2, 3) => 1,
31                (3, 2) | (2, 0) | (0, 1) | (1, 3) => -1,
32                _ => 0,
33            };
34        }
35        if ab != 3 {
36            return 0;
37        }
38        let step = if self.synchronized {
39            match self.quarters {
40                4 => 1,
41                -4 => -1,
42                _ => 0,
43            }
44        } else {
45            0
46        };
47        self.quarters = 0;
48        self.synchronized = true;
49        step
50    }
51}

53/// Stable-level debounce for the push button, distinct from quadrature.
54pub struct Debounce {
55    stable: bool,
56    observed: bool,
57    since: u64,
58}
59impl Debounce {
60    pub const fn new(pressed: bool) -> Self {
61        Self {
62            stable: pressed,
63            observed: pressed,
64            since: 0,
65        }
66    }
67    pub fn sample(&mut self, pressed: bool, now_ms: u64) -> Option<bool> {
68        if pressed != self.observed {
69            self.observed = pressed;
70            self.since = now_ms;
71        }
72        if self.observed != self.stable && now_ms.saturating_sub(self.since) >= 15 {
73            self.stable = self.observed;
74            Some(self.stable)
75        } else {
76            None
77        }
78    }
79
80    pub fn observed(&self) -> bool {
81        self.observed
82    }
83
84    pub fn deadline(&self) -> Option<u64> {
85        (self.observed != self.stable).then_some(self.since.saturating_add(15))
86    }
87}

89/// A gesture begins at the first contact, and ends only after a stable
90/// release, even if a bouncing press never became a debounced press.
91#[derive(Default)]
92pub struct PressLatch {
93    active: bool,
94    release_at: Option<u64>,
95}
96impl PressLatch {
97    pub fn observe(&mut self, pressed: bool, now_ms: u64) -> bool {
98        if pressed {
99            self.release_at = None;
100            let began = !self.active;
101            self.active = true;
102            began
103        } else {
104            if self.active {
105                let at = *self.release_at.get_or_insert(now_ms.saturating_add(15));
106                if now_ms >= at {
107                    self.active = false;
108                    self.release_at = None;
109                }
110            }
111            false
112        }
113    }
114
115    pub fn deadline(&self) -> Option<u64> {
116        self.release_at
117    }
118}

120pub const KEYBOARD_ADDRESS: u8 = 0x34;
121/// Backspace is row 2, column 9; TCA8418 FIFO key numbers start at one.
122pub const BACKSPACE_EVENT: u8 = 30;
123#[derive(Clone, Copy, Debug, PartialEq, Eq)]
124pub enum KeyEvent {
125    BackPress,
126    BackRelease,
127    Press,
128    Release,
129    Other,
130    Overflow,
131}
132
133pub struct Keyboard<I> {
134    i2c: I,
135    down: u64,
136}
137impl<I: I2c> Keyboard<I> {
138    pub fn new(i2c: I) -> Self {
139        Self { i2c, down: 0 }
140    }
141    pub async fn init(&mut self) -> Result<(), I::Error> {
142        // No auto-increment required: every register is addressed explicitly.
143        self.i2c.write(KEYBOARD_ADDRESS, &[0x01, 0]).await?;
144        for (reg, value) in [
145            (0x1d, 0x0f),
146            (0x1e, 0xff),
147            (0x1f, 3),
148            (0x29, 0),
149            (0x2a, 0),
150            (0x2b, 0),
151            (0x2c, 0),
152            (0x2d, 0),
153            (0x2e, 0),
154        ] {
155            self.i2c.write(KEYBOARD_ADDRESS, &[reg, value]).await?;
156        }
157        self.flush().await?;
158        // Key event and FIFO overflow interrupts; discard new entries on overflow.
159        self.i2c.write(KEYBOARD_ADDRESS, &[0x01, 0x09]).await
160    }
161    pub async fn flush(&mut self) -> Result<(), I::Error> {
162        // FIFO is ten entries deep. Bound recovery even if keys keep arriving.
163        for _ in 0..10 {
164            let _ = crate::read(&mut self.i2c, KEYBOARD_ADDRESS, 4).await?;
165        }
166        self.down = 0;
167        self.i2c.write(KEYBOARD_ADDRESS, &[2, 0x1f]).await
168    }
169    pub async fn next(&mut self) -> Result<Option<KeyEvent>, I::Error> {
170        let status = crate::read(&mut self.i2c, KEYBOARD_ADDRESS, 2).await?;
171        if status & 8 != 0 {
172            self.flush().await?;
173            return Ok(Some(KeyEvent::Overflow));
174        }
175        let event = crate::read(&mut self.i2c, KEYBOARD_ADDRESS, 4).await?;
176        if event == 0 {
177            self.i2c.write(KEYBOARD_ADDRESS, &[2, 1]).await?;
178            // Check after acknowledging, so an arrival racing the clear isn't lost.
179            if crate::read(&mut self.i2c, KEYBOARD_ADDRESS, 3).await? & 0x0f != 0 {
180                return Ok(Some(KeyEvent::Other));
181            }
182            return Ok(None);
183        }
184        Ok(Some(self.decode(event)))
185    }
186    pub fn decode(&mut self, event: u8) -> KeyEvent {
187        let key = event & 0x7f;
188        // The Pager configures a 4 x 10 matrix. Other FIFO codes are not keys.
189        if !(1..=40).contains(&key) {
190            return KeyEvent::Other;
191        }
192        let pressed = event & 0x80 != 0;
193        let mask = 1u64 << key;
194        if pressed == (self.down & mask != 0) {
195            return KeyEvent::Other;
196        }
197        if pressed {
198            self.down |= mask;
199        } else {
200            self.down &= !mask;
201        }
202        match (key == BACKSPACE_EVENT, pressed) {
203            (true, true) => KeyEvent::BackPress,
204            (true, false) => KeyEvent::BackRelease,
205            (false, true) => KeyEvent::Press,
206            (false, false) => KeyEvent::Release,
207        }
208    }
