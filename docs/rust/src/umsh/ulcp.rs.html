1//! Host-side client for the UMSH Local Control Protocol (ULCP).
2//!
3//! [`UlcpDevice`] drives a device over a frame-oriented [`FrameLink`], and
4//! exposes the link as a [`umsh_hal::Radio`] so the host can run the
5//! full MAC/node stack with the device acting purely as the PHY.
6//!
7//! The wire format lives in [`umsh_ulcp`] (re-exported as
8//! [`crate::ulcp_wire`]); this module owns the host-side session
9//! behavior: the reset/configure handshake, request/response
10//! transactions, and queueing of frames that arrive while a command is
11//! in flight.
12//!
13//! See `docs/protocol/src/ulcp.md` and the chapters under it for the
14//! protocol.

56#[derive(Debug)]
57pub enum UlcpError {
58    Io(io::Error),
59    /// The stream reached end-of-file; the device link is gone.
60    Disconnected,
61    /// The device violated the ULCP.
62    Protocol(&'static str),
63    /// The device reported a failure status for a command.
64    Status(Status),
65    /// The device reset outside of an initialization handshake, losing
66    /// its configuration. The radio must be re-initialized.
67    UnexpectedReset(Status),
68    /// The frame exceeds the device's advertised MTU.
69    FrameTooLarge(usize),
70    /// The device did not answer a command in time.
71    Timeout,
72    /// A non-stream transport failed.
73    Transport(String),
74    /// A host-domain write was attempted on an administrative handle.
75    /// Administering a device configures the device; it does not claim
76    /// the device (see [`AttachMode`]).
77    AdministrativeAttach,
78}

80impl core::fmt::Display for UlcpError {
81    fn fmt(&self, formatter: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
82        match self {
83            Self::Io(error) => write!(formatter, "io error: {error}"),
84            Self::Disconnected => write!(formatter, "ULCP link disconnected"),
85            Self::Protocol(message) => write!(formatter, "protocol error: {message}"),
86            Self::Status(status) => write!(formatter, "device reported {status:?}"),
87            Self::UnexpectedReset(status) => {
88                write!(formatter, "device reset unexpectedly ({status:?})")
89            }
90            Self::FrameTooLarge(len) => write!(formatter, "frame too large: {len} bytes"),
91            Self::Timeout => write!(formatter, "timed out waiting for device response"),
92            Self::Transport(message) => write!(formatter, "transport error: {message}"),
93            Self::AdministrativeAttach => write!(
94                formatter,
95                "host-domain writes need a tethered attach, not an administrative one"
96            ),
97        }
98    }

252/// `PROP_SAVED`: what the device reports about its stored snapshot.
253///
254/// The two failure values are the point of the property. An unattended
255/// repeater that comes back on stale configuration, or on none at all,
256/// looks identical to a healthy one from the outside — this is how a
257/// host that does eventually attach finds out.
258#[derive(Clone, Copy, Debug, PartialEq, Eq)]
259pub enum SavedSnapshot {
260    /// Nothing is saved.
261    None,
262    /// The newest saved generation is in effect.
263    Current,
264    /// A newer generation was rejected at boot and an older one is in
265    /// effect: the device is working, on stale configuration. Re-save to
266    /// clear it.
267    Fallback,
268    /// A snapshot exists but no generation could be read; the device
269    /// booted with post-reset defaults.
270    Unreadable,
271}
272
273impl SavedSnapshot {
274    fn from_octet(value: &[u8]) -> Result<Self, UlcpError> {
275        match value {
276            [ids::saved::NONE] => Ok(Self::None),
277            [ids::saved::CURRENT] => Ok(Self::Current),
278            [ids::saved::FALLBACK] => Ok(Self::Fallback),
279            [ids::saved::UNREADABLE] => Ok(Self::Unreadable),
280            _ => Err(UlcpError::Protocol("malformed PROP_SAVED")),
281        }
282    }
283
284    /// Whether a saved snapshot is in effect at all, in either
285    /// generation.
286    pub fn is_saved(self) -> bool {
287        matches!(self, Self::Current | Self::Fallback)
288    }
289}

463/// A cancel-safe, frame-oriented ULCP transport.
464#[allow(async_fn_in_trait)]
465pub trait FrameLink {
466    /// Send one complete ULCP frame.
467    async fn send_frame(&mut self, frame: &[u8]) -> Result<(), UlcpError>;
468
469    /// Poll for the next complete ULCP frame.
470    ///
471    /// Implementations keep all partial state in `self`, so cancellation cannot
472    /// discard a partial frame or unread bytes following a completed frame.
473    fn poll_recv_frame(
474        &mut self,
475        cx: &mut core::task::Context<'_>,
476    ) -> core::task::Poll<Result<Vec<u8>, UlcpError>>;
477
478    /// Receive the next complete ULCP frame.
479    async fn recv_frame(&mut self) -> Result<Vec<u8>, UlcpError> {
480        core::future::poll_fn(|cx| self.poll_recv_frame(cx)).await
481    }
482}

484/// HDLC-Lite framing over a reliable asynchronous byte stream.
485pub struct SerialFrameLink<IO> {
486    io: IO,
487    decoder: hdlc::Decoder<WIRE_BUF>,
488    read_buf: [u8; READ_CHUNK],
489    read_pos: usize,
490    read_len: usize,
491}
492
493impl<IO> SerialFrameLink<IO> {
494    /// Wrap a byte stream in ULCP HDLC framing.
495    pub fn new(io: IO) -> Self {
496        Self {
497            io,
498            decoder: hdlc::Decoder::new(),
499            read_buf: [0; READ_CHUNK],
500            read_pos: 0,
501            read_len: 0,
502        }
503    }
504
505    /// Recover the underlying byte stream.
506    pub fn into_inner(self) -> IO {
507        self.io
508    }
509}
510
511impl<IO> FrameLink for SerialFrameLink<IO>
512where
513    IO: AsyncRead + AsyncWrite + Unpin,
514{
515    async fn send_frame(&mut self, frame: &[u8]) -> Result<(), UlcpError> {
516        let mut wire = vec![0u8; hdlc::max_encoded_len(frame.len())];
517        let len = hdlc::encode_frame(frame, &mut wire).expect("buffer sized with max_encoded_len");
518        self.io.write_all(&wire[..len]).await?;
519        self.io.flush().await?;
520        Ok(())
521    }
522
523    fn poll_recv_frame(
524        &mut self,
525        cx: &mut core::task::Context<'_>,
526    ) -> core::task::Poll<Result<Vec<u8>, UlcpError>> {
527        loop {
528            while self.read_pos < self.read_len {
529                let byte = self.read_buf[self.read_pos];
530                self.read_pos += 1;
531                if let Some(Ok(frame)) = self.decoder.push(byte) {
532                    return core::task::Poll::Ready(Ok(frame.to_vec()));
533                }
534            }
535
536            self.read_pos = 0;
537            self.read_len = 0;
538            let mut read_buf = ReadBuf::new(&mut self.read_buf);
539            match core::pin::Pin::new(&mut self.io).poll_read(cx, &mut read_buf) {
540                core::task::Poll::Ready(Ok(())) => {
541                    self.read_len = read_buf.filled().len();
542                    if self.read_len == 0 {
543                        return core::task::Poll::Ready(Err(UlcpError::Disconnected));
544                    }
545                }
546                core::task::Poll::Ready(Err(error)) => {
547                    return core::task::Poll::Ready(Err(UlcpError::Io(error)));
548                }
549                core::task::Poll::Pending => return core::task::Poll::Pending,
550            }
551        }
552    }
553}

597                "BLE discovery, operation, and pairing timeouts must be nonzero",
598            ));
599        }
600        Ok(())

620        &mut self,
621        cx: &mut core::task::Context<'_>,
622    ) -> core::task::Poll<Result<Vec<u8>, UlcpError>> {
623        loop {
624            match self.notifications.poll_recv(cx) {
625                core::task::Poll::Ready(Some(segment)) => {
626                    if let Some(Ok(frame)) = self.reassembler.push(&segment) {
627                        return core::task::Poll::Ready(Ok(frame.to_vec()));
628                    }
629                    // Transport-level malformed/oversize segments are dropped.
630                }
631                core::task::Poll::Ready(None) => {
632                    self.reassembler.reset();
633                    return core::task::Poll::Ready(Err(UlcpError::Disconnected));
634                }
635                core::task::Poll::Pending => return core::task::Poll::Pending,
636            }
637        }
638    }

666    /// Scan for ULCP GATT Service peripherals without connecting.
667    ///
668    /// Runs discovery for the full `timeout` and reports every matching
669    /// peripheral seen, so nearby radios all get listed (unlike
670    /// [`connect`](Self::connect), which returns as soon as a match
671    /// appears).
672    pub async fn scan(timeout: Duration) -> Result<Vec<BleScanResult>, UlcpError> {
673        use btleplug::api::{Central, Manager as _, Peripheral as _, ScanFilter};
674
675        let manager = btleplug::platform::Manager::new()
676            .await
677            .map_err(ble_error)?;
678        let adapters = manager.adapters().await.map_err(ble_error)?;
679        let service = uuid::Uuid::from_u128(umsh_ulcp::gatt::SERVICE_UUID);
680        let deadline = Instant::now() + timeout;
681        let mut results: Vec<BleScanResult> = Vec::new();
682
683        for adapter in adapters {
684            adapter
685                .start_scan(ScanFilter {
686                    services: vec![service],
687                })
688                .await
689                .map_err(ble_error)?;
690            while Instant::now() < deadline {
691                tokio::time::sleep(Duration::from_millis(250)).await;
692                let peripherals =
693                    match tokio::time::timeout_at(deadline, adapter.peripherals()).await {
694                        Ok(result) => result.map_err(ble_error)?,
695                        Err(_) => break,
696                    };
697                for peripheral in peripherals {
698                    let properties =
699                        match tokio::time::timeout_at(deadline, peripheral.properties()).await {
700                            Ok(result) => result.map_err(ble_error)?,
701                            Err(_) => break,
702                        };
703                    let advertises_service = properties
704                        .as_ref()
705                        .is_some_and(|properties| properties.services.contains(&service));
706                    if !advertises_service {
707                        continue;
708                    }
709                    let id = peripheral.id().to_string();
710                    let name = properties
711                        .as_ref()
712                        .and_then(|properties| properties.local_name.clone());
713                    let rssi = properties.as_ref().and_then(|properties| properties.rssi);
714                    match results.iter_mut().find(|result| result.id == id) {
715                        Some(existing) => {
716                            existing.name = name.or(existing.name.take());
717                            existing.rssi = rssi.or(existing.rssi);
718                        }
719                        None => results.push(BleScanResult { id, name, rssi }),
720                    }
721                }
722            }
723            // CoreBluetooth operations can block indefinitely; bound the
724            // cleanup like connect does.
725            let _ = tokio::time::timeout(Duration::from_secs(1), adapter.stop_scan()).await;
726        }
727        Ok(results)
728    }

730    /// Discover and attach to a ULCP GATT Service peripheral.
731    ///
732    /// `selector` matches a local-name substring or the platform peripheral ID.
733    /// With no selector, discovery must yield exactly one companion radio.
734    pub async fn connect(
735        selector: Option<&str>,
736        config: BleFrameLinkConfig,
737    ) -> Result<Self, UlcpError> {
738        use btleplug::api::{Central, Manager as _, Peripheral as _, ScanFilter};
739        use futures_util::StreamExt;
740
741        config.validate()?;
742
743        let manager = btleplug::platform::Manager::new()
744            .await
745            .map_err(ble_error)?;
746        let adapters = manager.adapters().await.map_err(ble_error)?;
747        let service = uuid::Uuid::from_u128(umsh_ulcp::gatt::SERVICE_UUID);
748        let deadline = Instant::now() + config.discovery_timeout;
749        let mut matches = Vec::new();
750
751        for adapter in adapters {
752            adapter
753                .start_scan(ScanFilter {
754                    services: vec![service],
755                })
756                .await
757                .map_err(ble_error)?;
758            loop {
759                if Instant::now() >= deadline {
760                    break;
761                }
762                tokio::time::sleep(Duration::from_millis(250)).await;
763                matches.clear();
764                let peripherals =
765                    match tokio::time::timeout_at(deadline, adapter.peripherals()).await {
766                        Ok(result) => result.map_err(ble_error)?,
767                        Err(_) => break,
768                    };
769                for peripheral in peripherals {
770                    let properties =
771                        match tokio::time::timeout_at(deadline, peripheral.properties()).await {
772                            Ok(result) => result.map_err(ble_error)?,
773                            Err(_) => break,
774                        };
775                    let id = peripheral.id().to_string();
776                    let name = properties
777                        .as_ref()
778                        .and_then(|properties| properties.local_name.as_deref());
779                    let selected = selector.is_none_or(|selector| {
780                        id == selector || name.is_some_and(|name| name.contains(selector))
781                    });
782                    let advertises_service = properties
783                        .as_ref()
784                        .is_some_and(|properties| properties.services.contains(&service));
785                    if selected && advertises_service {
786                        matches.push(peripheral);
787                    }
788                }
789                if !matches.is_empty() || Instant::now() >= deadline {
790                    break;
791                }
792            }
793            // CoreBluetooth operations can block indefinitely. Discovery's
794            // configured deadline applies to every await, including cleanup.
795            let _ = tokio::time::timeout(Duration::from_secs(1), adapter.stop_scan()).await;
796            if !matches.is_empty() {
797                break;
798            }
799        }
800
801        let peripheral = match matches.len() {
802            0 => {
803                return Err(UlcpError::Transport(
804                    "no ULCP GATT Service peripheral found".into(),
805                ));
806            }
807            1 => matches.pop().unwrap(),
808            _ => {
809                return Err(UlcpError::Transport(
810                    "multiple companion radios found; provide a selector".into(),
811                ));
812            }
813        };
814
815        let setup = async {
816            let is_connected =
817                tokio::time::timeout(config.operation_timeout, peripheral.is_connected())
818                    .await
819                    .map_err(|_| ble_timeout("querying connection state"))?
820                    .map_err(ble_error)?;
821            if !is_connected {
822                tokio::time::timeout(config.operation_timeout, peripheral.connect())
823                    .await
824                    .map_err(|_| ble_timeout("connecting"))?
825                    .map_err(ble_error)?;
826            }
827            tokio::time::timeout(config.operation_timeout, peripheral.discover_services())
828                .await
829                .map_err(|_| ble_timeout("discovering services"))?
830                .map_err(ble_error)?;
831
832            let frame_in_uuid = uuid::Uuid::from_u128(umsh_ulcp::gatt::FRAME_IN_UUID);
833            let frame_out_uuid = uuid::Uuid::from_u128(umsh_ulcp::gatt::FRAME_OUT_UUID);
834            let characteristics = peripheral.characteristics();
835            let frame_in = characteristics
836                .iter()
837                .find(|characteristic| characteristic.uuid == frame_in_uuid)
838                .cloned()
839                .ok_or(UlcpError::Protocol("missing BLE Frame In"))?;
840            let frame_out = characteristics
841                .iter()
842                .find(|characteristic| characteristic.uuid == frame_out_uuid)
843                .cloned()
844                .ok_or(UlcpError::Protocol("missing BLE Frame Out"))?;
845
846            let mut stream =
847                tokio::time::timeout(config.operation_timeout, peripheral.notifications())
848                    .await
849                    .map_err(|_| ble_timeout("opening notifications"))?
850                    .map_err(ble_error)?;
851            let (tx, notifications) = tokio::sync::mpsc::channel(32);
852            tokio::spawn(async move {
853                while let Some(notification) = stream.next().await {
854                    if notification.uuid == frame_out_uuid
855                        && tx.send(notification.value).await.is_err()
856                    {
857                        break;
858                    }
859                }
860            });
861            // This security-gated CCCD write is the protocol attach edge.
862            // Pairing prompts are mediated by the host OS.
863            tokio::time::timeout(config.pairing_timeout, peripheral.subscribe(&frame_out))
864                .await
865                .map_err(|_| ble_timeout("subscribing to Frame Out"))?
866                .map_err(ble_error)?;
867            Ok::<_, UlcpError>((frame_in, notifications))
868        }
869        .await;
870
871        let (frame_in, notifications) = match setup {
872            Ok(setup) => setup,
873            Err(error) => {
874                // Failed setup must not leave the single-connection device
875                // occupied and invisible to the next retry.
876                let _ = tokio::time::timeout(Duration::from_secs(1), peripheral.disconnect()).await;
877                return Err(error);
878            }
879        };
880
881        Ok(Self {
882            peripheral,
883            frame_in,
884            receiver: BleNotificationReceiver::new(notifications),
885            segment_payload: config.segment_payload,
886            operation_timeout: config.operation_timeout,
887        })
888    }

890    /// Capture the backend's view of a failed link, then make a bounded
891    /// best-effort disconnect so a subsequent discovery does not inherit a
892    /// stale CoreBluetooth/BlueZ connection object.
893    async fn diagnose_and_disconnect(&self, failure: String) -> UlcpError {
894        use btleplug::api::Peripheral as _;
895
896        let connected = match tokio::time::timeout(
897            Duration::from_secs(2),
898            self.peripheral.is_connected(),
899        )
900        .await
901        {
902            Ok(Ok(value)) => value.to_string(),
903            Ok(Err(error)) => format!("error({error})"),
904            Err(_) => "query-timeout".into(),
905        };
906        let cleanup = match tokio::time::timeout(
907            Duration::from_secs(2),
908            self.peripheral.disconnect(),
909        )
910        .await
911        {
912            Ok(Ok(())) => "ok".into(),
913            Ok(Err(error)) => format!("error({error})"),
914            Err(_) => "timeout".into(),
915        };
916        UlcpError::Transport(format!(
917            "{failure}; backend is_connected={connected}; disconnect cleanup={cleanup}"
918        ))
919    }

924    async fn send_frame(&mut self, frame: &[u8]) -> Result<(), UlcpError> {
925        use btleplug::api::{Peripheral as _, WriteType};
926
927        for segment in umsh_ulcp::gatt::segments(frame, self.segment_payload) {
928            let mut value = vec![0; segment.payload().len() + 1];
929            segment
930                .write_to(&mut value)
931                .expect("segment destination is exactly sized");
932            let write = tokio::time::timeout(
933                self.operation_timeout,
934                self.peripheral
935                    .write(&self.frame_in, &value, WriteType::WithResponse),
936            )
937            .await;
938            match write {
939                Ok(Ok(())) => {}
940                Ok(Err(error)) => {
941                    return Err(self
942                        .diagnose_and_disconnect(format!("BLE Frame In write failed: {error}"))
943                        .await);
944                }
945                Err(_) => {
946                    return Err(self
947                        .diagnose_and_disconnect("BLE timed out while writing Frame In".into())
948                        .await);
949                }
950            }
951        }
952        Ok(())
953    }

995/// The two relationships a host can have with a device.
996///
997/// They are different things, and conflating them is how one phone
998/// administering ten repeaters ends up claiming all of them. Tethering is
999/// a transient local relationship — at most one at a time, re-established
1000/// on every attach, invisible to the mesh. Administration is
1001/// configuration of the device's own identity and behavior, which
1002/// outlives any particular host.
1003#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
1004pub enum AttachMode {
1005    /// This host is the device's tethered host: it provisions the host
1006    /// domain and the device filters, queues and acknowledges for it.
1007    #[default]
1008    Tethered,
1009    /// This host is administering the device without tethering to it.
1010    /// Host-domain writes are refused.
1011    Administrative,
1012}

1015where
1016    L: FrameLink,

1019        Self {

1021            config,
1022            rx_queue: VecDeque::new(),
1023            responses: VecDeque::new(),
1024            prop_events: VecDeque::new(),
1025            seen_reset: None,
1026            max_frame_size: 0,
1027            t_frame_ms: 0,
1028            dev_version: String::new(),
1029            dev_model: None,
1030            boot_status: Status::RESET_UNKNOWN,
1031            tids: TidAllocator::new(),
1032            trace: None,
1033            mode: AttachMode::Tethered,
1034        }
1035    }

1042    /// Refuse a host-domain write on an administrative handle.
1043    fn require_tethered(&self, key: u32) -> Result<(), UlcpError> {

1045            key,
1046            prop::HOST_KEY
1047                | prop::HOST_CHANNEL_KEYS

1059    /// the RF configuration, and enable the PHY.

1073    /// Attach to an already-operating device as its **tethered host**:
1074    /// the one host whose traffic it filters, queues and acknowledges.
1075    ///
1076    /// This is the full-protocol attach (spec §Attach, Detach, and
1077    /// Synchronization): attach implies no known state, so the host
1078    /// synchronizes by fetching. Only the identity handshake runs here
1079    /// — retained `PROP_LAST_STATUS` (the reset cause, preserved for
1080    /// [`Self::boot_status`] and [`Self::sync`]), the protocol version
1081    /// check, `PROP_DEV_VERSION`, and `PROP_PHY_MTU`. The PHY keeps
1082    /// whatever configuration and enable state it had; queued frames
1083    /// and provisioning are untouched. Follow with [`Self::sync`],
1084    /// [`Self::provision`], and drain the queue when ready.
1085    ///

1087    /// not intend to tether to — one phone administering ten repeaters
1088    /// must not write `PROP_HOST_KEY` on any of them.

1090        Self::attach_with_mode(link, config, AttachMode::Tethered).await
1091    }

1094    /// configure its own identity, radio, and behavior without becoming
1095    /// its host.
1096    ///
1097    /// Commissioning and tethering are different relationships and this
1098    /// is the difference made mechanical. The handle refuses every
1099    /// host-domain write — `PROP_HOST_KEY`, the host key tables, the
1100    /// filter table, the delegation policy, and [`Self::provision`] —
1101    /// with [`UlcpError::AdministrativeAttach`]. Everything
1102    /// else, including the device identity and the saved snapshot, works
1103    /// normally.
1104    ///
1105    /// A device may be administered by many hosts over its lifetime and
1106    /// tethered to at most one at a time; nothing about administering it
1107    /// disturbs whichever host it is currently serving.

1109        link: L,

1115    async fn attach_with_mode(
1116        link: L,
1117        config: UlcpDeviceConfig,
1118        mode: AttachMode,
1119    ) -> Result<Self, UlcpError> {
1120        let mut radio = Self::bare(link, config);
1121        radio.mode = mode;
1122        // Reading LAST_STATUS does not overwrite it, so sync() still
1123        // sees a retained reset code after this handshake.
1124        let boot_status = radio.get_prop(prop::LAST_STATUS).await?;
1125        radio.boot_status = decode_status(&boot_status);
1126
1127        let version = radio.get_prop(prop::PROTOCOL_VERSION).await?;
1128        if version.first().copied() != Some(ids::PROTOCOL_MAJOR_VERSION) {
1129            return Err(UlcpError::Protocol("protocol major version mismatch"));
1130        }
1131        let dev_version = radio.get_prop(prop::DEV_VERSION).await?;
1132        radio.dev_version = String::from_utf8_lossy(&dev_version)
1133            .trim_end_matches('\0')
1134            .to_owned();
1135        // `PROP_DEV_MODEL` is OPTIONAL, so a refusal is an answer — it
1136        // means "this device does not name its hardware" — and must not
1137        // fail the attach the way a missing DEV_VERSION would.
1138        radio.dev_model = radio.get_prop_string_opt(prop::DEV_MODEL).await;
1139
1140        let mtu = radio.get_prop(prop::PHY_MTU).await?;
1141        let [mtu_lo, mtu_hi, ..] = mtu[..] else {
1142            return Err(UlcpError::Protocol("malformed PROP_PHY_MTU"));
1143        };
1144        radio.max_frame_size = usize::from(u16::from_le_bytes([mtu_lo, mtu_hi]));
1145        if radio.max_frame_size == 0 {
1146            return Err(UlcpError::Protocol("device advertised zero MTU"));
1147        }
1148        radio.t_frame_ms = lora_airtime_ms(
1149            radio.config.spreading_factor,
1150            radio.config.bandwidth_hz,
1151            radio.config.coding_rate_denom,
1152            radio.max_frame_size,
1153        )
1154        .max(1);
1155        Ok(radio)
1156    }

1576    async fn initialize(&mut self) -> Result<(), UlcpError> {
1577        // The reset-status property is deliberately read before CMD_RST. The
1578        // protocol requires the device to retain its hardware boot cause for this
1579        // first query; CMD_RST would replace it with RESET_SOFTWARE.
1580        let boot_status = self.get_prop(prop::LAST_STATUS).await?;
1581        self.boot_status = decode_status(&boot_status);
1582
1583        // Reset and wait for the reset notification. The TID is
1584        // ignored for CMD_RST; the notification is unsolicited.
1585        let mut buf = [0u8; 2];
1586        let len = frame::reset(&mut buf, TID_UNSOLICITED)
1587            .map_err(|_| UlcpError::Protocol("frame encode"))?;
1588        self.send(&buf[..len]).await?;
1589        let deadline = Instant::now() + self.config.response_timeout;
1590        self.wait_reset(deadline).await?;
1591
1592        // Reject devices speaking an incompatible protocol revision.
1593        let version = self.get_prop(prop::PROTOCOL_VERSION).await?;
1594        if version.first().copied() != Some(ids::PROTOCOL_MAJOR_VERSION) {
1595            return Err(UlcpError::Protocol("protocol major version mismatch"));
1596        }
1597
1598        let dev_version = self.get_prop(prop::DEV_VERSION).await?;
1599        self.dev_version = String::from_utf8_lossy(&dev_version)
1600            .trim_end_matches('\0')
1601            .to_owned();
1602        self.dev_model = self.get_prop_string_opt(prop::DEV_MODEL).await;
1603
1604        let mtu = self.get_prop(prop::PHY_MTU).await?;
1605        let [mtu_lo, mtu_hi, ..] = mtu[..] else {
1606            return Err(UlcpError::Protocol("malformed PROP_PHY_MTU"));
1607        };
1608        self.max_frame_size = usize::from(u16::from_le_bytes([mtu_lo, mtu_hi]));
1609        if self.max_frame_size == 0 {
1610            return Err(UlcpError::Protocol("device advertised zero MTU"));
1611        }
1612
1613        let config = self.config.clone();
1614        self.set_prop(prop::PHY_FREQ, &config.freq_khz.to_le_bytes())
1615            .await?;
1616        self.set_prop(prop::PHY_LORA_BW, &config.bandwidth_hz.to_le_bytes())
1617            .await?;
1618        self.set_prop(prop::PHY_LORA_SF, &[config.spreading_factor])
1619            .await?;
1620        self.set_prop(prop::PHY_LORA_CR, &[config.coding_rate_denom])
1621            .await?;
1622        self.set_prop(prop::PHY_TX_POWER, &[config.tx_power_dbm as u8])
1623            .await?;
1624        self.set_prop(prop::PHY_LORA_SW, &config.sync_word.to_le_bytes())
1625            .await?;
1626        self.set_prop(prop::PHY_ENABLED, &[1]).await?;
1627
1628        self.t_frame_ms = lora_airtime_ms(
1629            config.spreading_factor,
1630            config.bandwidth_hz,
1631            config.coding_rate_denom,
1632            self.max_frame_size,
1633        )
1634        .max(1);
1635        Ok(())
1636    }

1698    /// Write several properties in order in one exchange via
1699    /// `CMD_PROP_MULTI_SET`.
1700    ///
1701    /// The device applies the writes strictly in order and stops at the
1702    /// first failure, so the returned entries cover a prefix of the
1703    /// request: each is either the authoritative value the device echoed
1704    /// or the status that ended the sequence. A short reply whose last
1705    /// entry succeeded means the device ran out of response space; the
1706    /// caller reissues the remainder.
1707    pub async fn set_props(
1708        &mut self,
1709        entries: &[(u32, Vec<u8>)],
1710    ) -> Result<Vec<MultiValue>, UlcpError> {
1711        let mut capacity = 8;
1712        for (key, value) in entries {
1713            self.require_tethered(*key)?;
1714            capacity += value.len() + pui::MAX_LEN * 2;
1715        }
1716        let borrowed: Vec<(u32, &[u8])> = entries
1717            .iter()
1718            .map(|(key, value)| (*key, value.as_slice()))
1719            .collect();
1720        let tid = self.alloc_tid();
1721        let mut buf = vec![0u8; capacity];
1722        let len = frame::prop_multi_set(&mut buf, tid, &borrowed)
1723            .map_err(|_| UlcpError::Protocol("frame encode"))?;
1724        self.send(&buf[..len]).await?;
1725        self.finish_multi_transaction(tid).await
1726    }

1731        let deadline = Instant::now() + self.config.response_timeout;

1821    /// As [`Self::queue_drain`], invoking `on_frame` with each frame
1822    /// (data, trailing metadata bytes) delivered before completion —
1823    /// buffered and interleaved live frames alike. The callback sees
1824    /// **every** such frame: an device queue larger than this driver's
1825    /// bounded receive buffer drains losslessly through it. Frames are
1826    /// additionally queued for [`Radio::poll_receive`], where the
1827    /// bounded buffer's oldest-dropped policy still applies.
1828    pub async fn queue_drain_with(
1829        &mut self,
1830        mut on_frame: impl FnMut(&[u8], &[u8]),
1831    ) -> Result<(), UlcpError> {
1832        let tid = self.alloc_tid();
1833        let mut buf = [0u8; 4];
1834        let len =
1835            frame::queue_drain(&mut buf, tid).map_err(|_| UlcpError::Protocol("frame encode"))?;
1836        self.send(&buf[..len]).await?;
1837
1838        let deadline = Instant::now() + self.config.response_timeout;
1839        loop {
1840            while let Some(response) = self.responses.pop_front() {
1841                if response.tid != tid {
1842                    continue;
1843                }
1844                if response.kind == ResponseKind::Is && response.key == prop::LAST_STATUS {
1845                    let status = decode_status(&response.value);
1846                    return if status == Status::OK {
1847                        Ok(())
1848                    } else {
1849                        Err(UlcpError::Status(status))
1850                    };
1851                }
1852                return Err(UlcpError::Protocol("unexpected drain response"));
1853            }
1854            if let Some(status) = self.seen_reset.take() {
1855                return Err(UlcpError::UnexpectedReset(status));
1856            }
1857            // Deliver at ingest time: each read that queued a stream
1858            // frame reports it immediately, so the callback cannot
1859            // miss frames the bounded receive buffer evicts mid-drain.

1863                    .back()

1987    /// contact), the capability list, the configured host identity —
1988    /// yielding an ownership verdict against `expected_host_key` — and
1989    /// the state each advertised capability grants, all in digest form.
1990    ///
1991    /// The host must decide ownership before treating queued data as
1992    /// its own: [`HostOwnership::OtherHost`] means the queue and
1993    /// provisioning belong to another identity.
1994    pub async fn sync(
1995        &mut self,
1996        expected_host_key: Option<&[u8; 32]>,
1997    ) -> Result<DeviceSync, UlcpError> {
1998        // Step 1: the retained status, before any other command can
1999        // overwrite a reset code.
2000        let last_status = decode_status(&self.get_prop(prop::LAST_STATUS).await?);
2001        let capabilities = self.capabilities().await?;
2002        let has = |capability: u32| capabilities.contains(&capability);
2003
2004        // Step 2: ownership.
2005        let (host_key, ownership) = if has(cap::HOST_FILTER) {
2006            let value = self.get_prop(prop::HOST_KEY).await?;

2107    /// Establish `desired` as the device's complete host domain,
2108    /// writing every part of it unconditionally.
2109    ///
2110    /// **This does not compare and patch, and that is deliberate.** Key
2111    /// tables read back in lossy form only: the device reports channel
2112    /// identifiers and peer public keys, never key material. An
2113    /// administrator can replace a peer's `K_enc`/`K_mic` without
2114    /// changing anything observable, so no comparison over the readable
2115    /// surface can detect it — and a digest over the secret state would
2116    /// mean deriving a readable value from key material, which is worse
2117    /// than the problem. The host asserts what it wants; it does not
2118    /// reason about what the device already holds.
2119    ///
2120    /// Removals still come from comparison, and that is not a
2121    /// contradiction: *membership* is readable even though key material
2122    /// is not. So this reads the digest lists, removes what `desired`
2123    /// omits, and writes everything `desired` contains regardless of
2124    /// what came back.
2125    ///
2126    /// The host domain is volatile across power cycles, so the usual
2127    /// case is a device that has just rebooted and holds nothing. When
2128    /// it has *not* rebooted the rewrite is redundant — that is the
2129    /// point. Correctness must not depend on detecting which case this
2130    /// is, because reboot detection would also have to cover partial
2131    /// provisioning, another administrator having intervened, and future
2132    /// device behavior changes.
2133    ///
2134    /// Each individual write is transactional on the device (spec
2135    /// §Mutation Atomicity); the *sequence* is not (see the ULCP
2136    /// transition plan, decision 7). An interrupted call leaves a
2137    /// mixture, which the next call repairs by rewriting everything.
2138    ///
2139    /// Provisioning is per item rather than per table wherever the table
2140    /// can grow: a whole peer table stops fitting in a frame at the
2141    /// fifth entry.
2142    pub async fn provision(
2143        &mut self,
2144        desired: &HostProvisioning,
2145    ) -> Result<ProvisionReport, UlcpError> {
2146        self.require_tethered(prop::HOST_KEY)?;
2147        let mut report = ProvisionReport::default();
2148        let current_key = self.get_prop(prop::HOST_KEY).await?;
2149        // A host-key write to a different value resets the whole host
2150        // domain on the device, so everything below lands on an empty
2151        // one; writing the same key is idempotent and has no effect.
2152        if current_key.as_slice() != desired.host_key.as_slice() {
2153            report.host_replaced = true;
2154        }
2155        self.set_prop(prop::HOST_KEY, &desired.host_key).await?;
2156
2157        // Filters: item and digest forms are identical and the whole
2158        // table is small, so one atomic write says everything.
2159        let mut table = Vec::new();
2160        for filter in &desired.filters {
2161            let mut item = [0u8; items::Filter::MAX_WIRE_LEN];
2162            let item_len = filter
2163                .encode(&mut item)
2164                .map_err(|_| UlcpError::Protocol("filter encode"))?;
2165            let mut prefixed = [0u8; items::Filter::MAX_WIRE_LEN + 2];
2166            let prefixed_len = items::encode_prefixed_item(&item[..item_len], &mut prefixed)
2167                .map_err(|_| UlcpError::Protocol("filter encode"))?;
2168            table.extend_from_slice(&prefixed[..prefixed_len]);
2169        }
2170        self.set_prop(prop::HOST_RX_FILTERS, &table).await?;
2171        report.filters_replaced = true;
2172
2173        // Channel keys: the remove selector is the key itself, which we
2174        // hold for everything we want and not for anything we do not, so
2175        // shedding an unknown channel needs the whole-table form. That
2176        // table is bounded and small enough to send.
2177        let engine = CryptoEngine::new(SoftwareAes, SoftwareSha256);
2178        let desired_ids: Vec<[u8; items::CHANNEL_ID_LEN]> = desired
2179            .channel_keys
2180            .iter()
2181            .map(|key| engine.derive_channel_id(&ChannelKey(*key)).0)
2182            .collect();

2184            Vec::new()
2185        } else {
2186            decode_fixed_list::<{ items::CHANNEL_ID_LEN }>(
2187                &self.get_prop(prop::HOST_CHANNEL_KEYS).await?,
2188                "malformed PROP_HOST_CHANNEL_KEYS digest",
2189            )?
2190        };
2191        if current_ids.iter().any(|id| !desired_ids.contains(id)) {
2192            let table: Vec<u8> = desired.channel_keys.concat();
2193            self.set_prop(prop::HOST_CHANNEL_KEYS, &table).await?;
2194            report.channels_replaced = true;
2195        } else {
2196            // Insert every desired key, including ones already reported.
2197            // A channel key *is* its own item, so a duplicate insert
2198            // asserts a state that already holds: `STATUS_ALREADY` says
2199            // "the entry is present", which is what was asked for, and
2200            // is treated as success. (Peers differ — a matching public
2201            // key replaces the entry's key material — so their inserts
2202            // never report it.)
2203            for key in &desired.channel_keys {
2204                match self.insert_prop_item(prop::HOST_CHANNEL_KEYS, key).await {
2205                    Ok(_) => report.channels_inserted += 1,
2206                    Err(UlcpError::Status(Status::ALREADY)) => {}
2207                    Err(error) => return Err(error),
2208                }
2209            }
2210        }
2211
2212        // Peers: remove by comparison over the readable public keys,
2213        // then insert every desired entry unconditionally. The device
2214        // reconciles rather than rebuilding, so re-inserting a peer it
2215        // already holds preserves that peer's replay baseline.
2216        let current_peers = if report.host_replaced {
2217            Vec::new()
2218        } else {
2219            decode_fixed_list::<{ items::PUBLIC_KEY_LEN }>(
2220                &self.get_prop(prop::HOST_PEER_KEYS).await?,
2221                "malformed PROP_HOST_PEER_KEYS digest",
2222            )?
2223        };
2224        for existing in &current_peers {
2225            if !desired
2226                .peer_keys
2227                .iter()
2228                .any(|entry| entry.public_key == *existing)
2229            {
2230                self.remove_prop_item(prop::HOST_PEER_KEYS, existing)
2231                    .await?;
2232                report.peers_removed += 1;
2233            }
2234        }
2235        for entry in &desired.peer_keys {
2236            let mut item = [0u8; items::PeerKeyEntry::WIRE_LEN];
2237            entry
2238                .encode(&mut item)
2239                .map_err(|_| UlcpError::Protocol("peer entry encode"))?;
2240            self.insert_prop_item(prop::HOST_PEER_KEYS, &item).await?;
2241            report.peers_inserted += 1;
2242        }
2243
2244        // Delegation policy last, once the keys it depends on exist.
2245        self.set_prop(prop::HOST_AUTO_ACK, &[desired.auto_ack as u8])
2246            .await?;
2247        report.auto_ack_changed = true;
2248        Ok(report)
2249    }

2272            .map_err(|_| UlcpError::Protocol("frame encode"))?;

2276            .await?;

2278            .map_err(|_| UlcpError::Protocol("malformed PROP_DEV_KEY"))
2279    }

2281    async fn finish_prop_transaction(
2282        &mut self,
2283        tid: u8,
2284        key: u32,
2285        policy: PropResponsePolicy,
2286    ) -> Result<Vec<u8>, UlcpError> {
2287        let deadline = Instant::now() + self.config.response_timeout;
2288        let response = self.wait_response(tid, deadline).await?;
2289        if response.kind != ResponseKind::Is {
2290            return Err(UlcpError::Protocol(

2292            ));
2293        }

2295            (PropResponsePolicy::Value, response_key) if response_key == key => Ok(response.value),
2296            (PropResponsePolicy::StatusOnly, prop::LAST_STATUS) => {
2297                let status = decode_status(&response.value);
2298                if status == Status::OK {
2299                    Ok(Vec::new())
2300                } else {
2301                    Err(UlcpError::Status(status))
2302                }
2303            }
2304            (PropResponsePolicy::Value, prop::LAST_STATUS) => {
2305                let status = decode_status(&response.value);
2306                if status == Status::OK {
2307                    Err(UlcpError::Protocol(
2308                        "unexpected status-only property response",
2309                    ))
2310                } else {
2311                    Err(UlcpError::Status(status))
2312                }
2313            }
2314            _ => Err(UlcpError::Protocol("response for unexpected property")),

2349    /// Sort a complete ULCP frame into the receive queue, response queue,
2350    /// or the reset flag; returns whether a stream frame was queued
2351    /// (the back of `rx_queue` is then the new packet). Malformed
2352    /// frames are dropped.
2353    fn ingest_frame(&mut self, frame_bytes: &[u8]) -> bool {
2354        if let Some(trace) = &mut self.trace {
2355            trace(TraceDirection::DeviceToHost, &describe_frame(frame_bytes));
2356        }
2357        let Ok(frame) = Frame::parse(frame_bytes) else {
2358            return false;
2359        };
2360        match frame.command() {
2361            Some(Cmd::StrRecv) => {
2362                let Ok(payload) = StreamPayload::parse(frame.payload) else {
2363                    return false;
2364                };
2365                if payload.stream != stream::PHY_RAW {
2366                    return false;
2367                }
2368                let meta = RxMeta::decode(payload.metadata).unwrap_or_default();
2369                if self.rx_queue.len() >= RX_QUEUE_DEPTH {
2370                    self.rx_queue.pop_front();
2371                }
2372                self.rx_queue.push_back(RxPacket {
2373                    data: payload.data.to_vec(),
2374                    meta,
2375                    raw_meta: payload.metadata.to_vec(),
2376                });
2377                return true;
2378            }
2379            Some(Cmd::PropIs) => self.ingest_prop_notification(ResponseKind::Is, &frame),
2380            Some(Cmd::PropInserted) => {
2381                self.ingest_prop_notification(ResponseKind::Inserted, &frame)
2382            }
2383            Some(Cmd::PropRemoved) => self.ingest_prop_notification(ResponseKind::Removed, &frame),
2384            // `CMD_PROP_ARE` carries entries rather than one key and
2385            // value, so it is queued whole and decoded by whichever
2386            // multi-property transaction is waiting on the TID. It is
2387            // never unsolicited, so a TID-0 one is dropped.
2388            Some(Cmd::PropAre) => {
2389                let tid = frame.header.tid();
2390                if tid != TID_UNSOLICITED {
2391                    if self.responses.len() >= RESPONSE_QUEUE_DEPTH {
2392                        self.responses.pop_front();
2393                    }
2394                    self.responses.push_back(Response {
2395                        tid,
2396                        kind: ResponseKind::Are,
2397                        key: prop::LAST_STATUS,
2398                        value: frame.payload.to_vec(),
2399                    });
2400                }
2401            }
2402            _ => {}
2403        }
2404        false
2405    }

2407    fn ingest_prop_notification(&mut self, kind: ResponseKind, frame: &Frame<'_>) {
2408        let Ok(notification) = PropertyNotification::from_frame(frame) else {
2409            return;
2410        };
2411        // The caller dispatches from the parsed command; keep that assertion
2412        // explicit so future command additions cannot be misclassified.
2413        if notification.kind != kind {
2414            return;
2415        }
2416        let tid = notification.tid;
2417        if tid != TID_UNSOLICITED {
2418            if self.responses.len() >= RESPONSE_QUEUE_DEPTH {
2419                self.responses.pop_front();
2420            }
2421            self.responses.push_back(Response {
2422                tid,
2423                kind,
2424                key: notification.key,
2425                value: notification.value.to_vec(),
2426            });
2427            return;
2428        }
2429        // Unsolicited `PROP_LAST_STATUS` is a reset notice or an
2430        // operation status, not a property update to retain.
2431        if kind == ResponseKind::Is && notification.key == prop::LAST_STATUS {
2432            let status = decode_status(notification.value);
2433            if status.is_reset() {
2434                self.seen_reset = Some(status);
2435            }
2436            return;
2437        }
2438        let event = match kind {
2439            ResponseKind::Is => PropEvent::Is {
2440                key: notification.key,
2441                value: notification.value.to_vec(),
2442            },
2443            ResponseKind::Inserted => PropEvent::Inserted {
2444                key: notification.key,
2445                digest: notification.value.to_vec(),
2446            },
2447            ResponseKind::Removed => PropEvent::Removed {
2448                key: notification.key,
2449                digest: notification.value.to_vec(),
2450            },
2451            // Unreachable: `PropertyNotification` refuses the
2452            // multi-property form, and this path only sees notifications
2453            // it parsed.
2454            ResponseKind::Are => return,
2455        };
2456        if self.prop_events.len() >= PROP_EVENT_DEPTH {
2457            self.prop_events.pop_front();
2458        }
2459        self.prop_events.push_back(event);
2460    }

2470    /// Read from the stream until the response for `tid` arrives.
2471    ///
2472    /// Frames received meanwhile are queued for [`Radio::poll_receive`].
2473    async fn wait_response(&mut self, tid: u8, deadline: Instant) -> Result<Response, UlcpError> {
2474        loop {
2475            // Drain responses before honoring a reset notice: if both
2476            // arrived in one read, the response was sent first and the
2477            // command did complete. The reset stays latched for the
2478            // next receive poll.
2479            while let Some(response) = self.responses.pop_front() {
2480                if response.tid == tid {
2481                    return Ok(response);
2482                }
2483                // A stale response from an earlier timed-out
2484                // transaction; drop it.
2485            }
2486            if let Some(status) = self.seen_reset.take() {
2487                return Err(UlcpError::UnexpectedReset(status));
2488            }
2489            self.read_more(deadline).await?;
2490        }
2491    }

2493    /// Read until the device announces a reset via `PROP_LAST_STATUS`.
2494    async fn wait_reset(&mut self, deadline: Instant) -> Result<Status, UlcpError> {
2495        loop {
2496            if let Some(status) = self.seen_reset.take() {
2497                return Ok(status);
2498            }
2499            // Accept a reset notice even if the device attached a TID.
2500            while let Some(response) = self.responses.pop_front() {
2501                if response.kind == ResponseKind::Is && response.key == prop::LAST_STATUS {
2502                    let status = decode_status(&response.value);
2503                    if status.is_reset() {
2504                        return Ok(status);
2505                    }
2506                }
2507            }
2508            self.read_more(deadline).await?;
2509        }
2510    }

2512    /// Read and ingest one frame before `deadline`; returns whether it
2513    /// was a stream frame that is now the back of `rx_queue`.
2514    async fn read_more(&mut self, deadline: Instant) -> Result<bool, UlcpError> {
2515        let now = Instant::now();
2516        if now >= deadline {
2517            return Err(UlcpError::Timeout);
2518        }
2519        let frame = match tokio::time::timeout(deadline - now, self.link.recv_frame()).await {
2520            Err(_elapsed) => return Err(UlcpError::Timeout),
2521            Ok(Err(error)) => return Err(error),
2522            Ok(Ok(frame)) => frame,
2523        };
2524        Ok(self.ingest_frame(&frame))
2525    }

2527    fn pop_rx(&mut self, buf: &mut [u8]) -> Option<RxInfo> {
2528        let packet = self.rx_queue.pop_front()?;
2529        let len = packet.data.len().min(buf.len());
2530        buf[..len].copy_from_slice(&packet.data[..len]);
2531        // `RX_FLAG_SELF_TX` says the *device* transmitted the frame. To
2532        // the host that is not a self-transmission at all — the device is
2533        // a separate node, and its frames are as real as any peer's. What
2534        // carries across is that nothing measured the frame on its way
2535        // here, which is exactly what `Backhaul` means on this side of
2536        // the link. Calling it `LocalTx` would tell the host's own MAC
2537        // that its antenna had already sent the frame, and it would
2538        // refuse to forward it.
2539        let self_tx = BufferedRxMeta::decode(&packet.raw_meta)
2540            .is_ok_and(|meta| meta.flags & RX_FLAG_SELF_TX != 0);
2541        Some(RxInfo {
2542            len,
2543            rssi: packet.meta.rssi_dbm.unwrap_or(0),
2544            snr: Snr::from_centibels(packet.meta.snr_cb.unwrap_or(0)),
2545            lqi: packet.meta.lqi,
2546            origin: if self_tx {
2547                RxOrigin::Backhaul
2548            } else {
2549                RxOrigin::Air
2550            },
2551        })
2552    }

2554    /// Issue one `CMD_STR_SEND` with an already-encoded metadata block
2555    /// and await its confirmation, retrying while CCA reports the
2556    /// channel busy and `cca_deadline` has not passed.
2557    async fn send_confirmed(
2558        &mut self,
2559        data: &[u8],
2560        metadata: &[u8],
2561        cca_deadline: Option<Instant>,
2562    ) -> Result<(), TxError<UlcpError>> {
2563        loop {
2564            let tid = self.alloc_tid();
2565            let mut frame_buf = vec![0u8; data.len() + metadata.len() + 16];
2566            let frame_len = frame::str_send(&mut frame_buf, tid, stream::PHY_RAW, data, metadata)
2567                .map_err(|_| TxError::Io(UlcpError::Protocol("frame encode")))?;
2568            self.send(&frame_buf[..frame_len])
2569                .await
2570                .map_err(TxError::Io)?;
2571
2572            // The confirmation arrives only after the frame is on the
2573            // air (or definitively failed), so allow for airtime.
2574            let deadline = Instant::now()
2575                + self.config.response_timeout
2576                + Duration::from_millis(u64::from(self.t_frame_ms) * 2);
2577            let response = self
2578                .wait_response(tid, deadline)
2579                .await
2580                .map_err(TxError::Io)?;
2581            if response.kind != ResponseKind::Is || response.key != prop::LAST_STATUS {
2582                return Err(TxError::Io(UlcpError::Protocol(
2583                    "unexpected transmit response",
2584                )));
2585            }
2586            match decode_status(&response.value) {
2587                Status::OK => return Ok(()),
2588                Status::CCA_FAILURE => match cca_deadline {
2589                    Some(deadline) if Instant::now() < deadline => {
2590                        tokio::time::sleep(CCA_RETRY_DELAY).await;
2591                    }
2592                    _ => return Err(TxError::CadTimeout),
2593                },
2594                status => return Err(TxError::Io(UlcpError::Status(status))),
2595            }
2596        }
2597    }

2600    /// block, byte for byte.
2601    ///
2602    /// [`Radio::transmit`] composes the metadata from [`TxOptions`],
2603    /// which is what a MAC wants. A bridge does not: it relays frames
2604    /// whose transmit parameters were decided elsewhere, and must be
2605    /// able to put exactly those bytes on the wire — including fields
2606    /// [`TxOptions`] has no vocabulary for, such as a power override.
2607    ///
2608    /// The channel-access retry budget comes from the metadata itself:
2609    /// with `TX_FLAG_NOCCA` clear the device performs CCA and a busy
2610    /// channel fails immediately with [`TxError::CadTimeout`], leaving
2611    /// the retry policy to the caller.
2612    pub async fn transmit_raw_with_meta(
2613        &mut self,
2614        data: &[u8],
2615        metadata: &[u8],
2616    ) -> Result<(), TxError<UlcpError>> {
2617        if data.len() > self.max_frame_size {
2618            return Err(TxError::Io(UlcpError::FrameTooLarge(data.len())));
2619        }
2620        let skips_cca = metadata
2621            .get(1)
2622            .is_some_and(|flags| flags & TX_FLAG_NOCCA != 0);
2623        let cca_deadline = (!skips_cca).then(Instant::now);
2624        self.send_confirmed(data, metadata, cca_deadline).await
2625    }

2627    /// Poll for one inbound frame, preserving its metadata bytes.
2628    ///
2629    /// [`Radio::poll_receive`] decodes the metadata into [`RxInfo`],
2630    /// which cannot represent it faithfully: the "unsupported" sentinels
2631    /// collapse to zero and the buffered-frame extension is discarded
2632    /// entirely. A bridge relays the metadata rather than interpreting
2633    /// it, so it needs the bytes.
2634    pub fn poll_receive_raw(
2635        &mut self,
2636        cx: &mut core::task::Context<'_>,
2637    ) -> core::task::Poll<Result<RawRxFrame, UlcpError>> {
2638        loop {
2639            if let Some(status) = self.seen_reset.take() {
2640                return core::task::Poll::Ready(Err(UlcpError::UnexpectedReset(status)));
2641            }
2642            if let Some(packet) = self.rx_queue.pop_front() {
2643                return core::task::Poll::Ready(Ok(RawRxFrame {
2644                    data: packet.data,
2645                    metadata: packet.raw_meta,
2646                }));
2647            }
2648
2649            match self.link.poll_recv_frame(cx) {
2650                core::task::Poll::Ready(Ok(frame)) => {
2651                    self.ingest_frame(&frame);
2652                }
2653                core::task::Poll::Ready(Err(error)) => return core::task::Poll::Ready(Err(error)),
2654                core::task::Poll::Pending => return core::task::Poll::Pending,
2655            }
2656        }
2657    }

2720    /// Transmit one frame and await the device's confirmation.
2721    ///
2722    /// A confirmed transmit blocks the caller for up to
2723    /// `response_timeout + 2 × t_frame_ms` while the frame goes out on air. This
2724    /// is inherent to the half-duplex [`Radio::transmit`] contract and a real
2725    /// radio behaves the same way. Frames the device receives during this window
2726    /// are not lost — they are queued (see [`wait_response`](Self::wait_response)
2727    /// → [`ingest`](Self::ingest)) and surface on the next
2728    /// [`poll_receive`](Radio::poll_receive). MAC-layer timers (ACK timeouts,
2729    /// retransmit deadlines) cannot advance while this future is pending, but
2730    /// they are only *delayed*, not missed: the coordinator re-evaluates every
2731    /// deadline against the current clock as soon as `transmit` returns, so a
2732    /// deadline that came due mid-transmit fires immediately afterward.
2733    async fn transmit(
2734        &mut self,
2735        data: &[u8],
2736        options: TxOptions,
2737    ) -> Result<(), TxError<Self::Error>> {
2738        if data.len() > self.max_frame_size {
2739            return Err(TxError::Io(UlcpError::FrameTooLarge(data.len())));
2740        }
2741
2742        // The device performs CCA itself; the CAD policy becomes a host-side
2743        // retry budget around `STATUS_CCA_FAILURE`.
2744        let mut meta = TxMeta::default();
2745        let cca_deadline = match options.cad {
2746            CadPolicy::Skip => {
2747                meta.flags |= TX_FLAG_NOCCA;
2748                None
2749            }
2750            // Gate is a single attempt: a zero-length budget, so a busy channel
2751            // fails immediately with CadTimeout.
2752            CadPolicy::Gate => Some(Instant::now()),
2753            CadPolicy::RetryFor { timeout_ms } => {
2754                Some(Instant::now() + Duration::from_millis(timeout_ms.into()))
2755            }
2756        };
2757        let mut meta_buf = [0u8; TxMeta::WIRE_LEN];
2758        let meta_len = meta
2759            .encode(&mut meta_buf)
2760            .expect("buffer sized with WIRE_LEN");
2761
2762        self.send_confirmed(data, &meta_buf[..meta_len], cca_deadline)
2763            .await
2764    }

2784                core::task::Poll::Pending => return core::task::Poll::Pending,
2785            }
2786        }

2860/// Decode a `PROP_ALERT` value: exactly one PUI naming a known state.
2861fn decode_alert(value: &[u8]) -> Result<AlertState, UlcpError> {
2862    const MALFORMED: &str = "malformed PROP_ALERT";
2863    let (code, consumed) = pui::decode(value).map_err(|_| UlcpError::Protocol(MALFORMED))?;
2864    if consumed != value.len() {
2865        return Err(UlcpError::Protocol(MALFORMED));
2866    }
2867    AlertState::from_code(code).ok_or(UlcpError::Protocol(MALFORMED))
2868}

2873    match value {

2875        [a, b, c, d] => Ok(Some(u32::from_le_bytes([*a, *b, *c, *d]))),

2877    }
