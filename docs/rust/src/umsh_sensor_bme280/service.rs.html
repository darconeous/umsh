1//! Cancellation-safe mailbox between a host request and a dedicated sensor
2//! worker. A canceled waiter leaves acquisition running; sequence tags prevent
3//! the next request from consuming its result. One onboard sensor per service.
4use core::sync::atomic::{AtomicU8, AtomicU32, Ordering};
5use embassy_sync::{blocking_mutex::raw::CriticalSectionRawMutex, mutex::Mutex, signal::Signal};
6
7pub struct Service {
8    address: AtomicU8,
9    sequence: AtomicU32,
10    caller: Mutex<CriticalSectionRawMutex, ()>,
11    request: Signal<CriticalSectionRawMutex, u32>,
12    result: Signal<CriticalSectionRawMutex, (u32, Option<u16>)>,
13}
14
15impl Default for Service {
16    fn default() -> Self {
17        Self::new()
18    }
19}
20impl Service {
21    pub const fn new() -> Self {
22        Self {
23            address: AtomicU8::new(0),
24            sequence: AtomicU32::new(0),
25            caller: Mutex::new(()),
26            request: Signal::new(),
27            result: Signal::new(),
28        }
29    }
30    /// Publish once at boot, after probing and before starting host service.
31    pub fn detected(&self, address: u8) {
32        self.address.store(address, Ordering::Release);
33    }
34    pub fn address(&self) -> Option<u8> {
35        match self.address.load(Ordering::Acquire) {
36            0 => None,
37            value => Some(value),
38        }
39    }
40    pub async fn sample(&self) -> Option<u16> {
41        self.address()?;
42        let _caller = self.caller.lock().await;
43        let id = self.sequence.fetch_add(1, Ordering::Relaxed);
44        self.request.signal(id);
45        loop {
46            let (completed, result) = self.result.wait().await;
47            if completed == id {
48                return result;
49            }
50        }
51    }
52    pub async fn request(&self) -> u32 {
53        self.request.wait().await
54    }
55    pub fn complete(&self, id: u32, value: Option<u16>) {
56        self.result.signal((id, value));
57    }
58}
