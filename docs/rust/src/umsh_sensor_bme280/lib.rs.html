2//! BME280 temperature-only forced acquisition. No background sampling.
3//!
4//! Register map, timing and compensation: Bosch BST-BME280-DS001-23,
5//! sections 4.2, 5 and 9.1:
6//! <https://www.bosch-sensortec.com/media/boschsensortec/downloads/datasheets/bst-bme280-ds002.pdf>
7//! Callers retain exclusive bus access for the complete operation and must not
8//! cancel an I2C transfer on controllers without cancellation-safe DMA.
9
10use embedded_hal_async::{delay::DelayNs, i2c::I2c};

14#[derive(Debug, PartialEq, Eq)]
15pub enum Error<E> {
16    Bus(E),
17    WrongChip,
18    Timeout,
19    InvalidReading,
20}
21
22/// Detection performs no measurement and accepts only BME280 (not BMP280).
23pub async fn probe<I: I2c>(bus: &mut I, address: u8) -> Result<(), Error<I::Error>> {
24    let mut id = [0];
25    bus.write_read(address, &[0xd0], &mut id)
26        .await
27        .map_err(Error::Bus)?;
28    if id[0] == 0x60 {
29        Ok(())
30    } else {
31        Err(Error::WrongChip)
32    }
33}
34
35async fn idle<I: I2c, D: DelayNs>(
36    bus: &mut I,
37    address: u8,
38    delay: &mut D,
39) -> Result<(), Error<I::Error>> {
40    // Also handles a previous full-oversampling conversion started by raw I2C.
41    // NVM copying (bit 0) and measurement (bit 3) must both have finished.
42    for _ in 0..40 {
43        let mut status = [0];
44        bus.write_read(address, &[0xf3], &mut status)
45            .await
46            .map_err(Error::Bus)?;
47        if status[0] & 9 == 0 {
48            return Ok(());
49        }
50        delay.delay_ms(5).await;
51    }
52    Err(Error::Timeout)
53}

55/// Fresh measurement in tenths of a kelvin. The chip returns to sleep after
56/// each conversion; pressure, humidity and the IIR filter are disabled.
57pub async fn sample<I: I2c, D: DelayNs>(
58    bus: &mut I,
59    address: u8,
60    delay: &mut D,
61) -> Result<u16, Error<I::Error>> {
62    probe(bus, address).await?;
63    bus.write(address, &[0xf4, 0]).await.map_err(Error::Bus)?;
64    idle(bus, address, delay).await?;
65    // Reload calibration after power loss/reset as well as on the initial get.
66    let mut trim = [0; 6];
67    bus.write_read(address, &[0x88], &mut trim)
68        .await
69        .map_err(Error::Bus)?;
70    bus.write(address, &[0xf5, 0]).await.map_err(Error::Bus)?;
71    bus.write(address, &[0xf2, 0]).await.map_err(Error::Bus)?;
72    bus.write(address, &[0xf4, 0x21])
73        .await
74        .map_err(Error::Bus)?;
75    // Temperature x1 requires at most 3.55 ms. Do not mistake the status
76    // register's initial idle value for completion immediately after forcing.
77    delay.delay_ms(5).await;
78    idle(bus, address, delay).await?;
79    let mut data = [0; 3];
80    bus.write_read(address, &[0xfa], &mut data)
81        .await
82        .map_err(Error::Bus)?;
83    let raw = (u32::from(data[0]) << 12) | (u32::from(data[1]) << 4) | u32::from(data[2] >> 4);
84    compensate(trim, raw).ok_or(Error::InvalidReading)
85}
86
87fn compensate(trim: [u8; 6], raw: u32) -> Option<u16> {
88    if raw == 0x80000 || raw > 0xfffff || trim == [0; 6] || trim == [0xff; 6] {
89        return None;
90    }
91    let t1 = i64::from(u16::from_le_bytes([trim[0], trim[1]]));
92    let t2 = i64::from(i16::from_le_bytes([trim[2], trim[3]]));
93    let t3 = i64::from(i16::from_le_bytes([trim[4], trim[5]]));
94    // Bosch integer compensation, widened so corrupt trims cannot overflow.
95    let adc = i64::from(raw);
96    let delta = (adc >> 4) - t1;
97    let fine = (((adc >> 3) - (t1 << 1)) * t2 >> 11) + (((delta * delta) >> 12) * t3 >> 14);
98    let hundredths_c = (fine * 5 + 128) >> 8;
99    if !(-4000..=8500).contains(&hundredths_c) {
100        return None;
101    }
102    Some(((hundredths_c + 27315 + 5) / 10) as u16)
103}
