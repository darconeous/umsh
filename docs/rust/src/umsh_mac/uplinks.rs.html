1//! Which repeaters have shown whether they hear this node.
2//!
3//! [`TransmitterObservations`](crate::TransmitterObservations) records the
4//! repeaters this node hears. A route is used in the other direction, and
5//! hearing a repeater well says nothing about whether it hears us: a
6//! hilltop site with a strong transmitter is audible long after a handheld
7//! has dropped out of its reach.
8//!
9//! The evidence for that direction comes from this node's own frames as
10//! repeaters forward them. A trace route lists the repeaters a frame
11//! crossed, the first one last, and the trace signal beside it carries how
12//! well each of them heard the transmitter before it. So when a frame this
13//! node originated comes back forwarded:
14//!
15//! - the last hint is the repeater that took the frame off this node's
16//!   transmission, and its trace-signal entry is how well it heard us;
17//! - on a flood, every other hint is a repeater that forwarded a copy it
18//!   took from another repeater. A flood repeater forwards the first copy
19//!   it accepts, and this node's own transmission comes first, so those
20//!   repeaters either did not hear it or heard it too weakly to use.
21
22use umsh_core::{RouterHint, options::TraceSignalEntry};
23
24/// How many repeaters the table remembers.
25pub(crate) const MAX_UPLINK_OBSERVATIONS: usize = 16;
26
27/// How long a piece of evidence stays worth acting on. Long enough to
28/// outlast the gaps in ordinary traffic, short enough that a node that has
29/// moved stops trusting what it learned before.
30const EVIDENCE_TTL_MS: u64 = 30 * 60 * 1000;
31
32/// What this node knows about whether one repeater hears it.
33#[derive(Clone, Copy, Debug, PartialEq, Eq)]
34pub(crate) enum Uplink {
35    /// The repeater recently took a frame off this node's transmission,
36    /// measured as the entry says when the frame carried a trace signal.
37    Heard(Option<TraceSignalEntry>),
38    /// The repeater recently forwarded a frame of ours it had taken from
39    /// another repeater instead, and has not been seen to hear us since.
40    Unheard,
41    /// Nothing recent either way.
42    Unknown,
43}
44
45#[derive(Clone, Copy, Debug)]
46struct UplinkObservation {
47    hint: RouterHint,
48    /// When the repeater last heard us, in whole seconds of the monotonic
49    /// clock, and how well when the frame said.
50    heard: Option<(u32, Option<TraceSignalEntry>)>,
51    /// When it last forwarded a frame of ours it had not heard from us.
52    unheard_s: Option<u32>,
53}
54
55impl UplinkObservation {
56    fn latest_s(&self) -> u32 {
57        let heard = self.heard.map_or(0, |(at, _)| at);
58        heard.max(self.unheard_s.unwrap_or(0))
59    }
60}
61
62/// A bounded table of uplink evidence, least recently updated dropped
63/// first.
64#[derive(Clone, Debug, Default)]
65pub(crate) struct UplinkObservations {
66    entries: heapless::Vec<UplinkObservation, MAX_UPLINK_OBSERVATIONS>,
67}
68
69fn seconds(now_ms: u64) -> u32 {
70    (now_ms / 1000) as u32
71}
72
73impl UplinkObservations {
74    pub(crate) const fn new() -> Self {
75        Self {
76            entries: heapless::Vec::new(),
77        }
78    }
79
80    /// Record that `hint` took a frame off this node's transmission.
81    pub(crate) fn note_heard(
82        &mut self,
83        hint: RouterHint,
84        signal: Option<TraceSignalEntry>,
85        now_ms: u64,
86    ) {
87        self.entry(hint, now_ms).heard = Some((seconds(now_ms), signal));
88    }
89
90    /// Record that `hint` forwarded a frame of ours it had taken from
91    /// another repeater.
92    pub(crate) fn note_unheard(&mut self, hint: RouterHint, now_ms: u64) {
93        self.entry(hint, now_ms).unheard_s = Some(seconds(now_ms));
94    }
95
96    /// What the table says about `hint` now. The more recent of the two
97    /// kinds of evidence wins; a tie goes to having been heard, since that
98    /// one is a positive observation.
99    pub(crate) fn uplink(&self, hint: &RouterHint, now_ms: u64) -> Uplink {
100        let Some(entry) = self.entries.iter().find(|entry| &entry.hint == hint) else {
101            return Uplink::Unknown;
102        };
103        let now_s = seconds(now_ms);
104        let fresh = |at_s: u32| u64::from(now_s.wrapping_sub(at_s)) * 1000 <= EVIDENCE_TTL_MS;
105        let heard = entry.heard.filter(|(at, _)| fresh(*at));
106        let unheard = entry.unheard_s.filter(|at| fresh(*at));
107        match (heard, unheard) {
108            (Some((heard_at, _)), Some(unheard_at)) if unheard_at > heard_at => Uplink::Unheard,
109            (Some((_, signal)), _) => Uplink::Heard(signal),
110            (None, Some(_)) => Uplink::Unheard,
111            (None, None) => Uplink::Unknown,
112        }
113    }
114
115    fn entry(&mut self, hint: RouterHint, now_ms: u64) -> &mut UplinkObservation {
116        if let Some(index) = self.entries.iter().position(|entry| entry.hint == hint) {
117            return &mut self.entries[index];
118        }
119        let fresh = UplinkObservation {
120            hint,
121            heard: None,
122            unheard_s: None,
123        };
124        if self.entries.push(fresh).is_err() {
125            // Full: the repeater whose evidence is oldest says the least.
126            let now_s = seconds(now_ms);
127            let oldest = self
128                .entries
129                .iter()
130                .enumerate()
131                .max_by_key(|(_, entry)| now_s.wrapping_sub(entry.latest_s()))
132                .map(|(index, _)| index)
133                .unwrap_or(0);
134            self.entries[oldest] = fresh;
135            return &mut self.entries[oldest];
136        }
137        let last = self.entries.len() - 1;
138        &mut self.entries[last]
139    }
140}

142#[cfg(test)]
143mod tests {
144    use super::*;
145
146    fn hint(seed: u8) -> RouterHint {
147        RouterHint([seed, seed])
148    }
149
150    #[test]
151    fn the_more_recent_evidence_wins() {
152        let mut table = UplinkObservations::new();
153        let signal = Some(TraceSignalEntry::new(-64, 80));
154        table.note_heard(hint(1), signal, 10_000);
155        assert_eq!(table.uplink(&hint(1), 10_000), Uplink::Heard(signal));
156
157        table.note_unheard(hint(1), 20_000);
158        assert_eq!(table.uplink(&hint(1), 20_000), Uplink::Unheard);
159
160        table.note_heard(hint(1), None, 30_000);
161        assert_eq!(table.uplink(&hint(1), 30_000), Uplink::Heard(None));
162        assert_eq!(table.uplink(&hint(2), 30_000), Uplink::Unknown);
163    }
164
165    #[test]
166    fn evidence_expires() {
167        let mut table = UplinkObservations::new();
168        table.note_unheard(hint(1), 1_000);
169        assert_eq!(
170            table.uplink(&hint(1), 1_000 + EVIDENCE_TTL_MS),
171            Uplink::Unheard
172        );
173        assert_eq!(
174            table.uplink(&hint(1), 2_000 + EVIDENCE_TTL_MS),
175            Uplink::Unknown
176        );
177    }
178
179    #[test]
180    fn a_full_table_drops_the_stalest_repeater() {
181        let mut table = UplinkObservations::new();
182        for seed in 0..MAX_UPLINK_OBSERVATIONS as u8 {
183            table.note_heard(hint(seed), None, 1_000 * (u64::from(seed) + 1));
184        }
185        // Refresh the first so insertion order would pick the wrong one.
186        table.note_unheard(hint(0), 100_000);
187        table.note_heard(hint(200), None, 101_000);
188        assert_eq!(table.entries.len(), MAX_UPLINK_OBSERVATIONS);
189        assert_eq!(table.uplink(&hint(0), 101_000), Uplink::Unheard);
190        assert_eq!(table.uplink(&hint(1), 101_000), Uplink::Unknown);
191        assert_eq!(table.uplink(&hint(200), 101_000), Uplink::Heard(None));
192    }
193}
