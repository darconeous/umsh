1//! Which repeaters this node has heard forwarding, and how well.
2//!
3//! Route learning records the way to a *peer*; this records the last
4//! reception from a neighboring *repeater*. A repeater prepends its router
5//! hint to the trace route of every frame it forwards, so the first hint of
6//! a trace names the transmitter just heard, and a frame with no such hint
7//! came off its originator and names no repeater at all. The two differ for
8//! exactly the traffic that makes a repeater useful: a frame forwarded past
9//! this node teaches nothing about a peer, and a frame overheard on the air
10//! never reaches the host, but both prove a neighboring repeater was on the
11//! air. That is what a
12//! [Peer Repeaters Response](../../docs/protocol/src/mac-commands.md) reports
13//! about the hops it names.
14//!
15//! A neighbor need not be on the air at all: a host attached over a
16//! point-to-point link forwards like any other repeater and is recorded like
17//! one, with no measurement, since no radio was between it and this node.
18
19use umsh_core::RouterHint;
20use umsh_hal::Snr;
21
22/// How many repeaters the table remembers.
23///
24/// A neighborhood larger than this is one where the least recently heard
25/// entries are the ones worth losing, and the whole table has to fit a
26/// single response page's worth of answers anyway.
27pub const MAX_TRANSMITTER_OBSERVATIONS: usize = 16;
28
29/// The most recent reception from one neighboring repeater.
30#[derive(Clone, Copy, Debug, PartialEq, Eq)]
31pub struct TransmitterObservation {
32    /// All a trace route reveals about the hop that forwarded a frame, and
33    /// so all this table can key on.
34    pub hint: RouterHint,
35    /// RSSI in dBm and SNR of the most recent reception, when a radio
36    /// measured it. `None` for a neighbor reached over a link with no radio
37    /// in it.
38    pub rssi_snr: Option<(i16, Snr)>,
39    /// When that reception was, on the monotonic clock.
40    pub last_seen_ms: u64,
41}
42
43/// A bounded, least-recently-heard table of repeater observations.
44#[derive(Clone, Debug, Default)]
45pub struct TransmitterObservations {
46    entries: heapless::Vec<TransmitterObservation, MAX_TRANSMITTER_OBSERVATIONS>,
47}
48
49impl TransmitterObservations {
50    pub const fn new() -> Self {
51        Self {
52            entries: heapless::Vec::new(),
53        }
54    }
55
56    /// Record a reception from `hint`, replacing whatever was known before.
57    ///
58    /// Only the latest reception is kept: a peer-repeater entry reports the
59    /// most recent measurement, and an average across a moving neighbor
60    /// would describe a link that no longer exists.
61    pub fn observe(&mut self, hint: RouterHint, rssi_snr: Option<(i16, Snr)>, now_ms: u64) {
62        if let Some(entry) = self.entries.iter_mut().find(|entry| entry.hint == hint) {
63            entry.rssi_snr = rssi_snr;
64            entry.last_seen_ms = now_ms;
65            return;
66        }
67        let observation = TransmitterObservation {
68            hint,
69            rssi_snr,
70            last_seen_ms: now_ms,
71        };
72        if self.entries.push(observation).is_ok() {
73            return;
74        }
75        // Full: the least recently heard transmitter is the one whose
76        // absence says the least.
77        let Some(oldest) = self
78            .entries
79            .iter_mut()
80            .min_by_key(|entry| entry.last_seen_ms)
81        else {
82            return;
83        };
84        *oldest = observation;
85    }
86
87    pub fn iter(&self) -> impl Iterator<Item = &TransmitterObservation> {
88        self.entries.iter()
89    }
90
91    pub fn len(&self) -> usize {
92        self.entries.len()
93    }
94
95    pub fn is_empty(&self) -> bool {
96        self.entries.is_empty()
97    }
98
99    /// The observation for one transmitter, if it is still held.
100    pub fn get(&self, hint: &RouterHint) -> Option<&TransmitterObservation> {
101        self.entries.iter().find(|entry| &entry.hint == hint)
102    }
103}

105#[cfg(test)]
106mod tests {
107    use super::*;
108
109    fn hint(seed: u8) -> RouterHint {
110        RouterHint([seed, seed])
111    }
112
113    fn measured(rssi_dbm: i16, snr_db: i8) -> Option<(i16, Snr)> {
114        Some((rssi_dbm, Snr::from_decibels(snr_db)))
115    }
116
117    #[test]
118    fn a_repeat_reception_replaces_what_was_known_rather_than_adding_to_it() {
119        let mut table = TransmitterObservations::new();
120        table.observe(hint(1), measured(-100, -9), 1_000);
121        table.observe(hint(1), measured(-70, 6), 2_000);
122        assert_eq!(table.len(), 1);
123        let entry = table.get(&hint(1)).unwrap();
124        assert_eq!(entry.rssi_snr, measured(-70, 6));
125        assert_eq!(entry.last_seen_ms, 2_000);
126
127        // A neighbor that moved onto a link with no radio in it is still
128        // the same neighbor, now without a reading.
129        table.observe(hint(1), None, 3_000);
130        let entry = table.get(&hint(1)).unwrap();
131        assert_eq!(entry.rssi_snr, None);
132        assert_eq!(entry.last_seen_ms, 3_000);
133    }
134
135    #[test]
136    fn a_full_table_drops_the_least_recently_heard_transmitter() {
137        let mut table = TransmitterObservations::new();
138        for seed in 0..MAX_TRANSMITTER_OBSERVATIONS as u8 {
139            table.observe(hint(seed), measured(-90, 0), 1_000 + u64::from(seed));
140        }
141        // Refresh the oldest so a plain insertion-order eviction would pick
142        // the wrong one.
143        table.observe(hint(0), measured(-80, 1), 9_000);
144        table.observe(hint(200), measured(-95, -2), 10_000);
145
146        assert_eq!(table.len(), MAX_TRANSMITTER_OBSERVATIONS);
147        assert!(table.get(&hint(200)).is_some());
148        assert!(
149            table.get(&hint(0)).is_some(),
150            "refreshed, so not the oldest"
151        );
152        assert!(table.get(&hint(1)).is_none(), "least recently heard");
153    }
154}
