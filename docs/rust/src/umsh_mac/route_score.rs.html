1//! How favorable a path looks, from what the frames that crossed it measured.
2//!
3//! A score ranks paths; it does not predict delivery. An endpoint cannot
4//! know the spreading factor, bandwidth, or conditions of a link it does not
5//! own—a bridge hides even that—so nothing here models the physics. Each hop
6//! is charged by how far its SNR falls short of a comfortable level, the
7//! charges add up along the path, and the path with the largest (least
8//! negative) score wins.
9//!
10//! Adding in decibels multiplies in power, which is why a sum of shortfalls
11//! behaves like a product of delivery odds without any arithmetic on
12//! probabilities. Past a knee the charge steepens: every digital link fails
13//! abruptly near its floor, so one hop barely above it must not tie with
14//! several hops that are only somewhat weak.
15//!
16//! Everything is in centibels, the unit the MAC measures SNR in.
17
18use umsh_core::options::TraceSignalEntry;
19
20/// SNR at or above which a hop is charged nothing.
21const COMFORT_SNR_CB: i32 = -50;
22/// Shortfall below comfort charged one-for-one before the charge steepens.
23const KNEE_CB: i32 = 40;
24/// How much each centibel of shortfall past the knee costs.
25const STEEP_SLOPE: i32 = 3;
26/// What every hop costs on top of its shortfall, so that a shorter path wins
27/// between two that are otherwise as good, and a longer one wins only when
28/// it has to.
29const HOP_COST_CB: i32 = 10;
30/// How much worse an unmeasured direction is assumed to be than the
31/// measured one. A link heard well one way may not work at all the other.
32const ASYMMETRY_CB: i32 = 60;
33/// The charge for an air hop no frame has measured.
34const UNMEASURED_CB: i32 = KNEE_CB;
35/// The extra charge for a first hop that has shown it does not hear us.
36const UNHEARD_CB: i32 = 120;
37
38/// How much an acknowledged route is credited when a new one is weighed
39/// against it. A credit rather than a rank: a route confirmed by an
40/// exchange whose flood tail may have done the delivering is still
41/// displaced by a much better path.
42const CONFIRMED_CREDIT_CB: i32 = 60;
43/// How much better a new path must score to displace the one in use, so
44/// that ordinary fading does not flip routes back and forth.
45const SWITCH_MARGIN_CB: i32 = 30;
46
47/// How long an acknowledgment vouches for the route it came back on.
48pub(crate) const CONFIRMATION_TTL_MS: u64 = 10 * 60 * 1000;
49
50/// What is known about one hop of a path, in the direction it would be
51/// sent.
52#[derive(Clone, Copy, Debug, PartialEq, Eq)]
53pub(crate) enum Hop {
54    /// Measured in the direction this node would send: SNR in centibels.
55    Outbound(i16),
56    /// Measured only in the direction toward this node.
57    Inbound(i16),
58    /// A first hop that has shown it does not hear this node, with what was
59    /// measured toward this node, if anything.
60    Unheard(Option<i16>),
61    /// A point-to-point link, with no radio in it.
62    PointToPoint,
63    /// Not measured in either direction.
64    Unmeasured,
65}
66
67impl Hop {
68    /// What a trace-signal entry says about the hop that wrote it: how well
69    /// that repeater heard the transmitter before it, which is the
70    /// direction toward this node.
71    pub(crate) fn from_trace_entry(entry: TraceSignalEntry) -> Self {
72        match entry.snr_centibels() {
73            Some(snr) => Self::Inbound(snr),
74            None => Self::PointToPoint,
75        }
76    }
77
78    fn charge_cb(self) -> i32 {
79        match self {
80            Self::Outbound(snr) => shortfall_charge(i32::from(snr)),
81            Self::Inbound(snr) => shortfall_charge(i32::from(snr) - ASYMMETRY_CB),
82            Self::Unheard(Some(snr)) => {
83                shortfall_charge(i32::from(snr) - ASYMMETRY_CB) + UNHEARD_CB
84            }
85            Self::Unheard(None) => UNMEASURED_CB + UNHEARD_CB,
86            Self::PointToPoint => 0,
87            Self::Unmeasured => UNMEASURED_CB,
88        }
89    }
90}
91
92/// The two-slope charge for a hop measured at `snr_cb`.
93fn shortfall_charge(snr_cb: i32) -> i32 {
94    let shortfall = (COMFORT_SNR_CB - snr_cb).max(0);
95    if shortfall <= KNEE_CB {
96        shortfall
97    } else {
98        KNEE_CB + STEEP_SLOPE * (shortfall - KNEE_CB)
99    }
100}
101
102/// Score a path from its hops, nearest first.
103pub(crate) fn path_score(hops: impl IntoIterator<Item = Hop>) -> i16 {
104    let total: i32 = hops
105        .into_iter()
106        .map(|hop| hop.charge_cb() + HOP_COST_CB)
107        .fold(0, i32::saturating_add);
108    i16::try_from(-total).unwrap_or(i16::MIN)
109}
110
111/// Whether a path scoring `candidate` should displace one scoring
112/// `current`, given whether an acknowledgment recently vouched for it.
113pub(crate) fn displaces(candidate: i16, current: i16, confirmed: bool) -> bool {
114    let credit = if confirmed { CONFIRMED_CREDIT_CB } else { 0 };
115    i32::from(candidate) > i32::from(current) + credit + SWITCH_MARGIN_CB
116}

118#[cfg(test)]
119mod tests {
120    use super::*;
121
122    fn db(db: i16) -> i16 {
123        db * 10
124    }
125
126    /// The example that motivated the steep segment: one hop barely above
127    /// its floor must lose to more hops that are each only somewhat weak.
128    #[test]
129    fn one_nearly_dead_hop_loses_to_several_weak_ones() {
130        let shortfall = |short_db: i16| Hop::Outbound(db(-5) - db(short_db));
131        let one_bad = path_score([shortfall(9), Hop::Outbound(db(10))]);
132        let three_weak = path_score([shortfall(3), shortfall(3), shortfall(3)]);
133        assert!(three_weak > one_bad, "{three_weak} vs {one_bad}");
134    }
135
136    #[test]
137    fn a_stronger_hop_scores_higher_until_comfort() {
138        let weak = path_score([Hop::Outbound(db(-10))]);
139        let better = path_score([Hop::Outbound(db(-7))]);
140        let comfortable = path_score([Hop::Outbound(db(-5))]);
141        let strong = path_score([Hop::Outbound(db(12))]);
142        assert!(better > weak);
143        assert!(comfortable > better);
144        assert_eq!(strong, comfortable, "past comfort, more SNR buys nothing");
145    }
146
147    /// Between two equally clean paths, the shorter one wins.
148    #[test]
149    fn every_hop_costs_something() {
150        let short = path_score([Hop::Outbound(db(10)), Hop::Inbound(db(10))]);
151        let long = path_score([
152            Hop::Outbound(db(10)),
153            Hop::Inbound(db(10)),
154            Hop::Inbound(db(10)),
155        ]);
156        assert!(short > long);
157    }
158
159    /// A hop measured only toward us is assumed worse the other way, so a
160    /// marginal reception cannot pass for a usable link.
161    #[test]
162    fn a_one_way_measurement_is_discounted() {
163        assert!(
164            path_score([Hop::Outbound(db(-6))]) > path_score([Hop::Inbound(db(-6))]),
165            "the same reading counts for more when it was taken our way"
166        );
167    }
168
169    #[test]
170    fn a_first_hop_that_does_not_hear_us_is_heavily_charged() {
171        let heard_well_unknown = path_score([Hop::Inbound(db(5)), Hop::Inbound(db(5))]);
172        let unheard = path_score([Hop::Unheard(Some(db(5))), Hop::Inbound(db(5))]);
173        let longer_but_proven = path_score([
174            Hop::Outbound(db(8)),
175            Hop::Inbound(db(4)),
176            Hop::Inbound(db(5)),
177        ]);
178        assert!(heard_well_unknown > unheard);
179        assert!(
180            longer_but_proven > unheard,
181            "a repeater that hears us beats a shortcut through one that does not"
182        );
183    }
184
185    #[test]
186    fn a_point_to_point_hop_costs_only_its_hop() {
187        assert_eq!(
188            path_score([Hop::PointToPoint]),
189            path_score([Hop::Outbound(db(20))])
190        );
191        assert_eq!(
192            Hop::from_trace_entry(TraceSignalEntry::POINT_TO_POINT),
193            Hop::PointToPoint
194        );
195        assert_eq!(
196            Hop::from_trace_entry(TraceSignalEntry::new(-98, 40)),
197            Hop::Inbound(40)
198        );
199    }
200
201    #[test]
202    fn displacing_a_route_takes_a_clear_margin_and_more_when_it_was_acknowledged() {
203        let current = db(-10);
204        assert!(
205            !displaces(current + db(2), current, false),
206            "within the margin"
207        );
208        assert!(displaces(current + db(4), current, false));
209        assert!(!displaces(current + db(4), current, true), "acknowledged");
210        assert!(displaces(current + db(10), current, true));
211    }
