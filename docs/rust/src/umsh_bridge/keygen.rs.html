1//! Identity issuance, so a deployment needs nothing but this binary.
2//!
3//! An identity's Ed25519 seed is the only credential a participant
4//! holds. Its public half — the UMSH address — is what the other end of
5//! the tunnel pins, and it is public: sharing it needs no confidential
6//! channel. The TLS certificates the handshake requires are minted in
7//! memory from the identity at startup and never stored.
8
9use std::path::Path;
10
11use anyhow::{Context, Result, bail};
12
13use crate::identity::{BridgeIdentity, format_seed};
14
15pub fn write_identity(path: &Path, force: bool) -> Result<()> {
16    let mut seed = [0u8; 32];
17    rand::fill(&mut seed);
18    let identity = BridgeIdentity::from_seed(&seed);
19
20    write_secret(path, &format!("{}\n", format_seed(&seed)), force)?;
21
22    println!("identity key: {}", path.display());
23    println!("address:      {}", identity.public_key());
24    println!("node hint:    {}", identity.node_hint());
25    println!("router hint:  {}", identity.router_hint());
26    println!();
27    println!("The address is public: configure it at the other end of the tunnel.");
28    Ok(())
29}
30
31/// Print the address of an existing identity — what the operator copies
32/// into the other end's configuration.
33pub fn print_address(path: &Path) -> Result<()> {
34    let identity = BridgeIdentity::load(path)?;
35    println!("{}", identity.public_key());
36    Ok(())
37}

39fn write_secret(path: &Path, contents: &str, force: bool) -> Result<()> {
40    guard_existing(path, force)?;
41    create_parent(path)?;
42    write_owner_only(path, contents).with_context(|| format!("writing {}", path.display()))
43}
44
45#[cfg(unix)]
46fn write_owner_only(path: &Path, contents: &str) -> std::io::Result<()> {
47    use std::io::Write;
48    use std::os::unix::fs::OpenOptionsExt;
49
50    // Mode goes on `open`, not a later `chmod`: between the two the key
51    // would exist world-readable.
52    let mut file = std::fs::OpenOptions::new()
53        .write(true)
54        .create(true)
55        .truncate(true)
56        .mode(0o600)
57        .open(path)?;
58    file.write_all(contents.as_bytes())
59}
60
61#[cfg(not(unix))]
62fn write_owner_only(path: &Path, contents: &str) -> std::io::Result<()> {
63    std::fs::write(path, contents)
64}
65
66fn guard_existing(path: &Path, force: bool) -> Result<()> {
67    if path.exists() && !force {
68        bail!(
69            "{} already exists; pass --force to replace it (the identity's address changes, \
70             and every peer pinning it must be updated)",
71            path.display()
72        );
73    }
74    Ok(())
75}
76
77fn create_parent(path: &Path) -> Result<()> {
78    if let Some(parent) = path
79        .parent()
80        .filter(|parent| !parent.as_os_str().is_empty())
81    {
82        std::fs::create_dir_all(parent)
83            .with_context(|| format!("creating {}", parent.display()))?;
84    }
85    Ok(())
86}

88#[cfg(test)]
89mod tests {
90    use super::*;
91
92    #[test]
93    fn an_existing_identity_is_not_replaced_without_force() {
94        let dir = tempfile::tempdir().unwrap();
95        let path = dir.path().join("identity.key");
96        write_identity(&path, false).unwrap();
97        let first = *BridgeIdentity::load(&path).unwrap().public_key();
98        assert!(write_identity(&path, false).is_err());
99        assert_eq!(
100            *BridgeIdentity::load(&path).unwrap().public_key(),
101            first,
102            "a refused overwrite leaves the identity untouched"
103        );
104        assert!(write_identity(&path, true).is_ok());
105    }
106
107    #[cfg(unix)]
108    #[test]
109    fn a_written_identity_key_is_readable_only_by_its_owner() {
110        use std::os::unix::fs::PermissionsExt;
111
112        let dir = tempfile::tempdir().unwrap();
113        let identity = dir.path().join("identity.key");
114        write_identity(&identity, false).unwrap();
115        let mode = std::fs::metadata(&identity).unwrap().permissions().mode() & 0o777;
116        assert_eq!(mode, 0o600, "identity key mode");
117        // And the loader accepts what the generator wrote.
118        BridgeIdentity::load(&identity).unwrap();
119        print_address(&identity).unwrap();
120    }
121}
