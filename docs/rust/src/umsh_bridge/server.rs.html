1//! The bridge server: listeners, per-client tunnels, and the one task
2//! that copies frames between them.
3//!
4//! A client is a configured name with a pinned identity, so the
5//! interface set is fixed for the run and a disconnected client is an
6//! interface that is down rather than one that has gone away. That is
7//! what lets the hub hold a stable interface index and lets a
8//! reconnecting client resume its own identity rather than acquire a new
9//! one.
10

30/// Frames waiting for the hub before an interface's reader blocks. The
31/// hub's work per frame is bounded and small, so this only ever absorbs
32/// a burst.
33const HUB_BACKLOG: usize = 64;
34
35pub async fn run(identity: BridgeIdentity, config: Config) -> Result<()> {
36    let mut server = config.server.expect("validated as a server configuration");
37    let credential =
38        Credential::for_identity(&identity).context("minting the server's TLS credential")?;
39
40    // A configured client with this server's own address would let the
41    // hub attribute the server's traffic to a client—and it can only
42    // be a mistake, because no client can hold this server's key.
43    for client in &server.clients {
44        if client.address.0 == *identity.public_key() {
45            anyhow::bail!(
46                "client \"{}\" lists this bridge's own identity as its address",
47                client.name
48            );
49        }
50    }
51
52    let interfaces = Arc::new(Interfaces::build(&server));
53    let policy = Policy::build(&server, &interfaces)?;
54
55    tracing::info!(
56        address = %identity.public_key(),
57        exit_clamp = server
58            .limits
59            .exit_clamp
60            .map_or_else(|| "off".to_string(), |clamp| clamp.to_string()),
61        "bridge server starting"
62    );
63    for iface in &interfaces.all {
64        tracing::info!(
65            iface = %iface.name,
66            fan_out = %policy.describe_fan_out(iface.id),
67            "interface"
68        );
69    }
70
71    let (ingress_tx, ingress_rx) = mpsc::channel(HUB_BACKLOG);
72
73    if let Some(radio) = interfaces.radio {
74        let iface = interfaces.get(radio).clone();
75        let inbound = Arc::new(TunnelQueue::new(
76            server.tunnel.queue_depth,
77            Duration::from_secs(server.tunnel.max_frame_age_secs),
78        ));
79        let relay = DeviceRelay::new(
80            std::mem::take(&mut server.radio),
81            inbound.clone(),
82            iface.egress.clone(),
83        );
84        tokio::task::spawn_local(async move { relay.run().await });
85        spawn_ingress(inbound, radio, ingress_tx.clone());
86    }
87
88    if let (Some(lan), Some(entry)) = (interfaces.lan, &server.lan) {
89        let iface = interfaces.get(lan).clone();
90        let inbound = Arc::new(TunnelQueue::new(
91            server.tunnel.queue_depth,
92            Duration::from_secs(server.tunnel.max_frame_age_secs),
93        ));
94        let relay = LanRelay::new(entry.socket_config(), inbound.clone(), iface.egress.clone());
95        tokio::task::spawn_local(async move { relay.run().await });
96        spawn_ingress(inbound, lan, ingress_tx.clone());
97    }
98
99    // Host interfaces need no credential and no radio: each is a socket
100    // presenting a simulated device whose radio is this hub.
101    for (index, entry) in server.hosts.iter().enumerate() {
102        let iface = interfaces
103            .by_host(index)
104            .ok_or_else(|| anyhow!("host {index} has no interface"))?
105            .clone();
106        let relay = crate::host::HostRelay::new(entry, iface, ingress_tx.clone());
107        let name = entry.name.clone();
108        tokio::task::spawn_local(async move {
109            if let Err(error) = relay.run().await {
110                tracing::error!(iface = %name, "host interface stopped: {error:#}");
111            }
112        });
113    }
114
115    let hub = Hub::new(interfaces.clone(), policy, server.limits.exit_clamp);
116    tokio::task::spawn_local(hub.run(ingress_rx));
117
118    // An identity names exactly one client; the configuration
119    // guarantees no two share one.
120    let by_address: HashMap<Address, usize> = server
121        .clients
122        .iter()
123        .enumerate()
124        .map(|(index, client)| (client.address, index))
125        .collect();
126    let accepted: Vec<Address> = by_address.keys().copied().collect();
127
128    let tls_config = tls::server_config(&credential, accepted)?;
129    let acceptor = TlsAcceptor::from(tls_config);
130    let shared = Arc::new(Shared {
131        interfaces,
132        by_address,
133        tunnel: server.tunnel,
134        client_names: server
135            .clients
136            .iter()
137            .map(|client| client.name.clone())
138            .collect(),
139        ingress: ingress_tx,
140    });
141
142    let mut listeners = Vec::new();
143    for address in &server.listen {
144        let listener = TcpListener::bind(address)
145            .await
146            .with_context(|| format!("listening on {address}"))?;
147        tracing::info!(%address, "listening");
148        listeners.push(listener);
149    }
150    if listeners.is_empty() {
151        return Err(anyhow!("no listener could be opened"));
152    }
153
154    for listener in listeners {
155        let acceptor = acceptor.clone();
156        let shared = shared.clone();
157        tokio::task::spawn_local(async move { accept_loop(listener, acceptor, shared).await });
158    }
159
160    std::future::pending().await
161}
162
163struct Shared {
164    interfaces: Arc<Interfaces>,
165    by_address: HashMap<Address, usize>,
166    tunnel: crate::config::TunnelConfig,
167    client_names: Vec<String>,
168    ingress: mpsc::Sender<Ingress>,
169}
170
171async fn accept_loop(listener: TcpListener, acceptor: TlsAcceptor, shared: Arc<Shared>) {
172    loop {
173        let (stream, peer) = match listener.accept().await {
174            Ok(accepted) => accepted,
175            Err(error) => {
176                // A per-connection accept failure (a descriptor limit, a
177                // connection reset between accept and return) must not
178                // take the listener down with it.
179                tracing::warn!("accept failed: {error}");
180                tokio::time::sleep(Duration::from_millis(100)).await;
181                continue;
182            }
183        };
184        stream.set_nodelay(true).ok();
185        let acceptor = acceptor.clone();
186        let shared = shared.clone();
187        tokio::task::spawn_local(async move {
188            if let Err(error) = serve(stream, acceptor, shared).await {
189                tracing::info!(%peer, "connection ended: {error:#}");
190            }
191        });
192    }
193}

195async fn serve(stream: TcpStream, acceptor: TlsAcceptor, shared: Arc<Shared>) -> Result<()> {
196    let peer = stream.peer_addr().ok();
197    let stream = acceptor.accept(stream).await.context("TLS handshake")?;
198
199    // The handshake already proved the peer holds a pinned identity's
200    // key, and proved it for the key its certificate names; this is
201    // only working out *which* client that was.
202    let address = {
203        let (_, connection) = stream.get_ref();
204        let certificate = connection
205            .peer_certificates()
206            .and_then(<[_]>::first)
207            .ok_or_else(|| anyhow!("the client presented no certificate"))?;
208        tls::certificate_key(certificate)
209            .ok_or_else(|| anyhow!("the client's certificate carries no key"))?
210    };
211    let client = *shared
212        .by_address
213        .get(&address)
214        .ok_or_else(|| anyhow!("no client is configured for {address}"))?;
215    let iface = shared
216        .interfaces
217        .by_client(client)
218        .ok_or_else(|| anyhow!("client {client} has no interface"))?
219        .clone();
220    let name = shared.client_names[client].clone();
221
222    // A second connection for the same credential displaces the first:
223    // the newer one is the live client, and the older is a session that
224    // has already lost its network path. Clearing the egress queue ends
225    // the old writer and discards its backlog in one move; the
226    // generation that clear establishes is this connection's claim on
227    // the interface.
228    if iface.is_connected() {
229        tracing::info!(client = %name, "displacing an earlier connection");
230    }
231    iface.egress.clear();
232    let session = iface.egress.generation();
233    iface.set_connected(true);
234    tracing::info!(client = %name, ?peer, "client attached");
235
236    let (read_half, write_half) = tokio::io::split(stream);
237    let mut reader = TunnelReader::new(
238        read_half,
239        Duration::from_secs(shared.tunnel.idle_timeout_secs),
240    );
241    let mut writer = TunnelWriter::new(
242        write_half,
243        Duration::from_secs(shared.tunnel.keepalive_secs),
244    );
245
246    let egress = iface.egress.clone();
247    let pump =
248        tokio::task::spawn_local(async move { pump_writer(&mut writer, &egress, session).await });
249
250    let outcome = loop {
251        match reader.read_frame().await {
252            Ok(frame) => {
253                let ingress = Ingress {
254                    iface: iface.id,
255                    frame,
256                };
257                if shared.ingress.send(ingress).await.is_err() {
258                    break Err(anyhow!("the hub stopped"));
259                }
260            }
261            Err(error) => break Err(anyhow!("{error}")),
262        }
263    };
264
265    // Only tear the interface down if this connection is still the one
266    // holding it. A displaced session lingers until its idle timeout
267    // fires, and by then a successor owns the interface—marking it
268    // down here would silence the live session until *its* idle timer
269    // noticed. The successor's clear bumped the generation, which is how
270    // this connection knows its claim has lapsed.
271    if iface.egress.generation() == session {
272        iface.set_connected(false);
273        iface.egress.clear();
274    }
275    pump.abort();
276    tracing::info!(
277        client = %name,
278        malformed = reader.malformed(),
279        "client detached"
280    );
281    outcome
282}

284/// Move an interface's received frames into the hub.
285fn spawn_ingress(inbound: Arc<TunnelQueue>, iface: InterfaceId, ingress: mpsc::Sender<Ingress>) {
286    tokio::task::spawn_local(async move {
287        loop {
288            match inbound.pop().await {
289                crate::tunnel::Dequeued::Frame(frame) => {
290                    if ingress.send(Ingress { iface, frame }).await.is_err() {
291                        return;
292                    }
293                }
294                crate::tunnel::Dequeued::SessionEnded => return,
295            }
296        }
297    });
298}

300/// Wire a validated configuration to whichever role it describes.
301pub async fn run_config(mut config: Config) -> Result<()> {
302    let identity = BridgeIdentity::load(
303        &config
304            .identity
305            .take()
306            .expect("validated: every role has an identity")
307            .key_file,
308    )?;
309    match config.client {
310        Some(client) => crate::client::run(identity, client).await,
311        None => run(identity, config).await,
312    }
