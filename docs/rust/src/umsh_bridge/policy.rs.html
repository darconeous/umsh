1//! Who may send how much, and where it may go.
2//!
3//! Two things live here, and they exist for the same reason. An
4//! authenticated but misbehaving client is the realistic failure mode of
5//! a bridge, so the per-client rate limit is policy rather than a
6//! backstop — the device's duty ledger is the backstop. And two clients
7//! whose radios share a segment double every fanned-out frame's airtime,
8//! which the egress allowlist is the place to prevent.
9//!
10//! Both are decisions about a connection, not about a packet: the bridge
11//! reads neither. What a packet is, whether it has been seen before, and
12//! whether it is worth putting on the air are questions for the repeater
13//! behind each participant's radio.
14
15use std::collections::HashSet;
16use std::time::Duration;
17
18use anyhow::{Result, bail};
19use tokio::time::Instant;
20
21use crate::config::ServerConfig;
22use crate::iface::{InterfaceId, InterfaceKind, Interfaces};
23
24pub struct Policy {
25    /// One entry per interface, parallel to [`Interfaces::all`].
26    interfaces: Vec<InterfacePolicy>,
27}
28
29struct InterfacePolicy {
30    name: String,
31    /// `None` is exempt. Only the server's own radio is: a bridge that
32    /// rate-limited its own segment would be throttling the mesh it is
33    /// part of, not a remote peer it is protecting itself from.
34    limit: Option<RateLimit>,
35    /// Interfaces this one's traffic may leave through. `None` means
36    /// all of them.
37    allow_to: Option<HashSet<InterfaceId>>,
38}
39
40impl Policy {
41    /// Resolve the configured names against the interfaces that exist.
42    pub fn build(config: &ServerConfig, interfaces: &Interfaces) -> Result<Self> {
43        let resolve = |owner: &str, names: &Option<Vec<String>>| {
44            names
45                .as_ref()
46                .map(|names| {
47                    names
48                        .iter()
49                        .map(|name| {
50                            interfaces
51                                .all
52                                .iter()
53                                .find(|iface| &iface.name == name)
54                                .map(|iface| iface.id)
55                                .ok_or_else(|| {
56                                    anyhow::anyhow!(
57                                        "\"{owner}\" allows forwarding to \"{name}\", which is \
58                                         not an interface"
59                                    )
60                                })
61                        })
62                        .collect::<Result<HashSet<_>>>()
63                })
64                .transpose()
65        };
66
67        let entries = interfaces
68            .all
69            .iter()
70            .map(|iface| {
71                let (limit, allow_to) = match iface.kind {
72                    InterfaceKind::Radio => (None, None),
73                    InterfaceKind::Client(index) => {
74                        let client = &config.clients[index];
75                        (
76                            client.max_frames_per_minute.map(RateLimit::per_minute),
77                            resolve(&client.name, &client.allow_to)?,
78                        )
79                    }
80                    // A host's limit is required by the configuration:
81                    // it spends no airtime of its own, but every frame
82                    // it injects is transmitted by every participant's
83                    // node.
84                    InterfaceKind::Host(index) => {
85                        let host = &config.hosts[index];
86                        (
87                            Some(RateLimit::per_minute(host.max_frames_per_minute)),
88                            resolve(&host.name, &host.allow_to)?,
89                        )
90                    }
91                };
92                Ok(InterfacePolicy {
93                    name: iface.name.clone(),
94                    limit,
95                    allow_to,
96                })
97            })
98            .collect::<Result<Vec<_>>>()?;
99
100        if config.radio.is_none()
101            && entries
102                .iter()
103                .all(|entry| entry.allow_to.as_ref().is_some_and(HashSet::is_empty))
104        {
105            bail!("every allow_to is empty; no frame could ever be forwarded");
106        }
107
108        Ok(Self {
109            interfaces: entries,
110        })
111    }
112
113    /// Spend one from the arrival interface's forwarding budget.
114    pub fn admit(&mut self, arrival: InterfaceId, now: Instant) -> bool {
115        let policy = &mut self.interfaces[arrival];
116        let Some(limit) = policy.limit.as_mut() else {
117            return true;
118        };
119        if limit.take(now) {
120            return true;
121        }
122        // Once per interval, not once per frame: a peer hammering the
123        // bridge must not also flood the log.
124        if limit.note_refusal(now) {
125            tracing::warn!(
126                iface = %policy.name,
127                "rate limit reached; frames from this interface are being dropped"
128            );
129        }
130        false
131    }
132
133    /// Whether a frame that arrived on `arrival` may leave through
134    /// `exit`.
135    pub fn may_forward(&self, arrival: InterfaceId, exit: InterfaceId) -> bool {
136        if arrival == exit {
137            return false;
138        }
139        match &self.interfaces[arrival].allow_to {
140            None => true,
141            Some(allowed) => allowed.contains(&exit),
142        }
143    }
144
145    /// How the configured egress rules read, for `check` and start-up
146    /// logging.
147    pub fn describe_fan_out(&self, arrival: InterfaceId) -> String {
148        let names: Vec<&str> = self
149            .interfaces
150            .iter()
151            .enumerate()
152            .filter(|(exit, _)| self.may_forward(arrival, *exit))
153            .map(|(_, entry)| entry.name.as_str())
154            .collect();
155        if names.is_empty() {
156            "nothing".to_string()
157        } else {
158            names.join(", ")
159        }
160    }
161}

163/// A token bucket that refills continuously, so a client may burst up to
164/// its whole minute's allowance and then settles to the steady rate.
165struct RateLimit {
166    capacity: f64,
167    per_second: f64,
168    tokens: f64,
169    updated: Option<Instant>,
170    last_complaint: Option<Instant>,
171}
172
173/// How often a rate-limited client is mentioned in the log.
174const COMPLAINT_INTERVAL: Duration = Duration::from_secs(60);
175
176impl RateLimit {
177    fn per_minute(frames: u32) -> Self {
178        let capacity = f64::from(frames.max(1));
179        Self {
180            capacity,
181            per_second: capacity / 60.0,
182            tokens: capacity,
183            updated: None,
184            last_complaint: None,
185        }
186    }
187
188    fn take(&mut self, now: Instant) -> bool {
189        if let Some(updated) = self.updated {
190            let elapsed = now.saturating_duration_since(updated).as_secs_f64();
191            self.tokens = (self.tokens + elapsed * self.per_second).min(self.capacity);
192        }
193        self.updated = Some(now);
194        if self.tokens >= 1.0 {
195            self.tokens -= 1.0;
196            true
197        } else {
198            false
199        }
200    }
201
202    fn note_refusal(&mut self, now: Instant) -> bool {
203        let due = self
204            .last_complaint
205            .is_none_or(|last| now.saturating_duration_since(last) >= COMPLAINT_INTERVAL);
206        if due {
207            self.last_complaint = Some(now);
208        }
209        due
210    }
211}

213#[cfg(test)]
214mod tests {
215    use super::*;
216    use crate::config::Config;
217
218    fn parse(extra: &str) -> (ServerConfig, Interfaces, Policy) {
219        let text = format!("[identity]\nkey_file = \"k\"\n[server]\n{extra}");
220        let config: Config = toml::from_str(&text).unwrap();
221        config.validate().unwrap();
222        let server = config.server.unwrap();
223        let interfaces = Interfaces::build(&server);
224        let policy = Policy::build(&server, &interfaces).unwrap();
225        (server, interfaces, policy)
226    }
227
228    /// A pinned-client address the fixtures can use.
229    fn address(seed: u8) -> String {
230        use umsh_crypto::NodeIdentity as _;
231        umsh_crypto::software::SoftwareIdentity::from_secret_bytes(&[seed; 32])
232            .public_key()
233            .to_string()
234    }
235
236    fn two_clients(extra: &str) -> (ServerConfig, Interfaces, Policy) {
237        parse(&format!(
238            "[server.radio]\ntype = \"ble\"\n\
239             [[server.clients]]\nname = \"cabin\"\naddress = \"{}\"\n{extra}\
240             [[server.clients]]\nname = \"summit\"\naddress = \"{}\"\n",
241            address(0xAA),
242            address(0xBB)
243        ))
244    }
245
246    #[test]
247    fn a_frame_never_goes_back_out_the_way_it_came() {
248        let (_, interfaces, policy) = two_clients("");
249        for iface in &interfaces.all {
250            assert!(!policy.may_forward(iface.id, iface.id));
251        }
252    }
253
254    #[test]
255    fn an_absent_allowlist_means_every_other_interface() {
256        let (_, interfaces, policy) = two_clients("");
257        let cabin = interfaces.by_client(0).unwrap().id;
258        assert_eq!(
259            policy.describe_fan_out(cabin),
260            "radio, summit",
261            "the arrival interface is excluded, the rest are not"
262        );
263    }
264
265    #[test]
266    fn an_allowlist_is_the_whole_list() {
267        let (_, interfaces, policy) = two_clients("allow_to = [\"radio\"]\n");
268        let cabin = interfaces.by_client(0).unwrap().id;
269        assert_eq!(policy.describe_fan_out(cabin), "radio");
270
271        // Two clients on the same segment: excluding one from the
272        // other's fan-out is how the duplicate airtime is avoided.
273        let summit = interfaces.by_client(1).unwrap().id;
274        assert!(!policy.may_forward(cabin, summit));
275        assert!(
276            policy.may_forward(summit, cabin),
277            "the rule is per-arrival, not symmetric"
278        );
279    }
280
281    #[test]
282    fn the_servers_own_radio_forwards_everywhere_regardless() {
283        let (_, interfaces, policy) = two_clients("allow_to = []\n");
284        let radio = interfaces.radio.unwrap();
285        assert_eq!(policy.describe_fan_out(radio), "cabin, summit");
286    }
287
288    #[tokio::test(start_paused = true)]
289    async fn a_client_spends_a_budget_that_refills_over_the_minute() {
290        let (_, interfaces, mut policy) = two_clients("max_frames_per_minute = 60\n");
291        let cabin = interfaces.by_client(0).unwrap().id;
292
293        // The full minute's allowance is available as a burst.
294        for _ in 0..60 {
295            assert!(policy.admit(cabin, Instant::now()));
296        }
297        assert!(!policy.admit(cabin, Instant::now()));
298
299        // One frame per second thereafter.
300        tokio::time::sleep(Duration::from_secs(1)).await;
301        assert!(policy.admit(cabin, Instant::now()));
302        assert!(!policy.admit(cabin, Instant::now()));
303    }
304
305    #[tokio::test(start_paused = true)]
306    async fn the_servers_own_radio_and_an_unlimited_client_are_never_refused() {
307        let (_, interfaces, mut policy) = two_clients("");
308        let radio = interfaces.radio.unwrap();
309        let cabin = interfaces.by_client(0).unwrap().id;
310        for _ in 0..1000 {
311            assert!(policy.admit(radio, Instant::now()));
312            assert!(policy.admit(cabin, Instant::now()));
313        }
314    }
