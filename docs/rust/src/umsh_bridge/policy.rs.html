1//! Step 5 of the forwarding procedure: local bridge policy.
2//!
3//! Two things live here, and they exist for the same reason. An
4//! authenticated but misbehaving client is the realistic failure mode of
5//! a bridge, so the per-client rate limit is policy rather than a
6//! backstop — the device's duty ledger is the backstop. And two clients
7//! whose radios share a segment double every fanned-out frame's airtime,
8//! which the egress allowlist is the place to prevent.
9//!
10//! Region matching is here too, and only ever *matches*. A bridge never
11//! rewrites a region code and never inserts one: its interfaces may sit
12//! in different regions, so a code added at one segment's exit would
13//! misdescribe the packet everywhere else it travels.
14
15use std::collections::HashSet;
16use std::time::Duration;
17
18use anyhow::{Result, bail};
19use tokio::time::Instant;
20use umsh_core::{ParsedOptions, RegionCode};
21
22use crate::config::ServerConfig;
23use crate::iface::{InterfaceId, Interfaces};
24
25pub struct Policy {
26    /// Indexed by client index, parallel to `ServerConfig::clients`.
27    clients: Vec<ClientPolicy>,
28    regions: Vec<RegionCode>,
29}
30
31struct ClientPolicy {
32    name: String,
33    limit: Option<RateLimit>,
34    /// Interfaces this client's traffic may leave through. `None` means
35    /// all of them.
36    allow_to: Option<HashSet<InterfaceId>>,
37    suppress_flood_confirmations: bool,
38}
39
40impl Policy {
41    /// Resolve the configured names against the interfaces that exist.
42    pub fn build(config: &ServerConfig, interfaces: &Interfaces) -> Result<Self> {
43        let clients = config
44            .clients
45            .iter()
46            .map(|client| {
47                let allow_to = client
48                    .allow_to
49                    .as_ref()
50                    .map(|names| {
51                        names
52                            .iter()
53                            .map(|name| {
54                                interfaces
55                                    .all
56                                    .iter()
57                                    .find(|iface| &iface.name == name)
58                                    .map(|iface| iface.id)
59                                    .ok_or_else(|| {
60                                        anyhow::anyhow!(
61                                            "client \"{}\" allows forwarding to \"{name}\", \
62                                             which is not an interface",
63                                            client.name
64                                        )
65                                    })
66                            })
67                            .collect::<Result<HashSet<_>>>()
68                    })
69                    .transpose()?;
70                Ok(ClientPolicy {
71                    name: client.name.clone(),
72                    limit: client.max_frames_per_minute.map(RateLimit::per_minute),
73                    allow_to,
74                    suppress_flood_confirmations: client.suppress_flood_confirmations,
75                })
76            })
77            .collect::<Result<Vec<_>>>()?;
78
79        if config.radio.is_none()
80            && clients
81                .iter()
82                .all(|client| client.allow_to.as_ref().is_some_and(HashSet::is_empty))
83        {
84            bail!("every client's allow_to is empty; no frame could ever be forwarded");
85        }
86
87        Ok(Self {
88            clients,
89            regions: config
90                .forwarding
91                .regions
92                .iter()
93                .map(|region| region.0)
94                .collect(),
95        })
96    }
97
98    /// Whether this bridge carries traffic tagged with the frame's
99    /// region. An untagged frame is carried: a bridge has no way to know
100    /// which region it came from, and refusing it would break every
101    /// sender that does not tag.
102    pub fn region_admits(&self, options: &ParsedOptions) -> bool {
103        if self.regions.is_empty() {
104            return true;
105        }
106        match options.region_code {
107            None => true,
108            Some(code) => self.regions.contains(&RegionCode::from_bytes(code)),
109        }
110    }
111
112    /// Spend one from the arrival interface's forwarding budget.
113    ///
114    /// The server's own radio has no budget: a bridge that rate-limited
115    /// its own segment would be throttling the mesh it is part of, not a
116    /// remote peer it is protecting itself from.
117    pub fn admit(&mut self, interfaces: &Interfaces, arrival: InterfaceId, now: Instant) -> bool {
118        let Some(client) = interfaces.get(arrival).client else {
119            return true;
120        };
121        let policy = &mut self.clients[client];
122        let Some(limit) = policy.limit.as_mut() else {
123            return true;
124        };
125        if limit.take(now) {
126            return true;
127        }
128        // Once per interval, not once per frame: a client hammering the
129        // tunnel must not also flood the log.
130        if limit.note_refusal(now) {
131            tracing::warn!(
132                client = %policy.name,
133                "rate limit reached; frames from this client are being dropped"
134            );
135        }
136        false
137    }
138
139    /// Whether a frame that arrived on `arrival` may leave through
140    /// `exit`.
141    pub fn may_forward(
142        &self,
143        interfaces: &Interfaces,
144        arrival: InterfaceId,
145        exit: InterfaceId,
146    ) -> bool {
147        if arrival == exit {
148            return false;
149        }
150        match interfaces.get(arrival).client {
151            None => true,
152            Some(client) => match &self.clients[client].allow_to {
153                None => true,
154                Some(allowed) => allowed.contains(&exit),
155            },
156        }
157    }
158
159    /// Whether the device behind this interface runs its own repeater
160    /// role, whose flood re-forward already confirms the previous hop.
161    pub fn suppresses_flood_confirmations(
162        &self,
163        interfaces: &Interfaces,
164        arrival: InterfaceId,
165    ) -> bool {
166        interfaces
167            .get(arrival)
168            .client
169            .is_some_and(|client| self.clients[client].suppress_flood_confirmations)
170    }
171
172    /// How the configured egress rules read, for `check` and start-up
173    /// logging.
174    pub fn describe_fan_out(&self, interfaces: &Interfaces, arrival: InterfaceId) -> String {
175        let names: Vec<&str> = interfaces
176            .all
177            .iter()
178            .filter(|iface| self.may_forward(interfaces, arrival, iface.id))
179            .map(|iface| iface.name.as_str())
180            .collect();
181        if names.is_empty() {
182            "nothing".to_string()
183        } else {
184            names.join(", ")
185        }
186    }
187}

189/// A token bucket that refills continuously, so a client may burst up to
190/// its whole minute's allowance and then settles to the steady rate.
191struct RateLimit {
192    capacity: f64,
193    per_second: f64,
194    tokens: f64,
195    updated: Option<Instant>,
196    last_complaint: Option<Instant>,
197}
198
199/// How often a rate-limited client is mentioned in the log.
200const COMPLAINT_INTERVAL: Duration = Duration::from_secs(60);
201
202impl RateLimit {
203    fn per_minute(frames: u32) -> Self {
204        let capacity = f64::from(frames.max(1));
205        Self {
206            capacity,
207            per_second: capacity / 60.0,
208            tokens: capacity,
209            updated: None,
210            last_complaint: None,
211        }
212    }
213
214    fn take(&mut self, now: Instant) -> bool {
215        if let Some(updated) = self.updated {
216            let elapsed = now.saturating_duration_since(updated).as_secs_f64();
217            self.tokens = (self.tokens + elapsed * self.per_second).min(self.capacity);
218        }
219        self.updated = Some(now);
220        if self.tokens >= 1.0 {
221            self.tokens -= 1.0;
222            true
223        } else {
224            false
225        }
226    }
227
228    fn note_refusal(&mut self, now: Instant) -> bool {
229        let due = self
230            .last_complaint
231            .is_none_or(|last| now.saturating_duration_since(last) >= COMPLAINT_INTERVAL);
232        if due {
233            self.last_complaint = Some(now);
234        }
235        due
236    }
237}

244    fn parse(extra: &str) -> (ServerConfig, Interfaces, Policy) {
245        let text = format!("[identity]\nkey_file = \"k\"\n[server]\n{extra}");
246        let config: Config = toml::from_str(&text).unwrap();
247        config.validate().unwrap();
248        let server = config.server.unwrap();
249        let interfaces = Interfaces::build(&server);
250        let policy = Policy::build(&server, &interfaces).unwrap();
251        (server, interfaces, policy)
252    }
253
254    /// A pinned-client address the fixtures can use.
255    fn address(seed: u8) -> String {
256        use umsh_crypto::NodeIdentity as _;
257        umsh_crypto::software::SoftwareIdentity::from_secret_bytes(&[seed; 32])
258            .public_key()
259            .to_string()
260    }
261
262    fn two_clients(extra: &str) -> (ServerConfig, Interfaces, Policy) {
263        parse(&format!(
264            "[server.radio]\ntype = \"ble\"\n\
265             [[server.clients]]\nname = \"cabin\"\naddress = \"{}\"\n{extra}\
266             [[server.clients]]\nname = \"summit\"\naddress = \"{}\"\n",
267            address(0xAA),
268            address(0xBB)
269        ))
270    }

272    #[test]
273    fn a_frame_never_goes_back_out_the_way_it_came() {
274        let (_, interfaces, policy) = two_clients("");
275        for iface in &interfaces.all {
276            assert!(!policy.may_forward(&interfaces, iface.id, iface.id));
277        }

280    #[test]
281    fn an_absent_allowlist_means_every_other_interface() {
282        let (_, interfaces, policy) = two_clients("");
283        let cabin = interfaces.by_client(0).unwrap().id;
284        assert_eq!(
285            policy.describe_fan_out(&interfaces, cabin),
286            "radio, summit",
287            "the arrival interface is excluded, the rest are not"

291    #[test]
292    fn an_allowlist_is_the_whole_list() {
293        let (_, interfaces, policy) = two_clients("allow_to = [\"radio\"]\n");
294        let cabin = interfaces.by_client(0).unwrap().id;
295        assert_eq!(policy.describe_fan_out(&interfaces, cabin), "radio");
296
297        // Two clients on the same segment: excluding one from the
298        // other's fan-out is how the duplicate airtime is avoided.
299        let summit = interfaces.by_client(1).unwrap().id;
300        assert!(!policy.may_forward(&interfaces, cabin, summit));
301        assert!(
302            policy.may_forward(&interfaces, summit, cabin),
303            "the rule is per-arrival, not symmetric"
304        );
305    }

310        let radio = interfaces.radio.unwrap();
311        assert_eq!(policy.describe_fan_out(&interfaces, radio), "cabin, summit");
312    }

314    #[tokio::test(start_paused = true)]
315    async fn a_client_spends_a_budget_that_refills_over_the_minute() {
316        let (_, interfaces, mut policy) = two_clients("max_frames_per_minute = 60\n");
317        let cabin = interfaces.by_client(0).unwrap().id;
318
319        // The full minute's allowance is available as a burst.
320        for _ in 0..60 {
321            assert!(policy.admit(&interfaces, cabin, Instant::now()));
322        }
323        assert!(!policy.admit(&interfaces, cabin, Instant::now()));
324
325        // One frame per second thereafter.
326        tokio::time::sleep(Duration::from_secs(1)).await;
327        assert!(policy.admit(&interfaces, cabin, Instant::now()));
328        assert!(!policy.admit(&interfaces, cabin, Instant::now()));
329    }

342    #[test]
343    fn region_matching_carries_untagged_traffic_and_filters_the_rest() {
344        let (_, _, policy) = parse(&format!(
345            "[server.radio]\ntype = \"ble\"\n\
346             [server.forwarding]\nregions = [\"SJC\"]\n\
347             [[server.clients]]\nname = \"cabin\"\naddress = \"{}\"\n",
348            address(0xAA)
349        ));
350        let sjc = RegionCode::from_iata("SJC").unwrap();
351        let mut options = ParsedOptions::default();
352        assert!(policy.region_admits(&options), "untagged");
353
354        options.region_code = Some(sjc.to_bytes());
355        assert!(policy.region_admits(&options));
356
357        options.region_code = Some(RegionCode::from_iata("LHR").unwrap().to_bytes());
358        assert!(!policy.region_admits(&options));
359
360        // With no regions configured there is no restriction at all.
361        let (_, _, policy) = two_clients("");
362        assert!(policy.region_admits(&options));
363    }
