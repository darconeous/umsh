1//! `umsh-bridge`: one end of a UMSH internet bridge.
2//!
3//! A bridge carries frames between radios that cannot hear each other.
4//! One process is the **server**, which authenticates participants and
5//! copies each frame it receives to all the others; the rest are
6//! **clients**, which relay frames byte for byte between their radio and
7//! the tunnel. Every participant's radio runs in backhaul mode, so the
8//! node behind it is a repeater whose point-to-point neighbor is the
9//! bridge: duplicate suppression, hop accounting, and forwarding policy
10//! are that node's, and the bridge itself has no opinion about what it
11//! carries.
12//!
13//! The protocol is specified in the [Internet Bridging appendix].
14//!
15//! This is a library only so that the integration tests can stand a
16//! whole bridge up in one process; the binary is its only consumer.
17//!
18//! [Internet Bridging appendix]: https://darconeous.github.io/umsh/docs/protocol/internet-bridging.html
19

41pub fn run(args: ToolArgs) -> Result<()> {
42    match args.command {
43        Command::Check(run_args) => check::check(&run_args.config),
44        Command::Run(run_args) => run_bridge(&run_args.config),
45        Command::Address { path } => keygen::print_address(&path),
46        Command::Keygen(KeygenCommand::Identity { path, force }) => {
47            keygen::write_identity(&path, force)
48        }
49    }
50}

52/// Run until interrupted.
53///
54/// A single-threaded runtime with a `LocalSet`: the work is at LoRa
55/// packet rates, the hub's state is deliberately not shared, and a BLE
56/// or serial device handle need not be `Send` to be spawned alongside
57/// everything else.
58fn run_bridge(path: &Path) -> Result<()> {
59    let config = config::Config::load(path)?;
60
61    let runtime = tokio::runtime::Builder::new_current_thread()
62        .enable_all()
63        .build()
64        .context("building the async runtime")?;
65    let local = tokio::task::LocalSet::new();
66
67    runtime.block_on(local.run_until(async move {
68        tokio::select! {
69            result = server::run_config(config) => result,
70            result = tokio::signal::ctrl_c() => {
71                result.context("waiting for an interrupt")?;
72                tracing::info!("interrupted; shutting down");
73                Ok(())
74            }
75        }
76    }))
77}
