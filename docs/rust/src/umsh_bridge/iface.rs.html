1//! The server's interfaces: its own radio, plus one per configured
2//! client.
3//!
4//! The set is static — clients are named in the configuration, and a
5//! disconnected client is an interface that is down, not an interface
6//! that is gone — so an interface is an index for the whole run and the
7//! engine can name one in a log line without a lookup.
8
9use std::sync::Arc;
10use std::sync::atomic::{AtomicBool, Ordering};
11use std::time::Duration;
12
13use crate::config::{RADIO_INTERFACE, ServerConfig};
14use crate::tunnel::{TunnelFrame, TunnelQueue};
15
16/// Index into [`Interfaces::all`].
17pub type InterfaceId = usize;
18
19pub struct Interface {
20    pub id: InterfaceId,
21    pub name: String,
22    /// Frames waiting to leave through this interface.
23    pub egress: Arc<TunnelQueue>,
24    /// The client this interface belongs to, as an index into
25    /// `ServerConfig::clients`. `None` for the server's own radio.
26    pub client: Option<usize>,
27    connected: AtomicBool,
28}
29
30impl Interface {
31    pub fn is_connected(&self) -> bool {
32        self.connected.load(Ordering::Relaxed)
33    }
34
35    pub fn set_connected(&self, connected: bool) {
36        self.connected.store(connected, Ordering::Relaxed);
37    }
38
39    /// Queue a frame for transmission, unless the interface is down —
40    /// in which case there is nothing to hold it for.
41    pub fn send(&self, frame: TunnelFrame) {
42        if !self.is_connected() {
43            return;
44        }
45        if self.egress.push(frame) {
46            tracing::warn!(iface = %self.name, "egress queue full; dropped the oldest frame");
47        }
48    }
49}
50
51pub struct Interfaces {
52    pub all: Vec<Arc<Interface>>,
53    /// The server's own radio, when it has one.
54    pub radio: Option<InterfaceId>,
55}
56
57impl Interfaces {
58    pub fn build(config: &ServerConfig) -> Self {
59        let max_age = Duration::from_secs(config.tunnel.max_frame_age_secs);
60        let depth = config.tunnel.queue_depth;
61        let mut all: Vec<Arc<Interface>> = Vec::new();
62
63        let radio = (!config.radio.is_none()).then(|| {
64            all.push(Arc::new(Interface {
65                id: 0,
66                name: RADIO_INTERFACE.to_string(),
67                egress: Arc::new(TunnelQueue::new(depth, max_age)),
68                client: None,
69                // Always up: the relay re-opens a failed device on its
70                // own, and the bounded, staleness-dropped queue holds
71                // what little accrues in the meantime.
72                connected: AtomicBool::new(true),
73            }));
74            0
75        });
76
77        for (index, client) in config.clients.iter().enumerate() {
78            all.push(Arc::new(Interface {
79                id: all.len(),
80                name: client.name.clone(),
81                egress: Arc::new(TunnelQueue::new(depth, max_age)),
82                client: Some(index),
83                connected: AtomicBool::new(false),
84            }));
85        }
86
87        Self { all, radio }
88    }
89
90    pub fn get(&self, id: InterfaceId) -> &Arc<Interface> {
91        &self.all[id]
92    }
93
94    pub fn by_client(&self, client: usize) -> Option<&Arc<Interface>> {
95        self.all.iter().find(|iface| iface.client == Some(client))
96    }
97
98    pub fn name(&self, id: InterfaceId) -> &str {
99        &self.all[id].name
100    }
101}
