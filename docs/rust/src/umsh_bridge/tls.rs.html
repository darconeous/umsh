1//! The tunnel's TLS 1.3 layer: mutual authentication by UMSH identity.
2//!
3//! Every participant holds an Ed25519 identity — the server's is the
4//! bridge's node identity, and a client's is its own, ready to become a
5//! mesh-addressable management identity later. Each side is configured
6//! with the peer public keys it will accept, written as canonical UMSH
7//! addresses.
8//!
9//! X.509 never enters the trust decision. TLS requires a certificate, so
10//! each endpoint mints a throwaway self-signed one around its identity
11//! key at startup; what the verifier actually checks is the TLS 1.3
12//! handshake signature — proof of possession of the identity key over a
13//! transcript both sides contributed randomness to — against the pinned
14//! public key. A certificate is a formality here: nothing it claims is
15//! believed, nothing about it is stored, and revoking a client is an
16//! edit to the server's configuration rather than a PKI operation.
17

35/// ALPN identifier from the spec. A future revision of the tunnel wire
36/// protocol picks a new one and fails the handshake against this one
37/// rather than misparsing its frames.
38pub const ALPN: &[u8] = b"umsh-bridge/1";

40/// A participant's identity in its public form: an Ed25519 public key,
41/// written and parsed as the canonical UMSH address (fixed-width base58,
42/// or 64 hex digits).
43#[derive(Clone, Copy, PartialEq, Eq, Hash)]
44pub struct Address(pub PublicKey);

58impl FromStr for Address {
59    type Err = anyhow::Error;
60
61    fn from_str(text: &str) -> Result<Self> {
62        let key: PublicKey = text
63            .trim()
64            .parse()
65            .map_err(|error| anyhow!("not a UMSH address: {error:?}"))?;
66        // Refused now rather than at connection time: a corrupted
67        // address can never verify a signature, so the config that
68        // carries it is broken however plausible it looks.
69        if !umsh_crypto::is_valid_ed25519_public_key(&key) {
70            bail!("not a valid Ed25519 public key");
71        }
72        Ok(Self(key))
73    }
74}

83/// The credential this participant presents: its identity key wrapped in
84/// a certificate minted in memory. Nothing here touches the filesystem.
85pub struct Credential {
86    pub chain: Vec<CertificateDer<'static>>,
87    pub key: PrivateKeyDer<'static>,
88    pub address: Address,
89}
90
91impl Credential {
92    pub fn for_identity(identity: &BridgeIdentity) -> Result<Self> {
93        let pkcs8 = pkcs8_of_seed(identity.seed());
94        let key_pair = rcgen::KeyPair::from_pkcs8_der_and_sign_algo(
95            &PrivatePkcs8KeyDer::from(pkcs8.as_slice()),
96            &rcgen::PKCS_ED25519,
97        )
98        .context("wrapping the identity key for certificate issuance")?;
99
100        let address = Address(*identity.public_key());
101        // rcgen's default validity window is 1975 to 4096; since the
102        // verifier never consults it, wider is better — a window that
103        // could expire would only ever be a scheduled outage.
104        let mut params = rcgen::CertificateParams::new(vec!["umsh-bridge.invalid".to_string()])
105            .context("building certificate params")?;
106        params
107            .distinguished_name
108            .push(rcgen::DnType::CommonName, address.to_string());
109        let certificate = params
110            .self_signed(&key_pair)
111            .context("self-signing the identity certificate")?;
112
113        Ok(Self {
114            chain: vec![certificate.der().clone().into_owned()],
115            key: PrivateKeyDer::Pkcs8(PrivatePkcs8KeyDer::from(pkcs8)),
116            address,
117        })
118    }
119}

121/// PKCS#8 v1 encoding of an Ed25519 seed — the fixed 16-byte prefix is
122/// the whole ASN.1 structure for a key this shape.
123fn pkcs8_of_seed(seed: &[u8; 32]) -> Vec<u8> {
124    const PREFIX: [u8; 16] = [
125        0x30, 0x2e, 0x02, 0x01, 0x00, 0x30, 0x05, 0x06, 0x03, 0x2b, 0x65, 0x70, 0x04, 0x22, 0x04,
126        0x20,
127    ];
128    let mut der = Vec::with_capacity(PREFIX.len() + seed.len());
129    der.extend_from_slice(&PREFIX);
130    der.extend_from_slice(seed);
131    der
132}

134/// The Ed25519 SubjectPublicKeyInfo is a fixed 44-byte structure whose
135/// 12-byte prefix is distinctive enough to scan for, which keeps a
136/// hostile certificate's DER out of any real parser. If a crafted
137/// certificate carries the pattern more than once, the first match wins
138/// — harmless, because the handshake signature must verify against the
139/// extracted key, so *naming* a key is worthless without holding it.
140pub fn certificate_key(cert: &CertificateDer<'_>) -> Option<Address> {
141    const SPKI_PREFIX: [u8; 12] = [
142        0x30, 0x2a, 0x30, 0x05, 0x06, 0x03, 0x2b, 0x65, 0x70, 0x03, 0x21, 0x00,
143    ];
144    let der = cert.as_ref();
145    let start = der
146        .windows(SPKI_PREFIX.len())
147        .position(|window| window == SPKI_PREFIX)?
148        + SPKI_PREFIX.len();
149    let key: [u8; 32] = der.get(start..start + 32)?.try_into().ok()?;
150    Some(Address(PublicKey(key)))
151}

153/// Server side: accept exactly the identities in `accepted`.
154pub fn server_config(credential: &Credential, accepted: Vec<Address>) -> Result<Arc<ServerConfig>> {
155    let verifier = Arc::new(IdentityClientVerifier { accepted });
156    let mut config = ServerConfig::builder_with_provider(provider())
157        .with_protocol_versions(&[&rustls::version::TLS13])
158        .context("TLS 1.3 is not available in this build")?
159        .with_client_cert_verifier(verifier)
160        .with_single_cert(credential.chain.clone(), credential.key.clone_key())
161        .context("the minted certificate and identity key do not go together")?;
162    config.alpn_protocols = vec![ALPN.to_vec()];
163    Ok(Arc::new(config))
164}
165
166/// Client side: accept exactly the server identity pinned in config.
167pub fn client_config(credential: &Credential, server: Address) -> Result<Arc<ClientConfig>> {
168    let verifier = Arc::new(IdentityServerVerifier { accepted: server });
169    let mut config = ClientConfig::builder_with_provider(provider())
170        .with_protocol_versions(&[&rustls::version::TLS13])
171        .context("TLS 1.3 is not available in this build")?
172        .dangerous()
173        .with_custom_certificate_verifier(verifier)
174        .with_client_auth_cert(credential.chain.clone(), credential.key.clone_key())
175        .context("the minted certificate and identity key do not go together")?;
176    config.alpn_protocols = vec![ALPN.to_vec()];
177    Ok(Arc::new(config))
178}

184/// The name presented in SNI. The pinned identity authenticates the
185/// server, so this is only a routing hint for a server that multiplexes
186/// on it — but rustls requires *some* name, and an IP literal is not
187/// one.
188pub fn server_name(name: &str) -> Result<ServerName<'static>> {
189    ServerName::try_from(name.to_string())
190        .map_err(|_| anyhow!("\"{name}\" is not a usable TLS server name"))
191}

193/// The load-bearing check, shared by both directions: the handshake
194/// signature must verify against `expected` — the pinned identity, not
195/// whatever key the certificate happens to carry. `message` already
196/// binds the whole transcript, so a passing signature is live proof of
197/// possession, not a replayable artifact.
198fn verify_identity_signature(
199    expected: &Address,
200    message: &[u8],
201    scheme: SignatureScheme,
202    signature: &[u8],
203) -> Result<HandshakeSignatureValid, rustls::Error> {
204    if scheme != SignatureScheme::ED25519 {
205        return Err(rustls::Error::PeerIncompatible(
206            rustls::PeerIncompatible::NoSignatureSchemesInCommon,
207        ));
208    }
209    let signature: &[u8; 64] = signature
210        .try_into()
211        .map_err(|_| rustls::Error::General("malformed Ed25519 signature".into()))?;
212    if umsh_crypto::verify_ed25519_signature(&expected.0, message, signature) {
213        Ok(HandshakeSignatureValid::assertion())
214    } else {
215        Err(rustls::Error::General(
216            "the handshake signature does not verify against the pinned identity".into(),
217        ))
218    }
219}

221#[derive(Debug)]
222struct IdentityClientVerifier {
223    accepted: Vec<Address>,
224}
225
226impl IdentityClientVerifier {
227    /// The certificate's key, if it is one this bridge pins. The
228    /// signature check is what makes the answer trustworthy; this only
229    /// selects which pinned key that check must pass against.
230    fn accepted_key(&self, cert: &CertificateDer<'_>) -> Result<Address, rustls::Error> {
231        let key = certificate_key(cert).ok_or_else(|| {
232            rustls::Error::General("the certificate carries no Ed25519 key".into())
233        })?;
234        if self.accepted.contains(&key) {
235            Ok(key)
236        } else {
237            Err(rustls::Error::General(
238                "client identity is not configured on this bridge".into(),
239            ))
240        }
241    }
242
243    /// The signature is checked against the key extracted from the
244    /// *same certificate* the connection is attributed to later — that
245    /// binding is what stops one pinned client from wearing another's
246    /// address.
247    fn check_signature(
248        &self,
249        message: &[u8],
250        cert: &CertificateDer<'_>,
251        scheme: SignatureScheme,
252        signature: &[u8],
253    ) -> Result<HandshakeSignatureValid, rustls::Error> {
254        let key = self.accepted_key(cert)?;
255        verify_identity_signature(&key, message, scheme, signature)
256    }
257}
258
259impl ClientCertVerifier for IdentityClientVerifier {
260    /// Empty: naming acceptable issuers would only help a client choose
261    /// among several certificates, and each client holds exactly one.
262    fn root_hint_subjects(&self) -> &[DistinguishedName] {
263        &[]
264    }
265
266    fn verify_client_cert(
267        &self,
268        end_entity: &CertificateDer<'_>,
269        _intermediates: &[CertificateDer<'_>],
270        _now: UnixTime,
271    ) -> Result<ClientCertVerified, rustls::Error> {
272        self.accepted_key(end_entity)?;
273        Ok(ClientCertVerified::assertion())
274    }
275
276    fn verify_tls12_signature(
277        &self,
278        _message: &[u8],
279        _cert: &CertificateDer<'_>,
280        _dss: &DigitallySignedStruct,
281    ) -> Result<HandshakeSignatureValid, rustls::Error> {
282        Err(rustls::Error::General("TLS 1.2 is not offered".into()))
283    }
284
285    fn verify_tls13_signature(
286        &self,
287        message: &[u8],
288        cert: &CertificateDer<'_>,
289        dss: &DigitallySignedStruct,
290    ) -> Result<HandshakeSignatureValid, rustls::Error> {
291        self.check_signature(message, cert, dss.scheme, dss.signature())
292    }
293
294    fn supported_verify_schemes(&self) -> Vec<SignatureScheme> {
295        vec![SignatureScheme::ED25519]
296    }
297}

299#[derive(Debug)]
300struct IdentityServerVerifier {
301    accepted: Address,
302}
303
304impl IdentityServerVerifier {
305    /// Deliberately ignores any certificate: whatever it says, the
306    /// transcript must be signed by the one identity this client pins.
307    fn check_signature(
308        &self,
309        message: &[u8],
310        scheme: SignatureScheme,
311        signature: &[u8],
312    ) -> Result<HandshakeSignatureValid, rustls::Error> {
313        verify_identity_signature(&self.accepted, message, scheme, signature)
314    }
315}
316
317impl ServerCertVerifier for IdentityServerVerifier {
318    fn verify_server_cert(
319        &self,
320        end_entity: &CertificateDer<'_>,
321        _intermediates: &[CertificateDer<'_>],
322        _server_name: &ServerName<'_>,
323        _ocsp_response: &[u8],
324        _now: UnixTime,
325    ) -> Result<ServerCertVerified, rustls::Error> {
326        // Only for a better error while the wrong end is still
327        // reachable; the signature check below is the decision.
328        match certificate_key(end_entity) {
329            Some(key) if key == self.accepted => Ok(ServerCertVerified::assertion()),
330            _ => Err(rustls::Error::General(
331                "the server is not the identity this client pins".into(),
332            )),
333        }
334    }
335
336    fn verify_tls12_signature(
337        &self,
338        _message: &[u8],
339        _cert: &CertificateDer<'_>,
340        _dss: &DigitallySignedStruct,
341    ) -> Result<HandshakeSignatureValid, rustls::Error> {
342        Err(rustls::Error::General("TLS 1.2 is not offered".into()))
343    }
344
345    fn verify_tls13_signature(
346        &self,
347        message: &[u8],
348        _cert: &CertificateDer<'_>,
349        dss: &DigitallySignedStruct,
350    ) -> Result<HandshakeSignatureValid, rustls::Error> {
351        self.check_signature(message, dss.scheme, dss.signature())
352    }
353
354    fn supported_verify_schemes(&self) -> Vec<SignatureScheme> {
355        vec![SignatureScheme::ED25519]
356    }
357}

374        assert_eq!(base58.parse::<Address>().unwrap(), address);

383        // search, since which y-coordinates decompress is not obvious.
384        let junk = (0u8..=255)

391            .expect("roughly half of all y-coordinates are not on the curve");
