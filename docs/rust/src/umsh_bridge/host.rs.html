1//! Host interfaces: a plain socket presenting a simulated ULCP device
2//! whose radio is the bridge.
3//!
4//! Every other interface fronts a real radio, and the node behind that
5//! radio is what makes a crossing safe. A host interface has neither. It
6//! exists so that a host with no radio at all — an iOS simulator build,
7//! `umshctl --tcp`, a test — can join the bridged medium as an ordinary
8//! node, running its own MAC over a device that is real code and a
9//! fictional PHY.
10//!
11//! The device is the production [`Session`], wrapped by
12//! [`SimulatedDevice`]. Two seams connect it to the bridge: what the
13//! session transmits is drained into the hub, and what the hub delivers
14//! is fed through the session's real receive path. Neither direction
15//! carries measurements, because nothing here measured anything — see
16//! the spec's [Host Interfaces].
17//!
18//! The session outlives any one connection, exactly as a radio outlives
19//! the host that unplugs from it: `attach`/`detach` follow the socket,
20//! and provisioned state survives a reconnect.
21//!
22//! [Host Interfaces]: https://darconeous.github.io/umsh/docs/protocol/internet-bridging.html#host-interfaces
23//! [`Session`]: umsh_ulcp_device::Session
24

41/// How much socket data to take in one read. ULCP frames are small; this
42/// only bounds a burst.
43const READ_CHUNK: usize = 4096;
44
45/// The shape of a bridge's soft device.
46///
47/// Deliberately spare. A board's capabilities describe hardware, and
48/// this has none: no battery to sample, no light sensor, no receiver, no
49/// Bluetooth to switch off, and above all no node — see
50/// [`SessionConfig::mac_node`], which is what keeps a participant from
51/// mistaking this for something it can backhaul through.
52///
53/// The duty limit is left wide open. A simulated transmission spends no
54/// airtime, so a duty ledger here would throttle a host against a budget
55/// it is not consuming; the interface's frame rate limit is the control
56/// that means something, and the configuration requires one.
57fn session_config() -> SessionConfig {
58    SessionConfig {
59        dev_version: concat!("umsh-bridge-host/", env!("CARGO_PKG_VERSION")),
60        dev_model: None,
61        default_device_name: "Bridged host interface",
62        mtu: 255,
63        sync_word: profiles::DEFAULT.sync_word,
64        min_tx_power_dbm: -9,
65        max_tx_power_dbm: 22,
66        freq_khz_min: 150_000,
67        freq_khz_max: 960_000,
68        defaults: RadioSettings {
69            enabled: false,
70            freq_khz: profiles::DEFAULT.freq_khz,
71            bw_hz: profiles::DEFAULT.bw_hz,
72            sf: profiles::DEFAULT.sf,
73            cr_denom: profiles::DEFAULT.cr_denom,
74            tx_power_dbm: profiles::DEFAULT_TX_POWER_DBM,
75        },
76        default_duty_limit: 0xFFFF,
77        duty: Box::leak(Box::new(DutyLedger::new())),
78        battery: None,
79        alert: None,
80        time: None,
81        gnss: None,
82        illuminance: false,
83        ble: false,
84        ble_pairing: false,
85        // A host interface is a task inside this process; there is no
86        // hardware for `CMD_REBOOT` to restart, and restarting the
87        // bridge is the operator's business.
88        reboot: false,
89        // A soft device on the bridge has no antenna of its own.
90        stats: None,
91        mac_node: false,
92    }
93}

95/// Metadata for a frame nobody measured.
96///
97/// A frame reaching a host crossed no air: it was measured on some other
98/// segment, by some other radio, and reporting that measurement here
99/// would attribute it to a reception that never happened.
100fn unmeasured() -> Vec<u8> {
101    let mut meta = [0u8; RxMeta::WIRE_LEN];
102    RxMeta {
103        rssi_dbm: None,
104        lqi: None,
105        snr_cb: None,
106    }
107    .encode(&mut meta)
108    .expect("buffer sized with WIRE_LEN");
109    meta.to_vec()
110}

112/// One host interface: its listener, its device, and the relay between
113/// them.
114pub struct HostRelay {
115    name: String,
116    listen: std::net::SocketAddr,
117    iface: Arc<Interface>,
118    ingress: mpsc::Sender<Ingress>,
119    /// Monotonic clock for the session, in milliseconds since start.
120    started: tokio::time::Instant,
121}
122
123impl HostRelay {
124    pub fn new(entry: &HostEntry, iface: Arc<Interface>, ingress: mpsc::Sender<Ingress>) -> Self {
125        Self {
126            name: entry.name.clone(),
127            listen: entry.listen,
128            iface,
129            ingress,
130            started: tokio::time::Instant::now(),
131        }
132    }
133
134    /// Serve this interface for as long as the process runs.
135    ///
136    /// One device, many connections: the session is built once here, and
137    /// each accepted socket attaches to it and detaches on the way out.
138    pub async fn run(self) -> Result<()> {
139        let listener = TcpListener::bind(self.listen)
140            .await
141            .with_context(|| format!("listening on {} for host \"{}\"", self.listen, self.name))?;
142        if self.listen.ip().is_loopback() {
143            tracing::info!(iface = %self.name, listen = %self.listen, "host interface listening");
144        } else {
145            tracing::warn!(
146                iface = %self.name,
147                listen = %self.listen,
148                "host interface listening off the loopback: this socket is unauthenticated and \
149                 whoever reaches it has an RF presence on every segment this bridge touches"
150            );
151        }
152
153        let mut device = SimulatedDevice::new(session_config());
154        let mut current: Option<TcpStream> = None;
155        let mut session = self.iface.egress.generation();
156        let mut buf = vec![0u8; READ_CHUNK];
157
158        loop {
159            let event = self
160                .next_event(&listener, current.as_mut(), session, &mut buf)
161                .await;
162
163            match event {
164                Event::Accepted(stream, peer) => {
165                    if current.is_some() {
166                        // Newest wins, as for a tunnel client: the older
167                        // connection has stopped being the live host,
168                        // and holding the interface for it would wedge
169                        // this listener behind a socket nobody is
170                        // reading.
171                        tracing::info!(iface = %self.name, "displacing an earlier host");
172                        device.detach();
173                    }
174                    stream.set_nodelay(true).ok();
175                    tracing::info!(iface = %self.name, ?peer, "host attached");
176
177                    // A socket is a cable: attaching resets exactly the
178                    // session-scoped state a DTR drop would, and the
179                    // egress queue starts empty rather than replaying
180                    // what the last host never read.
181                    self.iface.egress.clear();
182                    session = self.iface.egress.generation();
183                    device.attach();
184                    self.iface.set_connected(true);
185                    current = Some(stream);
186                }
187                Event::AcceptFailed(error) => {
188                    // A per-connection accept failure must not take the
189                    // listener down with it.
190                    tracing::warn!(iface = %self.name, "accept failed: {error}");
191                    tokio::time::sleep(Duration::from_millis(100)).await;
192                }
193                Event::Closed(reason) => {
194                    self.detach(&mut device, &mut current, reason);
195                    continue;
196                }
197                Event::Read(len) => {
198                    // A decode error is the host's problem, not the
199                    // link's: the decoder resynchronizes on the next
200                    // flag, so a garbled frame costs one frame.
201                    if let Err(error) = device.ingest(&buf[..len], self.now_ms()) {
202                        tracing::debug!(iface = %self.name, "{error}");
203                    }
204                    self.drain_transmissions(&mut device);
205                }
206                Event::Deliver(frame) => {
207                    device.inject_radio_rx_with_info(
208                        &frame.data,
209                        &RadioRxInfo::default(),
210                        self.now_ms(),
211                    );
212                    // Delivering a frame can make the session transmit —
213                    // an acknowledgement it was asked to send on the
214                    // host's behalf, say.
215                    self.drain_transmissions(&mut device);
216                }
217            }
218
219            // Whatever the device has to say goes out on the connection
220            // that is live now, which a displacement may have just
221            // replaced.
222            let Some(stream) = current.as_mut() else {
223                continue;
224            };
225            while let Some(wire) = device.take_outbound() {
226                if let Err(error) = stream.write_all(&wire).await {
227                    self.detach(&mut device, &mut current, format!("write failed: {error}"));
228                    break;
229                }
230            }
231        }
232    }
233
234    /// Wait for whichever comes first: a new host, bytes from the
235    /// current one, or a frame to deliver to it.
236    ///
237    /// The device is not borrowed here, and the writes that follow
238    /// happen after this returns, so both directions drive one
239    /// `&mut SimulatedDevice` without holding a borrow across an await.
240    async fn next_event(
241        &self,
242        listener: &TcpListener,
243        stream: Option<&mut TcpStream>,
244        session: u64,
245        buf: &mut [u8],
246    ) -> Event {
247        let queue: &TunnelQueue = &self.iface.egress;
248        match stream {
249            // No host attached: nothing to read, and nothing to deliver
250            // to. The interface is down, so the hub queues nothing for
251            // it anyway.
252            None => match listener.accept().await {
253                Ok((stream, peer)) => Event::Accepted(stream, peer),
254                Err(error) => Event::AcceptFailed(error),
255            },
256            Some(stream) => {
257                let accept = listener.accept();
258                let read = stream.read(buf);
259                tokio::pin!(accept, read);
260                core::future::poll_fn(|cx| {
261                    // The socket first: a host waiting on a response is
262                    // the common case, and draining its request is what
263                    // produces one. `read` is cancel-safe, so losing this
264                    // poll to either branch below loses no bytes.
265                    if let Poll::Ready(result) = read.as_mut().poll(cx) {
266                        return Poll::Ready(match result {
267                            Ok(0) => Event::Closed("host closed the connection".to_string()),
268                            Ok(len) => Event::Read(len),
269                            Err(error) => Event::Closed(format!("read failed: {error}")),
270                        });
271                    }
272                    if let Poll::Ready(result) = accept.as_mut().poll(cx) {
273                        return Poll::Ready(match result {
274                            Ok((stream, peer)) => Event::Accepted(stream, peer),
275                            Err(error) => Event::AcceptFailed(error),
276                        });
277                    }
278                    match queue.poll_pop(session, cx) {
279                        Poll::Ready(Dequeued::Frame(frame)) => Poll::Ready(Event::Deliver(frame)),
280                        // Only this task clears the queue, and only
281                        // between polls, so the generation cannot move
282                        // underneath a live connection.
283                        Poll::Ready(Dequeued::SessionEnded) => {
284                            Poll::Ready(Event::Closed("session ended".to_string()))
285                        }
286                        Poll::Pending => Poll::Pending,
287                    }
288                })
289                .await
290            }
291        }
292    }
293
294    /// Put the interface down and let go of the host that was on it.
295    fn detach(
296        &self,
297        device: &mut SimulatedDevice,
298        current: &mut Option<TcpStream>,
299        reason: String,
300    ) {
301        if current.take().is_none() {
302            return;
303        }
304        self.iface.set_connected(false);
305        self.iface.egress.clear();
306        device.detach();
307        tracing::info!(iface = %self.name, "host detached: {reason}");
308    }
309
310    /// Move everything the device transmitted into the hub.
311    fn drain_transmissions(&self, device: &mut SimulatedDevice) {
312        for data in device.take_transmitted() {
313            let frame = TunnelFrame::new(data, unmeasured());
314            // The hub is the only consumer and its work per frame is
315            // bounded; a full channel means the process is shutting
316            // down.
317            if self
318                .ingress
319                .try_send(Ingress {
320                    iface: self.iface.id,
321                    frame,
322                })
323                .is_err()
324            {
325                tracing::warn!(iface = %self.name, "the hub is not keeping up; dropped a frame");
326            }
327        }
328    }
329
330    fn now_ms(&self) -> u64 {
331        self.started.elapsed().as_millis() as u64
332    }
333}
334
335enum Event {
336    /// A host connected. If one was already attached, it is displaced.
337    Accepted(TcpStream, std::net::SocketAddr),
338    AcceptFailed(std::io::Error),
339    /// Bytes from the attached host, in the shared buffer.
340    Read(usize),
341    /// A frame from the hub, for the attached host.
342    Deliver(TunnelFrame),
343    /// The attached host is gone, for the reason given.
344    Closed(String),
345}

348pub type HostInterfaceId = InterfaceId;

350#[cfg(test)]
351mod tests {
352    use super::*;
353    use umsh_ulcp::meta::RxMeta;
354
355    #[test]
356    fn a_bridge_host_device_claims_no_node() {
357        // The whole safety argument for a host interface rests on this:
358        // the device has no node, says so, and a participant that would
359        // otherwise backhaul through it is refused by its own check.
360        assert!(!session_config().mac_node);
361    }
362
363    #[test]
364    fn a_bridge_host_device_claims_no_hardware_it_lacks() {
365        let config = session_config();
366        assert!(config.battery.is_none());
367        assert!(
368            config.alert.is_none(),
369            "no alert, so nothing polls a deadline"
370        );
371        assert!(config.time.is_none());
372        assert!(config.gnss.is_none());
373        assert!(!config.illuminance);
374        assert!(!config.ble);
375    }
376
377    #[test]
378    fn delivered_frames_carry_no_measurements() {
379        let meta = unmeasured();
380        let decoded = RxMeta::decode(&meta).expect("fixed-size metadata");
381        assert_eq!(decoded.rssi_dbm, None);
382        assert_eq!(decoded.snr_cb, None);
383        assert_eq!(decoded.lqi, None);
384    }
