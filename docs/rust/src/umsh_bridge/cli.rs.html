1//! Command-line surface.
2//!
3//! Deliberately thin: everything that describes a deployment lives in
4//! the config file, so the unit file, the `check` a change is validated
5//! with, and the process that runs all name the same artifact.
6
7use std::path::PathBuf;
8
9use clap::{Args, Parser, Subcommand};
10
11#[derive(Debug, Parser)]
12#[command(
13    name = "umsh-bridge",
14    version,
15    about = "Carry UMSH frames between radios that cannot hear each other",
16    long_about = "\
17Runs one end of a UMSH internet bridge: a server, which copies each frame
18it receives to every other participant, or a client, which relays frames
19byte for byte between its own radio and the server.
20
21Which end this is comes from the configuration file, not the command
22line, so one unit file fits either role and `umsh-bridge check` validates
23the exact artifact that will run.
24
25Every participant's radio runs in backhaul mode, which makes the node
26behind it a repeater whose point-to-point neighbor is the bridge. Hop
27accounting, duplicate suppression, and forwarding policy are that node's;
28the bridge has no opinion about what it carries and no presence on the
29mesh.
30
31Every participant holds an Ed25519 identity, issued once by `keygen
32identity`, which is what the tunnel authenticates with. Each side is
33configured with the other's public key — the UMSH address that `address`
34prints — which makes revoking a client an edit to the server's
35configuration. The TLS certificates the handshake requires are minted in
36memory from the identity at startup; none are stored or exchanged."
37)]
38pub struct ToolArgs {
39    #[command(subcommand)]
40    pub command: Command,
41
42    /// Raise the logging level; repeat for more (info, debug, trace).
43    #[arg(short = 'v', long, action = clap::ArgAction::Count, global = true)]
44    pub verbose: u8,
45
46    /// Log warnings and errors only; repeat for errors only.
47    #[arg(short = 'q', long, action = clap::ArgAction::Count, global = true)]
48    pub quiet: u8,
49
50    /// Per-target log filter, in `tracing-subscriber` syntax
51    /// (`umsh_bridge::hub=trace,info`). Overrides -v/-q entirely.
52    #[arg(long, value_name = "FILTER", env = "UMSH_BRIDGE_LOG", global = true)]
53    pub log_filter: Option<String>,
54}
55
56impl ToolArgs {
57    /// The filter -v/-q ask for, unless an explicit one was given.
58    pub fn log_filter(&self) -> String {
59        if let Some(filter) = &self.log_filter {
60            return filter.clone();
61        }
62        // Third-party crates are noisy at debug and above and none of
63        // their detail is about this bridge, so verbosity is spent on
64        // this crate first and the rest of the world stays at warn.
65        let level = match (self.verbose, self.quiet) {
66            (_, 2..) => return "error".into(),
67            (_, 1) => return "warn".into(),
68            (0, _) => "info",
69            (1, _) => "debug",
70            (2.., _) => "trace",
71        };
72        format!("warn,umsh_bridge={level}")
73    }
74}

76#[derive(Debug, Subcommand)]
77pub enum Command {
78    /// Run the bridge until interrupted.
79    Run(RunArgs),
80
81    /// Load and validate a configuration, reading the identity key it
82    /// names, without opening a socket or touching a radio.
83    Check(RunArgs),
84
85    /// Print the address of an identity key — the public name the
86    /// other end of the tunnel pins. Safe to share anywhere.
87    Address {
88        /// The identity key to read.
89        #[arg(default_value = "/etc/umsh-bridge/identity.key")]
90        path: PathBuf,
91    },
92
93    /// Issue the identity a participant needs.
94    #[command(subcommand)]
95    Keygen(KeygenCommand),
96}
97
98#[derive(Debug, Args)]
99pub struct RunArgs {
100    /// Configuration file.
101    #[arg(
102        short = 'c',
103        long,
104        value_name = "PATH",
105        env = "UMSH_BRIDGE_CONFIG",
106        default_value = "/etc/umsh-bridge/config.toml"
107    )]
108    pub config: PathBuf,
109}
110
111#[derive(Debug, Subcommand)]
112pub enum KeygenCommand {
113    /// Generate this participant's identity, once for its life: its
114    /// address is what the other end pins, and for a server it is also
115    /// the address the mesh knows the bridge by.
116    Identity {
117        /// Where to write the 64-hex Ed25519 seed.
118        #[arg(default_value = "/etc/umsh-bridge/identity.key")]
119        path: PathBuf,
120
121        /// Replace an existing key. The identity's address changes, and
122        /// every peer pinning it must be updated.
123        #[arg(long)]
124        force: bool,
125    },
126}

128#[cfg(test)]
129mod tests {
130    use super::*;
131    use clap::CommandFactory;
132
133    #[test]
134    fn the_grammar_is_internally_consistent() {
135        ToolArgs::command().debug_assert();
136    }
137
138    #[test]
139    fn verbosity_maps_to_a_filter_that_spares_the_rest_of_the_world() {
140        let filter = |argv: &[&str]| {
141            let mut args = vec!["umsh-bridge"];
142            args.extend_from_slice(argv);
143            ToolArgs::try_parse_from(args).unwrap().log_filter()
144        };
145        assert_eq!(filter(&["check"]), "warn,umsh_bridge=info");
146        assert_eq!(filter(&["check", "-v"]), "warn,umsh_bridge=debug");
147        assert_eq!(filter(&["check", "-vv"]), "warn,umsh_bridge=trace");
148        assert_eq!(filter(&["check", "-vvv"]), "warn,umsh_bridge=trace");
149        assert_eq!(filter(&["check", "-q"]), "warn");
150        assert_eq!(filter(&["check", "-qq"]), "error");
151        // An explicit filter is the whole answer, whatever else was
152        // asked for.
153        assert_eq!(
154            filter(&["check", "-vv", "--log-filter", "umsh_bridge::tunnel=trace"]),
155            "umsh_bridge::tunnel=trace"
156        );
157    }
158
159    #[test]
160    fn global_flags_work_before_or_after_the_command() {
161        let before = ToolArgs::try_parse_from(["umsh-bridge", "-v", "run", "-c", "a.toml"]);
162        let after = ToolArgs::try_parse_from(["umsh-bridge", "run", "-c", "a.toml", "-v"]);
163        assert_eq!(before.unwrap().verbose, after.unwrap().verbose);
164    }
165
166    #[test]
167    fn identity_paths_default_to_the_deployment_location() {
168        let parsed = ToolArgs::try_parse_from(["umsh-bridge", "address"]).unwrap();
169        let Command::Address { path } = parsed.command else {
170            panic!("parsed as the wrong command");
171        };
172        assert_eq!(path, PathBuf::from("/etc/umsh-bridge/identity.key"));
173
174        assert!(ToolArgs::try_parse_from(["umsh-bridge", "keygen", "identity"]).is_ok());
175        assert!(
176            ToolArgs::try_parse_from(["umsh-bridge", "keygen", "identity", "a.key", "--force"])
177                .is_ok()
178        );
179    }
180
181    #[test]
182    fn a_command_is_required() {
183        assert!(ToolArgs::try_parse_from(["umsh-bridge"]).is_err());
184    }
