1//! The bridge client: a byte-faithful relay between the server and its
2//! radio, its LAN bearer group, or both. With both, each frame from one
3//! of the three is copied to the other two.
4//!
5//! There is no forwarding logic here, deliberately—nor anywhere else
6//! in the bridge. A client does not parse what it carries, does not
7//! suppress duplicates, and does not decide anything about a frame. The
8//! node behind its radio does all of that, to bridged traffic as to
9//! anything else it hears.
10

24pub async fn run(identity: BridgeIdentity, mut config: ClientConfig) -> Result<()> {
25    let credential =
26        Credential::for_identity(&identity).context("minting this client's TLS credential")?;
27    let tls_config = tls::client_config(&credential, config.server_address)?;
28    let connector = tokio_rustls::TlsConnector::from(tls_config);
29
30    let max_age = Duration::from_secs(config.tunnel.max_frame_age_secs);
31    let queue = || Arc::new(TunnelQueue::new(config.tunnel.queue_depth, max_age));
32    let to_server = queue();
33
34    // Each local side's inbound and outbound queues, for whichever are
35    // configured.
36    let radio = (!config.radio.is_none()).then(|| (queue(), queue()));
37    let lan = config.lan.as_ref().map(|_| (queue(), queue()));
38    if let Some((inbound, outbound)) = &radio {
39        let relay = DeviceRelay::new(
40            std::mem::take(&mut config.radio),
41            inbound.clone(),
42            outbound.clone(),
43        );
44        tokio::task::spawn_local(async move { relay.run().await });
45    }
46    if let (Some((inbound, outbound)), Some(entry)) = (&lan, &config.lan) {
47        let relay = LanRelay::new(entry.socket_config(), inbound.clone(), outbound.clone());
48        tokio::task::spawn_local(async move { relay.run().await });
49    }
50    if let Some((inbound, _)) = &radio {
51        let mut targets = vec![to_server.clone()];
52        targets.extend(lan.as_ref().map(|(_, outbound)| outbound.clone()));
53        spawn_fan_out(inbound.clone(), targets);
54    }
55    if let Some((inbound, _)) = &lan {
56        let mut targets = vec![to_server.clone()];
57        targets.extend(radio.as_ref().map(|(_, outbound)| outbound.clone()));
58        spawn_fan_out(inbound.clone(), targets);
59    }
60    let to_local: Vec<Arc<TunnelQueue>> = radio
61        .iter()
62        .chain(&lan)
63        .map(|(_, outbound)| outbound.clone())
64        .collect();
65
66    tracing::info!(
67        server = %config.server,
68        address = %credential.address,
69        "bridge client starting"
70    );
71
72    let min = Duration::from_secs(config.tunnel.reconnect_min_secs.max(1));
73    let max = Duration::from_secs(config.tunnel.reconnect_max_secs.max(1));
74    let mut backoff = min;
75    loop {
76        let started = tokio::time::Instant::now();
77        match session(&connector, &config, &to_server, &to_local).await {
78            Ok(()) => tracing::info!("tunnel closed by the server"),
79            Err(error) => tracing::warn!("tunnel: {error:#}"),
80        }
81
82        // A connection that lived a while was working; the next failure
83        // deserves a fresh, short retry rather than the last one's
84        // accumulated patience.
85        if started.elapsed() > max {
86            backoff = min;
87        }
88        let delay = jittered(backoff);
89        tracing::debug!(?delay, "reconnecting");
90        tokio::time::sleep(delay).await;
91        backoff = (backoff * 2).min(max);
92    }
93}

95async fn session(
96    connector: &tokio_rustls::TlsConnector,
97    config: &ClientConfig,
98    to_server: &Arc<TunnelQueue>,
99    to_local: &[Arc<TunnelQueue>],
100) -> Result<()> {
101    let stream = connect_any(&config.server).await?;
102    stream.set_nodelay(true).ok();
103    let peer = stream.peer_addr().ok();
104
105    let name = match &config.server_name {
106        Some(name) => tls::server_name(name)?,
107        None => tls::server_name(host_of(&config.server))?,
108    };
109    let stream = connector
110        .connect(name, stream)
111        .await
112        .context("TLS handshake")?;
113    tracing::info!(?peer, "tunnel established");
114
115    let (read_half, write_half) = tokio::io::split(stream);
116    let mut reader = TunnelReader::new(
117        read_half,
118        Duration::from_secs(config.tunnel.idle_timeout_secs),
119    );
120    let mut writer = TunnelWriter::new(
121        write_half,
122        Duration::from_secs(config.tunnel.keepalive_secs),
123    );
124
125    // A re-established connection starts empty rather than flushing the
126    // old one's backlog: anything that waited through a reconnection has
127    // outlived every confirmation retry that could have wanted it. This
128    // is the last moment before the new session's writer starts, so it
129    // also covers whatever queued during the handshake.
130    to_server.clear();
131    let session = to_server.generation();
132
133    let queue = to_server.clone();
134    let pump =
135        tokio::task::spawn_local(async move { pump_writer(&mut writer, &queue, session).await });
136
137    let outcome = loop {
138        match reader.read_frame().await {
139            Ok(frame) => {
140                // Passed on unmodified—to the device as the body of a
141                // CMD_STR_SEND: the server composed it and the client
142                // has no business editing it.
143                for queue in to_local {
144                    if queue.push(frame.clone()) {
145                        tracing::warn!("a local side is not keeping up; dropped the oldest frame");
146                    }
147                }
148            }
149            Err(error) => break Err(anyhow!("{error}")),
150        }
151    };
152
153    // Ending the writer's session unparks it whether or not it is
154    // waiting on a frame.
155    to_server.clear();
156    pump.abort();
157    if reader.malformed() > 0 {
158        tracing::warn!(
159            malformed = reader.malformed(),
160            "the server sent frames this client could not read"
161        );
162    }
163    outcome
164}

166/// Copy each frame from `source` to every queue in `targets`.
167fn spawn_fan_out(source: Arc<TunnelQueue>, targets: Vec<Arc<TunnelQueue>>) {
168    tokio::task::spawn_local(async move {
169        loop {
170            // Nothing clears a local side's inbound queue, so a session
171            // ending here is only ever a spurious wake.
172            let Dequeued::Frame(frame) = source.pop().await else {
173                continue;
174            };
175            for target in &targets {
176                if target.push(frame.clone()) {
177                    tracing::warn!("a queue is not keeping up; dropped the oldest frame");
178                }
179            }
180        }
181    });
182}

184/// Try every address the name resolves to, in the order the resolver
185/// gave them—which is how a dual-stacked server is reached over
186/// whichever family actually works from here.
187async fn connect_any(server: &str) -> Result<TcpStream> {
188    let addresses: Vec<_> = tokio::net::lookup_host(server)
189        .await
190        .with_context(|| format!("resolving {server}"))?
191        .collect();
192    if addresses.is_empty() {
193        bail!("{server} resolved to no addresses");
194    }
195
196    let mut last = None;
197    for address in &addresses {
198        match TcpStream::connect(address).await {
199            Ok(stream) => return Ok(stream),
200            Err(error) => {
201                tracing::debug!(%address, "connect failed: {error}");
202                last = Some(error);
203            }
204        }
205    }
206    Err(anyhow!(last.expect("at least one address was tried"))
207        .context(format!("connecting to {server}")))
208}

210/// The host part of a `host:port`, including a bracketed IPv6 literal.
211fn host_of(server: &str) -> &str {
212    if let Some(rest) = server.strip_prefix('[') {
213        return rest.split(']').next().unwrap_or(server);
214    }
215    server
216        .rsplit_once(':')
217        .map(|(host, _)| host)
218        .unwrap_or(server)
219}

221/// Spread reconnections so a server coming back does not meet every
222/// client it lost at the same instant.
223fn jittered(base: Duration) -> Duration {
224    let millis = base.as_millis() as u64;
225    base + Duration::from_millis(rand::random_range(0..=millis / 2))
226}

228#[cfg(test)]
229mod tests {
230    use super::*;
231
232    #[test]
233    fn the_sni_name_comes_from_the_host_part_of_any_address_form() {
234        assert_eq!(host_of("bridge.example.net:21837"), "bridge.example.net");
235        assert_eq!(host_of("192.0.2.10:21837"), "192.0.2.10");
236        assert_eq!(host_of("[2001:db8::1]:21837"), "2001:db8::1");
237        // And every form yields a name rustls will accept.
238        for server in [
239            "bridge.example.net:21837",
240            "192.0.2.10:21837",
241            "[2001:db8::1]:21837",
242        ] {
243            tls::server_name(host_of(server)).unwrap();
244        }
245    }
246
247    #[test]
248    fn jitter_only_ever_adds() {
249        let base = Duration::from_secs(4);
250        for _ in 0..64 {
251            let delay = jittered(base);
252            assert!(
253                delay >= base && delay <= base + Duration::from_secs(2),
254                "{delay:?}"
255            );
256        }
257    }
