1//! The bridge client: a byte-faithful relay between one radio and the
2//! server.
3//!
4//! There is no forwarding logic here, deliberately. A client does not
5//! parse what it carries, does not suppress duplicates, and does not
6//! decide anything about a frame — the whole assembly is one virtual
7//! repeater and the server is where that repeater thinks.
8

21pub async fn run(identity: BridgeIdentity, mut config: ClientConfig) -> Result<()> {
22    let credential =
23        Credential::for_identity(&identity).context("minting this client's TLS credential")?;
24    let tls_config = tls::client_config(&credential, config.server_address)?;
25    let connector = tokio_rustls::TlsConnector::from(tls_config);
26
27    let max_age = Duration::from_secs(config.tunnel.max_frame_age_secs);
28    let to_server = Arc::new(TunnelQueue::new(config.tunnel.queue_depth, max_age));
29    let to_device = Arc::new(TunnelQueue::new(config.tunnel.queue_depth, max_age));
30
31    let relay = DeviceRelay::new(
32        std::mem::take(&mut config.radio),
33        to_server.clone(),
34        to_device.clone(),
35    );
36    tokio::task::spawn_local(async move { relay.run().await });
37
38    tracing::info!(
39        server = %config.server,
40        address = %credential.address,
41        "bridge client starting"
42    );
43
44    let min = Duration::from_secs(config.tunnel.reconnect_min_secs.max(1));
45    let max = Duration::from_secs(config.tunnel.reconnect_max_secs.max(1));
46    let mut backoff = min;
47    loop {
48        let started = tokio::time::Instant::now();
49        match session(&connector, &config, &to_server, &to_device).await {
50            Ok(()) => tracing::info!("tunnel closed by the server"),
51            Err(error) => tracing::warn!("tunnel: {error:#}"),
52        }
53
54        // A connection that lived a while was working; the next failure
55        // deserves a fresh, short retry rather than the last one's
56        // accumulated patience.
57        if started.elapsed() > max {
58            backoff = min;
59        }
60        let delay = jittered(backoff);
61        tracing::debug!(?delay, "reconnecting");
62        tokio::time::sleep(delay).await;
63        backoff = (backoff * 2).min(max);
64    }
65}

67async fn session(
68    connector: &tokio_rustls::TlsConnector,
69    config: &ClientConfig,
70    to_server: &Arc<TunnelQueue>,
71    to_device: &Arc<TunnelQueue>,
72) -> Result<()> {
73    let stream = connect_any(&config.server).await?;
74    stream.set_nodelay(true).ok();
75    let peer = stream.peer_addr().ok();
76
77    let name = match &config.server_name {
78        Some(name) => tls::server_name(name)?,
79        None => tls::server_name(host_of(&config.server))?,
80    };
81    let stream = connector
82        .connect(name, stream)
83        .await
84        .context("TLS handshake")?;
85    tracing::info!(?peer, "tunnel established");
86
87    let (read_half, write_half) = tokio::io::split(stream);
88    let mut reader = TunnelReader::new(
89        read_half,
90        Duration::from_secs(config.tunnel.idle_timeout_secs),
91    );
92    let mut writer = TunnelWriter::new(
93        write_half,
94        Duration::from_secs(config.tunnel.keepalive_secs),
95    );
96
97    // A re-established connection starts empty rather than flushing the
98    // old one's backlog: anything that waited through a reconnection has
99    // outlived every confirmation retry that could have wanted it. This
100    // is the last moment before the new session's writer starts, so it
101    // also covers whatever queued during the handshake.
102    to_server.clear();
103    let session = to_server.generation();
104
105    let queue = to_server.clone();
106    let pump =
107        tokio::task::spawn_local(async move { pump_writer(&mut writer, &queue, session).await });
108
109    let outcome = loop {
110        match reader.read_frame().await {
111            Ok(frame) => {
112                // Passed to the device unmodified, as the body of a
113                // CMD_STR_SEND: the server composed it and the client
114                // has no business editing it.
115                if to_device.push(frame) {
116                    tracing::warn!("the radio is not keeping up; dropped the oldest frame");
117                }
118            }
119            Err(error) => break Err(anyhow!("{error}")),
120        }
121    };
122
123    // Ending the writer's session unparks it whether or not it is
124    // waiting on a frame.
125    to_server.clear();
126    pump.abort();
127    if reader.malformed() > 0 {
128        tracing::warn!(
129            malformed = reader.malformed(),
130            "the server sent frames this client could not read"
131        );
132    }
133    outcome
134}

136/// Try every address the name resolves to, in the order the resolver
137/// gave them — which is how a dual-stacked server is reached over
138/// whichever family actually works from here.
139async fn connect_any(server: &str) -> Result<TcpStream> {
140    let addresses: Vec<_> = tokio::net::lookup_host(server)
141        .await
142        .with_context(|| format!("resolving {server}"))?
143        .collect();
144    if addresses.is_empty() {
145        bail!("{server} resolved to no addresses");
146    }
147
148    let mut last = None;
149    for address in &addresses {
150        match TcpStream::connect(address).await {
151            Ok(stream) => return Ok(stream),
152            Err(error) => {
153                tracing::debug!(%address, "connect failed: {error}");
154                last = Some(error);
155            }
156        }
157    }
158    Err(anyhow!(last.expect("at least one address was tried"))
159        .context(format!("connecting to {server}")))
160}

162/// The host part of a `host:port`, including a bracketed IPv6 literal.
163fn host_of(server: &str) -> &str {
164    if let Some(rest) = server.strip_prefix('[') {
165        return rest.split(']').next().unwrap_or(server);
166    }
167    server
168        .rsplit_once(':')
169        .map(|(host, _)| host)
170        .unwrap_or(server)
171}

173/// Spread reconnections so a server coming back does not meet every
174/// client it lost at the same instant.
175fn jittered(base: Duration) -> Duration {
176    let millis = base.as_millis() as u64;
177    base + Duration::from_millis(rand::random_range(0..=millis / 2))
178}

180#[cfg(test)]
181mod tests {
182    use super::*;
183
184    #[test]
185    fn the_sni_name_comes_from_the_host_part_of_any_address_form() {
186        assert_eq!(host_of("bridge.example.net:21837"), "bridge.example.net");
187        assert_eq!(host_of("192.0.2.10:21837"), "192.0.2.10");
188        assert_eq!(host_of("[2001:db8::1]:21837"), "2001:db8::1");
189        // And every form yields a name rustls will accept.
190        for server in [
191            "bridge.example.net:21837",
192            "192.0.2.10:21837",
193            "[2001:db8::1]:21837",
194        ] {
195            tls::server_name(host_of(server)).unwrap();
196        }
197    }
198
199    #[test]
200    fn jitter_only_ever_adds() {
201        let base = Duration::from_secs(4);
202        for _ in 0..64 {
203            let delay = jittered(base);
204            assert!(
205                delay >= base && delay <= base + Duration::from_secs(2),
206                "{delay:?}"
207            );
208        }
209    }
