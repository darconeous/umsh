1//! The radio a participant fronts, and the loop that relays frames
2//! between it and one queue in each direction.
3//!
4//! Both roles use this: a client's whole job is this relay, and the
5//! server runs one for its own radio alongside the hub.
6//!
7//! Attachment follows the spec's [Radio Attachment] rules — a tethered,
8//! non-resetting attach with `PROP_MAC_BACKHAUL` set, re-asserted after
9//! every reconnection because the property is session-scoped. Backhaul
10//! mode is what makes the device's own node the far end of a
11//! point-to-point link: what this relay sends is delivered to that node
12//! as though it had heard it, and what the node transmits comes back
13//! here. Transmits are confirmed, never `TX_FLAG_NODUTY`.
14//!
15//! [Radio Attachment]: https://darconeous.github.io/umsh/docs/protocol/internet-bridging.html#radio-attachment
16

40/// How long to wait before re-opening a device that failed. A radio
41/// unplugged and plugged back in should come back on its own, and a
42/// radio that is simply gone should not spin the CPU about it.
43const REOPEN_DELAY: Duration = Duration::from_secs(3);
44
45/// How many times a frame is offered to a device whose node has no room
46/// for it.
47const MAX_HANDOFF_ATTEMPTS: u8 = 5;
48
49/// How long to wait between those attempts. A backhauled send crosses a
50/// wire, so what is being waited out is the node draining its receive
51/// queue — a matter of one frame's processing, not one frame's airtime.
52const HANDOFF_RETRY: Duration = Duration::from_millis(20);
53const HANDOFF_RETRY_JITTER_MS: u64 = 80;
54
55/// The RF parameters carried by a device configuration are applied only
56/// by the resetting attach, which a bridge never performs: radio
57/// configuration is local to each participant and the device keeps
58/// whatever it was set up with. These values are therefore inert, and
59/// exist because the constructor asks for them.
60#[cfg(any(feature = "serial-radio", feature = "ble-radio"))]
61fn attach_config() -> UlcpDeviceConfig {
62    UlcpDeviceConfig::new(910_525, 62_500, 7, 5)
63}

65/// What a `PROP_MAC_PROMISCUOUS` set means for the bridge.
66///
67/// Backhaul mode already decides *which* frames reach this host — the
68/// node's transmissions, and nothing off the air — but they still pass
69/// the device's receive filtering on the way. A device with no host
70/// domain provisioned filters nothing, which is the ordinary case for a
71/// bridge; asking for promiscuous delivery anyway is what keeps a device
72/// that *has* been provisioned from hiding the repeats this bridge
73/// exists to carry. A device that refuses the property is left as it is.
74#[cfg(any(feature = "serial-radio", feature = "ble-radio"))]
75fn promiscuous_outcome(result: Result<Vec<u8>, UlcpError>) -> Result<()> {
76    match result {
77        Ok(_) => {
78            tracing::debug!("promiscuous delivery enabled");
79            Ok(())
80        }
81        Err(UlcpError::Status(status)) => {
82            tracing::warn!(
83                ?status,
84                "device refused promiscuous delivery; this interface carries only what the \
85                 device's own receive filtering admits"
86            );
87            Ok(())
88        }
89        Err(error) => Err(anyhow!("enabling promiscuous delivery: {error:?}")),
90    }
91}

93/// A ULCP device, or the fake radio that stands in for one.
94///
95/// The two ULCP transports are separate types rather than a trait
96/// object because `UlcpDevice` is generic over its link and its receive
97/// path is a `poll` function, not an `async fn` — which is exactly what
98/// lets the relay poll a receive and a transmit from the same `&mut`
99/// without holding a borrow across an await.
100enum Device {
101    #[cfg(feature = "serial-radio")]
102    Serial(UlcpDevice<umsh::ulcp::SerialFrameLink<tokio_serial::SerialStream>>),
103    #[cfg(feature = "ble-radio")]
104    Ble(Box<UlcpDevice<umsh::ulcp::BleFrameLink>>),
105    Udp(UdpFakeRadio),
106}
107
108macro_rules! with_device {
109    ($self:expr, $device:ident => $body:expr) => {
110        match $self {
111            #[cfg(feature = "serial-radio")]
112            Device::Serial($device) => $body,
113            #[cfg(feature = "ble-radio")]
114            Device::Ble($device) => $body,
115            Device::Udp($device) => $body,
116        }
117    };
118}

120impl Device {
121    async fn open(config: &RadioConfig) -> Result<Self> {
122        match config {
123            #[cfg(feature = "serial-radio")]
124            RadioConfig::Serial { port, baud } => {
125                let link = open_serial_link(port, *baud).await?;
126                let device = UlcpDevice::attach_existing(link, attach_config())
127                    .await
128                    .map_err(|error| anyhow!("attaching to {port}: {error:?}"))?;
129                Ok(Self::Serial(device))
130            }
131            #[cfg(not(feature = "serial-radio"))]
132            RadioConfig::Serial { .. } => {
133                bail!("this build has no serial support; rebuild with --features serial-radio")
134            }
135            #[cfg(feature = "ble-radio")]
136            RadioConfig::Ble { selector } => {
137                // Connect and attach separately rather than through
138                // `open_ble`, which resets the device: a bridge attaches
139                // to a radio that is already operating and must not
140                // disturb its configuration.
141                let link = umsh::ulcp::BleFrameLink::connect(
142                    selector.as_deref(),
143                    umsh::ulcp::BleFrameLinkConfig::default(),
144                )
145                .await
146                .map_err(|error| anyhow!("connecting over BLE: {error:?}"))?;
147                let device = UlcpDevice::attach_existing(link, attach_config())
148                    .await
149                    .map_err(|error| anyhow!("attaching over BLE: {error:?}"))?;
150                Ok(Self::Ble(Box::new(device)))
151            }
152            #[cfg(not(feature = "ble-radio"))]
153            RadioConfig::Ble { .. } => {
154                bail!("this build has no BLE support; rebuild with --features ble-radio")
155            }
156            RadioConfig::UdpMulticast {
157                group,
158                port,
159                rssi,
160                snr,
161            } => Ok(Self::Udp(
162                UdpFakeRadio::bind(*group, *port, *rssi, *snr).await?,
163            )),
164            RadioConfig::None => bail!("no radio is configured"),
165        }
166    }

168    /// Everything the spec asks of a bridge's attachment, in one place
169    /// so a reconnection cannot forget any of it.
170    async fn prepare(&mut self) -> Result<()> {
171        let capabilities = self.capabilities().await?;
172
173        if !capabilities.contains(&cap::MAC_BACKHAUL) {
174            // Fatal, and deliberately so. Without backhaul this host
175            // would sit on the shared medium, and every frame the bridge
176            // handed it would go on the air unrepeated: no duplicate
177            // suppression, no hop accounting, no forwarding policy. A
178            // bridge is not something to half-do.
179            bail!(
180                "device does not advertise CAP_MAC_BACKHAUL, so it cannot join a bridge; \
181                 update its firmware"
182            );
183        }
184        if !capabilities.contains(&cap::WRITABLE_RAW_STREAM) {
185            // Not fatal: a receive-only interface still carries what its
186            // node transmits to the rest of the bridge, and saying so
187            // once is more useful than refusing to start.
188            tracing::warn!(
189                "device does not advertise CAP_WRITABLE_RAW_STREAM; this interface can receive \
190                 but not transmit"
191            );
192        }
193
194        self.set_backhaul().await?;
195        self.set_promiscuous().await?;
196        self.report_repeater_state().await;
197        Ok(())
198    }
199
200    async fn capabilities(&mut self) -> Result<Vec<u32>> {
201        match self {
202            #[cfg(feature = "serial-radio")]
203            Self::Serial(device) => device
204                .capabilities()
205                .await
206                .map_err(|error| anyhow!("reading device capabilities: {error:?}")),
207            #[cfg(feature = "ble-radio")]
208            Self::Ble(device) => device
209                .capabilities()
210                .await
211                .map_err(|error| anyhow!("reading device capabilities: {error:?}")),
212            // The fake radio has no node behind it at all, so it claims
213            // what it needs to stand in for one.
214            Self::Udp(_) => Ok(vec![cap::MAC_BACKHAUL, cap::WRITABLE_RAW_STREAM]),
215        }
216    }
217
218    /// Move this host off the shared medium and onto a point-to-point
219    /// link to the device's own node. The property is session-scoped,
220    /// which is why this runs on every attach rather than once at
221    /// provisioning time.
222    async fn set_backhaul(&mut self) -> Result<()> {
223        let result = match self {
224            #[cfg(feature = "serial-radio")]
225            Self::Serial(device) => device.set_prop(prop::MAC_BACKHAUL, &[1]).await,
226            #[cfg(feature = "ble-radio")]
227            Self::Ble(device) => device.set_prop(prop::MAC_BACKHAUL, &[1]).await,
228            // The fake radio has no medium to leave.
229            Self::Udp(_) => return Ok(()),
230        };
231        result
232            .map(|_| ())
233            .map_err(|error| anyhow!("enabling backhaul mode: {error:?}"))
234    }
235
236    async fn set_promiscuous(&mut self) -> Result<()> {
237        match self {
238            #[cfg(feature = "serial-radio")]
239            Self::Serial(device) => {
240                promiscuous_outcome(device.set_prop(prop::MAC_PROMISCUOUS, &[1]).await)
241            }
242            #[cfg(feature = "ble-radio")]
243            Self::Ble(device) => {
244                promiscuous_outcome(device.set_prop(prop::MAC_PROMISCUOUS, &[1]).await)
245            }
246            // The fake radio has no filtering to bypass.
247            Self::Udp(_) => Ok(()),
248        }
249    }

251    /// Say whether the node behind this radio forwards, and never change
252    /// it.
253    ///
254    /// Whether a device repeats is persisted, device-domain
255    /// configuration — its owner's decision, not a bridge's, and one
256    /// that outlives the session. A device with its repeater off is a
257    /// leaf: it is reachable from everywhere the bridge reaches and its
258    /// own traffic crosses, but nothing is carried onward from its
259    /// segment. That is a deployment, not a fault, so it is worth one
260    /// line and no alarm.
261    async fn report_repeater_state(&mut self) {
262        let value = match self {
263            #[cfg(feature = "serial-radio")]
264            Self::Serial(device) => device.get_prop(prop::MAC_REPEATER_ENABLED).await,
265            #[cfg(feature = "ble-radio")]
266            Self::Ble(device) => device.get_prop(prop::MAC_REPEATER_ENABLED).await,
267            Self::Udp(_) => {
268                tracing::info!(
269                    "fake radio: no node behind it, so frames cross this interface with no \
270                     duplicate suppression and no hop accounting"
271                );
272                return;
273            }
274        };
275        match value.as_deref() {
276            Ok([0]) => tracing::info!(
277                "device repeater is disabled; this interface is a leaf — its node is reachable \
278                 across the bridge, but carries nothing onward from its own segment"
279            ),
280            Ok(_) => tracing::debug!("device repeater is enabled"),
281            Err(error) => tracing::debug!(?error, "could not read the device's repeater state"),
282        }
283    }

285    fn poll_receive_raw(&mut self, cx: &mut Context<'_>) -> Poll<Result<RawRxFrame, UlcpError>> {
286        with_device!(self, device => device.poll_receive_raw(cx))
287    }

289    /// Hand one frame to the node behind this radio.
290    ///
291    /// The metadata a frame carries across the tunnel describes how it
292    /// was *received* at the far end, and is what the staleness limit
293    /// reads on the way; a transmit needs metadata of its own. Building
294    /// it here, at the last possible moment, is what lets the tunnel stay
295    /// byte-faithful. The default asks for the device's configured power
296    /// and leaves `TX_FLAG_NODUTY` clear — a bridge does not get to spend
297    /// airtime the device's duty ledger has not granted.
298    async fn transmit(&mut self, frame: &TunnelFrame) -> Result<(), TxError<UlcpError>> {
299        let mut meta = [0u8; TxMeta::WIRE_LEN];
300        TxMeta::default()
301            .encode(&mut meta)
302            .expect("a fixed-size buffer holds fixed-size metadata");
303        with_device!(self, device => device
304            .transmit_raw_with_meta(&frame.data, &meta)
305            .await)
306    }
307
308    fn max_frame_size(&self) -> usize {
309        // `UlcpDevice` reports it through the `Radio` trait; the fake
310        // radio has it inherently.
311        #[cfg(any(feature = "serial-radio", feature = "ble-radio"))]
312        use umsh_hal::Radio as _;
313
314        with_device!(self, device => device.max_frame_size())
315    }

332/// How long to wait before offering a frame to a full node again.
333///
334/// Jittered so that several interfaces relieved by the same node do not
335/// all come back at once.
336fn handoff_retry() -> Duration {
337    HANDOFF_RETRY + Duration::from_millis(rand::random_range(0..=HANDOFF_RETRY_JITTER_MS))
338}

340/// One interface's two queues, from the relay's point of view.
341pub struct DeviceRelay {
342    config: RadioConfig,
343    /// Frames the device received, for whoever consumes this interface.
344    inbound: Arc<TunnelQueue>,
345    /// Frames to put on the air.
346    outbound: Arc<TunnelQueue>,
347}
348
349impl DeviceRelay {
350    pub fn new(config: RadioConfig, inbound: Arc<TunnelQueue>, outbound: Arc<TunnelQueue>) -> Self {
351        Self {
352            config,
353            inbound,
354            outbound,
355        }
356    }
357
358    /// Keep a device attached and relaying for as long as the process
359    /// runs, re-opening it after a failure.
360    ///
361    /// A device outage leaves no backlog to drain — backhaul mode ends
362    /// with the session, and a detached device queues nothing on this
363    /// host's behalf — so a fresh attachment starts clean by
364    /// construction and this loop needs no reconciliation of its own. It
365    /// is also what recovers a device whose firmware cannot bridge:
366    /// attachment fails, the loop waits, and an updated device joins on
367    /// its own.
368    pub async fn run(&self) -> ! {
369        loop {
370            match self.session().await {
371                Ok(()) => tracing::warn!(radio = %self.config.describe(), "device link ended"),
372                Err(error) => {
373                    tracing::warn!(radio = %self.config.describe(), "device: {error:#}");
374                }
375            }
376            // Whatever was queued for a device that is no longer there
377            // is as stale as anything queued for a dead tunnel.
378            self.outbound.clear();
379            tokio::time::sleep(REOPEN_DELAY).await;
380        }
381    }
382
383    async fn session(&self) -> Result<()> {
384        let mut device = Device::open(&self.config).await?;
385        device.prepare().await?;
386        let max_frame_size = device.max_frame_size();
387        tracing::info!(
388            radio = %self.config.describe(),
389            max_frame_size,
390            "radio attached"
391        );
392
393        let generation = self.outbound.generation();
394        loop {
395            // The device is borrowed only for the duration of this
396            // poll; the transmit below re-borrows it after the borrow
397            // has ended, which is what keeps both directions on one
398            // `&mut` without a `select!` borrow conflict.
399            let event = {
400                let device = &mut device;
401                core::future::poll_fn(|cx| {
402                    // Reception first: a frame not taken from the device
403                    // is a frame the device may have to drop.
404                    if let Poll::Ready(result) = device.poll_receive_raw(cx) {
405                        return Poll::Ready(Event::Received(result));
406                    }
407                    match self.outbound.poll_pop(generation, cx) {
408                        Poll::Ready(Dequeued::Frame(frame)) => Poll::Ready(Event::Send(frame)),
409                        Poll::Ready(Dequeued::SessionEnded) => Poll::Ready(Event::Ended),
410                        Poll::Pending => Poll::Pending,
411                    }
412                })
413                .await
414            };
415
416            match event {
417                Event::Ended => return Ok(()),
418                Event::Received(Err(error)) => {
419                    return Err(anyhow!("receiving from the device: {error:?}"));
420                }
421                Event::Received(Ok(raw)) => self.accept(raw),
422                Event::Send(frame) => {
423                    if frame.data.len() > max_frame_size {
424                        tracing::debug!(
425                            len = frame.data.len(),
426                            max_frame_size,
427                            "dropping a frame larger than this radio can transmit"
428                        );
429                        continue;
430                    }
431                    self.transmit(&mut device, &frame).await?;
432                }
433            }
434        }
435    }
436
437    /// Hand one frame to the device's node, retrying a few times if the
438    /// node has nowhere to put it.
439    ///
440    /// A backhauled send never contends for the channel — it crosses a
441    /// wire, spends no airtime, and waits for no one. What it can meet is
442    /// a node whose receive queue is full, which ULCP reports the only
443    /// way it can, as `STATUS_CCA_FAILURE`. The answer is to wait briefly
444    /// and offer it again: the node drains as it works, and the waiting
445    /// is done listening, which is itself what relieves the queue.
446    ///
447    /// Whether the frame then goes on the air is the node's decision. A
448    /// duplicate it has already seen, or one whose flood budget is spent,
449    /// ends there — that is the bridging policy doing its job, not a
450    /// failure of this hand-off.
451    async fn transmit(&self, device: &mut Device, frame: &TunnelFrame) -> Result<()> {
452        for attempt in 1..=MAX_HANDOFF_ATTEMPTS {
453            match device.transmit(frame).await {
454                Ok(()) => {
455                    tracing::trace!(len = frame.data.len(), attempt, "device tx");
456                    return Ok(());
457                }
458                Err(TxError::CadTimeout) => {
459                    if attempt < MAX_HANDOFF_ATTEMPTS {
460                        self.listen_during(device, handoff_retry()).await?;
461                    }
462                }
463                // A refusal with a status is the device declining for a
464                // reason of its own — detached, or handed something it
465                // will not send. Re-offering would not change its mind.
466                Err(TxError::Io(UlcpError::Status(status))) => {
467                    tracing::debug!(?status, "device refused a transmit");
468                    return Ok(());
469                }
470                Err(error) => bail!("transmitting: {error:?}"),
471            }
472        }
473
474        tracing::debug!(
475            attempts = MAX_HANDOFF_ATTEMPTS,
476            len = frame.data.len(),
477            "hand-off abandoned: the node had no room for it"
478        );
479        Ok(())
480    }
481
482    /// Wait without going deaf.
483    ///
484    /// Reception cannot pause while the relay waits: a frame left sitting
485    /// in the device is a frame the device may have to drop, and when the
486    /// wait is for a full node receive queue, draining it is precisely
487    /// what makes room for the frame being retried.
488    async fn listen_during(&self, device: &mut Device, wait: Duration) -> Result<()> {
489        let sleep = tokio::time::sleep(wait);
490        tokio::pin!(sleep);
491
492        core::future::poll_fn(|cx| {
493            loop {
494                match device.poll_receive_raw(cx) {
495                    Poll::Ready(Ok(raw)) => self.accept(raw),
496                    Poll::Ready(Err(error)) => {
497                        return Poll::Ready(Err(anyhow!("receiving from the device: {error:?}")));
498                    }
499                    Poll::Pending => break,
500                }
501            }
502            sleep.as_mut().poll(cx).map(Ok)
503        })
504        .await
505    }
506
507    /// One received frame, on its way to whoever consumes this interface.
508    fn accept(&self, raw: RawRxFrame) {
509        tracing::trace!(len = raw.data.len(), "device rx");
510        if self.inbound.push(TunnelFrame::new(raw.data, raw.metadata)) {
511            tracing::warn!("inbound queue full; dropped the oldest frame");
512        }
513    }
514}
515
516enum Event {
517    Received(Result<RawRxFrame, UlcpError>),
518    Send(TunnelFrame),
519    Ended,
520}

522#[cfg(test)]
523mod tests {
524    use super::*;
525
526    #[test]
527    fn a_hand_off_retry_is_short_and_jittered() {
528        // Short because what is being waited out is a node draining its
529        // receive queue, not a frame finishing in the air; jittered so
530        // that interfaces sharing a node do not synchronize on it.
531        let mut seen_low = false;
532        let mut seen_high = false;
533        for _ in 0..512 {
534            let wait = handoff_retry();
535            assert!(wait >= HANDOFF_RETRY);
536            assert!(wait <= HANDOFF_RETRY + Duration::from_millis(HANDOFF_RETRY_JITTER_MS));
537            seen_low |= wait < HANDOFF_RETRY + Duration::from_millis(HANDOFF_RETRY_JITTER_MS / 4);
538            seen_high |= wait > HANDOFF_RETRY + Duration::from_millis(HANDOFF_RETRY_JITTER_MS / 2);
539        }
540        assert!(seen_low && seen_high, "the range is not being covered");
541    }
542
543    #[test]
544    fn the_whole_hand_off_is_bounded_well_inside_the_staleness_limit() {
545        // Frames are discarded once they are ten seconds old, so the
546        // retries must not be able to hold one anywhere near that long.
547        let worst = (HANDOFF_RETRY + Duration::from_millis(HANDOFF_RETRY_JITTER_MS))
548            * u32::from(MAX_HANDOFF_ATTEMPTS - 1);
549        assert!(worst < Duration::from_secs(1), "{worst:?}");
550    }
551
552    #[test]
553    fn a_transmit_asks_for_no_favors() {
554        // Default power, and TX_FLAG_NODUTY clear: the device's duty
555        // ledger is the backstop against a bridge that would otherwise
556        // spend a segment's whole airtime budget on somebody else's
557        // traffic.
558        let meta = TxMeta::default();
559        assert_eq!(meta.flags, 0);
560        assert_eq!(meta.power, umsh_ulcp::meta::TX_POWER_DEFAULT);
561    }
