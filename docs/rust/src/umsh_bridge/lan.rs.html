1//! A participant's membership in a LAN bearer group, and the loop that
2//! relays frames between it and one queue in each direction.
3//!
4//! Every datagram is a frame, with no header and no measurements: what
5//! arrives from the group crosses as an unmeasured reception, and what
6//! leaves for it is the frame alone.
7//!
8//! No repeater stands between the group and the bridge, so nothing
9//! else stops a frame circulating between two bridge participants on
10//! the same LAN, or between this one and its own multicast echo. The
11//! relay therefore carries a frame across at most once, in either
12//! direction, within [`CROSSING_WINDOW`]. That drops nothing a
13//! repeater would have carried: every frame it suppresses is one the
14//! far side's duplicate caches have already seen.
15
16use std::collections::VecDeque;
17use std::sync::Arc;
18use std::time::Duration;
19
20use anyhow::{Context as _, Result};
21use tokio::time::Instant;
22use umsh_lan_bearer::socket::{LanConfig, LanSocket};
23use umsh_lan_bearer::{Datagram, MAX_FRAME, classify};
24use umsh_ulcp::meta::RxMeta;
25
26use crate::tunnel::{Dequeued, TunnelFrame, TunnelQueue};

28/// How long a frame that crossed in either direction is kept from
29/// crossing again.
30pub const CROSSING_WINDOW: Duration = Duration::from_secs(30);
31
32/// The most crossings remembered. A group busy enough to exceed this
33/// within the window shortens the window rather than growing memory.
34const CROSSINGS_CAP: usize = 256;
35
36/// How long to wait before rejoining a group that failed—a network
37/// not up yet, or an interface that went away.
38const REJOIN_DELAY: Duration = Duration::from_secs(3);
39
40pub struct LanRelay {
41    config: LanConfig,
42    /// Frames heard from the group, for whoever consumes this interface.
43    inbound: Arc<TunnelQueue>,
44    /// Frames to send to the group.
45    outbound: Arc<TunnelQueue>,
46}
47
48impl LanRelay {
49    pub fn new(config: LanConfig, inbound: Arc<TunnelQueue>, outbound: Arc<TunnelQueue>) -> Self {
50        Self {
51            config,
52            inbound,
53            outbound,
54        }
55    }
56
57    /// Stay joined and relaying for as long as the process runs.
58    pub async fn run(&self) -> ! {
59        loop {
60            if let Err(error) = self.session().await {
61                tracing::warn!(group = %self.config.group, "lan: {error:#}");
62            }
63            self.outbound.clear();
64            tokio::time::sleep(REJOIN_DELAY).await;
65        }
66    }
67
68    async fn session(&self) -> Result<()> {
69        let socket = LanSocket::bind(self.config)
70            .with_context(|| format!("joining {}", self.config.group))?;
71        tracing::info!(group = %self.config.group, "lan joined");
72
73        let metadata = unmeasured();
74        let mut crossings = Crossings::default();
75        let mut datagram = [0u8; MAX_FRAME + 1];
76        let generation = self.outbound.generation();
77        loop {
78            tokio::select! {
79                received = socket.recv(&mut datagram) => {
80                    let (len, source) = received.context("receiving from the group")?;
81                    let frame = &datagram[..len];
82                    if classify(frame) == Datagram::Malformed {
83                        tracing::debug!(%source, len, "dropping a datagram that is no frame");
84                        continue;
85                    }
86                    if !crossings.admit(frame, Instant::now()) {
87                        continue;
88                    }
89                    tracing::trace!(%source, len, "lan rx");
90                    if self.inbound.push(TunnelFrame::new(frame.to_vec(), metadata.clone())) {
91                        tracing::warn!("inbound queue full; dropped the oldest frame");
92                    }
93                }
94                dequeued = core::future::poll_fn(|cx| self.outbound.poll_pop(generation, cx)) => {
95                    let Dequeued::Frame(frame) = dequeued else {
96                        return Ok(());
97                    };
98                    if frame.data.is_empty() || frame.data.len() > MAX_FRAME {
99                        continue;
100                    }
101                    if !crossings.admit(&frame.data, Instant::now()) {
102                        continue;
103                    }
104                    // A send that fails is lost the way a collided
105                    // transmission is; a network that is gone shows up
106                    // on the receive side.
107                    if let Err(error) = socket.send(&frame.data).await {
108                        tracing::debug!("lan send failed: {error}");
109                    }
110                }
111            }
112        }
113    }
114}

116/// Receive metadata for a frame from the group: nothing was measured.
117fn unmeasured() -> Vec<u8> {
118    let mut metadata = [0u8; RxMeta::WIRE_LEN];
119    RxMeta {
120        rssi_dbm: None,
121        lqi: None,
122        snr_cb: None,
123    }
124    .encode(&mut metadata)
125    .expect("buffer sized with WIRE_LEN");
126    metadata.to_vec()
127}
128
129/// The frames that crossed recently, in either direction.
130#[derive(Default)]
131struct Crossings {
132    recent: VecDeque<(Instant, Vec<u8>)>,
133}
134
135impl Crossings {
136    /// Whether `frame` may cross now, recording that it did.
137    fn admit(&mut self, frame: &[u8], now: Instant) -> bool {
138        while let Some((at, _)) = self.recent.front()
139            && now.saturating_duration_since(*at) > CROSSING_WINDOW
140        {
141            self.recent.pop_front();
142        }
143        if self.recent.iter().any(|(_, seen)| seen == frame) {
144            return false;
145        }
146        if self.recent.len() == CROSSINGS_CAP {
147            self.recent.pop_front();
148        }
149        self.recent.push_back((now, frame.to_vec()));
150        true
151    }
152}

154#[cfg(test)]
155mod tests {
156    use super::*;
157
158    #[test]
159    fn a_frame_crosses_once_in_either_direction_until_the_window_passes() {
160        let mut crossings = Crossings::default();
161        let start = Instant::now();
162        assert!(crossings.admit(&[0xC0, 1], start));
163        // Its own echo, or the same frame back from another bridge.
164        assert!(!crossings.admit(&[0xC0, 1], start + Duration::from_secs(1)));
165        assert!(crossings.admit(&[0xC0, 2], start + Duration::from_secs(1)));
166        assert!(crossings.admit(&[0xC0, 1], start + CROSSING_WINDOW + Duration::from_secs(1)));
167    }
168
169    #[test]
170    fn the_memory_is_bounded() {
171        let mut crossings = Crossings::default();
172        let now = Instant::now();
173        for tag in 0..=CROSSINGS_CAP as u16 {
174            assert!(crossings.admit(&tag.to_be_bytes(), now));
175        }
176        assert_eq!(crossings.recent.len(), CROSSINGS_CAP);
177        // The oldest made room.
178        assert!(crossings.admit(&0u16.to_be_bytes(), now));
179    }
