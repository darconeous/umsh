1//! Static configuration: the whole of what the daemon knows.
2//!
3//! The role — server or client — is a property of the file rather than
4//! the invocation, so a deployment is one unit file and one config, and
5//! `umsh-bridge check` validates the exact artifact that will run.

16/// Default tunnel port: `0x554D`, big-endian ASCII "UM". Unassigned by
17/// IANA, and far enough from the usual hand-picked numbers to be
18/// unlikely to collide with something else on a shared host.
19pub const DEFAULT_PORT: u16 = 21837;

21/// Default host-interface port: one above the tunnel, so a bench
22/// deployment of both needs no configuration at all.
23pub const DEFAULT_HOST_PORT: u16 = 21838;

25/// Interface name of a participant's own radio. Reserved: a client
26/// entry may not take it.
27pub const RADIO_INTERFACE: &str = "radio";

80/// A host interface: a plain socket presenting a simulated ULCP device
81/// whose radio is the bridge itself. Whoever connects attaches to the
82/// bridged medium directly, as a host — no radio, no repeater, no
83/// tunnel credential.
84///
85/// The binding is unauthenticated and grants what a serial cable
86/// grants, key provisioning included, which is why it stays on the
87/// loopback unless the deployment says otherwise in so many words.
88#[derive(Debug, Deserialize)]
89#[serde(deny_unknown_fields)]
90pub struct HostEntry {
91    /// Interface name, used in log lines and in `allow_to`.
92    pub name: String,
93    /// Where the ULCP socket listens.
94    #[serde(default = "default_host_listen")]
95    pub listen: SocketAddr,
96    /// Forwarding budget for frames this host injects. Required, not
97    /// optional as for a client: a host spends no airtime of its own,
98    /// but every frame it injects is transmitted by every participant's
99    /// node.
100    pub max_frames_per_minute: u32,
101    /// Interfaces this host's traffic may be fanned out to. Absent
102    /// means all of them.
103    pub allow_to: Option<Vec<String>>,
104    /// Confirm a non-loopback `listen` on purpose. Reaching this socket
105    /// is an RF presence on every segment the bridge touches, plus a
106    /// physically-secure ULCP attachment to the simulated device.
107    #[serde(default)]
108    pub allow_remote: bool,
109}

131/// What the bridge does to traffic beyond carrying it.
132///
133/// Region and signal policy are not here: they belong to each
134/// participant's own repeater, which is what puts bridged traffic on the
135/// air, and are configured on the device.
136#[derive(Debug, Default, Deserialize)]
137#[serde(deny_unknown_fields)]
138pub struct LimitsConfig {
139    /// Ceiling applied to `FHOPS_REM` as a frame crosses. Absent leaves
140    /// every frame's budget exactly as it arrived, which is the default:
141    /// a crossing already spends two hops, one at each end's repeater.
142    /// Set it to hold traffic closer to home — 0 stops a bridged flood
143    /// at the far side's own segment.
144    pub exit_clamp: Option<u8>,
145}

168/// Which device this participant fronts.
169#[derive(Debug, Default)]
170pub enum RadioConfig {
171    /// A ULCP device on a serial port.
172    Serial { port: String, baud: u32 },
173    /// A ULCP device over BLE, named or discovered.
174    Ble {
175        /// Radio name or scan id. Absent discovers, which is only
176        /// unambiguous where exactly one radio is in range — name the
177        /// radio for an unattended deployment.
178        selector: Option<String>,
179    },
180    /// The UDP-multicast fake radio the examples use. For testing a
181    /// bridge without hardware, or without airtime.
182    UdpMulticast {
183        group: Ipv4Addr,
184        port: u16,
185        /// Signal quality to synthesize for received frames, since UDP
186        /// measures none.
187        rssi: i16,
188        snr: i8,
189    },
190    /// No radio at all: a server that only joins clients together.
191    #[default]
192    None,
193}

195/// The radio table as written, before it is narrowed to one variant.
196///
197/// A tagged enum would be the obvious shape, but serde buffers an
198/// internally tagged enum's content and `deny_unknown_fields` never
199/// reaches the variant — a misspelled key would be silently ignored,
200/// which for a daemon's configuration is the worst possible outcome.
201/// Reading the table flat also lets a key belonging to the wrong `type`
202/// be named in the error.
203#[derive(Deserialize)]
204#[serde(deny_unknown_fields)]
205struct RadioTable {
206    #[serde(rename = "type")]
207    kind: String,
208    /// A device path for `serial`, a UDP port number for
209    /// `udp-multicast`.
210    port: Option<toml::Value>,
211    baud: Option<u32>,
212    selector: Option<String>,
213    group: Option<Ipv4Addr>,
214    rssi: Option<i16>,
215    snr: Option<i8>,
216}
217
218impl<'de> Deserialize<'de> for RadioConfig {
219    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
220        use serde::de::Error as _;
221
222        let table = RadioTable::deserialize(deserializer)?;
223        let reject = |unexpected: &[(&str, bool)]| -> Result<(), D::Error> {
224            for (name, present) in unexpected {
225                if *present {
226                    return Err(D::Error::custom(format!(
227                        "\"{name}\" has no meaning for a {} radio",
228                        table.kind
229                    )));
230                }
231            }
232            Ok(())
233        };
234
235        match table.kind.as_str() {
236            "serial" => {
237                reject(&[
238                    ("selector", table.selector.is_some()),
239                    ("group", table.group.is_some()),
240                    ("rssi", table.rssi.is_some()),
241                    ("snr", table.snr.is_some()),
242                ])?;
243                let port = table
244                    .port
245                    .as_ref()
246                    .and_then(toml::Value::as_str)
247                    .ok_or_else(|| D::Error::custom("a serial radio needs a string \"port\""))?;
248                Ok(Self::Serial {
249                    port: port.to_string(),
250                    baud: table.baud.unwrap_or_else(default_baud),
251                })
252            }
253            "ble" => {
254                reject(&[
255                    ("port", table.port.is_some()),
256                    ("baud", table.baud.is_some()),
257                    ("group", table.group.is_some()),
258                    ("rssi", table.rssi.is_some()),
259                    ("snr", table.snr.is_some()),
260                ])?;
261                Ok(Self::Ble {
262                    selector: table.selector,
263                })
264            }
265            "udp-multicast" => {
266                reject(&[
267                    ("baud", table.baud.is_some()),
268                    ("selector", table.selector.is_some()),
269                ])?;
270                let port = match &table.port {
271                    None => default_udp_port(),
272                    Some(value) => value
273                        .as_integer()
274                        .and_then(|port| u16::try_from(port).ok())
275                        .ok_or_else(|| {
276                            D::Error::custom("a udp-multicast \"port\" is a number, 1..=65535")
277                        })?,
278                };
279                let group = table.group.unwrap_or_else(default_udp_group);
280                if !group.is_multicast() {
281                    return Err(D::Error::custom(format!(
282                        "\"{group}\" is not an IPv4 multicast address"
283                    )));
284                }
285                Ok(Self::UdpMulticast {
286                    group,
287                    port,
288                    rssi: table.rssi.unwrap_or_else(default_udp_rssi),
289                    snr: table.snr.unwrap_or_else(default_udp_snr),
290                })
291            }
292            "none" => {
293                reject(&[
294                    ("port", table.port.is_some()),
295                    ("baud", table.baud.is_some()),
296                    ("selector", table.selector.is_some()),
297                    ("group", table.group.is_some()),
298                    ("rssi", table.rssi.is_some()),
299                    ("snr", table.snr.is_some()),
300                ])?;
301                Ok(Self::None)
302            }
303            other => Err(D::Error::custom(format!(
304                "unknown radio type \"{other}\"; expected serial, ble, udp-multicast, or none"
305            ))),
306        }
307    }
308}

315    /// How the interface is named in logs.
316    pub fn describe(&self) -> String {
317        match self {
318            Self::Serial { port, baud } => format!("serial {port} @ {baud}"),

385impl Config {
386    pub fn load(path: &Path) -> Result<Self> {
387        let text =
388            std::fs::read_to_string(path).with_context(|| format!("reading {}", path.display()))?;
389        let config: Self =
390            toml::from_str(&text).with_context(|| format!("parsing {}", path.display()))?;
391        config.validate()?;
392        Ok(config)
393    }
394
395    /// Everything the type system and serde cannot say.
396    pub fn validate(&self) -> Result<()> {
397        match (&self.server, &self.client) {
398            (Some(_), Some(_)) => {
399                bail!("give either [server] or [client], not both: one process is one role")
400            }
401            (None, None) => bail!("no role configured: add a [server] or [client] section"),
402            (Some(server), None) => {
403                if self.identity.is_none() {
404                    bail!("a bridge server owns the bridge's node identity; add [identity]");
405                }
406                server.validate()
407            }
408            (None, Some(client)) => {
409                if self.identity.is_none() {
410                    bail!(
411                        "a client authenticates the tunnel with its own identity; add [identity]"
412                    );
413                }
414                client.validate()
415            }
416        }
417    }
418}

420impl ServerConfig {
421    fn validate(&self) -> Result<()> {
422        if self.listen.is_empty() {
423            bail!("[server] listen is empty; the server would accept nothing");
424        }
425        if self.clients.is_empty() && self.hosts.is_empty() && self.radio.is_none() {
426            bail!("[server] has no clients, no hosts, and no radio; there is nothing to bridge");
427        }
428        if let Some(clamp) = self.limits.exit_clamp
429            && clamp > 15
430        {
431            bail!(
432                "[server.limits] exit_clamp is {clamp}; FHOPS_REM is four bits and holds at most 15"
433            );
434        }
435        if self.tunnel.queue_depth == 0 {
436            bail!("[server.tunnel] queue_depth must be at least 1");
437        }
438        if self.tunnel.idle_timeout_secs <= self.tunnel.keepalive_secs {
439            bail!(
440                "[server.tunnel] idle_timeout_secs ({}) must exceed keepalive_secs ({}), or a \
441                 healthy peer times out between its own keepalives",
442                self.tunnel.idle_timeout_secs,
443                self.tunnel.keepalive_secs
444            );
445        }
446
447        let mut names = HashSet::new();
448        let mut addresses = HashSet::new();
449        for client in &self.clients {
450            if client.name == RADIO_INTERFACE {
451                bail!("client name \"{RADIO_INTERFACE}\" is reserved for the server's own radio");
452            }
453            if client.name.trim().is_empty() {
454                bail!("every [[server.clients]] needs a name; it is how policy refers to it");
455            }
456            if !names.insert(client.name.as_str()) {
457                bail!("two [[server.clients]] are both named \"{}\"", client.name);
458            }
459            if !addresses.insert(client.address) {
460                bail!(
461                    "client \"{}\" shares an address with another; an identity names exactly \
462                     one client",
463                    client.name
464                );
465            }
466        }
467
468        let mut listens = HashSet::new();
469        for host in &self.hosts {
470            if host.name == RADIO_INTERFACE {
471                bail!("host name \"{RADIO_INTERFACE}\" is reserved for the server's own radio");
472            }
473            if host.name.trim().is_empty() {
474                bail!("every [[server.hosts]] needs a name; it is how policy refers to it");
475            }
476            if !names.insert(host.name.as_str()) {
477                bail!(
478                    "\"{}\" names two interfaces; clients and hosts share one namespace",
479                    host.name
480                );
481            }
482            if !listens.insert(host.listen) {
483                bail!(
484                    "host \"{}\" listens on {}, which another host already holds",
485                    host.name,
486                    host.listen
487                );
488            }
489            if !host.listen.ip().is_loopback() && !host.allow_remote {
490                bail!(
491                    "host \"{}\" listens on {}, which is not the loopback. The socket is \
492                     unauthenticated and reaching it is an RF presence on every segment this \
493                     bridge touches; set allow_remote = true to do this on purpose",
494                    host.name,
495                    host.listen
496                );
497            }
498        }
499
500        let allowlists = self
501            .clients
502            .iter()
503            .map(|client| (client.name.as_str(), &client.allow_to))
504            .chain(
505                self.hosts
506                    .iter()
507                    .map(|host| (host.name.as_str(), &host.allow_to)),
508            );
509        for (name, allow_to) in allowlists {
510            let Some(allow_to) = allow_to else {
511                continue;
512            };
513            for target in allow_to {
514                if target == RADIO_INTERFACE {
515                    if self.radio.is_none() {
516                        bail!(
517                            "\"{name}\" allows forwarding to \"{RADIO_INTERFACE}\", but \
518                             [server.radio] is none"
519                        );
520                    }
521                    continue;
522                }
523                if !names.contains(target.as_str()) {
524                    bail!(
525                        "\"{name}\" allows forwarding to \"{target}\", which is not a \
526                         configured interface"
527                    );
528                }
529            }
530        }
531        Ok(())
532    }
533
534    /// Every interface name this server can forward to, radio first.
535    pub fn interface_names(&self) -> Vec<String> {
536        let mut names = Vec::new();
537        if !self.radio.is_none() {
538            names.push(RADIO_INTERFACE.to_string());
539        }
540        names.extend(self.clients.iter().map(|client| client.name.clone()));
541        names.extend(self.hosts.iter().map(|host| host.name.clone()));
542        names
543    }
544}

546impl ClientConfig {
547    fn validate(&self) -> Result<()> {
548        if self.radio.is_none() {
549            bail!("[client.radio] is none; a client with no radio relays nothing");
550        }
551        if self.tunnel.queue_depth == 0 {
552            bail!("[client.tunnel] queue_depth must be at least 1");
553        }
554        if self.tunnel.idle_timeout_secs <= self.tunnel.keepalive_secs {
555            bail!(
556                "[client.tunnel] idle_timeout_secs ({}) must exceed keepalive_secs ({})",
557                self.tunnel.idle_timeout_secs,
558                self.tunnel.keepalive_secs
559            );
560        }
561        if self.tunnel.reconnect_max_secs < self.tunnel.reconnect_min_secs {
562            bail!("[client.tunnel] reconnect_max_secs is below reconnect_min_secs");
563        }
564        if !self.server.contains(':') {
565            bail!(
566                "[client] server \"{}\" needs a port; the bridge default is {DEFAULT_PORT}",
567                self.server
568            );
569        }
570        Ok(())
571    }
572}
