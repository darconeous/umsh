1//! The forwarding procedure, and the one duplicate cache the whole
2//! bridge shares.
3//!
4//! The bridge is one virtual repeater whose radios happen to sit in
5//! different places: it suppresses duplicates once, accounts for a hop
6//! once, and prepends its router hint to a trace route once, no matter
7//! how many interfaces took part. That is why every decision is made
8//! here, in one task, and clients relay bytes without an opinion.
9//!
10//! [`Engine::evaluate`] walks the [thirteen steps] in order, including
11//! the one that is a no-op for a forwarding-only bridge, so the code and
12//! the specification can be read side by side.
13//!
14//! [thirteen steps]: https://darconeous.github.io/umsh/docs/protocol/internet-bridging.html#forwarding-procedure
15

33/// How long a duplicate key stays suppressed. The same hour the MAC
34/// uses: long enough to collapse a burst of retransmissions of one
35/// packet, short enough that a node re-announcing itself is heard again
36/// well within the time anyone would wait for it.
37const CACHE_TTL: Duration = Duration::from_millis(umsh_mac::DUP_CACHE_TTL_MS);
38
39/// How many times a flood confirmation copy defers to an overheard
40/// forwarding before giving up on it. Any overheard forwarding confirms
41/// the previous hop just as well as ours would have.
42const MAX_DEFERRALS: u8 = 3;
43
44/// Largest frame the bridge will compose. A rewritten frame that would
45/// exceed this is dropped at step 10, and one this large is already far
46/// beyond what any LoRa PHY carries.
47const MAX_FRAME_SIZE: usize = 255;
48
49/// How often the bridge says what it has been doing. Slow enough to
50/// leave on in a deployment, frequent enough to see a problem develop.
51const STATS_INTERVAL: Duration = Duration::from_secs(300);

67/// What happened to a frame, for logs and tests.
68#[derive(Clone, Copy, Debug, PartialEq, Eq)]
69pub enum Verdict {
70    /// The frame did not parse as a UMSH packet at all.
71    Unparseable,
72    /// Step 1: already seen, and outside the re-confirmation window or
73    /// on a different interface.
74    Duplicate,
75    /// Step 1 / Re-confirmation: a previous hop retrying, answered with
76    /// another confirmation copy and nothing else.
77    Reconfirmed,
78    /// Step 2: the bridge is the source or the destination.
79    Local,
80    /// Step 4: an unknown critical option.
81    UnknownCriticalOption,
82    /// Step 5: policy declined it.
83    Policy(&'static str),
84    /// Step 6: the source route names someone else next.
85    NotOurHop,
86    /// Step 7: the flood budget is spent.
87    HopsExhausted,
88    /// Step 8: too weak to relay.
89    TooWeak(&'static str),
90    /// Step 10 or the rewrite: the frame cannot be composed.
91    Undeliverable(RewriteError),
92    /// Steps 11-13: accepted, and transmitted on this many interfaces
93    /// (not counting the confirmation copy).
94    Forwarded(usize),
95}

133    /// Drive the bridge: frames in, transmissions out, deferred
134    /// confirmation copies fired when their window elapses.
135    pub async fn run(mut self, mut ingress: mpsc::Receiver<Ingress>) {
136        let mut report = tokio::time::interval(STATS_INTERVAL);
137        report.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
138        report.tick().await;
139
140        loop {
141            let deadline = self.pending.iter().map(|pending| pending.at).min();
142            tokio::select! {
143                biased;
144                arrival = ingress.recv() => match arrival {
145                    Some(arrival) => {
146                        let verdict = self.evaluate(&arrival, Instant::now());
147                        tracing::debug!(
148                            iface = %self.interfaces.name(arrival.iface),
149                            len = arrival.frame.data.len(),
150                            ?verdict,
151                        );
152                    }
153                    // Every interface task is gone, which cannot happen
154                    // while the process is running.
155                    None => return,
156                },
157                () = sleep_until(deadline) => self.fire_due(Instant::now()),
158                _ = report.tick() => self.report(),
159            }
160        }
161    }

163    /// What the bridge has been doing, at an interval slow enough to
164    /// leave in a running deployment's log.
165    fn report(&self) {
166        let counters = self.counters;
167        tracing::info!(
168            received = counters.received,
169            forwarded = counters.forwarded,
170            duplicates = counters.duplicates,
171            reconfirmations = counters.reconfirmations,
172            dropped = counters.dropped,
173            cached = self.cache.entries.len(),
174            "bridge"
175        );
176        for iface in &self.interfaces.all {
177            let (full, stale) = iface.egress.dropped();
178            if full == 0 && stale == 0 {
179                continue;
180            }
181            tracing::info!(
182                iface = %iface.name,
183                dropped_backpressure = full,
184                dropped_stale = stale,
185                "egress"
186            );
187        }
188    }

190    /// One pass of the forwarding procedure, in specification order.
191    pub fn evaluate(&mut self, arrival: &Ingress, now: Instant) -> Verdict {
192        self.counters.received += 1;
193        let verdict = self.classify(arrival, now);
194        match verdict {
195            Verdict::Forwarded(_) => self.counters.forwarded += 1,
196            Verdict::Duplicate => self.counters.duplicates += 1,
197            Verdict::Reconfirmed => self.counters.reconfirmations += 1,
198            _ => self.counters.dropped += 1,
199        }
200        verdict
201    }
202
203    fn classify(&mut self, arrival: &Ingress, now: Instant) -> Verdict {
204        let src = &arrival.frame.data;
205        let Ok(header) = PacketHeader::parse(src) else {
206            return Verdict::Unparseable;
207        };
208        let Ok(options) = ParsedOptions::extract(src, header.options_range.clone()) else {
209            return Verdict::Unparseable;
210        };
211
212        // Step 1: duplicate suppression, against the one cache the whole
213        // bridge shares. A Route Retry variant keys differently and so
214        // forwards normally, which is the point of it.
215        let key = forwarding_dup_key_parsed(&header, src);
216        if let Some(key) = &key
217            && let Some(entry) = self.cache.get(key, now)
218        {
219            return self.on_duplicate(entry, key, arrival, &header, now);
220        }
221
222        // Step 2: local origin and local destination. This is also what
223        // keeps the bridge's own transmissions, overheard through
224        // another of its interfaces, from being bridged back.
225        if self.is_local(src, &header) {
226            return Verdict::Local;
227        }
228
229        // Step 3: locally handled unicast. A forwarding-only bridge runs
230        // no node logic, so nothing is ever handled locally and this
231        // step never declines a frame. It is kept as a step so the
232        // procedure can be read against the specification.
233
234        // Step 4: an option the bridge does not understand, marked
235        // critical, means the frame must not be relayed blind.
236        if options.has_unknown_critical {
237            return Verdict::UnknownCriticalOption;
238        }
239
240        // Step 5: policy.
241        if !self.policy.region_admits(&options) {
242            return Verdict::Policy("region");
243        }
244        if !self.policy.admit(&self.interfaces, arrival.iface, now) {
245            return Verdict::Policy("rate limit");
246        }
247
248        // Step 6: a source route names its next hop, and if that is not
249        // us the frame is not ours to carry.
250        let routed = match rewrite::next_source_route_hop(src, &options) {
251            Some(next) if next == self.router_hint => true,
252            Some(_) => return Verdict::NotOurHop,
253            None => false,
254        };
255
256        if !routed {
257            // Step 7: flood hop accounting.
258            if header.flood_hops.is_some_and(|hops| hops.remaining() == 0) {
259                return Verdict::HopsExhausted;
260            }
261
262            // Step 8: signal-quality thresholds, against the
263            // measurements of the radio that actually heard the frame.
264            if let Some(reason) = self.too_weak(&arrival.frame) {
265                return Verdict::TooWeak(reason);
266            }
267        }
268
269        // Steps 9 and 10 happen inside the rewrite.
270        let plan = RewritePlan {
271            router_hint: self.router_hint,
272            consume_source_route: routed,
273            decrement_flood_hops: !routed,
274            exit_clamp: self.exit_clamp,
275        };
276        let rewritten = match rewrite::rewrite(src, &header, &options, plan, MAX_FRAME_SIZE) {
277            Ok(rewritten) => rewritten,
278            Err(error) => return Verdict::Undeliverable(error),
279        };
280
281        // Step 11: accept. The insertion happens before any
282        // transmission, which is normative for a bridge — it is what
283        // stops the bridge's own zero-hop confirmation copy from
284        // sterilizing the live flood at a repeater that hears the copy
285        // first.
286        let Some(key) = key else {
287            // A frame with no key cannot be suppressed, re-confirmed, or
288            // recognized when it comes back. Forwarding it would risk a
289            // loop the cache cannot break.
290            return Verdict::Undeliverable(RewriteError::Malformed);
291        };
292        self.cache.insert(
293            key.clone(),
294            CacheEntry {
295                arrival: arrival.iface,
296                accepted_at: now,
297                confirmation: rewritten.confirmation().to_vec(),
298                hop_byte: header.flood_hops.map(|hops| hops.0),
299                source_routed: routed,
300            },
301            now,
302        );
303
304        // Step 12: fan out on every other interface policy allows.
305        // These introduce the packet to segments that have not carried
306        // it, so the flood contention window does not apply.
307        let mut sent = 0;
308        for iface in &self.interfaces.all {
309            if !self
310                .policy
311                .may_forward(&self.interfaces, arrival.iface, iface.id)
312            {
313                continue;
314            }
315            iface.send(TunnelFrame::new(
316                rewritten.forward.clone(),
317                transmit_metadata(),
318            ));
319            sent += 1;
320        }
321
322        // Step 13: the confirmation copy, back the way the frame came.
323        self.schedule_confirmation(arrival.iface, key, rewritten.confirmation(), routed, now);
324        Verdict::Forwarded(sent)
325    }

327    /// A duplicate is silence, with one exception: a previous hop
328    /// retrying because our confirmation copy was lost.
329    fn on_duplicate(
330        &mut self,
331        entry: CacheEntry,
332        key: &DupCacheKey,
333        arrival: &Ingress,
334        header: &PacketHeader,
335        now: Instant,
336    ) -> Verdict {
337        if entry.arrival != arrival.iface {
338            // An ordinary suppressed duplicate: the frame reached us by
339            // a second path. Nothing is transmitted anywhere.
340            return Verdict::Duplicate;
341        }
342        if now.saturating_duration_since(entry.accepted_at) > self.reconfirmation_window() {
343            return Verdict::Duplicate;
344        }
345
346        // On the arrival interface, inside the window, two very
347        // different things look alike. A byte-identical hop count is the
348        // previous hop retrying — it has not heard our confirmation and
349        // will declare the route failed if it never does. A different
350        // one is somebody else's forwarding of the same packet, which
351        // confirms the previous hop just as well as ours would, so our
352        // copy can stand down.
353        //
354        // Only for a flood hop. A source-routed hop's copy is the
355        // previous hop's *only* confirmation — nobody else is going to
356        // forward that packet — so it is never abandoned.
357        let arriving_hop_byte = header.flood_hops.map(|hops| hops.0);
358        if !entry.source_routed && arriving_hop_byte != entry.hop_byte {
359            self.defer_confirmation(arrival.iface, key, now);
360            return Verdict::Duplicate;
361        }
362
363        // The copy may still be waiting out its contention window; this
364        // retransmission replaces it rather than stacking a second one.
365        //
366        // A retry answered here happens even on an interface whose flood
367        // confirmations are suppressed, deliberately: the retry means
368        // the previous hop heard nothing — the co-located device's own
369        // re-forward was lost too — so the stored copy is the
370        // confirmation of last resort.
371        self.cancel_pending(arrival.iface, key);
372        self.transmit_confirmation(arrival.iface, &entry.confirmation);
373        Verdict::Reconfirmed
374    }

376    fn is_local(&self, src: &[u8], header: &PacketHeader) -> bool {
377        if header.dst == Some(self.node_hint) {
378            return true;
379        }
380        match header.source {
381            SourceAddrRef::Hint(hint) => hint == self.node_hint,
382            SourceAddrRef::FullKeyAt { offset } => src
383                .get(offset..offset + 3)
384                .is_some_and(|key| key == self.node_hint.0),
385            // A blinded source cannot be compared without the channel
386            // key, which a forwarding-only bridge does not hold.
387            SourceAddrRef::Encrypted { .. } | SourceAddrRef::None => false,
388        }
389    }

391    /// Each threshold is applied only where the radio reported the
392    /// measurement it needs. A sentinel is "this radio cannot tell you",
393    /// not "zero", and treating it as a reading would silence a segment
394    /// whose hardware simply does not measure.
395    fn too_weak(&self, frame: &TunnelFrame) -> Option<&'static str> {
396        let Ok(meta) = RxMeta::decode(&frame.metadata) else {
397            return None;
398        };
399        if let Some(floor) = self.min_rssi {
400            match meta.rssi_dbm {
401                Some(rssi) if rssi < floor => return Some("rssi"),
402                None => tracing::trace!("no RSSI reported; the minimum-RSSI check is skipped"),
403                _ => {}
404            }
405        }
406        if let Some(floor) = self.min_snr_cb {
407            match meta.snr_cb {
408                Some(snr) if snr < floor => return Some("snr"),
409                None => tracing::trace!("no SNR reported; the minimum-SNR check is skipped"),
410                _ => {}
411            }
412        }
413        None
414    }

416    fn reconfirmation_window(&self) -> Duration {
417        self.cache.reconfirmation_window
418    }
419
420    /// A source-routed hop's confirmation copy is the previous hop's
421    /// only confirmation, so it goes out under ordinary channel access
422    /// and is never abandoned. A flood hop's is one forwarding among
423    /// several on a segment already carrying the packet, so it takes the
424    /// contention window and stands down if someone else forwards first.
425    fn schedule_confirmation(
426        &mut self,
427        iface: InterfaceId,
428        key: DupCacheKey,
429        confirmation: &[u8],
430        source_routed: bool,
431        now: Instant,
432    ) {
433        if source_routed {
434            self.transmit_confirmation(iface, confirmation);
435            return;
436        }
437        if self
438            .policy
439            .suppresses_flood_confirmations(&self.interfaces, iface)
440        {
441            // The device behind this interface re-forwards the flood
442            // itself, and that re-forward confirms the previous hop.
443            tracing::trace!(
444                iface = %self.interfaces.name(iface),
445                "flood confirmation copy suppressed; the device's own repeater role covers it"
446            );
447            return;
448        }
449        self.pending.push_back(PendingConfirmation {
450            iface,
451            key,
452            frame: confirmation.to_vec(),
453            at: now + self.contention_delay(),
454            deferrals: 0,
455        });
456    }
457
458    /// Defer this packet's pending flood confirmation on this interface,
459    /// or abandon it once it has stood down often enough. An overheard
460    /// forwarding speaks only for its own packet, which is why the match
461    /// is on the key and not the interface alone.
462    fn defer_confirmation(&mut self, iface: InterfaceId, key: &DupCacheKey, now: Instant) {
463        let Some(index) = self
464            .pending
465            .iter()
466            .position(|pending| pending.iface == iface && pending.key == *key)
467        else {
468            return;
469        };
470        if self.pending[index].deferrals >= MAX_DEFERRALS {
471            tracing::trace!(
472                iface = %self.interfaces.name(iface),
473                "abandoning a flood confirmation copy; the segment is carrying the packet"
474            );
475            self.pending.remove(index);
476            return;
477        }
478        self.pending[index].deferrals += 1;
479        self.pending[index].at = now + self.contention_delay();
480    }
481
482    /// Drop a pending confirmation that a direct retransmission has just
483    /// made redundant.
484    fn cancel_pending(&mut self, iface: InterfaceId, key: &DupCacheKey) {
485        self.pending
486            .retain(|pending| !(pending.iface == iface && pending.key == *key));
487    }
488
489    fn fire_due(&mut self, now: Instant) {
490        while let Some(index) = self.pending.iter().position(|pending| pending.at <= now) {
491            let pending = self.pending.remove(index).expect("index from position");
492            self.transmit_confirmation(pending.iface, &pending.frame);
493        }
494    }
495
496    fn transmit_confirmation(&self, iface: InterfaceId, frame: &[u8]) {
497        tracing::trace!(
498            iface = %self.interfaces.name(iface),
499            "confirmation copy"
500        );
501        self.interfaces
502            .get(iface)
503            .send(TunnelFrame::new(frame.to_vec(), transmit_metadata()));
504    }
505
506    /// A uniform spread over the contention window.
507    ///
508    /// An ordinary repeater weights this by signal quality so the
509    /// weakest-placed hearer speaks first. A bridge cannot: the
510    /// measurements it has for a tunneled frame belong to a radio
511    /// somewhere else, and may be sentinels.
512    fn contention_delay(&self) -> Duration {
513        if self.contention.is_zero() {
514            return Duration::ZERO;
515        }
516        Duration::from_millis(rand::random_range(0..=self.contention.as_millis() as u64))
517    }
518}

520/// The transmit metadata the server composes for every frame it sends.
521///
522/// Default power — the device's own configuration is the local decision,
523/// not the bridge's — and no flags. `TX_FLAG_NODUTY` is clear and stays
524/// clear: the device's duty-cycle enforcement is the backstop against a
525/// bridge that would otherwise spend a whole segment's airtime budget.
526fn transmit_metadata() -> Vec<u8> {
527    let mut buf = [0u8; TxMeta::WIRE_LEN];
528    TxMeta::default()
529        .encode(&mut buf)
530        .expect("buffer sized with WIRE_LEN");
531    buf.to_vec()
532}

541struct PendingConfirmation {
542    iface: InterfaceId,
543    /// Routing identity of the packet this copy confirms, so overheard
544    /// forwardings and retries touch only their own packet's copy.
545    key: DupCacheKey,
546    frame: Vec<u8>,
547    at: Instant,
548    deferrals: u8,
549}

551#[derive(Clone)]
552struct CacheEntry {
553    /// The interface the frame was accepted from — the only one a
554    /// re-confirmation may go back out on.
555    arrival: InterfaceId,
556    accepted_at: Instant,
557    /// The copy to re-transmit if the previous hop retries. Stored
558    /// rather than recomputed, so a re-confirmation is a queue push.
559    confirmation: Vec<u8>,
560    /// The raw hop-count byte of the frame as accepted, which is how a
561    /// previous hop's retry is told from somebody else's forwarding.
562    hop_byte: Option<u8>,
563    source_routed: bool,
564}
565
566/// The bridge's duplicate cache.
567///
568/// [`umsh_mac::DuplicateCache`] holds keys and nothing else, which is
569/// all a single-radio repeater needs. A bridge has to remember where a
570/// frame came from and what it sent back, so it keeps its own store —
571/// keyed identically, since that key is the interop surface.
572struct DuplicateCache {
573    entries: VecDeque<(DupCacheKey, CacheEntry)>,
574    capacity: usize,
575    reconfirmation_window: Duration,
576}
577
578impl DuplicateCache {
579    fn new(capacity: usize) -> Self {
580        Self {
581            entries: VecDeque::with_capacity(capacity.max(1)),
582            capacity: capacity.max(1),
583            reconfirmation_window: Duration::from_secs(30),
584        }
585    }
586
587    fn with_window(mut self, window: Duration) -> Self {
588        self.reconfirmation_window = window;
589        self
590    }
591
592    fn get(&mut self, key: &DupCacheKey, now: Instant) -> Option<CacheEntry> {
593        self.expire(now);
594        self.entries
595            .iter()
596            .find(|(candidate, _)| candidate == key)
597            .map(|(_, entry)| entry.clone())
598    }
599
600    /// First insert wins: a key already present keeps its original
601    /// arrival interface and acceptance time, which is what the
602    /// re-confirmation window is measured against.
603    fn insert(&mut self, key: DupCacheKey, entry: CacheEntry, now: Instant) {
604        self.expire(now);
605        if self.entries.iter().any(|(candidate, _)| *candidate == key) {
606            return;
607        }
608        while self.entries.len() >= self.capacity {
609            self.entries.pop_front();
610        }
611        self.entries.push_back((key, entry));
612    }
613
614    /// Insertion-ordered, so expiry prunes from the front without a
615    /// scan.
616    fn expire(&mut self, now: Instant) {
617        while let Some((_, entry)) = self.entries.front() {
618            if now.saturating_duration_since(entry.accepted_at) < CACHE_TTL {
619                break;
620            }
621            self.entries.pop_front();
622        }
623    }
624}

666    fn ingress(
667        iface: InterfaceId,
668        data: Vec<u8>,
669        rssi: Option<i16>,
670        snr_cb: Option<i16>,
671    ) -> Ingress {
672        let mut metadata = [0u8; RxMeta::WIRE_LEN];
673        RxMeta {
674            rssi_dbm: rssi,
675            lqi: None,
676            snr_cb,
677        }
678        .encode(&mut metadata)
679        .unwrap();
680        Ingress {
681            iface,
682            frame: TunnelFrame::new(data, metadata.to_vec()),
683        }
684    }

690    fn broadcast(remaining: u8, source: NodeHint, options: &[(OptionNumber, Vec<u8>)]) -> Vec<u8> {
691        let mut buf = [0u8; 255];
692        let mut builder = PacketBuilder::new(&mut buf)
693            .broadcast()
694            .source_hint(source)
695            .flood_hops(remaining);
696        for (number, value) in options {
697            builder = builder.option(*number, value);
698        }
699        builder.payload(b"hello").build().unwrap().to_vec()
700    }

702    /// Drain what each interface was asked to transmit.
703    fn sent(interfaces: &Interfaces) -> Vec<(String, Vec<u8>)> {
704        let mut out = Vec::new();
705        for iface in &interfaces.all {
706            while let Some(crate::tunnel::Dequeued::Frame(frame)) = poll_once(&iface.egress) {
707                out.push((iface.name.clone(), frame.data));
708            }
709        }
710        out
711    }
712
713    fn poll_once(queue: &crate::tunnel::TunnelQueue) -> Option<crate::tunnel::Dequeued> {
714        let waker = std::task::Waker::noop();
715        let mut cx = std::task::Context::from_waker(waker);
716        match queue.poll_pop(queue.generation(), &mut cx) {
717            std::task::Poll::Ready(dequeued) => Some(dequeued),
718            std::task::Poll::Pending => None,
719        }
720    }
