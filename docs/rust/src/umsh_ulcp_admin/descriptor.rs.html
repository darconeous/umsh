1//! The properties the model covers.
2//!
3//! A descriptor says how to read a property's octets (its codec, from
4//! [`umsh_ulcp::describe`]), whether the device has it at all (the
5//! capability that gates it), and what may be done to it (its access and
6//! kind). The device is the authority on everything else: a value that
7//! fails to decode is held as the octets the device reported, and a write
8//! the device adjusts is reported as the device adjusted it.
9
10use umsh_ulcp::PropertyType;
11use umsh_ulcp::ids::{cap, prop};
12
13/// What may be done to a property.
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15pub enum Access {
16    /// Read once: it never changes while the device runs.
17    Constant,
18    /// Read, never written.
19    ReadOnly,
20    /// Read and written whole.
21    ReadWrite,
22    /// A set of items, read whole and changed an item at a time.
23    Table,
24}
25
26/// The shape of a property's value.
27#[derive(Clone, Copy, Debug, PartialEq, Eq)]
28pub enum Kind {
29    /// One value.
30    Scalar,
31    /// A list of packed unsigned integers.
32    Puis,
33    /// Fixed-length items with no length prefix. An item is inserted and
34    /// selected for removal in its item form, and reported—in the
35    /// property's value and in `CMD_PROP_INSERTED` and
36    /// `CMD_PROP_REMOVED`—in its reported form, which omits any key
37    /// material.
38    Items { item_len: u8, reported_len: u8 },
39}
40
41/// One covered property.
42#[derive(Clone, Copy, Debug, PartialEq, Eq)]
43pub struct Descriptor {
44    pub key: u32,
45    /// How its octets read.
46    pub codec: PropertyType,
47    /// The capability a device advertises when it has this property, or
48    /// `None` for one every device has.
49    pub gate: Option<u32>,
50    pub access: Access,
51    pub kind: Kind,
52    /// The longest value or item the model writes, in octets.
53    pub max_len: usize,
54}
55
56impl Descriptor {
57    pub const fn writable(&self) -> bool {
58        matches!(self.access, Access::ReadWrite)
59    }
60
61    pub const fn is_table(&self) -> bool {
62        matches!(self.access, Access::Table)
63    }
64}
65
66const fn scalar(
67    key: u32,
68    codec: PropertyType,
69    gate: Option<u32>,
70    access: Access,
71    max_len: usize,
72) -> Descriptor {
73    Descriptor {
74        key,
75        codec,
76        gate,
77        access,
78        kind: Kind::Scalar,
79        max_len,
80    }
81}
82
83const fn table(key: u32, gate: u32, item_len: u8, reported_len: u8) -> Descriptor {
84    Descriptor {
85        key,
86        codec: PropertyType::Bytes,
87        gate: Some(gate),
88        access: Access::Table,
89        kind: Kind::Items {
90            item_len,
91            reported_len,
92        },
93        max_len: item_len as usize,
94    }
95}

232/// The descriptor for `key`, if the model covers it.
233pub fn descriptor(key: u32) -> Option<&'static Descriptor> {
234    COVERED.iter().find(|descriptor| descriptor.key == key)
235}

237/// A value read through its codec.
238#[derive(Clone, Copy, Debug, PartialEq, Eq)]
239pub enum Value<'a> {
240    Bool(bool),
241    U8(u8),
242    U16(u16),
243    U32(u32),
244    I8(i8),
245    I16(i16),
246    I32(i32),
247    Text(&'a str),
248    Key32(&'a [u8; 32]),
249    Bytes(&'a [u8]),
250}
251
252/// Read `raw` as `codec` says. `None` when the octets are not that
253/// codec's: a value of the wrong width, or text that is not UTF-8.
254pub fn decode(codec: PropertyType, raw: &[u8]) -> Option<Value<'_>> {
255    Some(match codec {
256        PropertyType::Bool => match raw {
257            [0] => Value::Bool(false),
258            [1] => Value::Bool(true),
259            _ => return None,
260        },
261        PropertyType::U8 => Value::U8(u8::from_le_bytes(raw.try_into().ok()?)),
262        PropertyType::U16 => Value::U16(u16::from_le_bytes(raw.try_into().ok()?)),
263        PropertyType::U32 => Value::U32(u32::from_le_bytes(raw.try_into().ok()?)),
264        PropertyType::I8 => Value::I8(i8::from_le_bytes(raw.try_into().ok()?)),
265        PropertyType::I16 => Value::I16(i16::from_le_bytes(raw.try_into().ok()?)),
266        PropertyType::I32 => Value::I32(i32::from_le_bytes(raw.try_into().ok()?)),
267        PropertyType::Text => Value::Text(core::str::from_utf8(raw).ok()?),
268        PropertyType::Key32 => Value::Key32(raw.try_into().ok()?),
269        PropertyType::Status | PropertyType::BatteryDiagnostic | PropertyType::Bytes => {
270            Value::Bytes(raw)
271        }
272    })
273}

275#[cfg(test)]
276mod tests {
277    use super::*;
278
279    /// The codecs are umsh-ulcp's, not this crate's opinion of them.
280    #[test]
281    fn every_codec_is_the_one_umsh_ulcp_describes() {
282        for descriptor in COVERED {
283            assert_eq!(
284                umsh_ulcp::property_type(descriptor.key),
285                Some(descriptor.codec),
286                "{:?}",
287                umsh_ulcp::property_name(descriptor.key)
288            );
289        }
290    }
291
292    #[test]
293    fn each_property_is_covered_once() {
294        for (index, descriptor) in COVERED.iter().enumerate() {
295            assert!(
296                COVERED[index + 1..]
297                    .iter()
298                    .all(|other| other.key != descriptor.key)
299            );
300        }
301    }
302
303    /// The administrator's binding refuses some writes outright, and
304    /// the device's own interface has the same surface; nothing the model
305    /// writes may be one of them.
306    #[test]
307    fn everything_the_model_writes_is_the_device_interfaces_to_write() {
308        for descriptor in COVERED {
309            if descriptor.writable() || descriptor.is_table() {
310                assert!(umsh_ulcp::ids::device_writable(descriptor.key));
311            }
312        }
313    }
314
315    #[test]
316    fn values_decode_through_their_codec() {
317        assert_eq!(
318            decode(PropertyType::U32, &910_525u32.to_le_bytes()),
319            Some(Value::U32(910_525))
320        );
321        assert_eq!(decode(PropertyType::I8, &[0xF7]), Some(Value::I8(-9)));
322        assert_eq!(decode(PropertyType::Bool, &[2]), None);
323        assert_eq!(decode(PropertyType::U16, &[1]), None);
324        assert_eq!(
325            decode(PropertyType::Text, b"Pocket"),
326            Some(Value::Text("Pocket"))
327        );
328        assert_eq!(decode(PropertyType::Text, &[0xFF]), None);
329    }
