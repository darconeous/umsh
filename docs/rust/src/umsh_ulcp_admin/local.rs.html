1//! The device's own session, reached in process.
2//!
3//! A local binding answers every request that carries a non-zero TID
4//! exactly once, so a [`LocalExchange`] holds a request until its answer
5//! comes and never gives up on one: a missing answer would be a bug in the
6//! device, not a condition to recover from. What the device publishes
7//! arrives on the same port, in order with the answers.
8//!
9//! The exchange is not an attach. It resets no session state and
10//! displaces no host, and it reaches the device domain only.
11
12use umsh_ulcp::Frame;
13use umsh_ulcp::frame::{TID_MAX, TID_UNSOLICITED};
14
15use crate::exchange::{Arrival, Busy, Exchange, Request, Ticket};
16
17/// Largest frame a local binding carries: the most a session emits in one
18/// frame, and what the firmware driver's frame buffers hold.
19pub const FRAME_MAX: usize = 300;
20
21/// The device's port into its own session: where requests go in and the
22/// session's frames for the interface come out.
23pub trait LocalPort {
24    /// Hand one request frame to the device.
25    fn send(&mut self, frame: &[u8]) -> Result<(), PortBusy>;
26
27    /// The next frame the device emitted for its interface, copied into
28    /// `out`.
29    fn receive(&mut self, out: &mut [u8]) -> Option<PortFrame>;
30}
31
32/// The device cannot take a request right now.
33#[derive(Clone, Copy, Debug, PartialEq, Eq)]
34pub struct PortBusy;
35
36/// What a [`LocalPort`] hands back.
37#[derive(Clone, Copy, Debug, PartialEq, Eq)]
38pub enum PortFrame {
39    /// A frame of this many octets: an answer, or a publication.
40    Frame(usize),
41    /// The port dropped publications because the interface fell behind.
42    Overrun,
43}
44
45/// An [`Exchange`] over the device's own session.
46///
47/// Requests are told apart by TID, so at most seven are outstanding at
48/// once; an eighth waits for an answer to free one. TIDs are handed out in
49/// rotation, so one is reused as late as possible.
50pub struct LocalExchange<P> {
51    port: P,
52    /// The ticket outstanding under each TID. TID 0 is never used.
53    outstanding: [Option<Ticket>; TID_MAX as usize + 1],
54    next_tid: u8,
55    next_ticket: u32,
56}
57
58impl<P: LocalPort> LocalExchange<P> {
59    pub const fn new(port: P) -> Self {
60        Self {
61            port,
62            outstanding: [None; TID_MAX as usize + 1],
63            next_tid: 1,
64            next_ticket: 0,
65        }
66    }
67
68    pub fn port(&self) -> &P {
69        &self.port
70    }
71
72    pub fn port_mut(&mut self) -> &mut P {
73        &mut self.port
74    }
75
76    /// Requests handed to the device and not yet answered.
77    pub fn outstanding(&self) -> usize {
78        self.outstanding.iter().flatten().count()
79    }
80
81    fn free_tid(&self) -> Option<u8> {
82        (0..TID_MAX)
83            .map(|offset| (self.next_tid - 1 + offset) % TID_MAX + 1)
84            .find(|&tid| self.outstanding[usize::from(tid)].is_none())
85    }
86}
87
88impl<P: LocalPort> Exchange for LocalExchange<P> {
89    const ANSWER_MAX: usize = FRAME_MAX;
90
91    fn submit(&mut self, request: &Request<'_>) -> Result<Ticket, Busy> {
92        let tid = self.free_tid().ok_or(Busy::Full)?;
93        let mut frame = [0u8; FRAME_MAX];
94        let len = request
95            .encode(tid, &mut frame)
96            .map_err(|_| Busy::TooLarge)?;
97        self.port
98            .send(&frame[..len])
99            .map_err(|PortBusy| Busy::Full)?;
100        let ticket = Ticket(self.next_ticket);
101        self.next_ticket = self.next_ticket.wrapping_add(1);
102        self.outstanding[usize::from(tid)] = Some(ticket);
103        self.next_tid = tid % TID_MAX + 1;
104        Ok(ticket)
105    }
106
107    fn receive<'b>(&mut self, out: &'b mut [u8]) -> Option<Arrival<'b>> {
108        loop {
109            let len = match self.port.receive(out)? {
110                PortFrame::Overrun => return Some(Arrival::Overrun),
111                PortFrame::Frame(len) => len,
112            };
113            let Ok(parsed) = Frame::parse(&out[..len]) else {
114                debug_assert!(false, "the session emitted an unparseable frame");
115                continue;
116            };
117            let tid = parsed.header.tid();
118            if tid == TID_UNSOLICITED {
119                return Some(Arrival::Publication { frame: &out[..len] });
120            }
121            match self.outstanding[usize::from(tid)].take() {
122                Some(ticket) => {
123                    return Some(Arrival::Answer {
124                        ticket,
125                        frame: &out[..len],
126                    });
127                }
128                None => debug_assert!(false, "an answer under TID {tid}, which nothing asked"),
129            }
130        }
131    }
132}

134/// The port into the firmware ULCP driver: requests go in as
135/// [`InEvent::Device`] on the driver's input channel, and the driver
136/// answers on the board's [`DeviceLink`].
137///
138/// The port never waits. A full input channel is [`PortBusy`], and its
139/// owner awaits [`DriverPort::ready`] to learn when the link has
140/// something, which is also when a slot may have come free.
141///
142/// [`InEvent::Device`]: umsh_ulcp_runtime::driver::InEvent::Device
143/// [`DeviceLink`]: umsh_ulcp_runtime::driver::DeviceLink
144#[cfg(any(feature = "driver", test))]
145pub mod driver {
146    use embassy_sync::blocking_mutex::raw::RawMutex;
147    use umsh_ulcp_runtime::driver::{DeviceFrame, DeviceLink, FrameBuf, InEvent, InputChannel};
148
149    use super::{LocalPort, PortBusy, PortFrame};
150
151    pub struct DriverPort<M: RawMutex + 'static> {
152        input: &'static InputChannel<M>,
153        link: &'static DeviceLink<M>,
154    }
155
156    impl<M: RawMutex + 'static> DriverPort<M> {
157        pub const fn new(input: &'static InputChannel<M>, link: &'static DeviceLink<M>) -> Self {
158            Self { input, link }
159        }
160
161        /// Wait until the link has something to receive.
162        pub async fn ready(&self) {
163            self.link.ready().await
164        }
165    }
166
167    impl<M: RawMutex + 'static> LocalPort for DriverPort<M> {
168        fn send(&mut self, frame: &[u8]) -> Result<(), PortBusy> {
169            let frame = FrameBuf::from_slice(frame).map_err(|()| PortBusy)?;
170            self.input
171                .try_send(InEvent::Device(frame))
172                .map_err(|_| PortBusy)
173        }
174
175        fn receive(&mut self, out: &mut [u8]) -> Option<PortFrame> {
176            match self.link.try_receive()? {
177                DeviceFrame::Frame(frame) => {
178                    debug_assert!(frame.len() <= out.len(), "the receive buffer is too small");
179                    let len = frame.len().min(out.len());
180                    out[..len].copy_from_slice(&frame[..len]);
181                    Some(PortFrame::Frame(len))
182                }
183                DeviceFrame::Overrun => Some(PortFrame::Overrun),
184            }
185        }
186    }
187}

189#[cfg(test)]
190mod tests {
191    use super::*;

193    /// A port that takes everything and answers whatever the test queues.
194    #[derive(Default)]
195    struct Script {
196        sent: std::vec::Vec<std::vec::Vec<u8>>,
197        replies: std::collections::VecDeque<Option<std::vec::Vec<u8>>>,
198    }
199
200    impl LocalPort for Script {
201        fn send(&mut self, frame: &[u8]) -> Result<(), PortBusy> {
202            self.sent.push(frame.to_vec());
203            Ok(())
204        }
205
206        fn receive(&mut self, out: &mut [u8]) -> Option<PortFrame> {
207            match self.replies.pop_front()? {
208                Some(frame) => {
209                    out[..frame.len()].copy_from_slice(&frame);
210                    Some(PortFrame::Frame(frame.len()))
211                }
212                None => Some(PortFrame::Overrun),
213            }
214        }
215    }
216
217    fn tid_of(frame: &[u8]) -> u8 {
218        Frame::parse(frame).unwrap().header.tid()
219    }

221    #[test]
222    fn the_frame_ceiling_is_what_the_session_and_driver_carry() {
223        assert_eq!(FRAME_MAX, umsh_ulcp_device::MULTI_MAX);
224        assert_eq!(FRAME_MAX, umsh_ulcp_runtime::driver::FRAME_OUT_MAX);
225        assert_eq!(FRAME_MAX, umsh_ulcp_runtime::driver::FRAME_IN_MAX);
226    }

228    /// Seven TIDs, never zero, handed out in rotation; an eighth request
229    /// waits for an answer.
230    #[test]
231    fn requests_take_the_seven_tids_in_rotation() {
232        let mut exchange = LocalExchange::new(Script::default());
233        let tickets: std::vec::Vec<_> = (0..7)
234            .map(|_| exchange.submit(&Request::Get(5)).unwrap())
235            .collect();
236        assert_eq!(exchange.submit(&Request::Get(5)), Err(Busy::Full));
237        let tids: std::vec::Vec<_> = exchange.port().sent.iter().map(|f| tid_of(f)).collect();
238        assert_eq!(tids, [1, 2, 3, 4, 5, 6, 7]);
239
240        // TID 3 is answered and comes free, but the rotation goes on from
241        // where it was before reusing it.
242        let mut answer = [0u8; 16];
243        let len = umsh_ulcp::frame::prop_is(&mut answer, 3, 5, &[]).unwrap();
244        exchange
245            .port_mut()
246            .replies
247            .push_back(Some(answer[..len].to_vec()));
248        let mut out = [0u8; FRAME_MAX];
249        assert_eq!(
250            exchange.receive(&mut out),
251            Some(Arrival::Answer {
252                ticket: tickets[2],
253                frame: &answer[..len]
254            })
255        );
256        exchange.submit(&Request::Get(5)).unwrap();
257        assert_eq!(tid_of(exchange.port().sent.last().unwrap()), 3);
258        assert_eq!(exchange.outstanding(), 7);
259    }

261    /// The driver's side of the link—answers on it, publications and an
262    /// overrun through it—is the runtime's to test. What is left is that
263    /// the port hands requests to the driver's input and never waits.
264    #[test]
265    fn the_driver_port_hands_requests_in_and_never_waits() {
266        use embassy_sync::blocking_mutex::raw::CriticalSectionRawMutex;
267        use umsh_ulcp_runtime::driver::{DeviceLink, InEvent, InputChannel};
268
269        static INPUT: InputChannel<CriticalSectionRawMutex> = InputChannel::new();
270        static LINK: DeviceLink<CriticalSectionRawMutex> = DeviceLink::new();
271        let mut exchange = LocalExchange::new(driver::DriverPort::new(&INPUT, &LINK));
272
273        exchange.submit(&Request::Get(prop_key())).unwrap();
274        let Ok(InEvent::Device(frame)) = INPUT.try_receive() else {
275            panic!("the request reached the driver as a device request");
276        };
277        assert_eq!(tid_of(&frame), 1);
278
279        // The board's other tasks fill the input. A full input is a link
280        // busy for now: the request is not taken, and holds no TID.
281        while INPUT.try_send(InEvent::Device(frame.clone())).is_ok() {}
282        assert_eq!(exchange.submit(&Request::Get(prop_key())), Err(Busy::Full));
283        assert_eq!(exchange.outstanding(), 1);
284        INPUT.clear();
285        exchange.submit(&Request::Get(prop_key())).unwrap();
286        assert_eq!(exchange.outstanding(), 2);
287
288        let mut out = [0u8; FRAME_MAX];
289        assert_eq!(exchange.receive(&mut out), None);
290    }
291
292    fn prop_key() -> u32 {
293        umsh_ulcp::ids::prop::PHY_TX_POWER
294    }

296    #[test]
297    fn publications_and_overruns_pass_through_in_order() {
298        let mut exchange = LocalExchange::new(Script::default());
299        let mut publication = [0u8; 16];
300        let len = umsh_ulcp::frame::prop_is(&mut publication, 0, 77, &[1]).unwrap();
301        let port = exchange.port_mut();
302        port.replies.push_back(Some(publication[..len].to_vec()));
303        port.replies.push_back(None);
304        let mut out = [0u8; FRAME_MAX];
305        assert_eq!(
306            exchange.receive(&mut out),
307            Some(Arrival::Publication {
308                frame: &publication[..len]
309            })
310        );
311        assert_eq!(exchange.receive(&mut out), Some(Arrival::Overrun));
312        assert_eq!(exchange.receive(&mut out), None);
313    }
