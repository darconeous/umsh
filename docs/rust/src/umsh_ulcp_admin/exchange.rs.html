1//! What a link to a device does: take a request, hand back its answer,
2//! and pass on whatever the device publishes.
3//!
4//! The trait is synchronous and never waits. A link that is waiting on
5//! something—a slot for the next request, a frame from the device—says
6//! so by returning nothing, and its owner waits on whatever the link is
7//! built from. That keeps the trait free of any executor, and lets one
8//! model serve a device in process and a device across the mesh.
9
10use umsh_ulcp::frame::{self, WriteError};
11
12/// One request for a device.
13#[derive(Clone, Copy, Debug, PartialEq, Eq)]
14pub enum Request<'a> {
15    /// `CMD_PROP_GET`.
16    Get(u32),
17    /// `CMD_PROP_MULTI_GET`: several properties, answered in request
18    /// order.
19    MultiGet(&'a [u32]),
20    /// `CMD_PROP_SET`.
21    Set { key: u32, value: &'a [u8] },
22    /// `CMD_PROP_MULTI_SET`: writes applied in order, stopping at the
23    /// first that fails.
24    MultiSet(&'a [(u32, &'a [u8])]),
25    /// `CMD_PROP_INSERT`: one item, in the property's item form.
26    Insert { key: u32, item: &'a [u8] },
27    /// `CMD_PROP_REMOVE`: one item selector.
28    Remove { key: u32, selector: &'a [u8] },
29    /// `CMD_SAVE`.
30    Save,
31}
32
33impl Request<'_> {
34    /// Encode as one ULCP frame under `tid`, returning its length.
35    pub fn encode(&self, tid: u8, out: &mut [u8]) -> Result<usize, WriteError> {
36        match *self {
37            Self::Get(key) => frame::prop_get(out, tid, key),
38            Self::MultiGet(keys) => frame::prop_multi_get(out, tid, keys),
39            Self::Set { key, value } => frame::prop_set(out, tid, key, value),
40            Self::MultiSet(entries) => frame::prop_multi_set(out, tid, entries),
41            Self::Insert { key, item } => frame::prop_insert(out, tid, key, item),
42            Self::Remove { key, selector } => frame::prop_remove(out, tid, key, selector),
43            Self::Save => frame::save(out, tid),
44        }
45    }
46}
47
48/// A link's name for a request it took, by which it reports the answer.
49#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
50pub struct Ticket(pub u32);
51
52/// Why a link did not take a request.
53#[derive(Clone, Copy, Debug, PartialEq, Eq)]
54pub enum Busy {
55    /// Not now: every slot is in use, or the device's queue is full. The
56    /// request is worth offering again once something arrives.
57    Full,
58    /// Never: the request does not fit the link's frame.
59    TooLarge,
60}
61
62/// What came back from the device.
63#[derive(Clone, Copy, Debug, PartialEq, Eq)]
64pub enum Arrival<'b> {
65    /// The answer to `ticket`: one ULCP frame, as the device emitted it.
66    Answer { ticket: Ticket, frame: &'b [u8] },
67    /// The link gave up on `ticket`, so whether a write it carried took
68    /// effect is unknown. A local link never says this.
69    Lost { ticket: Ticket },
70    /// A frame the device published unasked.
71    Publication { frame: &'b [u8] },
72    /// The link dropped publications, so anything learned from them may
73    /// be out of date.
74    Overrun,
75}
76
77/// A link to one device.
78pub trait Exchange {
79    /// The largest answer this link hands back, which is the buffer
80    /// [`Exchange::receive`] needs.
81    const ANSWER_MAX: usize;
82
83    /// Hand a request to the link.
84    fn submit(&mut self, request: &Request<'_>) -> Result<Ticket, Busy>;
85
86    /// The next thing to arrive, written into `out`.
87    fn receive<'b>(&mut self, out: &'b mut [u8]) -> Option<Arrival<'b>>;
88}
