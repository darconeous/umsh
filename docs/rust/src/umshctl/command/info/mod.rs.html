1//! `info`: what the device says about itself, in as few exchanges as it
2//! can be asked in.
3//!
4//! The report is a list of topics ([`topics::TOPICS`]), each a set of
5//! properties and two renderings of them. Everything a run needs is
6//! fetched in one `CMD_PROP_MULTI_GET` — plus the two that cannot share
7//! one — so a whole report costs about what a single property used to.
8//!
9//! Naming a topic asks for that topic alone, which over the mesh is the
10//! difference between a question and an errand.
11
12pub mod props;
13pub mod topics;

28#[derive(Debug, clap::Args)]
29pub struct InfoArgs {
30    /// Report host ownership relative to this host identity public key.
31    #[arg(long, value_name = "KEY")]
32    pub expect_host_key: Option<KeyArg>,
33
34    /// Print `NAME=VALUE` lines a shell can `eval`, rather than prose.
35    #[arg(long)]
36    pub env: bool,
37
38    /// Report one subject rather than all of them.
39    ///
40    /// With no topic every subject the device supports is reported, and
41    /// each one named here reports that subject alone.
42    #[arg(value_name = "TOPIC", value_parser = topic_parser())]
43    pub topic: Option<String>,
44}
45
46/// The topic names, as a value parser — so clap rejects a misspelling,
47/// `--help` lists what there is, and the shell completes them, all from
48/// the one table that defines them.
49fn topic_parser() -> clap::builder::PossibleValuesParser {
50    clap::builder::PossibleValuesParser::new(topics::TOPICS.iter().map(|topic| topic.name))
51}

53pub async fn run<L: FrameLink>(device: &mut UlcpDevice<L>, args: InfoArgs) -> Result<()> {
54    // Two properties cannot travel with the rest and must come first.
55    //
56    // `PROP_LAST_STATUS` reports a refused position by putting itself
57    // into it, so its value and a refusal are the same bytes; asking for
58    // it alone is what tells them apart. `PROP_CAPS` decides which
59    // topics exist and therefore what the batch contains, so it cannot
60    // be in that batch.
61    let mut set = PropSet::new();
62    set.insert(
63        prop::LAST_STATUS,
64        Ok(device.get_prop(prop::LAST_STATUS).await?),
65    );
66    let raw_caps = device.get_prop(prop::CAPS).await?;
67    let caps = umsh::ulcp::decode_capabilities(&raw_caps)?;
68    set.insert(prop::CAPS, Ok(raw_caps));
69
70    let ctx = Context {
71        caps,
72        remote: device.is_remote(),
73        expect_host_key: args.expect_host_key.map(|key| key.0),
74        dev_version: device.dev_version().to_owned(),
75        // A device that answers the optional model property with an
76        // empty string has not named its hardware any more than one that
77        // refuses the read has.
78        dev_model: device
79            .dev_model()
80            .filter(|model| !model.is_empty())
81            .map(str::to_owned),
82    };
83
84    let selected: Vec<&Topic> = match &args.topic {
85        // clap has already checked the name against the same table.
86        Some(name) => {
87            let topic = topics::topic(name).expect("clap accepted an unknown topic");
88            if !(topic.gate)(&ctx) {
89                bail!(
90                    "this device has nothing to report under {name:?} — see `info` for what it \
91                     does report"
92                );
93            }
94            vec![topic]
95        }
96        None => topics::TOPICS
97            .iter()
98            .filter(|topic| (topic.gate)(&ctx))
99            .collect(),
100    };
101
102    // Every property every selected topic wants, asked for at once.
103    let mut keys: Vec<u32> = Vec::new();
104    for topic in &selected {
105        for key in (topic.keys)(&ctx) {
106            if !keys.contains(&key) {
107                keys.push(key);
108            }
109        }
110    }
111    let fetched = props::fetch(device, &keys, props::batched(&ctx.caps)).await?;
112    for &key in &keys {
113        if let Some(value) = fetched.bytes(key) {
114            set.insert(key, Ok(value.to_vec()));
115        } else if let Some(status) = fetched.refusal(key) {
116            set.insert(key, Err(status));
117        }
118    }
119
120    if args.env {
121        for topic in &selected {
122            for (name, value) in (topic.env)(&set, &ctx) {
123                println!("{}_{name}={}", topic.prefix, topics::shell_quote(&value));
124            }
125        }
126        return Ok(());
127    }
128
129    // One topic on its own reads as a report about that subject, so its
130    // lines stand at the left margin. A whole report needs the subject
131    // named, and its lines indented under it.
132    let single = selected.len() == 1;
133    for topic in &selected {
134        let lines = (topic.render)(&set, &ctx);
135        if single {
136            for (label, value) in lines {
137                field(&label, value);
138            }
139        } else {
140            println!("{}:", topic.name);
141            for (label, value) in lines {
142                subfield(&label, value);
143            }
144        }
145        for (key, status) in topics::refusals(&set, &(topic.keys)(&ctx)) {
146            let label = property_name(key).map_or_else(|| format!("prop {key}"), str::to_owned);
147            let refused = format!("refused: {status:?}");
148            if single {
149                field(&label, refused);
150            } else {
151                subfield(&label, refused);
152            }
153        }
154    }
155    Ok(())
156}

158pub fn battery_display(status: &BatteryStatus) -> String {
159    if status.is_empty() {
160        return "unsupported reporting".to_string();
161    }
162    let voltage = status
163        .voltage_mv
164        .map_or("voltage unsupported".to_string(), |mv| format!("{mv} mV"));
165    let level = status
166        .level_percent
167        .map_or("level unsupported".to_string(), |percent| {
168            format!("{percent}%")
169        });
170    let state = match status.charge_state {
171        Some(BatteryChargeState::Discharging) => "discharging",
172        Some(BatteryChargeState::Charging) => "charging",
173        Some(BatteryChargeState::Charged) => "charged",
174        None => "charge state unsupported",
175    };
176    format!("{voltage}, {level}, {state}")
177}

179/// `illuminance`: one ambient light reading, on its own.
180///
181/// Separate from `info` because calibrating the sensor against a
182/// reference meter means taking readings in a tight loop, and a full
183/// report per data point is unusable for that.
184pub async fn illuminance<L: FrameLink>(device: &mut UlcpDevice<L>) -> Result<()> {
185    if !device.capabilities().await?.contains(&cap::ILLUMINANCE) {
186        field("illuminance", "unsupported (no CAP_ILLUMINANCE)");
187        return Ok(());
188    }
189    match device.illuminance().await? {
190        Some(millilux) => field(
191            "illuminance",
192            format!("{} ({millilux} mlux)", format_millilux(millilux)),
193        ),
194        None => field("illuminance", "no reading".to_string()),
195    }
196    Ok(())
197}
198
199/// Millilux as lux to three decimal places. Fixed rather than adaptive
200/// because the readings this is used to calibrate span moonlight to
201/// daylight, and a column that keeps its shape is easier to compare
202/// against a meter than one that keeps rescaling.
203pub fn format_millilux(millilux: u32) -> String {
204    format!("{}.{:03} lux", millilux / 1000, millilux % 1000)
205}

223    fn props_of(entries: &[(u32, &[u8])]) -> PropSet {
224        let mut set = PropSet::new();
225        for (key, value) in entries {
226            set.insert(*key, Ok(value.to_vec()));
227        }
228        set
229    }

231    fn value(lines: &[(String, String)], label: &str) -> Option<String> {
232        lines
233            .iter()
234            .find(|(name, _)| name == label)
235            .map(|(_, value)| value.clone())
236    }
