1//! `dev-channel` / `dev-peer` / `dev-admin`: the device identity's own
2//! key tables.
3//!
4//! The digest form differs by table (a 2-byte channel id versus the
5//! 32-byte peer key itself), so listings and mutation reports print
6//! whatever digest the device quotes.
7
8use anyhow::{Result, bail};
9
10use umsh::ulcp_wire::ids::prop;
11
12use super::persist;
13use super::values::KeyArg;
14use crate::App;
15use crate::output::hex;
16
17#[derive(Debug, clap::Subcommand)]
18pub enum TableOp {
19    /// List the entries the device holds, in digest form.
20    List,
21    /// Add an entry.
22    Add {
23        #[arg(value_name = "KEY")]
24        key: KeyArg,
25    },
26    /// Remove an entry.
27    Remove {
28        #[arg(value_name = "KEY")]
29        key: KeyArg,
30    },
31}
32
33pub async fn run(app: &mut App, key: u32, noun: &str, op: Option<TableOp>) -> Result<()> {
34    let no_save = app.no_save;
35    let device = app.device()?;
36    match op.unwrap_or(TableOp::List) {
37        TableOp::List => {
38            let value = device.get_prop(key).await?;
39            let digest_len = if key == prop::DEV_CHANNEL_KEYS { 2 } else { 32 };
40            if value.is_empty() {
41                println!("no device {noun}s provisioned");
42            } else if !value.len().is_multiple_of(digest_len) {
43                bail!("malformed device {noun} listing");
44            } else {
45                for digest in value.chunks(digest_len) {
46                    println!("{}", hex(digest));
47                }
48            }
49            Ok(())
50        }
51        TableOp::Add { key: item } => {
52            let digest = device.insert_prop_item(key, &item.0).await?;
53            println!("device {noun} added (digest {})", hex(&digest));
54            persist(device, no_save).await
55        }
56        TableOp::Remove { key: item } => {
57            let digest = device.remove_prop_item(key, &item.0).await?;
58            println!("device {noun} removed (digest {})", hex(&digest));
59            persist(device, no_save).await
60        }
61    }
62}
