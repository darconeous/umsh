1//! Classic-pcap sink for captured frames.
2//!
3//! Radio frames and ULCP control frames land in one file using the
4//! repository's established synthetic Ethernet/IPv4/UDP encapsulation,
5//! so stock Wireshark opens a capture containing both layers. Raw LoRa
6//! bytes are also available, at the cost of a link type the user has to
7//! name.
8//!
9//! LoRaTap carries the radio frame under a header describing the
10//! channel it arrived on, so the RSSI and SNR the receiver reported
11//! survive into the capture instead of being dropped on the floor. It
12//! is the encapsulation the Wireshark extcap interface uses, because it
13//! invents none of the addressing the Ethernet form has to.
14
15use std::fs::File;
16use std::io::{BufWriter, Write};
17use std::path::Path;
18use std::time::{SystemTime, UNIX_EPOCH};

20use umsh::hal::RxInfo;

22/// Which layers a pcap file records.
23#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, clap::ValueEnum)]
24pub enum CaptureLayers {
25    /// Frames the radio heard over the air.
26    #[default]
27    Radio,
28    /// Control traffic between this host and the device.
29    #[value(alias = "companion")]
30    Ulcp,
31    Both,
32}
33
34impl CaptureLayers {
35    pub fn radio(self) -> bool {
36        matches!(self, Self::Radio | Self::Both)
37    }
38
39    pub fn ulcp(self) -> bool {
40        matches!(self, Self::Ulcp | Self::Both)
41    }
42}
43
44#[derive(Clone, Copy, Debug, PartialEq, Eq)]
45pub enum PcapDirection {
46    HostToDevice,
47    DeviceToHost,
48}
49
50const PCAP_LINKTYPE_ETHERNET: u32 = 1;
51const PCAP_LINKTYPE_LORATAP: u32 = 270;
52pub const PCAP_LINKTYPE_IPV4: u32 = 228;
53const RADIO_UDP_PORT: u16 = 15056;
54const ULCP_HOST_UDP_PORT: u16 = 15057;
55const ULCP_DEVICE_UDP_PORT: u16 = 15058;
56
57/// Fixed part of a LoRaTap v0 header, in bytes.
58const LORATAP_V0_LEN: u16 = 15;
59
60/// LoRaTap reports signal strength as an unsigned byte biased by this
61/// much, so -139 dBm is zero.
62const LORATAP_RSSI_BIAS: i32 = 139;
63
64#[derive(Clone, Copy, Debug)]
65pub enum PcapEncapsulation {
66    Ethernet,
67    RawLoRa {
68        linktype: u32,
69    },
70    LoRaTap,
71    /// Raw IPv4 (`LINKTYPE_IPV4`), for datagrams recorded with the
72    /// addresses they actually carried.
73    Ipv4,
74}
75
76/// The channel a capture is listening on, read back from the device.
77///
78/// LoRaTap describes every frame in terms of the channel it arrived on,
79/// so these travel with each record even though the radio holds them
80/// still for the whole capture.
81#[derive(Clone, Copy, Debug, Default)]
82pub struct RfParams {
83    pub freq_hz: u32,
84    pub bw_hz: u32,
85    pub sf: u8,
86    pub sync_word: u8,
87}

89pub struct PcapWriter {
90    output: BufWriter<Box<dyn Write>>,
91    layers: CaptureLayers,
92    encapsulation: PcapEncapsulation,
93    packet_id: u16,
94}
95
96impl PcapWriter {
97    pub fn create(
98        path: &Path,
99        layers: CaptureLayers,
100        encapsulation: PcapEncapsulation,
101    ) -> std::io::Result<Self> {
102        Self::to_writer(Box::new(File::create(path)?), layers, encapsulation)
103    }
104
105    /// Write the capture into an arbitrary sink.
106    ///
107    /// The extcap interface hands us the FIFO Wireshark is reading, which
108    /// is opened for writing rather than created.
109    pub fn to_writer(
110        sink: Box<dyn Write>,
111        layers: CaptureLayers,
112        encapsulation: PcapEncapsulation,
113    ) -> std::io::Result<Self> {
114        let mut output = BufWriter::new(sink);
115        output.write_all(&0xa1b2_c3d4u32.to_le_bytes())?;
116        output.write_all(&2u16.to_le_bytes())?;
117        output.write_all(&4u16.to_le_bytes())?;
118        output.write_all(&0i32.to_le_bytes())?;
119        output.write_all(&0u32.to_le_bytes())?;
120        output.write_all(&65_535u32.to_le_bytes())?;
121        let linktype = match encapsulation {
122            PcapEncapsulation::Ethernet => PCAP_LINKTYPE_ETHERNET,
123            PcapEncapsulation::RawLoRa { linktype } => linktype,
124            PcapEncapsulation::LoRaTap => PCAP_LINKTYPE_LORATAP,
125            PcapEncapsulation::Ipv4 => PCAP_LINKTYPE_IPV4,
126        };
127        output.write_all(&linktype.to_le_bytes())?;
128        output.flush()?;
129        Ok(Self {
130            output,
131            layers,
132            encapsulation,
133            packet_id: 0,
134        })
135    }

137    pub fn write_radio(&mut self, frame: &[u8]) -> std::io::Result<()> {
138        if self.layers.radio() {
139            match self.encapsulation {
140                PcapEncapsulation::Ethernet => self.write_udp(
141                    PcapDirection::DeviceToHost,
142                    RADIO_UDP_PORT,
143                    RADIO_UDP_PORT,
144                    frame,
145                )?,
146                PcapEncapsulation::RawLoRa { .. } => self.write_record(frame)?,
147                PcapEncapsulation::Ipv4 => {
148                    return Err(std::io::Error::other(
149                        "an IPv4 capture records datagrams, not radio frames",
150                    ));
151                }
152                // Without reception metadata the header would be all
153                // zeroes, which reads as a real -139 dBm measurement.
154                PcapEncapsulation::LoRaTap => {
155                    return Err(std::io::Error::other(
156                        "LoRaTap capture requires per-frame reception metadata",
157                    ));
158                }
159            }
160        }
161        Ok(())
162    }

164    /// Record a radio frame along with how the receiver heard it.
165    ///
166    /// Only LoRaTap has somewhere to put the metadata; the other
167    /// encapsulations quietly ignore it and record the frame alone.
168    pub fn write_radio_with_info(
169        &mut self,
170        rf: &RfParams,
171        info: &RxInfo,
172        frame: &[u8],
173    ) -> std::io::Result<()> {
174        if !self.layers.radio() {
175            return Ok(());
176        }
177        if !matches!(self.encapsulation, PcapEncapsulation::LoRaTap) {
178            return self.write_radio(frame);
179        }
180
181        let mut packet = Vec::with_capacity(usize::from(LORATAP_V0_LEN) + frame.len());
182        packet.push(0); // version
183        packet.push(0); // padding
184        packet.extend_from_slice(&LORATAP_V0_LEN.to_be_bytes());
185        packet.extend_from_slice(&rf.freq_hz.to_be_bytes());
186        packet.push(loratap_bandwidth(rf.bw_hz));
187        packet.push(rf.sf);
188        // A frame the device transmitted itself was never received, and
189        // LoRaTap v0 has no way to say so—its signal bytes are always
190        // readings. The rails (-139 dBm, -32 dB) are at least values no
191        // real link here produces, where the collapsed 0 dBm would chart
192        // as the strongest signal in the capture.
193        let (rssi, snr) = if info.origin.is_measured() {
194            (
195                loratap_rssi(i32::from(info.rssi)),
196                loratap_snr(info.snr.as_centibels()),
197            )
198        } else {
199            (0, loratap_snr(i16::MIN))
200        };
201        packet.push(rssi); // packet RSSI
202        packet.push(rssi); // max RSSI
203        packet.push(rssi); // current RSSI
204        packet.push(snr);
205        packet.push(rf.sync_word);
206        packet.extend_from_slice(frame);
207
208        self.write_record(&packet)
209    }

211    pub fn write_ulcp(&mut self, direction: PcapDirection, frame: &[u8]) -> std::io::Result<()> {
212        if !self.layers.ulcp() {
213            return Ok(());
214        }
215        debug_assert!(matches!(self.encapsulation, PcapEncapsulation::Ethernet));
216        let (src_port, dst_port) = match direction {
217            PcapDirection::HostToDevice => (ULCP_HOST_UDP_PORT, ULCP_DEVICE_UDP_PORT),
218            PcapDirection::DeviceToHost => (ULCP_DEVICE_UDP_PORT, ULCP_HOST_UDP_PORT),
219        };
220        self.write_udp(direction, src_port, dst_port, frame)
221    }
222
223    /// Record a UDP datagram as raw IPv4, from `source` to `destination`.
224    pub fn write_datagram(
225        &mut self,
226        source: std::net::SocketAddrV4,
227        destination: std::net::SocketAddrV4,
228        payload: &[u8],
229    ) -> std::io::Result<()> {
230        debug_assert!(matches!(self.encapsulation, PcapEncapsulation::Ipv4));
231        let mut packet = Vec::new();
232        self.push_ipv4_udp(
233            &mut packet,
234            (source.ip().octets(), source.port()),
235            (destination.ip().octets(), destination.port()),
236            payload,
237        )?;
238        self.write_record(&packet)
239    }
240
241    fn write_udp(
242        &mut self,
243        direction: PcapDirection,
244        src_port: u16,
245        dst_port: u16,
246        payload: &[u8],
247    ) -> std::io::Result<()> {
248        let mut packet = Vec::new();
249
250        // Synthetic Ethernet and loopback IPv4 endpoints. Direction remains
251        // visible in both endpoint addresses and ULCP UDP ports.
252        packet.extend_from_slice(&[0x02, 0, 0, 0, 0, 2]);
253        packet.extend_from_slice(&[0x02, 0, 0, 0, 0, 1]);
254        packet.extend_from_slice(&0x0800u16.to_be_bytes());
255        let (src_ip, dst_ip) = match direction {
256            PcapDirection::HostToDevice => ([127, 0, 0, 1], [127, 0, 0, 2]),
257            PcapDirection::DeviceToHost => ([127, 0, 0, 2], [127, 0, 0, 1]),
258        };
259        self.push_ipv4_udp(&mut packet, (src_ip, src_port), (dst_ip, dst_port), payload)?;
260        self.write_record(&packet)
261    }
262
263    /// Append an IPv4 header, a UDP header, and `payload` to `packet`.
264    fn push_ipv4_udp(
265        &mut self,
266        packet: &mut Vec<u8>,
267        (src_ip, src_port): ([u8; 4], u16),
268        (dst_ip, dst_port): ([u8; 4], u16),
269        payload: &[u8],
270    ) -> std::io::Result<()> {
271        let udp_len = 8usize
272            .checked_add(payload.len())
273            .and_then(|len| u16::try_from(len).ok())
274            .ok_or_else(|| std::io::Error::other("capture payload exceeds IPv4 UDP size"))?;
275        let ip_len = 20u16
276            .checked_add(udp_len)
277            .ok_or_else(|| std::io::Error::other("capture packet exceeds IPv4 size"))?;
278        packet.reserve(usize::from(ip_len));
279        let ip_start = packet.len();
280        packet.extend_from_slice(&[
281            0x45,
282            0,
283            (ip_len >> 8) as u8,
284            ip_len as u8,
285            (self.packet_id >> 8) as u8,
286            self.packet_id as u8,
287            0,
288            0,
289            64,
290            17,
291            0,
292            0,
293            src_ip[0],
294            src_ip[1],
295            src_ip[2],
296            src_ip[3],
297            dst_ip[0],
298            dst_ip[1],
299            dst_ip[2],
300            dst_ip[3],
301        ]);
302        let checksum = ipv4_checksum(&packet[ip_start..ip_start + 20]);
303        packet[ip_start + 10..ip_start + 12].copy_from_slice(&checksum.to_be_bytes());
304        packet.extend_from_slice(&src_port.to_be_bytes());
305        packet.extend_from_slice(&dst_port.to_be_bytes());
306        packet.extend_from_slice(&udp_len.to_be_bytes());
307        packet.extend_from_slice(&0u16.to_be_bytes());
308        packet.extend_from_slice(payload);
309        self.packet_id = self.packet_id.wrapping_add(1);
310        Ok(())
311    }
312
313    fn write_record(&mut self, packet: &[u8]) -> std::io::Result<()> {
314        let timestamp = SystemTime::now()
315            .duration_since(UNIX_EPOCH)
316            .unwrap_or_default();
317        let seconds = u32::try_from(timestamp.as_secs()).unwrap_or(u32::MAX);
318        let captured_len = u32::try_from(packet.len())
319            .map_err(|_| std::io::Error::other("capture record exceeds pcap size"))?;
320        self.output.write_all(&seconds.to_le_bytes())?;
321        self.output
322            .write_all(&timestamp.subsec_micros().to_le_bytes())?;
323        self.output.write_all(&captured_len.to_le_bytes())?;
324        self.output.write_all(&captured_len.to_le_bytes())?;
325        self.output.write_all(packet)?;
326        // Keep the file usable by Wireshark during a long-running capture.
327        self.output.flush()
328    }

331/// Map a bandwidth to LoRaTap's enumerated byte.
332///
333/// The field is an enumeration of the three classic LoRa bandwidths
334/// rather than a scale factor, so it cannot express the 62.5 kHz UMSH
335/// normally runs at. Anything it cannot name is reported as zero, which
336/// Wireshark renders as "Unknown"—a narrower bandwidth silently
337/// labeled as one of the three would misdescribe the radio.
338fn loratap_bandwidth(bw_hz: u32) -> u8 {
339    match bw_hz {
340        125_000 => 1,
341        250_000 => 2,
342        500_000 => 4,
343        _ => 0,
344    }
345}
346
347/// Bias a dBm reading into LoRaTap's unsigned byte, saturating rather
348/// than wrapping so an implausible reading stays at the rail.
349fn loratap_rssi(dbm: i32) -> u8 {
350    (dbm + LORATAP_RSSI_BIAS).clamp(0, u8::MAX as i32) as u8
351}
352
353/// Convert centibels to LoRaTap's signed quarter-dB byte, rounding to
354/// nearest.
355fn loratap_snr(centibels: i16) -> u8 {
356    let scaled = i32::from(centibels) * 4;
357    let quarters = if scaled >= 0 {
358        (scaled + 5) / 10
359    } else {
360        (scaled - 5) / 10
361    };
362    quarters.clamp(i8::MIN as i32, i8::MAX as i32) as i8 as u8
363}
364
365fn ipv4_checksum(header: &[u8]) -> u16 {
366    let mut sum = 0u32;
367    for word in header.chunks_exact(2) {
368        sum += u32::from(u16::from_be_bytes([word[0], word[1]]));
369    }
370    while sum > 0xffff {
371        sum = (sum & 0xffff) + (sum >> 16);
372    }
373    !(sum as u16)
374}

471    /// The scalings here are the ones Wireshark's LoRaTap dissector
472    /// actually applies: RSSI is biased by 139 and SNR is in quarter-dB
473    /// steps, both verified against `tshark -V` output.
474    #[test]
475    fn loratap_records_channel_and_reception_metadata() {
476        let sink = SharedSink::default();
477        let bytes = sink.0.clone();
478        let mut writer = PcapWriter::to_writer(
479            Box::new(sink),
480            CaptureLayers::Radio,
481            PcapEncapsulation::LoRaTap,
482        )
483        .unwrap();
484        let frame = [0xc0, 0xa1, 0xb2, 0x03, 0x11, 0x22];
485        writer
486            .write_radio_with_info(&sample_rf(), &sample_info(frame.len(), -52, 95), &frame)
487            .unwrap();
488        drop(writer);
489
490        let bytes = bytes.borrow();
491        assert_eq!(
492            u32::from_le_bytes(bytes[20..24].try_into().unwrap()),
493            PCAP_LINKTYPE_LORATAP,
494        );
495        let packet = &bytes[40..];
496        assert_eq!(packet.len(), 15 + frame.len());
497        assert_eq!(packet[0], 0, "version");
498        assert_eq!(u16::from_be_bytes(packet[2..4].try_into().unwrap()), 15);
499        assert_eq!(
500            u32::from_be_bytes(packet[4..8].try_into().unwrap()),
501            910_525_000,
502        );
503        assert_eq!(packet[8], 1, "125 kHz in 125 kHz steps");
504        assert_eq!(packet[9], 7, "spreading factor");
505        assert_eq!(packet[10], 87, "-52 dBm biased by 139");
506        assert_eq!(packet[13], 38, "9.5 dB in quarter-dB steps");
507        assert_eq!(packet[14], 0x2b, "sync word");
508        assert_eq!(&packet[15..], &frame);
509    }
