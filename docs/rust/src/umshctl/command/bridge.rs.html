1//! Configure the device's own bridge client, using its existing identity.
2use super::{persist, values::KeyArg};
3use crate::{App, output::subfield};
4use anyhow::{Result, bail};
5use umsh::ulcp_wire::{
6    bridge::{DEFAULT_PORT, Link, valid_host},
7    ids::prop,
8};
9
10#[derive(Debug, clap::Subcommand)]
11pub enum BridgeOp {
12    Show,
13    On,
14    Off,
15    Configure {
16        host: String,
17        /// Bridge TCP port (the global --port selects the serial device).
18        #[arg(long = "server-port", default_value_t = DEFAULT_PORT, value_parser = clap::value_parser!(u16).range(1..))]
19        bridge_port: u16,
20        #[arg(long)]
21        server_key: KeyArg,
22    },
23}
24
25pub async fn run(app: &mut App, op: Option<BridgeOp>) -> Result<()> {
26    let no_save = app.no_save;
27    let device = app.device()?;
28    match op.unwrap_or(BridgeOp::Show) {
29        BridgeOp::Show => {
30            let [enabled, host, port, server_key, device_key, link, repeater] = device
31                .read_all(&[
32                    prop::BRIDGE_ENABLED,
33                    prop::BRIDGE_HOST,
34                    prop::BRIDGE_PORT,
35                    prop::BRIDGE_SERVER_KEY,
36                    prop::DEV_KEY,
37                    prop::BRIDGE_LINK,
38                    prop::MAC_REPEATER_ENABLED,
39                ])
40                .await?;
41            subfield(
42                "bridge",
43                if enabled == [1] {
44                    "enabled"
45                } else {
46                    "disabled"
47                },
48            );
49            subfield(
50                "host",
51                String::from_utf8_lossy(host.strip_suffix(&[0]).unwrap_or(&host)),
52            );
53            if let Ok(port) = <[u8; 2]>::try_from(port.as_slice()) {
54                subfield("port", u16::from_le_bytes(port).to_string());
55            }
56            for (name, bytes) in [
57                ("server identity", server_key),
58                ("device identity", device_key),
59            ] {
60                subfield(
61                    name,
62                    match <[u8; 32]>::try_from(bytes.as_slice()) {
63                        Ok(bytes) => umsh::core::PublicKey(bytes).to_string(),
64                        Err(_) => "unset".into(),
65                    },
66                );
67            }
68            if let Some(link) = Link::decode(&link) {
69                subfield(
70                    "connection",
71                    format!("{:?} ({:?})", link.state, link.reason),
72                );
73            }
74            subfield("node", if repeater == [1] { "repeater" } else { "leaf" });
75            return Ok(());
76        }
77        BridgeOp::On => {
78            device.set_prop(prop::BRIDGE_ENABLED, &[1]).await?;
79        }
80        BridgeOp::Off => {
81            device.set_prop(prop::BRIDGE_ENABLED, &[0]).await?;
82        }
83        BridgeOp::Configure {
84            host,
85            bridge_port,
86            server_key,
87        } => {
88            if host.is_empty() || !valid_host(&host) {
89                bail!("host must be a DNS hostname or IP literal without scheme or port");
90            }
91            let mut bytes = host.into_bytes();
92            bytes.push(0);
93            device.set_prop(prop::BRIDGE_HOST, &bytes).await?;
94            device
95                .set_prop(prop::BRIDGE_PORT, &bridge_port.to_le_bytes())
96                .await?;
97            device
98                .set_prop(prop::BRIDGE_SERVER_KEY, &server_key.0)
99                .await?;
100        }
101    }
102    persist(device, no_save).await
103}
