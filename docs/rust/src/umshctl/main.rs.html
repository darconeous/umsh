1//! `umshctl`: the host tool for ULCP radio devices—inspection,
2//! device identity, persistence, pairing, radio configuration, and
3//! packet capture.
4//!
5//! Every attach is administrative, with the non-resetting full-protocol
6//! handshake, so pointing this tool at an autonomously operating board
7//! never disturbs its configuration: only the command explicitly given
8//! changes anything.
9//!
10//! With no command it opens a shell—one attach, many commands—which
11//! is worth a great deal over BLE, where each fresh attach costs a
12//! discovery pass plus a handshake.
13

15    not(any(feature = "serial-radio", feature = "ble-radio")),
16    allow(unused_variables, dead_code)

40#[command(
41    name = "umshctl",
42    version,
43    about = "Inspect, configure, and capture from ULCP radio devices",
44    long_about = "\
45Manages a ULCP radio device without disturbing it: attaches with the
46non-resetting full-protocol handshake, so an autonomously operating
47board keeps its configuration unless the command changes it.
48
49With no command, opens an interactive shell against one attachment.
50With no connection, discovers a radio over BLE: one match is used, and
51several offer a numbered choice, which --pick asks for outright. A
52serial port is used only when named,
53because identifying one means opening it, and opening a port can reset
54or DFU-trigger hardware that is not a ULCP radio at all. --tcp reaches a
55radio whose port has been bridged to a socket, which carries the same
56framing a wire does; there is nothing to discover, so it is always named.
57
58A link that drops under a long-running command, a shell, or a mesh
59session is reopened behind the scenes and the session carries on; the
60reattach is narrated on stderr only when it takes more than a moment.
61--no-reconnect fails instead. A one-shot command whose link drops fails
62either way.
63
64KEY values are 44-character base58 or 64-character hex. Secrets are
65never echoed in output or traces—though shell history keeps whatever
66was typed, the same as any shell.
67
68CODE values are a short code of one to three letters or digits—an IATA
69airport code (SJC) or a country or state (US, WA)—a raw 2-byte code
70(0x7853), or any other text, which is hashed as a region name
71(\"Rogue Valley\"). Only an all-letter short code reads back as itself.
72Region codes are routing-domain tags, not RF band plans, so every
73repeater in an area must agree on the same spelling."
74)]

187/// Everything a command may need beyond the device itself: the
188/// attachment, the settings, and the difference between a shell and a
189/// one-shot invocation.
190pub struct App {
191    pub session: Option<Session>,
192    pub prefs: Prefs,
193    /// True in the shell, where a question can be asked and a failure
194    /// returns to a prompt.
195    pub interactive: bool,
196    pub trace: bool,
197    pub no_save: bool,
198    pub baud: u32,
199    /// How a bare `connect` resolves a scan—`--pick` at launch keeps
200    /// asking for the rest of the shell session.
201    pub discovery: Discovery,
202    /// The last `ble-scan` listing, so `connect <N>` can refer to it.
203    pub last_scan: Vec<Found>,
204    /// What a mesh session borrowed, while one is open. Present exactly
205    /// when `session` reaches its device over the air.
206    pub mesh: Option<mesh::MeshHome>,
207    /// How a lost link is brought back, or `None` to fail instead.
208    pub recovery: Option<Recovery>,
209}

211impl App {
212    pub fn session(&mut self) -> Result<&mut Session> {
213        match &mut self.session {
214            Some(session) => Ok(session),
215            None => bail!("not attached—try `ble-scan` or `connect`"),
216        }
217    }
218
219    pub fn device(&mut self) -> Result<&mut UlcpDevice<SessionLink>> {
220        Ok(&mut self.session()?.device)
221    }
222
223    /// Make sure the attached session's link is live before a command
224    /// uses it.
225    ///
226    /// Being attached is a logical state: a link that dropped between
227    /// two commands does not make the session any less the user's, so
228    /// it is brought back here, underneath, before the command runs. A
229    /// mesh session's link is the radio inside its driver, which keeps
230    /// itself; nothing to do from here. Not attached at all is not a
231    /// failure here either—the command says so in its own words.
232    pub async fn ensure_attached(&mut self) -> Result<()> {
233        let recovery = self.recovery.clone();
234        let Some(session) = &mut self.session else {
235            return Ok(());
236        };
237        if session.is_mesh() || !session.device.link_lost() {
238            return Ok(());
239        }
240        let Some(policy) = recovery else {
241            bail!(
242                "the link to {} was lost; `connect` reattaches (or drop --no-reconnect)",
243                session.label
244            );
245        };
246        match session.recover(&policy).await? {
247            Recovered::Attached => Ok(()),
248            Recovered::Interrupted => bail!("reconnect interrupted"),
249        }
250    }
251
252    /// Drop the attachment, returning what it was called. Dropping the
253    /// link is what reverts session-scoped device state.
254    ///
255    /// Ending a mesh session drops its device handle first, which is what
256    /// closes the link and lets the driver wind down and hand the
257    /// borrowed radio back—so the caller is left attached to the local
258    /// radio again rather than to nothing.
259    pub async fn detach(&mut self) -> Option<String> {
260        let session = self.session.take();
261        let label = session.as_ref().map(|session| session.label.clone());
262        if let Some(session) = session {
263            session.close().await;
264        }
265        if let Some(home) = self.mesh.take() {
266            mesh::restore_local(self, home).await;
267        }
268        label
269    }
270
271    pub fn rename(&mut self, label: String) {
272        if let Some(session) = &mut self.session {
273            session.label = label;
274        }
275    }

277    /// An attached, non-interactive app for the Wireshark extcap
278    /// interface.
279    ///
280    /// `interactive` is false in both its senses here: there is no
281    /// prompt to return to, and nobody to answer a question.
282    pub fn for_extcap(
283        session: Session,
284        prefs: Prefs,
285        baud: u32,
286        recovery: Option<Recovery>,
287    ) -> Self {
288        Self {
289            session: Some(session),
290            prefs,
291            interactive: false,
292            trace: false,
293            no_save: true,
294            baud,
295            discovery: Discovery::Auto,
296            last_scan: Vec::new(),
297            mesh: None,
298            recovery,
299        }
300    }
301
302    pub async fn attach(&mut self, target: Target) -> Result<()> {
303        let session = connection::connect(target, self.trace).await?;
304        announce_attached(&session);
305        self.session = Some(session);
306        Ok(())
307    }
308
309    pub fn prompt(&self) -> String {
310        match &self.session {
311            Some(session) => format!("{} ({})> ", session.label, session.target.transport()),
312            None => "(unattached)> ".to_string(),
313        }
314    }
315}

317fn announce_attached(session: &Session) {
318    eprintln!(
319        "attached: {} ({}) device={}{} boot_status={:?}",
320        session.label,
321        session.target.transport(),
322        session.device.dev_version(),
323        // `PROP_DEV_MODEL` is optional; say nothing rather than "unknown"
324        // when the device does not name its hardware—and a device that
325        // answers with an empty string has named it no better than one
326        // that refuses the read.
327        session
328            .device
329            .dev_model()
330            .filter(|model| !model.is_empty())
331            .map(|model| format!(" on {model}"))
332            .unwrap_or_default(),
333        session.device.boot_status(),
334    );
335}

337/// Work out which radio to talk to, from the flags, the environment,
338/// the saved default, and finally the air.
339async fn resolve(args: &ToolArgs, prefs: &Prefs) -> Result<Option<Target>> {
340    if args.pick
341        && (args.port.is_some()
342            || args.tcp.is_some()
343            || args.lan.is_some()
344            || matches!(args.ble, Some(Some(_))))
345    {
346        bail!(
347            "--pick chooses a radio from a listing; --port, --tcp, --lan, and --ble=SELECTOR \
348             already name one"
349        );
350    }
351    if let Some(group) = &args.lan {
352        if args.port.is_some() || args.ble.is_some() || args.tcp.is_some() {
353            bail!("--lan, --tcp, --port, and --ble name different radios; give one");
354        }
355        let group = group.as_deref().map(parse_group).transpose()?;
356        return Ok(Some(Target::Lan { group }));
357    }
358    if let Some(endpoint) = &args.tcp {
359        if args.port.is_some() || args.ble.is_some() {
360            bail!("--tcp, --port, and --ble name different radios; give one");
361        }
362        let (host, port) = connection::parse_endpoint(endpoint)?;
363        return Ok(Some(Target::Tcp { host, port }));
364    }
365    if let Some(port) = &args.port {
366        if args.ble.is_some() {
367            bail!("--port and --ble name different radios; give one");
368        }
369        return Ok(Some(Target::Serial {
370            port: port.clone(),
371            baud: args.baud,
372        }));
373    }
374    if let Some(Some(selector)) = &args.ble {
375        return Ok(Some(Target::Ble {
376            selector: selector.clone(),
377            name: None,
378        }));
379    }
380    // Bare `--ble`, or nothing at all: the tool finds the radio itself.
381    let target =
382        connection::discover(prefs, std::io::stdin().is_terminal(), args.discovery()).await?;
383    if let Some(target) = &target {
384        // A mutating one-shot must never act on a silently chosen radio.
385        eprintln!("discovered: {}", target.provisional_label());
386    }
387    Ok(target)
388}

396async fn run(args: ToolArgs) -> Result<()> {
397    output::set_color(args.color.enabled());
398    let interactive = args.command.is_none();
399    let mut app = App {
400        session: None,
401        prefs: Prefs::load(),
402        interactive,
403        trace: args.trace,
404        no_save: args.no_save,
405        baud: args.baud,
406        discovery: args.discovery(),
407        last_scan: Vec::new(),
408        mesh: None,
409        recovery: (!args.no_reconnect).then(Recovery::default),
410    };
411
412    // Everything clap's grammar cannot express is checked before a
413    // radio is opened.
414    if let Some(command) = &args.command {
415        command.validate()?;
416        // A command a mesh session cannot carry should say so now, not
417        // after a discovery pass and an attach handshake.
418        if args.node.is_some()
419            && let Some(refusal) = command.mesh_refusal()
420        {
421            bail!("{refusal}");
422        }
423    }
424    let needs_device = args
425        .command
426        .as_ref()
427        .is_none_or(|command| command.needs_device());
428    if needs_device {
429        match resolve(&args, &app.prefs).await? {
430            Some(target) => {
431                let session = connection::connect(target, app.trace).await?;
432                // The shell says what it is talking to before its first
433                // prompt; a one-shot's output is the command's own.
434                if interactive {
435                    announce_attached(&session);
436                }
437                app.session = Some(session);
438            }
439            // The shell can still scan and connect; a one-shot cannot.
440            None if interactive => {}
441            None => bail!("no ULCP radios found; name one with --port, --tcp, or --ble=SELECTOR"),
442        }
443    }
444
445    // `--node` borrows the radio just attached and opens a session to
446    // the named device across the mesh. With a command that is the
447    // one-shot form of the shell's `remote`: run it and hand the radio
448    // back, so a script gets the same session the shell would have. With
449    // no command it is simply where the shell starts—the prompt opens
450    // already talking to the far node.
451    //
452    // A one-shot spends nothing on the air before the command itself: a
453    // script asking for the battery should pay for the battery and
454    // nothing else. The shell greets the node by name, because a prompt
455    // that says which device it is answers a question every subsequent
456    // command would otherwise raise.
457    if let Some(node) = args.node {
458        let greeting = if interactive {
459            mesh::Greeting::Named
460        } else {
461            mesh::Greeting::Silent
462        };
463        mesh::open_remote(&mut app, umsh::core::PublicKey(node.0), greeting).await?;
464        let result = match args.command {
465            Some(command) => command.run(&mut app).await,
466            None => repl::run(&mut app).await,
467        };
468        app.detach().await;
469        return result;
470    }
471
472    match args.command {
473        Some(command) => command.run(&mut app).await,
474        None => repl::run(&mut app).await,
475    }
476}

478#[tokio::main(flavor = "current_thread")]
479async fn main() {
480    // Wireshark drives this binary through its own argument vocabulary,
481    // which is checked before the tool's parser so the two never have to
482    // agree on a shared grammar.
483    //
484    // Everything here is `!Send`—the node layer, the pcap tap, the
485    // whole session—so the one task a mesh session spawns to hold the
486    // borrowed radio needs a `LocalSet` to be spawned onto.
487    let local = tokio::task::LocalSet::new();
488    let result = local
489        .run_until(async {
490            if extcap::is_extcap_invocation() {
491                extcap::run().await
492            } else {
493                run(ToolArgs::parse()).await
494            }
495        })
496        .await;
497    if let Err(error) = result {
498        eprintln!("error: {error:#}");
499        std::process::exit(1);
500    }
501}
