1//! `umshctl`: the host tool for ULCP radio devices—inspection,
2//! device identity, persistence, pairing, radio configuration, and
3//! packet capture.
4//!
5//! Every attach is administrative, with the non-resetting full-protocol
6//! handshake, so pointing this tool at an autonomously operating board
7//! never disturbs its configuration: only the command explicitly given
8//! changes anything.
9//!
10//! With no command it opens a shell—one attach, many commands—which
11//! is worth a great deal over BLE, where each fresh attach costs a
12//! discovery pass plus a handshake.
13

15    not(any(feature = "serial-radio", feature = "ble-radio")),
16    allow(unused_variables, dead_code)
17)]

40#[command(
41    name = "umshctl",
42    version,
43    about = "Inspect, configure, and capture from ULCP radio devices",
44    long_about = "\
45Manages a ULCP radio device without disturbing it: attaches with the
46non-resetting full-protocol handshake, so an autonomously operating
47board keeps its configuration unless the command changes it.
48
49With no command, opens an interactive shell against one attachment.
50With no connection, discovers a radio over BLE: one match is used, and
51several offer a numbered choice, which --pick asks for outright. A
52serial port is used only when named,
53because identifying one means opening it, and opening a port can reset
54or DFU-trigger hardware that is not a ULCP radio at all. --tcp reaches a
55radio whose port has been bridged to a socket, which carries the same
56framing a wire does; there is nothing to discover, so it is always named.
57
58A link that drops under a long-running command, a shell, or a mesh
59session is reopened behind the scenes and the session carries on; the
60reattach is narrated on stderr only when it takes more than a moment.
61--no-reconnect fails instead. A one-shot command whose link drops fails
62either way.
63
64KEY values are 44-character base58 or 64-character hex. Secrets are
65never echoed in output or traces—though shell history keeps whatever
66was typed, the same as any shell.
67
68CODE values are a short code of one to three letters or digits—an IATA
69airport code (SJC) or a country or state (US, WA)—a raw 2-byte code
70(0x7853), or any other text, which is hashed as a region name
71(\"Rogue Valley\"). Only an all-letter short code reads back as itself.
72Region codes are routing-domain tags, not RF band plans, so every
73repeater in an area must agree on the same spelling."
74)]

186/// Everything a command may need beyond the device itself: the
187/// attachment, the settings, and the difference between a shell and a
188/// one-shot invocation.
189pub struct App {
190    pub session: Option<Session>,
191    pub prefs: Prefs,
192    /// True in the shell, where a question can be asked and a failure
193    /// returns to a prompt.
194    pub interactive: bool,
195    pub trace: bool,
196    pub no_save: bool,
197    pub baud: u32,
198    /// How a bare `connect` resolves a scan—`--pick` at launch keeps
199    /// asking for the rest of the shell session.
200    pub discovery: Discovery,
201    /// The last `ble-scan` listing, so `connect <N>` can refer to it.
202    pub last_scan: Vec<Found>,
203    /// What a mesh session borrowed, while one is open. Present exactly
204    /// when `session` reaches its device over the air.
205    pub mesh: Option<mesh::MeshHome>,
206    /// How a lost link is brought back, or `None` to fail instead.
207    pub recovery: Option<Recovery>,
208}

210impl App {
211    pub fn session(&mut self) -> Result<&mut Session> {
212        match &mut self.session {
213            Some(session) => Ok(session),
214            None => bail!("not attached—try `ble-scan` or `connect`"),
215        }
216    }
217
218    pub fn device(&mut self) -> Result<&mut UlcpDevice<SessionLink>> {
219        Ok(&mut self.session()?.device)
220    }
221
222    /// Make sure the attached session's link is live before a command
223    /// uses it.
224    ///
225    /// Being attached is a logical state: a link that dropped between
226    /// two commands does not make the session any less the user's, so
227    /// it is brought back here, underneath, before the command runs. A
228    /// mesh session's link is the radio inside its driver, which keeps
229    /// itself; nothing to do from here. Not attached at all is not a
230    /// failure here either—the command says so in its own words.
231    pub async fn ensure_attached(&mut self) -> Result<()> {
232        let recovery = self.recovery.clone();
233        let Some(session) = &mut self.session else {
234            return Ok(());
235        };
236        if session.is_mesh() || !session.device.link_lost() {
237            return Ok(());
238        }
239        let Some(policy) = recovery else {
240            bail!(
241                "the link to {} was lost; `connect` reattaches (or drop --no-reconnect)",
242                session.label
243            );
244        };
245        match session.recover(&policy).await? {
246            Recovered::Attached => Ok(()),
247            Recovered::Interrupted => bail!("reconnect interrupted"),
248        }
249    }
250
251    /// Drop the attachment, returning what it was called. Dropping the
252    /// link is what reverts session-scoped device state.
253    ///
254    /// Ending a mesh session drops its device handle first, which is what
255    /// closes the link and lets the driver wind down and hand the
256    /// borrowed radio back—so the caller is left attached to the local
257    /// radio again rather than to nothing.
258    pub async fn detach(&mut self) -> Option<String> {
259        let session = self.session.take();
260        let label = session.as_ref().map(|session| session.label.clone());
261        if let Some(session) = session {
262            session.close().await;
263        }
264        if let Some(home) = self.mesh.take() {
265            mesh::restore_local(self, home).await;
266        }
267        label
268    }
269
270    pub fn rename(&mut self, label: String) {
271        if let Some(session) = &mut self.session {
272            session.label = label;
273        }
274    }

276    /// An attached, non-interactive app for the Wireshark extcap
277    /// interface.
278    ///
279    /// `interactive` is false in both its senses here: there is no
280    /// prompt to return to, and nobody to answer a question.
281    pub fn for_extcap(
282        session: Session,
283        prefs: Prefs,
284        baud: u32,
285        recovery: Option<Recovery>,
286    ) -> Self {
287        Self {
288            session: Some(session),
289            prefs,
290            interactive: false,
291            trace: false,
292            no_save: true,
293            baud,
294            discovery: Discovery::Auto,
295            last_scan: Vec::new(),
296            mesh: None,
297            recovery,
298        }
299    }
300
301    pub async fn attach(&mut self, target: Target) -> Result<()> {
302        let session = connection::connect(target, self.trace).await?;
303        announce_attached(&session);
304        self.session = Some(session);
305        Ok(())
306    }
307
308    pub fn prompt(&self) -> String {
309        match &self.session {
310            Some(session) => format!("{} ({})> ", session.label, session.target.transport()),
311            None => "(unattached)> ".to_string(),
312        }
313    }

316fn announce_attached(session: &Session) {
317    eprintln!(
318        "attached: {} ({}) device={}{} boot_status={:?}",
319        session.label,
320        session.target.transport(),
321        session.device.dev_version(),
322        // `PROP_DEV_MODEL` is optional; say nothing rather than "unknown"
323        // when the device does not name its hardware—and a device that
324        // answers with an empty string has named it no better than one
325        // that refuses the read.
326        session
327            .device
328            .dev_model()
329            .filter(|model| !model.is_empty())
330            .map(|model| format!(" on {model}"))
331            .unwrap_or_default(),
332        session.device.boot_status(),
333    );
334}

336/// Work out which radio to talk to, from the flags, the environment,
337/// the saved default, and finally the air.
338async fn resolve(args: &ToolArgs, prefs: &Prefs) -> Result<Option<Target>> {
339    if args.pick
340        && (args.port.is_some()
341            || args.tcp.is_some()
342            || args.lan.is_some()
343            || matches!(args.ble, Some(Some(_))))
344    {
345        bail!(
346            "--pick chooses a radio from a listing; --port, --tcp, --lan, and --ble=SELECTOR \
347             already name one"
348        );
349    }
350    if let Some(group) = &args.lan {
351        if args.port.is_some() || args.ble.is_some() || args.tcp.is_some() {
352            bail!("--lan, --tcp, --port, and --ble name different radios; give one");
353        }
354        let group = match group {
355            Some(spec) => parse_group(spec)?,
356            None => std::net::SocketAddrV4::new(
357                umsh_lan_bearer::DEFAULT_GROUP.into(),
358                umsh_lan_bearer::DEFAULT_PORT,
359            ),
360        };
361        return Ok(Some(Target::Lan { group }));
362    }
363    if let Some(endpoint) = &args.tcp {
364        if args.port.is_some() || args.ble.is_some() {
365            bail!("--tcp, --port, and --ble name different radios; give one");
366        }
367        let (host, port) = connection::parse_endpoint(endpoint)?;
368        return Ok(Some(Target::Tcp { host, port }));
369    }
370    if let Some(port) = &args.port {
371        if args.ble.is_some() {
372            bail!("--port and --ble name different radios; give one");
373        }
374        return Ok(Some(Target::Serial {
375            port: port.clone(),
376            baud: args.baud,
377        }));
378    }
379    if let Some(Some(selector)) = &args.ble {
380        return Ok(Some(Target::Ble {
381            selector: selector.clone(),
382            name: None,
383        }));
384    }
385    // Bare `--ble`, or nothing at all: the tool finds the radio itself.
386    let target =
387        connection::discover(prefs, std::io::stdin().is_terminal(), args.discovery()).await?;
388    if let Some(target) = &target {
389        // A mutating one-shot must never act on a silently chosen radio.
390        eprintln!("discovered: {}", target.provisional_label());
391    }
392    Ok(target)
393}

397pub fn parse_group(spec: &str) -> Result<std::net::SocketAddrV4> {
398    umsh_lan_bearer::parse_group(spec).map_err(anyhow::Error::msg)
399}

401async fn run(args: ToolArgs) -> Result<()> {
402    output::set_color(args.color.enabled());
403    let interactive = args.command.is_none();
404    let mut app = App {
405        session: None,
406        prefs: Prefs::load(),
407        interactive,
408        trace: args.trace,
409        no_save: args.no_save,
410        baud: args.baud,
411        discovery: args.discovery(),
412        last_scan: Vec::new(),
413        mesh: None,
414        recovery: (!args.no_reconnect).then(Recovery::default),
415    };
416
417    // Everything clap's grammar cannot express is checked before a
418    // radio is opened.
419    if let Some(command) = &args.command {
420        command.validate()?;
421        // A command a mesh session cannot carry should say so now, not
422        // after a discovery pass and an attach handshake.
423        if args.node.is_some()
424            && let Some(refusal) = command.mesh_refusal()
425        {
426            bail!("{refusal}");
427        }
428    }
429    let needs_device = args
430        .command
431        .as_ref()
432        .is_none_or(|command| command.needs_device());
433    if needs_device {
434        match resolve(&args, &app.prefs).await? {
435            Some(target) => {
436                let session = connection::connect(target, app.trace).await?;
437                announce_attached(&session);
438                app.session = Some(session);
439            }
440            // The shell can still scan and connect; a one-shot cannot.
441            None if interactive => {}
442            None => bail!("no ULCP radios found; name one with --port, --tcp, or --ble=SELECTOR"),
443        }
444    }
445
446    // `--node` borrows the radio just attached and opens a session to
447    // the named device across the mesh. With a command that is the
448    // one-shot form of the shell's `remote`: run it and hand the radio
449    // back, so a script gets the same session the shell would have. With
450    // no command it is simply where the shell starts—the prompt opens
451    // already talking to the far node.
452    //
453    // A one-shot spends nothing on the air before the command itself: a
454    // script asking for the battery should pay for the battery and
455    // nothing else. The shell greets the node by name, because a prompt
456    // that says which device it is answers a question every subsequent
457    // command would otherwise raise.
458    if let Some(node) = args.node {
459        let greeting = if interactive {
460            mesh::Greeting::Named
461        } else {
462            mesh::Greeting::Silent
463        };
464        mesh::open_remote(&mut app, umsh::core::PublicKey(node.0), greeting).await?;
465        let result = match args.command {
466            Some(command) => command.run(&mut app).await,
467            None => repl::run(&mut app).await,
468        };
469        app.detach().await;
470        return result;
471    }
472
473    match args.command {
474        Some(command) => command.run(&mut app).await,
475        None => repl::run(&mut app).await,
476    }
477}

479#[tokio::main(flavor = "current_thread")]
480async fn main() {
481    // Wireshark drives this binary through its own argument vocabulary,
482    // which is checked before the tool's parser so the two never have to
483    // agree on a shared grammar.
484    //
485    // Everything here is `!Send`—the node layer, the pcap tap, the
486    // whole session—so the one task a mesh session spawns to hold the
487    // borrowed radio needs a `LocalSet` to be spawned onto.
488    let local = tokio::task::LocalSet::new();
489    let result = local
490        .run_until(async {
491            if extcap::is_extcap_invocation() {
492                extcap::run().await
493            } else {
494                run(ToolArgs::parse()).await
495            }
496        })
497        .await;
498    if let Err(error) = result {
499        eprintln!("error: {error:#}");
500        std::process::exit(1);
501    }
502}

989    fn ping_rejects_contradictions_and_out_of_range_values() {
990        const KEY: &str = "c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4c4";

1000        assert!(parse(&["ping", KEY, "--hops", "16"]).is_err());
