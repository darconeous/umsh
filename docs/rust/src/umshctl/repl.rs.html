1//! The interactive shell: one attach, many commands.
2//!
3//! Every line is re-parsed with the same `clap` tree the one-shot
4//! grammar uses, so `help`, `help phy`, and `phy --help` all work and
5//! the completer can walk the tree instead of maintaining a word list.

18/// What marks a `connect` selector as a TCP endpoint rather than a
19/// radio name.
20const TCP_SCHEME: &str = "tcp://";
21
22/// The REPL's grammar: everything the one-shot tree has, plus the few
23/// verbs that only mean something inside a session.
24#[derive(Debug, clap::Subcommand)]
25pub enum ReplCommand {
26    #[command(flatten)]
27    Shared(Command),
28
29    /// Attach to a radio, replacing the current attachment. With no
30    /// argument, rediscovers the way a bare launch does.
31    Connect {
32        /// A scan-listing number, a BLE name or id, a serial port
33        /// path, or `tcp://HOST:PORT`.
34        #[arg(value_name = "SELECTOR")]
35        selector: Option<String>,
36
37        /// Choose from a numbered listing, ignoring the saved default.
38        #[arg(long, conflicts_with = "selector")]
39        pick: bool,
40    },
41
42    /// Detach from the current radio without leaving the shell.
43    Disconnect,
44
45    /// Leave the shell.
46    #[command(alias = "quit")]
47    Exit,
48}
49
50/// Wrapper used to parse REPL lines and to build the completion tree.
51#[derive(Debug, Parser)]
52#[command(name = "", no_binary_name = true, disable_help_flag = false)]
53pub struct ReplCommandLine {
54    #[command(subcommand)]
55    pub command: ReplCommand,
56}

58/// Rustyline helper providing tab completion driven by the clap tree.
59#[derive(rustyline::Helper, rustyline::Hinter, rustyline::Highlighter, rustyline::Validator)]
60struct ReplHelper;
61
62impl Completer for ReplHelper {
63    type Candidate = Pair;
64
65    fn complete(
66        &self,
67        line: &str,
68        pos: usize,
69        _ctx: &Context<'_>,
70    ) -> rustyline::Result<(usize, Vec<Self::Candidate>)> {
71        let line_to_cursor = &line[..pos];
72
73        // Tokenize up to the cursor; give up quietly on an unclosed
74        // quote rather than completing something surprising.
75        let Some(mut tokens) = shlex::split(line_to_cursor) else {
76            return Ok((pos, Vec::new()));
77        };
78
79        // Separate the partial word being typed from the tokens behind it.
80        let (partial, complete_start) =
81            if !line_to_cursor.ends_with(char::is_whitespace) && !tokens.is_empty() {
82                let partial = tokens.pop().unwrap_or_default();
83                let start = pos.saturating_sub(partial.len());
84                (partial, start)
85            } else {
86                (String::new(), pos)
87            };
88
89        // Walk the tree with the finished tokens, tracking whether the
90        // next one will be swallowed as a flag's value.
91        let mut cmd = ReplCommandLine::command();
92        let mut expect_value_for: Option<String> = None;
93        for token in &tokens {
94            if expect_value_for.take().is_some() {
95                continue;
96            }
97            if let Some(flag) = token.strip_prefix("--").or_else(|| token.strip_prefix('-')) {
98                if let Some(arg) = cmd.get_arguments().find(|arg| {
99                    arg.get_long() == Some(flag)
100                        || arg.get_short().is_some_and(|c| c.to_string() == flag)
101                }) && takes_value(arg)
102                {
103                    expect_value_for = Some(flag.to_string());
104                }
105            } else if let Some(sub) = cmd.find_subcommand(token) {
106                cmd = sub.clone();
107            }
108        }
109
110        let candidates: Vec<String> = if let Some(flag) = expect_value_for {
111            cmd.get_arguments()
112                .find(|arg| arg.get_long() == Some(flag.as_str()))
113                .map(possible_values)
114                .unwrap_or_default()
115        } else if partial.starts_with('-') {
116            cmd.get_arguments()
117                .filter_map(|arg| arg.get_long().map(|long| format!("--{long}")))
118                .collect()
119        } else {
120            // Subcommand names and their aliases, then the possible
121            // values of whichever positional comes next.
122            let mut candidates: Vec<String> = cmd
123                .get_subcommands()
124                .flat_map(|sub| {
125                    std::iter::once(sub.get_name().to_string())
126                        .chain(sub.get_all_aliases().map(str::to_string))
127                })
128                .collect();
129            candidates.extend(cmd.get_positionals().flat_map(possible_values));
130            candidates
131        };
132
133        let pairs = candidates
134            .into_iter()
135            .filter(|candidate| candidate.starts_with(&partial))
136            .map(|candidate| Pair {
137                display: candidate.clone(),
138                replacement: format!("{candidate} "),
139            })
140            .collect();
141        Ok((complete_start, pairs))
142    }
143}

145fn possible_values(arg: &clap::Arg) -> Vec<String> {
146    arg.get_possible_values()
147        .iter()
148        .map(|value| value.get_name().to_string())
149        .collect()
150}
151
152/// Whether a clap argument consumes the next token as its value.
153fn takes_value(arg: &clap::Arg) -> bool {
154    matches!(
155        arg.get_action(),
156        clap::ArgAction::Set | clap::ArgAction::Append
157    )
158}

160pub async fn run(app: &mut App) -> Result<()> {
161    let mut editor = Editor::<ReplHelper, rustyline::history::DefaultHistory>::new()?;
162    editor.set_helper(Some(ReplHelper));
163    let history = connection::history_path();
164    if let Some(path) = &history {
165        // A missing history file is the normal first run.
166        let _ = editor.load_history(path);
167    }
168
169    banner(app);
170    let mut failed = false;
171    loop {
172        let mut prompt = app.prompt();
173        if failed {
174            prompt.insert_str(0, "❌ ");
175        }
176        let line = match editor.readline(&prompt) {
177            Ok(line) => line,
178            Err(ReadlineError::Interrupted | ReadlineError::Eof) => break,
179            Err(error) => {
180                eprintln!("error: {error}");
181                break;
182            }
183        };
184        if line.trim().is_empty() {
185            continue;
186        }
187        editor.add_history_entry(line.as_str())?;
188        match process_line(app, &line).await {
189            Ok(true) => failed = false,
190            Ok(false) => break,
191            Err(error) => {
192                // `help` renders through clap's error channel, and a
193                // help request is not a failure.
194                failed =
195                    !line.trim_start().starts_with("help") && !line.trim_end().ends_with("help");
196                eprintln!("{error}");
197            }
198        }
199    }
200
201    if let Some(path) = &history {
202        if let Some(parent) = path.parent() {
203            let _ = std::fs::create_dir_all(parent);
204        }
205        if let Err(error) = editor.save_history(path) {
206            warn(format!("could not save history: {error}"));
207        }
208    }
209    Ok(())
210}

212fn banner(app: &App) {
213    println!("umshctl — `help` lists commands, `exit` leaves.");
214    if app.session.is_none() {
215        println!("not attached: `scan` to look for radios, `connect` to attach.");
216    }
217}

219/// Parse and run one line. `Ok(false)` means the user asked to leave.
220async fn process_line(app: &mut App, line: &str) -> Result<bool> {
221    let Some(args) = shlex::split(line) else {
222        anyhow::bail!("unbalanced quotes");
223    };
224    let command = ReplCommandLine::try_parse_from(args)?.command;
225    match command {
226        ReplCommand::Exit => return Ok(false),
227        ReplCommand::Disconnect => {
228            match app.detach() {
229                Some(label) => println!("detached from {label}"),
230                None => println!("not attached"),
231            }
232            return Ok(true);
233        }
234        ReplCommand::Connect { selector, pick } => {
235            connect(app, selector, pick).await?;
236            return Ok(true);
237        }
238        ReplCommand::Shared(command) => {
239            command.validate()?;
240            if command.needs_device() && app.session.is_none() {
241                anyhow::bail!("not attached — try `scan` or `connect`");
242            }
243            command.run(app).await?;
244        }
245    }
246    Ok(true)
247}

249/// `connect`: rediscover, or attach to the radio the user named.
250async fn connect(app: &mut App, selector: Option<String>, pick: bool) -> Result<()> {
251    let target = match selector {
252        // A bare number refers to the last `scan` listing, which is the
253        // whole reason the REPL keeps it.
254        Some(selector) => match selector.parse::<usize>() {
255            Ok(index) if (1..=app.last_scan.len()).contains(&index) => {
256                Target::from(&app.last_scan[index - 1])
257            }
258            Ok(index) => anyhow::bail!(
259                "no radio {index} in the last scan ({} listed); run `scan` again",
260                app.last_scan.len()
261            ),
262            Err(_) => named_target(selector, app.baud)?,
263        },
264        None => {
265            let how = if pick {
266                connection::Discovery::Ask
267            } else {
268                app.discovery
269            };
270            let Some(target) = connection::discover(&app.prefs, app.interactive, how).await? else {
271                anyhow::bail!("no ULCP radios found");
272            };
273            target
274        }
275    };
276    // Detaching first reverts session-scoped device state (promiscuous
277    // mode) on the radio being left behind.
278    app.detach();
279    app.attach(target).await
280}

282/// Classify a `connect` selector that is not a scan-listing number.
283///
284/// The `tcp://` test comes first: the scheme's own slashes would
285/// satisfy the path rule below it. A bare `host:port` stays a BLE
286/// selector — the scheme is what separates an endpoint from a name.
287fn named_target(selector: String, baud: u32) -> Result<Target> {
288    if let Some(endpoint) = selector.strip_prefix(TCP_SCHEME) {
289        let (host, port) = connection::parse_endpoint(endpoint)?;
290        return Ok(Target::Tcp { host, port });
291    }
292    if selector.contains('/') {
293        return Ok(Target::Serial {
294            port: selector,
295            baud,
296        });
297    }
298    Ok(Target::Ble {
299        selector,
300        name: None,
301    })
302}

319    #[test]
320    fn a_tcp_selector_beats_the_serial_path_rule() {
321        // `tcp://` contains the slashes the path rule looks for, so the
322        // order of the two tests is what makes this work at all.
323        assert_eq!(
324            named_target("tcp://127.0.0.1:9000".into(), 115_200).unwrap(),
325            Target::Tcp {
326                host: "127.0.0.1".into(),
327                port: 9000,
328            }
329        );
330        assert_eq!(
331            named_target("/dev/cu.usbmodem101".into(), 115_200).unwrap(),
332            Target::Serial {
333                port: "/dev/cu.usbmodem101".into(),
334                baud: 115_200,
335            }
336        );
337        // A bare endpoint is still a name: BLE radios are named freely,
338        // and the scheme is the only reliable signal.
339        assert_eq!(
340            named_target("127.0.0.1:9000".into(), 115_200).unwrap(),
341            Target::Ble {
342                selector: "127.0.0.1:9000".into(),
343                name: None,
344            }
345        );
346        // A malformed endpoint is reported, not silently taken for a
347        // radio name.
348        assert!(named_target("tcp://127.0.0.1".into(), 115_200).is_err());
349    }
