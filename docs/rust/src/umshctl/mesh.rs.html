1//! The host stack this tool becomes to reach a device over the mesh.
2//!
3//! Every other part of this tool speaks ULCP down the wire to the radio
4//! it is attached to. Reaching a device across the room means borrowing
5//! that radio and becoming a node: running a host MAC over the
6//! attachment, and speaking the same ULCP grammar to the far end,
7//! carried by the Node Management binding
8//! (`docs/protocol/src/app-node-management.md`).
9//!
10//! Two identities are in play and it is worth keeping them apart. The
11//! attached radio has its own device identity, which `identity` shows.
12//! This tool has a separate, persistent administrator identity—
13//! `admin-key` prints it—and a device is managed from here only once
14//! that key is listed in its `PROP_DEV_ADMINS`, which `dev-admin add`
15//! does over a bench link.

48/// One identity—the administrator's. Channels and queues are sized for
49/// a tool that talks to one device at a time and holds one exchange open
50/// while it does.
51///
52/// Peers are sized for `discover` instead, which is the one command that
53/// meets a crowd: every stranger that answers a solicitation is
54/// auto-registered from the full key it sends, and the registry evicts
55/// the least recently used once it is full. Four slots would have made a
56/// discovery of any size report the last four nodes to speak. This is a
57/// host tool with a host's memory.
58const IDENTITIES: usize = 1;
59const PEERS: usize = 32;
60const CHANNELS: usize = 1;
61const ACKS: usize = 8;
62const TX: usize = 8;
63const FRAME: usize = 256;
64const DUP: usize = 32;
65
66/// The radio this tool runs its MAC over: the attached device, whatever
67/// transport it was attached over. One type, because the transport is
68/// what recovery replaces, and only this type knows how.
69pub type CtlRadio = UlcpDevice<SessionLink>;
70pub type CtlPlatform = TokioPlatform<CtlRadio, TokioFileCounterStore, TokioFileKeyValueStore>;
71pub type CtlMac = Mac<CtlPlatform, IDENTITIES, PEERS, CHANNELS, ACKS, TX, FRAME, DUP>;
72pub type CtlHandle<'a> =
73    MacHandle<'a, CtlPlatform, IDENTITIES, PEERS, CHANNELS, ACKS, TX, FRAME, DUP>;
74pub type CtlHost<'a> = Host<CtlHandle<'a>>;

76/// How long the whole operation may take before the tool gives up.

78/// The exchange engine has its own attempt budget; this bounds the
79/// continuation loop of a large read, where each fragment restarts that

85/// Load the administrator identity, generating one the first time.
86///
87/// The seed is a plain 32-byte file, written where the rest of this
88/// tool's state lives.
89pub fn admin_identity() -> Result<SoftwareIdentity> {
90    let path = connection::admin_identity_path()
91        .ok_or_else(|| anyhow!("no HOME directory to keep an administrator identity in"))?;
92    load_or_create_identity(&path)
93}
94
95fn load_or_create_identity(path: &Path) -> Result<SoftwareIdentity> {
96    match std::fs::read(path) {
97        Ok(bytes) => {
98            let secret: [u8; 32] = bytes.try_into().map_err(|_| {
99                anyhow!(
100                    "{} is not a 32-byte identity seed; move it aside to start over",
101                    path.display()
102                )
103            })?;
104            Ok(SoftwareIdentity::from_secret_bytes(&secret))
105        }
106        Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
107            if let Some(parent) = path.parent() {
108                std::fs::create_dir_all(parent)
109                    .with_context(|| format!("creating {}", parent.display()))?;
110            }
111            let mut secret = [0u8; 32];
112            rng().fill_bytes(&mut secret);
113            std::fs::write(path, secret).with_context(|| format!("writing {}", path.display()))?;
114            note("generated a new administrator identity");
115            Ok(SoftwareIdentity::from_secret_bytes(&secret))
116        }
117        Err(error) => Err(error).with_context(|| format!("reading {}", path.display())),
118    }
119}

135/// The device's live PHY, as a radio configuration.
136///
137/// The host MAC paces itself on modeled airtime, and the attach this
138/// tool uses deliberately leaves the radio's configuration alone, so the
139/// defaults it carries may describe some other radio entirely. Read the
140/// real ones before borrowing the link.
141///
142/// A radio whose PHY is switched off refuses every transmission, and it
143/// does so far downstream—as an `INVALID_STATE` on the first frame the
144/// MAC tries to send, long after the radio has been borrowed. Reading
145/// the flag here turns that into an answer the caller can act on, while
146/// it still has its attachment.
147pub async fn adopt_phy(device: &mut CtlRadio) -> Result<UlcpDeviceConfig> {
148    let keys = [
149        prop::PHY_FREQ,
150        prop::PHY_LORA_BW,
151        prop::PHY_LORA_SF,
152        prop::PHY_LORA_CR,
153        prop::PHY_TX_POWER,
154        prop::PHY_ENABLED,
155    ];
156    let answers = device.get_props(&keys).await?;
157    let mut config = connection::attach_config();
158    for (requested, answer) in keys.iter().zip(&answers) {
159        let Ok((_, value)) = answer else { continue };
160        match *requested {
161            prop::PHY_ENABLED => {
162                if value.first() == Some(&0) {
163                    bail!("the radio's PHY is switched off; turn it on with `set phy-enabled on`");
164                }
165            }
166            prop::PHY_FREQ => {
167                if let Ok(bytes) = <[u8; 4]>::try_from(&value[..]) {
168                    config.freq_khz = u32::from_le_bytes(bytes);
169                }
170            }
171            prop::PHY_LORA_BW => {
172                if let Ok(bytes) = <[u8; 4]>::try_from(&value[..]) {
173                    config.bandwidth_hz = u32::from_le_bytes(bytes);
174                }
175            }
176            prop::PHY_LORA_SF => {
177                if let Some(&sf) = value.first() {
178                    config.spreading_factor = sf;
179                }
180            }
181            prop::PHY_LORA_CR => {
182                if let Some(&cr) = value.first() {
183                    config.coding_rate_denom = cr;
184                }
185            }
186            prop::PHY_TX_POWER => {
187                if let Some(&power) = value.first() {
188                    config.tx_power_dbm = power as i8;
189                }
190            }
191            _ => {}
192        }
193    }
194    Ok(config)
195}

197/// This host owns the MAC and does its own filtering, so the radio's
198/// provisioned receive filters must not gate delivery. The mode is
199/// session-scoped and touches no provisioning—which is also why it is
200/// asserted again after every reattach.
201pub async fn prepare_radio(device: &mut CtlRadio) -> Result<()> {
202    match device.set_prop(prop::MAC_PROMISCUOUS, &[1]).await {
203        Ok(_) => Ok(()),
204        Err(UlcpError::Status(status)) => {
205            note(format!(
206                "radio refused promiscuous mode ({status:?}); reception follows its own filtering"
207            ));
208            Ok(())
209        }
210        Err(error) => Err(error.into()),
211    }
212}

225/// What a node stack needs to know about the radio it runs over, beyond
226/// the radio itself: how to reach it again, and whether to.
227///
228/// The session that lent the radio is described here rather than
229/// carried—its device handle is inside the MAC for the duration—so the
230/// stack can put a lost link back under that handle without the session
231/// being any the wiser.
232#[derive(Clone)]
233pub struct StackContext {
234    pub target: connection::Target,
235    pub tap: FrameTap,
236    pub label: String,
237    /// The policy [`NodeStack::pump_until`] heals a lost link with, or
238    /// `None` when the caller does its own healing (the mesh driver) or
239    /// wants the failure (`--no-reconnect`).
240    pub self_heal: Option<Recovery>,
241}
242
243impl StackContext {
244    /// Describe `session`'s radio for a stack that is about to borrow it.
245    fn for_session(session: &Session, self_heal: Option<Recovery>) -> Self {
246        Self {
247            target: session.target.clone(),
248            tap: session.tap.clone(),
249            label: session.label.clone(),
250            self_heal,
251        }
252    }
253
254    fn relink(&self) -> Relink<'_> {
255        Relink {
256            target: &self.target,
257            tap: &self.tap,
258            label: &self.label,
259        }
260    }
261}

263/// Something this tool runs as a node on its own radio.
264///
265/// `manage`, the messaging commands, and `discover` all want the same
266/// preamble—read the device's PHY, take the attachment over, turn it
267/// into a MAC—and the same guarantee afterwards, that the attachment
268/// comes back whether the errand worked or not. That is
269/// [`borrowing_the_radio`]; this is the part that differs.
270pub trait RadioErrand {
271    async fn run(
272        self,
273        mac: &AsyncRefCell<CtlMac>,
274        identity: SoftwareIdentity,
275        ctx: &StackContext,
276    ) -> Result<()>;
277}

279/// Take the attachment over as this tool's radio, run `errand` on it,
280/// and hand the attachment back.
281///
282/// This tool has one radio. A command that kept it on failure would
283/// leave the session holding nothing, so the give-back is unconditional
284///—every early return here happens before the link is consumed.
285pub async fn borrowing_the_radio<E: RadioErrand>(app: &mut App, errand: E) -> Result<()> {
286    let identity = admin_identity()?;
287    // Read the device's own PHY before taking the link over, so the host
288    // MAC paces itself the way the radio is actually configured.
289    let config = adopt_phy(app.device()?).await?;
290
291    let Some(session) = app.session.take() else {
292        bail!("not attached—try `ble-scan` or `connect`");
293    };
294    let ctx = StackContext::for_session(&session, app.recovery.clone());
295    let Session {
296        device,
297        target,
298        label,
299        tap,
300    } = session;
301    // Re-attaching is the only way to give the handle a configuration; it
302    // costs a handful of property reads and no reconnect. The link is
303    // consumed, so a failure here really does end the attachment.
304    let mut device = UlcpDevice::attach_administrative(device.into_link(), config)
305        .await
306        .context("re-attaching the radio with its own PHY")?;
307    if app.trace {
308        connection::install_trace(&mut device);
309    }
310    prepare_radio(&mut device).await?;
311
312    let (device, result) = match counter_store() {
313        Ok(store) => {
314            let mac = build_mac(device, store);
315            let result = errand.run(&mac, identity, &ctx).await;
316            (mac.into_inner().into_radio(), result)
317        }
318        Err(error) => (device, Err(error)),
319    };
320    app.session = Some(Session {
321        device,
322        target,
323        label,
324        tap,
325    });
326    result
327}

342/// The tool as a node: a host MAC over the borrowed radio, one local
343/// node standing for the administrator identity, and the pump that keeps
344/// both moving.
345pub struct NodeStack<'a> {
346    pub host: CtlHost<'a>,
347    pub node: LocalNode<CtlHandle<'a>>,
348    pub handle: CtlHandle<'a>,
349    mac: &'a AsyncRefCell<CtlMac>,
350    ctx: StackContext,
351    /// How many times the link under the radio has been put back. A
352    /// caller that had an exchange in flight compares this before and
353    /// after to learn whether the exchange straddled a recovery.
354    recoveries: u32,
355    started: Instant,
356}
357
358impl<'a> NodeStack<'a> {
359    /// Register `identity` on the borrowed MAC and stand a node up on it,
360    /// returning the stack and the administrator's public key.
361    pub async fn build(
362        mac: &'a AsyncRefCell<CtlMac>,
363        identity: SoftwareIdentity,
364        ctx: &StackContext,
365    ) -> Result<(Self, PublicKey)> {
366        let handle = MacHandle::new(mac);
367        let local_key = *identity.public_key();
368        let identity_id = handle
369            .add_identity(identity)
370            .await
371            .map_err(|error| anyhow!("registering the administrator identity: {error:?}"))?;
372        // A frame counter that restarted at zero would be rejected as a
373        // replay by every device that has heard this identity before.
374        handle
375            .load_persisted_counter(identity_id)
376            .await
377            .map_err(|error| anyhow!("loading persisted frame counters: {error:?}"))?;
378
379        let mut host: CtlHost<'a> = Host::new(handle);
380        let node = host.add_node(identity_id);
381        Ok((
382            Self {
383                host,
384                node,
385                handle,
386                mac,
387                ctx: ctx.clone(),
388                recoveries: 0,
389                started: Instant::now(),
390            },
391            local_key,
392        ))
393    }
394
395    /// When this stack came up. The exchange engine's deadlines are
396    /// expressed against [`Self::now_ms`], which counts from here.
397    pub fn started(&self) -> Instant {
398        self.started
399    }
400
401    pub fn now_ms(&self) -> u64 {
402        self.started.elapsed().as_millis() as u64
403    }
404
405    /// How many times the link under the radio has been put back.
406    pub fn recoveries(&self) -> u32 {
407        self.recoveries
408    }
409
410    /// Whether the radio's transport has reported itself gone.
411    pub async fn radio_link_lost(&self) -> bool {
412        self.mac.borrow().await.radio().link_lost()
413    }
414
415    /// Put the link back under the radio, attempt after attempt, until
416    /// it is back or `abort` resolves.
417    ///
418    /// The MAC and everything standing on it survive: identities,
419    /// peers, routes, pending acknowledgments. Only the transport under
420    /// the device handle is replaced, and the session-scoped state this
421    /// host asserts on the radio is asserted again. The exclusive borrow
422    /// is uncontended here—nothing is pumping while the radio is known
423    /// to be gone.
424    pub async fn recover(
425        &mut self,
426        policy: &Recovery,
427        abort: impl Future<Output = ()>,
428    ) -> Result<Recovered> {
429        let mut mac = self.mac.borrow_mut().await;
430        let radio = mac.radio_mut();
431        let outcome = connection::recover_device(radio, &self.ctx.relink(), policy, abort).await?;
432        if outcome == Recovered::Attached {
433            prepare_radio(radio).await?;
434            self.recoveries += 1;
435        }
436        Ok(outcome)
437    }
438
439    /// Drive the MAC until it has nothing to do or `deadline` arrives.
440    ///
441    /// A quiet radio produces no MAC wake, so the timeouts that retire an
442    /// unanswered acknowledgment need their own nudge afterwards.
443    ///
444    /// A radio whose link has gone is put back here, when the stack was
445    /// told to: the pump is where the loss is noticed, and the caller's
446    /// own deadline loop simply comes round again once the link is back.
447    /// Ctrl-C ends the wait, not the process.
448    pub async fn pump_until(&mut self, deadline: Instant) -> Result<()> {
449        let pumped = tokio::select! {
450            result = self.host.pump_once() => result,
451            _ = tokio::time::sleep_until(deadline) => Ok(()),
452        };
453        if let Err(error) = pumped {
454            let policy = self.ctx.self_heal.clone();
455            if let Some(policy) = policy
456                && self.radio_link_lost().await
457            {
458                let interrupted = async {
459                    let _ = tokio::signal::ctrl_c().await;
460                };
461                return match self.recover(&policy, interrupted).await? {
462                    Recovered::Attached => Ok(()),
463                    Recovered::Interrupted => bail!("reconnect interrupted"),
464                };
465            }
466            bail!("the radio stopped answering: {error:?}");
467        }
468        self.host.service_protocol_timeouts().await;
469        let _ = self.handle.service_counter_persistence().await;
470        Ok(())
471    }
472
473    /// Carry one exchange to its end, pumping the host in between, giving
474    /// up at `give_up`.
475    ///
476    /// An exchange that ends without an outcome—this tool ran out of
477    /// patience, or the radio underneath went away—is abandoned, so the
478    /// manager is free for the next one.
479    pub async fn exchange(
480        &mut self,
481        manager: &mut NodeManager<CtlHandle<'a>>,
482        request: &[u8],
483        give_up: Instant,
484    ) -> Result<Outcome> {
485        manager
486            .begin(request, self.now_ms())
487            .map_err(|error| anyhow!("{error:?}"))?;
488        let outcome = self.carry(manager, give_up).await;
489        if outcome.is_err() {
490            manager.abandon().await;
491        }
492        outcome
493    }
494
495    async fn carry(
496        &mut self,
497        manager: &mut NodeManager<CtlHandle<'a>>,
498        give_up: Instant,
499    ) -> Result<Outcome> {
500        loop {
501            if Instant::now() > give_up {
502                bail!("gave up after {} s", OPERATION_TIMEOUT.as_secs());
503            }
504            let progress = manager
505                .service(self.now_ms())
506                .await
507                .map_err(|error| anyhow!("sending to the device: {error:?}"))?;
508            let deadline_ms = match progress {
509                Progress::Done(outcome) => return Ok(outcome),
510                Progress::Waiting { deadline_ms } => deadline_ms,
511            };
512            let wait = Duration::from_millis(deadline_ms.saturating_sub(self.now_ms()));
513            self.pump_until(Instant::now() + wait).await?;
514        }
515    }
516}

518/// What a failed exchange means to somebody holding the tool.
519pub fn describe(failure: Failure) -> anyhow::Error {
520    match failure {
521        Failure::TimedOut => anyhow!(
522            "the device received the request but did not answer—this tool may not be one of \
523             its administrators (`admin-key` prints the key it would have to list)"
524        ),
525        Failure::Unreachable => anyhow!(
526            "the device did not acknowledge the request—it may be out of range or switched off"
527        ),
528        Failure::CursorInvalid => {
529            anyhow!("the device lost its place in the read; run the command again")
530        }
531        Failure::TooLarge => anyhow!("the answer is larger than this tool reassembles"),
532        Failure::Malformed => anyhow!("the device's answer could not be read"),
533        Failure::UnknownCriticalOption(number) => {
534            anyhow!("the device's answer carries option {number}, which this tool does not know")
535        }
536    }
537}

541/// What a mesh session borrowed, and how to give it back.
542///
543/// The driver task owns the radio for as long as the session lasts; its
544/// join handle is how the radio comes home, and the rest is what the
545/// local session was called before it was lent out.
546pub struct MeshHome {
547    pub driver: tokio::task::JoinHandle<CtlRadio>,
548    pub local_target: connection::Target,
549    pub local_label: String,
550    pub tap: connection::FrameTap,
551}
552
553/// Whether opening a session should say hello first.
554#[derive(Clone, Copy, PartialEq, Eq)]
555pub enum Greeting {
556    /// Ask the device its name, so the prompt is something a person
557    /// recognizes and an unreachable node is reported now rather than by
558    /// whatever they type first. One exchange.
559    Named,
560    /// Say nothing. The key is the label, and the command the caller
561    /// came for is the first thing on the air.
562    Silent,
563}
564
565/// Borrow the attached radio and open a ULCP session to `target` over the
566/// mesh.
567///
568/// On success `app` is attached to the remote device and holds the
569/// [`MeshHome`] that ends the session. On failure the local attachment is
570/// restored, because a tool that loses your radio for mistyping a key is
571/// not one you would use twice.
572pub async fn open_remote(app: &mut App, target: PublicKey, greeting: Greeting) -> Result<()> {
573    if app.mesh.is_some() {
574        bail!("already on a mesh session—`disconnect` returns to the radio");
575    }
576    let identity = admin_identity()?;
577
578    // Read the device's own PHY before taking the link over, and open the
579    // counter store while there is still an attachment to hand back.
580    let phy = adopt_phy(app.device()?).await?;
581    let store = counter_store()?;
582
583    let Some(session) = app.session.take() else {
584        bail!("not attached—try `ble-scan` or `connect`");
585    };
586    // The driver heals the link itself, so the stack under it does not.
587    let ctx = StackContext::for_session(&session, None);
588    let recovery = app.recovery.clone();
589    let connection::Session {
590        device,
591        target: local_target,
592        label: local_label,
593        tap,
594    } = session;
595
596    // Re-attaching is the only way to give the handle the configuration
597    // the host MAC paces itself on. The link is consumed, so a failure
598    // here really does end the attachment.
599    let mut radio = UlcpDevice::attach_administrative(device.into_link(), phy.clone())
600        .await
601        .context("re-attaching the radio with its own PHY")?;
602    if app.trace {
603        connection::install_trace(&mut radio);
604    }
605    prepare_radio(&mut radio).await?;
606
607    let (link, endpoint) = mesh_link();
608    let driver = tokio::task::spawn_local(drive(
609        radio, store, identity, target, endpoint, ctx, recovery,
610    ));
611
612    // From here the radio belongs to the driver, and the only way back to
613    // it is through the join handle.
614    let home = MeshHome {
615        driver,
616        local_target,
617        local_label,
618        tap,
619    };
620    match open_session(link, phy, target, greeting).await {
621        Ok(session) => {
622            app.session = Some(session);
623            app.mesh = Some(home);
624            Ok(())
625        }
626        Err(error) => {
627            // The link is already dropped, so the driver is winding down;
628            // wait for the radio and put the local session back.
629            restore_local(app, home).await;
630            Err(error)
631        }
632    }
633}

635/// Open the device handle at the far end of `link`.
636///
637/// Opening costs nothing on the air—the binding needs no session and
638/// the device is told nothing—so the only exchange here is the name,
639/// and only when one was asked for.
640async fn open_session(
641    link: MeshFrameLink,
642    phy: UlcpDeviceConfig,
643    target: PublicKey,
644    greeting: Greeting,
645) -> Result<connection::Session> {
646    let tap = connection::new_tap();
647    let session_link = SessionLink::new(connection::AnyLink::Mesh(link), tap.clone());
648    let mut device = UlcpDevice::open_remote(session_link, connection::mesh_attach_config(phy));
649    // A device that will not say its name is still perfectly usable; the
650    // key it answers to will do as a label.
651    let label = match greeting {
652        Greeting::Silent => target.to_string(),
653        Greeting::Named => {
654            note("reaching the device over the mesh—an exchange can take a while");
655            match device.device_name().await {
656                Ok(name) if !name.is_empty() => name,
657                Ok(_) => target.to_string(),
658                Err(error) => {
659                    return Err(
660                        anyhow::Error::new(error).context("reaching the device over the mesh")
661                    );
662                }
663            }
664        }
665    };
666    Ok(connection::Session {
667        device,
668        target: connection::Target::Mesh { key: target.0 },
669        label,
670        tap,
671    })
672}

674/// End a mesh session and put the local attachment back.
675///
676/// The session's device handle must already be dropped: that is what
677/// closes the link and tells the driver to wind down.
678pub async fn restore_local(app: &mut App, home: MeshHome) {
679    let MeshHome {
680        driver,
681        local_target,
682        local_label,
683        tap,
684    } = home;
685    match driver.await {
686        Ok(radio) => {
687            app.session = Some(connection::Session {
688                device: radio,
689                target: local_target,
690                label: local_label,
691                tap,
692            });
693        }
694        Err(error) => {
695            // The driver panicked or was cancelled, and the radio went
696            // with it. Say so rather than leaving a session that is not
697            // attached to anything.
698            note(format!(
699                "the mesh session ended badly ({error}); the radio was not recovered"
700            ));
701        }
702    }
703}

705/// The driver task: owns the borrowed radio for the life of the session,
706/// carries every request the link hands it, and gives the radio back.
707async fn drive(
708    radio: CtlRadio,
709    store: TokioFileCounterStore,
710    identity: SoftwareIdentity,
711    target: PublicKey,
712    endpoint: MeshEndpoint,
713    ctx: StackContext,
714    recovery: Option<Recovery>,
715) -> CtlRadio {
716    let mac = build_mac(radio, store);
717    serve(&mac, identity, target, endpoint, &ctx, recovery.as_ref()).await;
718    mac.into_inner().into_radio()
719}
720
721/// One step of the driver loop: what the select settled on.
722enum Step {
723    /// A request to carry.
724    Carry(MeshRequest),
725    /// The MAC made progress on its own.
726    Pumped,
727    /// The MAC could not be pumped; whether that is the link is decided
728    /// once the borrow is released.
729    Failed(anyhow::Error),
730    /// The link is gone; the session is over.
731    Closed,
732}
733
734/// Whether an exchange that straddled a recovery deserves a second try.
735///
736/// The exchange engine's own retries absorb a short outage; a long one
737/// ends the exchange with no outcome, or with a timeout that was really
738/// the radio's absence. Either is worth one more try—except for a
739/// reset-class request, which may already have been delivered and must
740/// not be delivered twice.
741fn should_rerun(recovered: bool, outcome: &Result<Outcome>, reset_class: bool) -> bool {
742    recovered
743        && !reset_class
744        && match outcome {
745            Err(_) | Ok(Outcome::Failed(Failure::TimedOut | Failure::Unreachable)) => true,
746            Ok(_) => false,
747        }
748}
749
750/// Everything the driver holds while serving a session, so carrying a
751/// request reads the same wherever it happens.
752struct Driver<'a> {
753    stack: NodeStack<'a>,
754    manager: NodeManager<CtlHandle<'a>>,
755    routes: crate::routes::RouteCache,
756}
757
758impl Driver<'_> {
759    /// Run one exchange, reporting whether a recovery happened under it.
760    async fn attempt(&mut self, request: &MeshRequest) -> (bool, Result<Outcome>) {
761        let before = self.stack.recoveries();
762        let give_up = Instant::now() + OPERATION_TIMEOUT;
763        let outcome = self
764            .stack
765            .exchange(&mut self.manager, request.frame(), give_up)
766            .await;
767        (self.stack.recoveries() != before, outcome)
768    }
769
770    /// Carry one request and hand its outcome back through `endpoint`.
771    async fn carry(&mut self, endpoint: &mut MeshEndpoint, request: MeshRequest) {
772        let (recovered, mut outcome) = self.attempt(&request).await;
773        if should_rerun(recovered, &outcome, request.is_reset_class()) {
774            outcome = self.attempt(&request).await.1;
775        }
776        match outcome {
777            Ok(Outcome::Replied { .. }) => {
778                let reply = self.manager.reply().to_vec();
779                endpoint.deliver(&request, DeliveredOutcome::Replied(&reply));
780            }
781            Ok(Outcome::NoResponse) => endpoint.deliver(&request, DeliveredOutcome::NoResponse),
782            Ok(Outcome::Failed(failure)) => {
783                endpoint.deliver(&request, DeliveredOutcome::Failed(failure))
784            }
785            // This command could not be carried—it ran out of patience,
786            // or the engine would not take it. The session survives: a
787            // radio that has actually died fails the pump on the very
788            // next turn of the loop, and that is dealt with there.
789            Err(error) => endpoint.refuse(format!("{error:#}")),
790        }
791        // An exchange is where a route is learned, so this is where
792        // there is something new to remember. Writing per exchange
793        // rather than at the end also means the file is current while a
794        // shell session is still open, which is what lets `routes`
795        // report on one.
796        self.routes.harvest(&self.stack.handle).await;
797        if let Err(error) = self.routes.store() {
798            crate::output::warn(format!("could not save learned routes: {error:#}"));
799        }
800    }
801}
802
803async fn serve(
804    mac: &AsyncRefCell<CtlMac>,
805    identity: SoftwareIdentity,
806    target: PublicKey,
807    mut endpoint: MeshEndpoint,
808    ctx: &StackContext,
809    recovery: Option<&Recovery>,
810) {
811    let (stack, _local_key) = match NodeStack::build(mac, identity, ctx).await {
812        Ok(built) => built,
813        Err(error) => return endpoint.fail(MeshFault::Radio(format!("{error:#}"))),
814    };
815    let peer = match stack.node.peer(target).await {
816        Ok(peer) => peer,
817        Err(error) => {
818            return endpoint.fail(MeshFault::Radio(format!(
819                "registering the device as a peer: {error:?}"
820            )));
821        }
822    };
823    // Whatever an earlier invocation learned about reaching this device,
824    // put back before the first frame goes out. A wrong guess costs one
825    // exchange and the MAC's own retry finds the path again; not
826    // guessing costs a flood every time the tool is run.
827    let routes = crate::routes::RouteCache::load();
828    if let Some(record) = routes.get(&target) {
829        peer.restore_route(record.route.clone()).await;
830    }
831
832    // The first token has only to be unpredictable; the manager keeps
833    // every one after it distinct from all of its own.
834    let mut seed = [0u8; 2];
835    rng().fill_bytes(&mut seed);
836    let manager = NodeManager::new(peer, u16::from_be_bytes(seed));
837    let mut driver = Driver {
838        stack,
839        manager,
840        routes,
841    };
842
843    let mut fatal = None;
844    loop {
845        // The arms only settle what happened. Everything that needs the
846        // stack runs below, once the futures borrowing it are gone.
847        let step = tokio::select! {
848            request = endpoint.next() => match request {
849                Some(request) => Step::Carry(request),
850                None => Step::Closed,
851            },
852            result = driver.stack.host.pump_once() => match result {
853                Ok(()) => Step::Pumped,
854                Err(error) => Step::Failed(anyhow!("the radio stopped answering: {error:?}")),
855            },
856        };
857        match step {
858            Step::Closed => break,
859            Step::Pumped => {
860                driver.stack.host.service_protocol_timeouts().await;
861                let _ = driver.stack.handle.service_counter_persistence().await;
862            }
863            Step::Failed(error) => {
864                let Some(policy) = recovery else {
865                    fatal = Some(MeshFault::Radio(format!("{error:#}")));
866                    break;
867                };
868                if !driver.stack.radio_link_lost().await {
869                    fatal = Some(MeshFault::Radio(format!("{error:#}")));
870                    break;
871                }
872                // The session goes on while the link is put back: what
873                // arrives meanwhile waits its turn, and the handle being
874                // dropped—the user leaving—is what ends the wait.
875                let mut deferred = VecDeque::new();
876                let outcome = {
877                    let closed = async {
878                        while let Some(request) = endpoint.next().await {
879                            deferred.push_back(request);
880                        }
881                    };
882                    driver.stack.recover(policy, closed).await
883                };
884                match outcome {
885                    Ok(Recovered::Attached) => {
886                        for request in deferred {
887                            driver.carry(&mut endpoint, request).await;
888                        }
889                    }
890                    Ok(Recovered::Interrupted) => break,
891                    Err(error) => {
892                        fatal = Some(MeshFault::Radio(format!("{error:#}")));
893                        break;
894                    }
895                }
896            }
897            Step::Carry(request) => driver.carry(&mut endpoint, request).await,
898        }
899    }
900    let _ = driver.stack.handle.service_counter_persistence().await;
901    driver.routes.harvest(&driver.stack.handle).await;
902    if let Err(error) = driver.routes.store() {
903        crate::output::warn(format!("could not save learned routes: {error:#}"));
904    }
905    if let Some(fault) = fatal {
906        endpoint.fail(fault);
907    }
908}

910#[cfg(test)]
911mod tests {
912    use super::*;
913
914    #[test]
915    fn an_exchange_is_rerun_once_only_when_a_recovery_ate_it() {
916        let lost: Result<Outcome> = Err(anyhow!("gave up"));
917        let timed_out: Result<Outcome> = Ok(Outcome::Failed(Failure::TimedOut));
918        let answered: Result<Outcome> = Ok(Outcome::NoResponse);
919        let malformed: Result<Outcome> = Ok(Outcome::Failed(Failure::Malformed));
920
921        assert!(should_rerun(true, &lost, false));
922        assert!(should_rerun(true, &timed_out, false));
923        // An exchange that reached an answer is not repeated.
924        assert!(!should_rerun(true, &answered, false));
925        // Nor one the device answered badly: that was not the link.
926        assert!(!should_rerun(true, &malformed, false));
927        // Nothing is repeated when the link never went away.
928        assert!(!should_rerun(false, &lost, false));
929        // A reset may already have landed; it is never sent twice.
930        assert!(!should_rerun(true, &lost, true));
931    }
