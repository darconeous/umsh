12/// Classify authentication refusals when the TLS library preserves their cause.
13/// Post-handshake alerts collapsed into InternalError remain TLS failures.
14pub fn failure_reason(error: &TlsError) -> umsh_ulcp::bridge::Reason {
15    use embedded_tls::alert::AlertDescription as Alert;
16    use umsh_ulcp::bridge::Reason;
17    match error {
18        TlsError::InvalidSignature | TlsError::InvalidCertificate | TlsError::InvalidPrivateKey => {
19            Reason::Authentication
20        }
21        TlsError::HandshakeAborted(
22            _,
23            Alert::BadCertificate
24            | Alert::UnsupportedCertificate
25            | Alert::CertificateRevoked
26            | Alert::CertificateExpired
27            | Alert::CertificateUnknown
28            | Alert::UnknownCa
29            | Alert::AccessDenied
30            | Alert::CertificateRequired
31            | Alert::DecryptError,
32        ) => Reason::Authentication,
33        TlsError::IoError | TlsError::Io(_) => Reason::Tcp,
34        _ => Reason::Tls,
35    }
36}

38/// Private to the client task. Never expose this value through ULCP or logging.
39pub struct IdentityProvider<R> {
40    rng: R,
41    key: SigningKey,
42    certificate: Vec<u8, 512>,
43    verifier: PinnedVerifier,
44    signed: bool,
45}
46
47impl<R: CryptoRngCore> IdentityProvider<R> {
48    pub fn new(seed: &[u8; 32], server: &[u8; 32], rng: R) -> Result<Self, TlsError> {
49        let key = SigningKey::from_bytes(seed);
50        let certificate = certificate(&key);
51        Ok(Self {
52            rng,
53            key,
54            certificate,
55            verifier: PinnedVerifier::new(server)?,
56            signed: false,
57        })
58    }
59
60    pub fn public_key(&self) -> [u8; 32] {
61        self.key.verifying_key().to_bytes()
62    }
63
64    /// Check after `open`: an abbreviated handshake must not bypass mutual auth.
65    pub fn authenticated(&self) -> bool {
66        self.signed && self.verifier.verified
67    }
68
69    pub fn authentication_failed(&self) -> bool {
70        self.verifier.failed
71    }
72}

74impl<R: CryptoRngCore> CryptoProvider for IdentityProvider<R> {
75    type CipherSuite = Aes128GcmSha256;
76    type Signature = [u8; 64];
77
78    fn rng(&mut self) -> impl CryptoRngCore {
79        &mut self.rng
80    }
81
82    fn verifier(&mut self) -> Result<&mut impl TlsVerifier<Aes128GcmSha256>, TlsError> {
83        Ok(&mut self.verifier)
84    }
85
86    fn signer(
87        &mut self,
88    ) -> Result<(impl signature::SignerMut<[u8; 64]>, SignatureScheme), TlsError> {
89        Ok((
90            DeviceSigner {
91                key: &self.key,
92                signed: &mut self.signed,
93            },
94            SignatureScheme::Ed25519,
95        ))
96    }
97
98    fn client_cert(&mut self) -> Option<Certificate<impl AsRef<[u8]>>> {
99        Some(Certificate::X509(self.certificate.as_slice()))
100    }
101}

103struct DeviceSigner<'a> {
104    key: &'a SigningKey,
105    signed: &'a mut bool,
106}
107
108impl signature::SignerMut<[u8; 64]> for DeviceSigner<'_> {
109    fn try_sign(&mut self, message: &[u8]) -> Result<[u8; 64], signature::Error> {
110        let signature: Signature = self.key.try_sign(message)?;
111        *self.signed = true;
112        Ok(signature.to_bytes())
113    }
114}

116pub struct PinnedVerifier {
117    key: VerifyingKey,
118    transcript: Option<[u8; 32]>,
119    verified: bool,
120    failed: bool,
121}
122
123impl PinnedVerifier {
124    pub fn new(key: &[u8; 32]) -> Result<Self, TlsError> {
125        let key = VerifyingKey::from_bytes(key).map_err(|_| TlsError::InvalidSignature)?;
126        if key.is_weak() || key.to_edwards().compress().to_bytes() != key.to_bytes() {
127            return Err(TlsError::InvalidSignature);
128        }
129        Ok(Self {
130            key,
131            transcript: None,
132            verified: false,
133            failed: false,
134        })
135    }
136}
137
138impl TlsVerifier<Aes128GcmSha256> for PinnedVerifier {
139    fn set_hostname_verification(&mut self, _: &str) -> Result<(), TlsError> {
140        // The hostname supplies SNI; the pin supplies authentication.
141        Ok(())
142    }
143
144    fn verify_certificate(
145        &mut self,
146        transcript: &Sha256,
147        _: CertificateRef,
148    ) -> Result<(), TlsError> {
149        self.transcript = Some(transcript.clone().finalize().into());
150        self.verified = false;
151        Ok(())
152    }
153
154    fn verify_signature(&mut self, verify: CertificateVerifyRef) -> Result<(), TlsError> {
155        self.failed = true;
156        self.verified = false;
157        if verify.signature_scheme != SignatureScheme::Ed25519 {
158            return Err(TlsError::InvalidSignatureScheme);
159        }
160        let hash = self.transcript.take().ok_or(TlsError::InvalidSignature)?;
161        // RFC 8446 section 4.4.3: spaces || context || zero || transcript hash.
162        let mut message = Vec::<u8, 130>::new();
163        message.extend_from_slice(&[0x20; 64]).unwrap();
164        message
165            .extend_from_slice(b"TLS 1.3, server CertificateVerify\0")
166            .unwrap();
167        message.extend_from_slice(&hash).unwrap();
168        let signature =
169            Signature::from_slice(verify.signature).map_err(|_| TlsError::InvalidSignature)?;
170        self.key
171            .verify_strict(&message, &signature)
172            .map_err(|_| TlsError::InvalidSignature)?;
173        self.verified = true;
174        self.failed = false;
175        Ok(())
176    }
177}

179// A fixed-shape, self-signed X.509 v1 certificate. No untrusted DER is parsed
180// here. Lengths depend only on these constants and fixed-size keys/signatures.
181fn tlv(tag: u8, body: &[u8]) -> Vec<u8, 512> {
182    let mut out = Vec::new();
183    out.push(tag).unwrap();
184    if body.len() < 128 {
185        out.push(body.len() as u8).unwrap();
186    } else if body.len() < 256 {
187        out.extend_from_slice(&[0x81, body.len() as u8]).unwrap();
188    } else {
189        out.extend_from_slice(&[0x82, (body.len() >> 8) as u8, body.len() as u8])
190            .unwrap();
191    }
192    out.extend_from_slice(body).unwrap();
193    out
194}

196fn certificate(key: &SigningKey) -> Vec<u8, 512> {
197    const ALGORITHM: &[u8] = &[0x30, 0x05, 0x06, 0x03, 0x2b, 0x65, 0x70];
198    const NAME: &[u8] = &[
199        0x30, 0x0f, 0x31, 0x0d, 0x30, 0x0b, 0x06, 0x03, 0x55, 0x04, 0x03, 0x0c, 0x04, b'U', b'M',
200        b'S', b'H',
201    ];
202    let mut validity = Vec::<u8, 512>::new();
203    validity
204        .extend_from_slice(&tlv(0x17, b"240101000000Z"))
205        .unwrap();
206    validity
207        .extend_from_slice(&tlv(0x17, b"491231235959Z"))
208        .unwrap();
209    let mut public = Vec::<u8, 512>::new();
210    public.push(0).unwrap();
211    public
212        .extend_from_slice(&key.verifying_key().to_bytes())
213        .unwrap();
214    let mut spki = Vec::<u8, 512>::from_slice(ALGORITHM).unwrap();
215    spki.extend_from_slice(&tlv(0x03, &public)).unwrap();
216    let mut tbs = Vec::<u8, 512>::from_slice(&[0x02, 0x01, 0x01]).unwrap();
217    for part in [
218        ALGORITHM,
219        NAME,
220        &tlv(0x30, &validity),
221        NAME,
222        &tlv(0x30, &spki),
223    ] {
224        tbs.extend_from_slice(part).unwrap();
225    }
226    let tbs = tlv(0x30, &tbs);
227    let signature: Signature = key.sign(&tbs);
228    let mut bits = Vec::<u8, 512>::new();
229    bits.push(0).unwrap();
230    bits.extend_from_slice(&signature.to_bytes()).unwrap();
231    let mut body = tbs;
232    body.extend_from_slice(ALGORITHM).unwrap();
233    body.extend_from_slice(&tlv(0x03, &bits)).unwrap();
234    tlv(0x30, &body)
235}

237#[cfg(test)]
238mod tests {
239    use super::*;
240
241    #[test]
242    fn pinned_verifier_rejects_forgery_and_wrong_handshake_context() {
243        let key = SigningKey::from_bytes(&[7; 32]);
244        let mut verifier = PinnedVerifier::new(&key.verifying_key().to_bytes()).unwrap();
245        let hash = [8; 32];
246        let mut message = Vec::<u8, 130>::from_slice(&[0x20; 64]).unwrap();
247        message
248            .extend_from_slice(b"TLS 1.3, server CertificateVerify\0")
249            .unwrap();
250        message.extend_from_slice(&hash).unwrap();
251        let signature = key.sign(&message).to_bytes();
252        verifier.transcript = Some(hash);
253        verifier
254            .verify_signature(CertificateVerifyRef {
255                signature_scheme: SignatureScheme::Ed25519,
256                signature: &signature,
257            })
258            .unwrap();
259        assert!(verifier.verified);
260        // Proof cannot be reused without a fresh Certificate transcript.
261        assert!(
262            verifier
263                .verify_signature(CertificateVerifyRef {
264                    signature_scheme: SignatureScheme::Ed25519,
265                    signature: &signature
266                })
267                .is_err()
268        );
269        verifier.transcript = Some([9; 32]);
270        assert!(
271            verifier
272                .verify_signature(CertificateVerifyRef {
273                    signature_scheme: SignatureScheme::Ed25519,
274                    signature: &signature
275                })
276                .is_err()
277        );
278        verifier.transcript = Some(hash);
279        let mut forged = signature;
280        forged[0] ^= 1;
281        assert!(
282            verifier
283                .verify_signature(CertificateVerifyRef {
284                    signature_scheme: SignatureScheme::Ed25519,
285                    signature: &forged
286                })
287                .is_err()
288        );
289        assert!(verifier.failed);
290    }
