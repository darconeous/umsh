1//! The device side of an exchange: tokens, retained replies, and
2//! cursors.
3//!
4//! The engine performs no I/O and runs no ULCP command. It reads an
5//! arriving request and decides whether the request needs to be executed
6//! at all. When it does, the caller dispatches the embedded frame through
7//! whatever machinery serves its local link and hands back the whole
8//! reply, and the engine wraps the share one payload carries.
9//!
10//! A reply is retained only where executing the request again would be
11//! wrong. A command that changes something is retained so that its
12//! retransmission is answered rather than executed twice. A read too
13//! large for one payload is retained whole, so that every continuation is
14//! cut from it and the read comes back as one execution produced it. A
15//! read that fits one payload is not retained at all: running it again
16//! changes nothing, so its retransmission is simply answered afresh.
17//!
18//! Authorization is not the engine's concern: a request reaches it only
19//! after its source has been checked against the administrator list.
20
21use umsh_ulcp::frame::{self, Cmd, Frame};

27/// An Ed25519 public key naming a node.
28pub type PublicKey = [u8; 32];
29
30/// How many administrators the reference engine retains a reply for at
31/// once.
32///
33/// The spec requires only the most recently active administrator's, and
34/// bounds nothing above that. An entry is occupied only by a write or a
35/// read in progress, and costs a small header—the replies themselves
36/// share the engine's pool—so four covers a device managed by a handful
37/// of people at once.
38pub const CACHE_ENTRIES: usize = 4;
39
40/// Why an arriving payload produced nothing.
41#[derive(Clone, Copy, Debug, PartialEq, Eq)]
42pub enum DropReason {
43    /// The payload was too short to hold a token, so there is nothing to
44    /// correlate a response with and no way to report the problem.
45    NoToken,
46    /// The output buffer could not hold the response. The caller sized
47    /// it below the transport's own payload limit.
48    NoRoom,
49}
50
51/// What the caller must do with an arriving payload.
52#[derive(Clone, Copy, Debug, PartialEq, Eq)]
53pub enum Ingress<'p> {
54    /// Send nothing and account for it.
55    Drop(DropReason),
56    /// `out[..len]` is a complete response payload, and nothing was
57    /// executed: a retransmission, the next fragment of a continued read,
58    /// or an error the engine answered on its own.
59    Respond { len: usize },
60    /// Dispatch the frame, then call [`DeviceEngine::complete`].
61    Dispatch(Dispatch<'p>),
62}
63
64/// A request the engine wants executed.
65#[derive(Clone, Copy, Debug, PartialEq, Eq)]
66pub struct Dispatch<'p> {
67    /// Exactly one ULCP frame, in the grammar of the local bindings. Its
68    /// TID is zero and receivers ignore it.
69    pub frame: &'p [u8],
70    /// The largest reply frame [`DeviceEngine::complete`] takes back. A
71    /// read may answer with as much as the engine retains, because its
72    /// continuations are cut from the retained reply. Anything else must
73    /// fit one response payload, the envelope's worst case already
74    /// deducted, because there is no continuing it.
75    pub budget: usize,
76    /// A reset-class command, which is answered by no response payload
77    /// at all. The caller executes it and completes with an empty reply.
78    pub resets: bool,
79}

82    /// The parsed command, absent when this build does not define it.
83    /// Such a frame is the caller's to answer `STATUS_INVALID_COMMAND`;
84    /// the engine only needed enough of it to apply the cursor rules.
85    pub fn command(&self) -> Option<Cmd> {
86        Frame::parse(self.frame).ok().and_then(|f| f.command())
87    }

90/// Why a response could not be assembled.
91#[derive(Clone, Copy, Debug, PartialEq, Eq)]
92pub enum CompleteError {
93    /// [`DeviceEngine::complete`] was called without a dispatch in
94    /// flight.
95    NotDispatched,
96    /// The reply exceeds the budget the dispatch handed out, or its
97    /// leading bytes leave a payload no room to carry any of it.
98    TooLarge,
99}

101/// A cursor as this engine issues them.
102///
103/// ```text
104/// +-------+--------+--------+
105/// | NONCE | SERIAL | OFFSET |
106/// +-------+--------+--------+
107///    2 B     2 B      2 B
108/// ```
109///
110/// Opaque to the administrator, which returns it byte for byte. A cursor
111/// is a position in a reply the engine retains, so each field answers one
112/// way that reply can be gone: NONCE a reboot, and SERIAL the
113/// administrator's retained entry having moved on to another exchange,
114/// or to nobody after an eviction or a reset. OFFSET is where in the
115/// reply's trailing content the next fragment begins.
116const CURSOR_LEN: usize = 6;
117
118fn encode_cursor(nonce: u16, serial: u16, offset: u16) -> [u8; CURSOR_LEN] {
119    let mut out = [0u8; CURSOR_LEN];
120    out[0..2].copy_from_slice(&nonce.to_be_bytes());
121    out[2..4].copy_from_slice(&serial.to_be_bytes());
122    out[4..6].copy_from_slice(&offset.to_be_bytes());
123    out
124}
125
126/// A 16-bit FNV-1a of the request frame, binding a continuation to the
127/// read whose reply it is cut from. A continuation repeats the frame
128/// verbatim—the cursor rides in the envelope, not the frame—so an
129/// honest continuation always matches.
130fn request_tag(frame: &[u8]) -> u16 {
131    let mut hash: u32 = 0x811C_9DC5;
132    for &byte in frame {
133        hash ^= u32::from(byte);
134        hash = hash.wrapping_mul(0x0100_0193);
135    }
136    // Fold to 16 bits rather than truncate, so every input byte reaches
137    // the result.
138    ((hash >> 16) ^ hash) as u16
139}

141/// Whether this command initiates a reset, and so is answered by no
142/// response payload.
143///
144/// `CMD_RESTORE` qualifies because a device that has a saved snapshot
145/// resets into it; one that has none answers normally, but the
146/// administrator cannot know which in advance, so the binding treats the
147/// command as reset-class either way and the administrator confirms
148/// delivery with a MAC acknowledgment. `CMD_REBOOT` qualifies for the
149/// same reason: a board restarts and says nothing, while one that cannot
150/// answers `STATUS_UNIMPLEMENTED`.
151const fn reset_class(cmd: Cmd) -> bool {
152    matches!(
153        cmd,
154        Cmd::Reset | Cmd::Restore | Cmd::FactoryReset | Cmd::Reboot
155    )
156}

158/// One administrator's most recent exchange, where it is one that has to
159/// be retained.
160///
161/// What is kept is the whole reply, not the payload that carried it, and
162/// it lives in the engine's shared pool at its own length. A write's
163/// reply fits one payload and goes out whole; a read that does not fit
164/// goes out a fragment per exchange, every fragment cut from this one
165/// reply. The fragment fields record which share the exchange under
166/// `token` carried, so a retransmission repeats exactly that.
167#[derive(Clone, Copy, Default)]
168struct Entry {
169    key: PublicKey,
170    token: Token,
171    /// Where the reply lies in the pool.
172    start: u16,
173    len: u16,
174    /// Leading octets every fragment repeats: all of them, for a reply
175    /// with no trailing content.
176    prefix: u16,
177    /// The share of the trailing content `token`'s exchange carried.
178    resume: u16,
179    take: u16,
180    /// [`request_tag`] of the frame the reply answers.
181    tag: u16,
182    /// What this reply's cursors name it by.
183    serial: u16,
184    /// When this administrator was last heard from, for eviction.
185    last_ms: u64,
186    occupied: bool,
187}
188
189impl Entry {
190    fn reply<'p>(&self, pool: &'p [u8]) -> &'p [u8] {
191        let start = usize::from(self.start);
192        &pool[start..start + usize::from(self.len)]
193    }
194
195    fn trailing_len(&self) -> usize {
196        usize::from(self.len - self.prefix)
197    }
198
199    /// Make `cut` the share this entry's exchange carries.
200    fn carry(&mut self, cut: Cut) {
201        // Every position lies within the reply, which `DeviceEngine::new`
202        // holds to what a cursor's offset addresses.
203        self.prefix = cut.prefix as u16;
204        self.resume = cut.resume as u16;
205        self.take = cut.take as u16;
206    }
207
208    /// Write the response payload carrying this entry's share of the
209    /// reply it keeps in `pool`.
210    fn encode(&self, pool: &[u8], nonce: u16, out: &mut [u8]) -> Option<usize> {
211        let (prefix, trailing) = self.reply(pool).split_at(usize::from(self.prefix));
212        let start = usize::from(self.resume);
213        let end = start + usize::from(self.take);
214        let fragment = &trailing[start..end];
215        let remaining = trailing.len() - end;
216
217        let cursor = encode_cursor(nonce, self.serial, end as u16);
218        let mut envelope = Envelope::new(self.token, &[]);
219        if remaining > 0 {
220            envelope = envelope
221                .with_cursor(&cursor)
222                .with_remaining(remaining as u32);
223        }
224        // The frame is the prefix followed by the fragment, which are not
225        // contiguous in the reply, so the envelope goes first and the two
226        // parts after it.
227        let head = envelope.encode(out).ok()?;
228        let len = head + prefix.len() + fragment.len();
229        let (lead, rest) = out.get_mut(head..len)?.split_at_mut(prefix.len());
230        lead.copy_from_slice(prefix);
231        rest.copy_from_slice(fragment);
232        Some(len)
233    }
234}

236/// The exchange between [`DeviceEngine::begin`] and
237/// [`DeviceEngine::complete`].
238#[derive(Clone, Copy)]
239struct InFlight {
240    key: PublicKey,
241    token: Token,
242    tag: u16,
243    /// A read, whose reply is retained only when it does not fit one
244    /// payload.
245    read: bool,
246    budget: usize,
247    /// Frame octets one response payload carries.
248    room: usize,
249    now_ms: u64,
250}
251
252/// The device half of the Node Management binding.
253///
254/// `PAYLOAD` is the largest response payload the transport can carry.
255/// `POOL` is the octets every retained reply shares, each held at its own
256/// length and the least recently active administrator's evicted when a
257/// new one needs the room. It is also the largest read the engine can
258/// serve, since a read too large for one payload is retained whole so
259/// that its continuations can be cut from it. `ENTRIES` is how many
260/// administrators can hold a retained reply at once.
261pub struct DeviceEngine<
262    const PAYLOAD: usize,
263    const POOL: usize,
264    const ENTRIES: usize = CACHE_ENTRIES,
265> {
266    nonce: u16,
267    /// The serial the next retained reply takes.
268    serial: u16,
269    entries: [Entry; ENTRIES],
270    /// Every retained reply, packed from the start by [`Self::compact`].
271    pool: [u8; POOL],
272    in_flight: Option<InFlight>,
273}
274
275impl<const PAYLOAD: usize, const POOL: usize, const ENTRIES: usize>
276    DeviceEngine<PAYLOAD, POOL, ENTRIES>
277{
278    const ADDRESSABLE: () = assert!(
279        POOL <= u16::MAX as usize,
280        "a cursor addresses at most 64 KiB of reply"
281    );
282
283    /// Build an engine. `nonce` is drawn once per boot from the
284    /// cryptographic RNG: it is what stops a cursor issued before a
285    /// reboot from being honored after one, when the serials have
286    /// started over.
287    pub fn new(nonce: u16) -> Self {
288        let () = Self::ADDRESSABLE;
289        Self {
290            nonce,
291            serial: 0,
292            entries: [Entry::default(); ENTRIES],
293            pool: [0; POOL],
294            in_flight: None,
295        }
296    }

298    /// Read an arriving Node Management Request payload, the payload type
299    /// byte already stripped.
300    ///
301    /// `now_ms` orders the retained entries for eviction.
302    pub fn begin<'p>(
303        &mut self,
304        from: &PublicKey,
305        payload: &'p [u8],
306        now_ms: u64,
307        out: &mut [u8],
308    ) -> Ingress<'p> {
309        self.in_flight = None;
310
311        // The token leads the payload, so it survives anything the
312        // option block does. Every envelope that has one is answered;
313        // only a payload too short to hold one is dropped, since a
314        // response nothing can be correlated with is no answer at all.
315        let &[token0, token1, ..] = payload else {
316            return Ingress::Drop(DropReason::NoToken);
317        };
318        let token = [token0, token1];
319
320        // A retransmission of a retained exchange is answered from the
321        // retained reply without executing anything—before the request
322        // is even looked at, since the point is not to look at it again.
323        if let Some(index) = self.retained(from, token) {
324            self.entries[index].last_ms = now_ms;
325            return self.respond(index, out);
326        }
327
328        let request = match Envelope::parse(payload) {
329            Ok(request) => request,
330            Err(EnvelopeError::UnknownCritical(_)) => {
331                return self.answer(from, token, Status::UNIMPLEMENTED, out);
332            }
333            Err(EnvelopeError::InvalidOptionValue(crate::envelope::OPT_CURSOR)) => {
334                return self.answer(from, token, Status::CURSOR_INVALID, out);
335            }
336            Err(_) => return self.answer(from, token, Status::PARSE_ERROR, out),
337        };
338
339        // A frame that does not parse is answered rather than dropped:
340        // the administrator learns its request was heard and malformed.
341        let Ok(parsed) = Frame::parse(request.frame) else {
342            return self.answer(from, token, Status::PARSE_ERROR, out);
343        };
344        let cmd = parsed.command();
345        let read = cmd.is_some_and(continuable);
346        let room = Self::room(out);
347
348        if let Some(cursor) = request.cursor {
349            // A cursor on something that is not a read at all, including
350            // a command this build does not define.
351            if !read {
352                return self.answer(from, token, Status::INVALID_ARGUMENT, out);
353            }
354            let Some((index, offset)) = self.continued(from, cursor, request.frame) else {
355                return self.answer(from, token, Status::CURSOR_INVALID, out);
356            };
357            // A continuation executes nothing. It is the next fragment of
358            // the reply its read began with, so the read comes back as
359            // one execution produced it, however the device has changed
360            // since.
361            let Some(next) = cut(self.entries[index].reply(&self.pool), offset, room) else {
362                return Ingress::Drop(DropReason::NoRoom);
363            };
364            let entry = &mut self.entries[index];
365            entry.token = token;
366            entry.last_ms = now_ms;
367            entry.carry(next);
368            return self.respond(index, out);
369        }
370
371        // A new exchange ends the one retained before it, whether or not
372        // this one is retained in turn. Tokens need only differ from the
373        // previous exchange's, so an older retained token is one the
374        // administrator may legitimately use again.
375        self.release(from);
376        let budget = if read { POOL } else { room.min(POOL) };
377        self.in_flight = Some(InFlight {
378            key: *from,
379            token,
380            // Binds any cursor this exchange issues to the read that
381            // issued it.
382            tag: request_tag(request.frame),
383            read,
384            budget,
385            room,
386            now_ms,
387        });
388        Ingress::Dispatch(Dispatch {
389            frame: request.frame,
390            budget,
391            resets: cmd.is_some_and(reset_class),
392        })
393    }
394
395    /// Write the response payload carrying the first share of the reply
396    /// the dispatch produced into `out`, retaining the reply unless it is
397    /// a read that fits whole.
398    ///
399    /// `reply` is empty for a reset-class command, which is answered by
400    /// no payload at all, and the result is then `None`; otherwise it is
401    /// the payload's length.
402    pub fn complete(
403        &mut self,
404        reply: &[u8],
405        out: &mut [u8],
406    ) -> Result<Option<usize>, CompleteError> {
407        let in_flight = self.in_flight.take().ok_or(CompleteError::NotDispatched)?;
408
409        if reply.is_empty() {
410            // A reset takes the retained entries with it: after one, a
411            // retransmitted request is executed again, with the same
412            // result, and a cursor has nothing left to continue.
413            self.forget_retained();
414            return Ok(None);
415        }
416
417        if reply.len() > in_flight.budget {
418            return Err(CompleteError::TooLarge);
419        }
420        let first = cut(reply, 0, in_flight.room).ok_or(CompleteError::TooLarge)?;
421        if in_flight.read && first.remaining == 0 {
422            // Running a read again changes nothing, so its retransmission
423            // is simply answered afresh, with current values.
424            return Envelope::new(in_flight.token, reply)
425                .encode(out)
426                .map(Some)
427                .map_err(|_| CompleteError::TooLarge);
428        }
429        let index = self
430            .retain(
431                &in_flight.key,
432                in_flight.token,
433                reply,
434                first,
435                in_flight.tag,
436                in_flight.now_ms,
437            )
438            .ok_or(CompleteError::TooLarge)?;
439        self.entries[index]
440            .encode(&self.pool, self.nonce, out)
441            .map(Some)
442            .ok_or(CompleteError::TooLarge)
443    }
444
445    /// Forget every retained reply, as a reset does.
446    pub fn forget_retained(&mut self) {
447        self.entries = [Entry::default(); ENTRIES];
448    }
449
450    /// Replace an unsent lifecycle success with a failure for the same token.
451    /// A retry must not replay OK after its pending handoff was canceled.
452    pub fn fail_retained(&mut self, from: &PublicKey, token: Token) {
453        let Some(index) = self.retained(from, token) else {
454            return;
455        };
456        let last_ms = self.entries[index].last_ms;
457        let mut buf = [0; 8];
458        let len = frame::last_status(&mut buf, 0, Status::FAILURE).expect("status fits");
459        let failure = &buf[..len];
460        // Retaining afresh releases the success first, so even a failure
461        // that could not be retained leaves no success to replay.
462        let _ = cut(failure, 0, failure.len())
463            .and_then(|whole| self.retain(from, token, failure, whole, 0, last_ms));
464    }

466    /// Frame octets one response payload carries, measured against the
467    /// worst envelope rather than any particular one, so a fragment that
468    /// acquires a cursor still fits the payload it was cut for.
469    fn room(out: &[u8]) -> usize {
470        PAYLOAD.min(out.len()).saturating_sub(OVERHEAD_MAX)
471    }
472
473    /// Answer a request the engine can decide on its own. Like any new
474    /// exchange it ends the retained one; nothing of its own is retained,
475    /// since the same request always draws the same answer.
476    fn answer(
477        &mut self,
478        from: &PublicKey,
479        token: Token,
480        status: Status,
481        out: &mut [u8],
482    ) -> Ingress<'static> {
483        self.release(from);
484        let mut buf = [0u8; 8];
485        let Ok(len) = frame::last_status(&mut buf, frame::TID_UNSOLICITED, status) else {
486            return Ingress::Drop(DropReason::NoRoom);
487        };
488        match Envelope::new(token, &buf[..len]).encode(out) {
489            Ok(len) => Ingress::Respond { len },
490            Err(_) => Ingress::Drop(DropReason::NoRoom),
491        }
492    }

494    /// Forget whatever `key` has retained.
495    fn release(&mut self, key: &PublicKey) {
496        for entry in self.entries.iter_mut() {
497            if entry.occupied && &entry.key == key {
498                entry.occupied = false;
499            }
500        }
501    }
502
503    /// Write the response payload for the share entry `index` carries.
504    fn respond(&self, index: usize, out: &mut [u8]) -> Ingress<'static> {
505        match self.entries[index].encode(&self.pool, self.nonce, out) {
506            Some(len) => Ingress::Respond { len },
507            None => Ingress::Drop(DropReason::NoRoom),
508        }
509    }

511    /// The retained entry a cursor continues, and where in its reply's
512    /// trailing content, or `None` if the cursor cannot be honored.
513    fn continued(&self, from: &PublicKey, cursor: &[u8], frame: &[u8]) -> Option<(usize, usize)> {
514        let cursor: [u8; CURSOR_LEN] = cursor.try_into().ok()?;
515        let nonce = u16::from_be_bytes([cursor[0], cursor[1]]);
516        let serial = u16::from_be_bytes([cursor[2], cursor[3]]);
517        let offset = usize::from(u16::from_be_bytes([cursor[4], cursor[5]]));
518        if nonce != self.nonce {
519            return None;
520        }
521        // Retained under the administrator that began the read, so no
522        // other administrator's cursor reaches it.
523        let tag = request_tag(frame);
524        let index = self.entries.iter().position(|entry| {
525            entry.occupied && &entry.key == from && entry.serial == serial && entry.tag == tag
526        })?;
527        // Strictly inside the trailing content: a cursor is issued only
528        // while something remains.
529        (offset < self.entries[index].trailing_len()).then_some((index, offset))
530    }
531
532    fn retained(&self, key: &PublicKey, token: Token) -> Option<usize> {
533        self.entries
534            .iter()
535            .position(|entry| entry.occupied && &entry.key == key && entry.token == token)
536    }
537
538    /// Hold `reply` as `key`'s most recent exchange, carrying `cut`, and
539    /// say where.
540    ///
541    /// The administrator's earlier entry gives way first. Then the least
542    /// recently active administrators are evicted, one at a time, until
543    /// both an entry and the reply's octets are free—which by
544    /// construction never evicts the administrator being answered.
545    fn retain(
546        &mut self,
547        key: &PublicKey,
548        token: Token,
549        reply: &[u8],
550        cut: Cut,
551        tag: u16,
552        now_ms: u64,
553    ) -> Option<usize> {
554        if reply.len() > POOL {
555            // Unretainable. Nothing this crate builds gets here; a caller
556            // that overruns its own budget does.
557            return None;
558        }
559        self.release(key);
560        loop {
561            let used: usize = self
562                .entries
563                .iter()
564                .filter(|entry| entry.occupied)
565                .map(|entry| usize::from(entry.len))
566                .sum();
567            let unoccupied = self.entries.iter().any(|entry| !entry.occupied);
568            if unoccupied && used + reply.len() <= POOL {
569                break;
570            }
571            let stalest = self
572                .entries
573                .iter()
574                .enumerate()
575                .filter(|(_, entry)| entry.occupied)
576                .min_by_key(|(_, entry)| entry.last_ms)
577                .map(|(index, _)| index)?;
578            self.entries[stalest].occupied = false;
579        }
580        let start = self.compact();
581        let slot = self.entries.iter().position(|entry| !entry.occupied)?;
582        self.pool[start..start + reply.len()].copy_from_slice(reply);
583
584        let serial = self.serial;
585        self.serial = serial.wrapping_add(1);
586        let entry = &mut self.entries[slot];
587        entry.key = *key;
588        entry.token = token;
589        // Within the pool, which `new` holds to what a `u16` addresses.
590        entry.start = start as u16;
591        entry.len = reply.len() as u16;
592        entry.carry(cut);
593        entry.tag = tag;
594        // A new serial for every reply, so a cursor issued against the
595        // one this replaces names nothing.
596        entry.serial = serial;
597        entry.last_ms = now_ms;
598        entry.occupied = true;
599        Some(slot)
600    }
601
602    /// Pack every retained reply against the start of the pool, and say
603    /// where the free space begins.
604    fn compact(&mut self) -> usize {
605        let mut packed = [false; ENTRIES];
606        let mut next = 0;
607        // Lowest start first, so each reply moves down over space already
608        // vacated and never over one still waiting to move.
609        while let Some(index) = (0..ENTRIES)
610            .filter(|&index| self.entries[index].occupied && !packed[index])
611            .min_by_key(|&index| self.entries[index].start)
612        {
613            let entry = &mut self.entries[index];
614            let start = usize::from(entry.start);
615            let len = usize::from(entry.len);
616            self.pool.copy_within(start..start + len, next);
617            entry.start = next as u16;
618            next += len;
619            packed[index] = true;
620        }
621        next
622    }

648    /// The status a `PROP_LAST_STATUS` response reports.
649    fn reported_status(payload: &[u8]) -> Status {
650        let envelope = Envelope::parse(payload).expect("envelope");
651        let parsed = Frame::parse(envelope.frame).expect("frame");
652        assert_eq!(parsed.command(), Some(Cmd::PropIs));
653        let (key, consumed) = umsh_ulcp::pui::decode(parsed.payload).expect("key");
654        assert_eq!(key, prop::LAST_STATUS);
655        let (status, _) = umsh_ulcp::pui::decode(&parsed.payload[consumed..]).expect("status");
656        Status(status)
657    }

659    /// What came of one exchange.
660    #[derive(Debug, PartialEq, Eq)]
661    enum Answer {
662        /// The engine had the request executed; this is the response.
663        Executed(Vec<u8>),
664        /// Answered without executing anything.
665        Answered(Vec<u8>),
666    }
667
668    impl Answer {
669        fn payload(&self) -> &[u8] {
670            match self {
671                Self::Executed(payload) | Self::Answered(payload) => payload,
672            }
673        }
674    }
675
676    /// Run one exchange, executing it—if the engine asks—as a device
677    /// whose whole reply is `reply`.
678    fn exchange(
679        engine: &mut Engine,
680        from: &PublicKey,
681        token: Token,
682        frame: &[u8],
683        cursor: Option<&[u8]>,
684        reply: &[u8],
685        now_ms: u64,
686    ) -> Answer {
687        let mut payload = [0u8; PAYLOAD];
688        let mut envelope = Envelope::new(token, frame);
689        if let Some(cursor) = cursor {
690            envelope = envelope.with_cursor(cursor);
691        }
692        let len = envelope.encode(&mut payload).expect("encode");
693        let mut out = [0u8; PAYLOAD];
694        match engine.begin(from, &payload[..len], now_ms, &mut out) {
695            Ingress::Respond { len } => Answer::Answered(out[..len].to_vec()),
696            Ingress::Dispatch(_) => {
697                let len = engine
698                    .complete(reply, &mut out)
699                    .expect("complete")
700                    .expect("a response");
701                Answer::Executed(out[..len].to_vec())
702            }
703            Ingress::Drop(reason) => panic!("dropped: {reason:?}"),
704        }
705    }

707    /// The peer table read whole: a request, and a reply too large for
708    /// one payload whose value counts up from `seed`, so that two
709    /// executions can be told apart.
710    fn peers_read(len: usize, seed: u8) -> (Vec<u8>, Vec<u8>) {
711        let mut request = [0u8; 8];
712        let request_len = get(prop::DEV_PEERS, &mut request);
713        let value: Vec<u8> = (0..len).map(|i| seed.wrapping_add(i as u8)).collect();
714        let mut reply = vec![0u8; len + 8];
715        let reply_len = frame::prop_is(&mut reply, 0, prop::DEV_PEERS, &value).unwrap();
716        reply.truncate(reply_len);
717        (request[..request_len].to_vec(), reply)
718    }

720    /// A write of the device name and the reply that echoes it: an
721    /// exchange that changes something, and so is retained.
722    fn rename(name: &[u8]) -> (Vec<u8>, Vec<u8>) {
723        let mut request = [0u8; 64];
724        let request_len = frame::prop_set(&mut request, 0, prop::DEV_NAME, name).unwrap();
725        let mut reply = [0u8; 64];
726        let reply_len = frame::prop_is(&mut reply, 0, prop::DEV_NAME, name).unwrap();
727        (request[..request_len].to_vec(), reply[..reply_len].to_vec())
728    }
