1//! Managing a device over the mesh, from a node.
2//!
3//! [`Exchange`] knows the binding and nothing else: it hands out payloads
4//! to send and reads the ones that come back. This binds it to a
5//! [`PeerConnection`], so an administrator's operation becomes a matter of
6//! servicing it until it finishes.
7//!
8//! The node layer delivers packets from inside the coordinator's borrow,
9//! so a response is queued where it arrives and read where it can be
10//! acted on. The caller owns the clock and the pump: it services the
11//! exchange, then drives its `Host` until the deadline the service call
12//! named, whichever comes first.
13

30/// Responses held between the subscription that takes them in and the
31/// service call that reads them.
32///
33/// An administrator has one exchange outstanding at a time, so anything
34/// beyond a couple of payloads is a duplicate or a stray; a short queue
35/// keeps a talkative peer from growing this without bound.
36const INBOX: usize = 4;
37
38/// How large a reply this reassembles before giving up.
39///
40/// A continued read costs one exchange per fragment, so a reply of this
41/// size is already a slow operation over LoRa; the ceiling is what keeps
42/// a device that never stops issuing cursors from consuming the host.
43pub const REPLY_MAX: usize = 8 * 1024;
44
45/// Why an operation could not be started.
46#[derive(Clone, Copy, Debug, PartialEq, Eq)]
47pub enum BeginError {
48    /// The request frame is larger than a Node Management payload holds.
49    RequestTooLarge,
50    /// An exchange is already outstanding. An administrator may have only
51    /// one with a given device.
52    Busy,
53}
54
55/// Why an outstanding exchange could not be carried further.
56#[derive(Clone, Copy, Debug, PartialEq, Eq)]
57pub enum ManagementError<E> {
58    /// The node could not send.
59    Transport(E),
60    /// Nothing is outstanding to service.
61    Idle,
62}
63
64/// How often a service call looks again at an attempt still with the MAC.
65/// Checking costs nothing on the air; it only bounds how long a caller
66/// that sleeps until the deadline goes without noticing an acknowledgment.
67const DELIVERY_RECHECK_MS: u64 = 1_000;
68
69/// What the caller should do next.
70#[derive(Clone, Copy, Debug, PartialEq, Eq)]
71pub enum Progress {
72    /// Nothing further until `deadline_ms`. Pump the host until then; a
73    /// response or acknowledgment arriving earlier makes the next service
74    /// call productive sooner.
75    Waiting { deadline_ms: u64 },
76    /// The exchange is over.
77    Done(Outcome),
78}

80/// One administrator's dealings with one device.
81pub struct NodeManager<M: MacBackend> {
82    peer: PeerConnection<LocalNode<M>>,
83    inbox: Rc<RefCell<VecDeque<Vec<u8>>>>,
84    /// Payloads taken in that turned out to belong to nobody: another
85    /// exchange's token, an unauthenticated sender, or more than the
86    /// inbox holds.
87    stray: Rc<Cell<u32>>,
88    _subscription: Subscription,
89    exchange: Option<Exchange<REQUEST_MAX>>,
90    /// The reply being reassembled, and how much of it is assembled. Held
91    /// apart from [`Reassembly`], whose borrow cannot outlive a service
92    /// call.
93    reply: Vec<u8>,
94    assembled: usize,
95    /// The most recent send. Its acknowledgment, or the MAC giving up on
96    /// it, is what moves the exchange on, and it is canceled once nothing
97    /// depends on it.
98    ticket: Option<SendProgressTicket>,
99    /// Distinguishes each exchange's token from the last.
100    counter: u16,
101    options: SendOptions,
102}
103
104impl<M: MacBackend> NodeManager<M> {
105    /// Watch `peer` for Node Management responses.
106    ///
107    /// `seed` picks the first token; anything unpredictable will do.
108    pub fn new(peer: PeerConnection<LocalNode<M>>, seed: u16) -> Self {
109        let inbox = Rc::new(RefCell::new(VecDeque::with_capacity(INBOX)));
110        let stray = Rc::new(Cell::new(0u32));
111        let subscription = {
112            let inbox = inbox.clone();
113            let stray = stray.clone();
114            peer.on_receive(move |packet: &ReceivedPacketRef<'_>| {
115                if packet.payload_type() != PayloadType::NodeManagementResponse {
116                    return false;
117                }
118                // The binding rests on what secure unicast guarantees, so
119                // an unauthenticated packet claiming to be a response is
120                // not one—and it is still this handler's to consume,
121                // since nothing else wants it either.
122                let room = inbox.borrow().len() < INBOX;
123                if !packet.source_authenticated() || !room {
124                    stray.set(stray.get().saturating_add(1));
125                    return true;
126                }
127                inbox.borrow_mut().push_back(packet.payload().to_vec());
128                true
129            })
130        };
131
132        Self {
133            peer,
134            inbox,
135            stray,
136            _subscription: subscription,
137            exchange: None,
138            reply: vec![0u8; REPLY_MAX],
139            assembled: 0,
140            ticket: None,
141            counter: seed,
142            options: SendOptions::default().with_ack_requested(true),
143        }
144    }
145
146    /// The device being managed.
147    pub fn device(&self) -> &PublicKey {
148        self.peer.peer()
149    }
150
151    /// How requests go out. An acknowledgment is requested whatever these
152    /// say: it is what paces retransmission and completes a reset.
153    pub fn send_options_mut(&mut self) -> &mut SendOptions {
154        &mut self.options
155    }
156
157    /// Payloads taken in that belonged to no outstanding exchange.
158    pub fn stray(&self) -> u32 {
159        self.stray.get()
160    }
161
162    /// The counter behind the last token any exchange here consumed,
163    /// the outstanding one included.
164    ///
165    /// A caller that builds one manager per operation seeds the next one
166    /// from this, so no token is ever issued twice against a device that
167    /// holds answered tokens against retransmission.
168    pub fn counter(&self) -> u16 {
169        self.exchange
170            .as_ref()
171            .map_or(self.counter, Exchange::counter)
172    }
173
174    /// The device's most recent estimate of octets not yet returned,
175    /// present only during a continued read.
176    pub fn remaining(&self) -> Option<u32> {
177        self.exchange.as_ref().and_then(Exchange::remaining)
178    }
179
180    /// Whether an exchange is outstanding.
181    pub fn is_busy(&self) -> bool {
182        self.exchange.is_some()
183    }
184
185    /// Begin an operation carrying `request`, one ULCP frame.
186    ///
187    /// The frame's TID is ignored over this binding; the envelope token
188    /// is what correlates the response.
189    pub fn begin(&mut self, request: &[u8], now_ms: u64) -> Result<(), BeginError> {
190        if self.exchange.is_some() {
191            return Err(BeginError::Busy);
192        }
193        self.counter = self.counter.wrapping_add(1);
194        let exchange = Exchange::new(request, self.counter, now_ms)
195            .map_err(|_| BeginError::RequestTooLarge)?;
196        // Anything still queued belongs to an exchange that is over.
197        self.inbox.borrow_mut().clear();
198        self.exchange = Some(exchange);
199        self.assembled = 0;
200        self.ticket = None;
201        Ok(())
202    }
203
204    /// Carry the outstanding exchange as far as it will go right now:
205    /// take in whatever arrived, then send if an attempt is due.
206    ///
207    /// A `Done` clears the exchange, leaving [`Self::reply`] readable
208    /// until the next [`Self::begin`].
209    pub async fn service(
210        &mut self,
211        now_ms: u64,
212    ) -> Result<Progress, ManagementError<<LocalNode<M> as Transport>::Error>> {
213        let Some(exchange) = self.exchange.as_mut() else {
214            return Err(ManagementError::Idle);
215        };
216
217        let mut payload = [0u8; 1 + PAYLOAD_MAX];
218        payload[0] = PayloadType::NodeManagementRequest as u8;
219
220        let mut step = None;
221        while let Some(response) = pop(&self.inbox) {
222            let mut reassembly = Reassembly::resume(&mut self.reply, self.assembled);
223            let next = exchange.receive(&response, &mut reassembly, now_ms, &mut payload[1..]);
224            self.assembled = reassembly.len();
225            match next {
226                Some(next) => {
227                    step = Some(next);
228                    break;
229                }
230                None => self.stray.set(self.stray.get().saturating_add(1)),
231            }
232        }
233
234        // A response in hand settles more than the MAC's verdict could, so
235        // the verdict is read only when nothing arrived.
236        let step = match step {
237            Some(step) => step,
238            None => {
239                if exchange.is_delivering()
240                    && let Some(ticket) = &self.ticket
241                {
242                    if ticket.was_acked() {
243                        exchange.acknowledged(now_ms);
244                    } else if ticket.has_failed() {
245                        exchange.undelivered();
246                    }
247                }
248                exchange.poll(now_ms, &mut payload[1..])
249            }
250        };
251
252        match step {
253            Step::Send { len } => {
254                // The attempt before this one may still be with the MAC:
255                // an earlier fragment whose response arrived ahead of its
256                // acknowledgment. Left alone, it would go on retrying, and
257                // eventually flood, a request that has been answered.
258                cancel_unfinished(&self.peer, self.ticket.take()).await;
259                self.options.ack_requested = true;
260                let ticket = self
261                    .peer
262                    .send(&payload[..1 + len], &self.options)
263                    .await
264                    .map_err(ManagementError::Transport)?;
265                self.ticket = Some(ticket);
266                Ok(Progress::Waiting {
267                    deadline_ms: now_ms + DELIVERY_RECHECK_MS,
268                })
269            }
270            Step::Wait { deadline_ms } => Ok(Progress::Waiting { deadline_ms }),
271            Step::Delivering => Ok(Progress::Waiting {
272                deadline_ms: now_ms + DELIVERY_RECHECK_MS,
273            }),
274            Step::Done(outcome) => Ok(self.settle(outcome).await),
275        }
276    }
277
278    /// Give up the outstanding exchange without an outcome, so the next
279    /// [`Self::begin`] is not refused as busy.
280    ///
281    /// For a caller that stopped waiting on its own terms—its patience
282    /// ran out, or the radio underneath it was replaced—rather than the
283    /// engine's. The token count is carried forward exactly as
284    /// [`settle`](Self::settle) carries it, so a device that does answer
285    /// the abandoned exchange late answers a token nothing will reissue,
286    /// and the MAC stops retrying the request. A no-op when nothing is
287    /// outstanding.
288    pub async fn abandon(&mut self) {
289        if let Some(exchange) = &self.exchange {
290            self.counter = exchange.counter();
291        }
292        self.exchange = None;
293        cancel_unfinished(&self.peer, self.ticket.take()).await;
294        self.inbox.borrow_mut().clear();
295    }
296
297    /// The reply frame of the exchange that just finished: one whole ULCP
298    /// frame, its trailing content the concatenation of every fragment.
299    pub fn reply(&self) -> &[u8] {
300        &self.reply[..self.assembled]
301    }
302
303    async fn settle(&mut self, outcome: Outcome) -> Progress {
304        // A continued read rotated tokens the manager's own counter never
305        // saw. Taking the exchange's final count back is what keeps the
306        // next `begin` from reissuing one of them—which the device
307        // would answer with the old exchange's retained response.
308        if let Some(exchange) = &self.exchange {
309            self.counter = exchange.counter();
310        }
311        self.exchange = None;
312        // A response that overtook its request's acknowledgment leaves the
313        // MAC waiting on one nobody needs.
314        cancel_unfinished(&self.peer, self.ticket.take()).await;
315        if let Outcome::Replied { len } = outcome {
316            self.assembled = len;
317        }
318        Progress::Done(outcome)
319    }
320}

322/// Stop the MAC working on a send whose outcome no longer matters.
323async fn cancel_unfinished<M: MacBackend>(
324    peer: &PeerConnection<LocalNode<M>>,
325    ticket: Option<SendProgressTicket>,
326) {
327    if let Some(ticket) = ticket
328        && !ticket.is_finished()
329        && let Some(token) = ticket.token()
330    {
331        peer.cancel_send(token).await;
332    }
333}

335/// Take the next queued response without holding the borrow across the
336/// work that follows it.
337fn pop(inbox: &Rc<RefCell<VecDeque<Vec<u8>>>>) -> Option<Vec<u8>> {
338    inbox.borrow_mut().pop_front()
