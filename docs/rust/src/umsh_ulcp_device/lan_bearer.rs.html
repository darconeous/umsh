1//! Saved device-domain configuration for the LAN bearer.
2use umsh_ulcp::{
3    Status,
4    ids::prop,
5    lan::{DEFAULT_INGRESS_LIMIT, Group},
6};
7
8/// The bearer's settings. The group and the ingress limit are held as
9/// what an operator chose, if anything: an unchosen one follows this
10/// firmware's default, and is saved as unchosen so that a later firmware
11/// with a different default brings it along.
12#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
13pub struct LanBearerConfig {
14    pub enabled: bool,
15    group: Option<Group>,
16    ingress_limit: Option<u16>,
17}
18
19impl LanBearerConfig {
20    /// The group in effect.
21    pub fn group(&self) -> Group {
22        self.group.unwrap_or_default()
23    }
24
25    /// Datagrams per minute carried from the LAN onto the radio.
26    pub fn ingress_limit(&self) -> u16 {
27        self.ingress_limit.unwrap_or(DEFAULT_INGRESS_LIMIT)
28    }
29
30    /// Whether `key` is one of the saved configuration properties.
31    pub const fn is_config_prop(key: u32) -> bool {
32        matches!(
33            key,
34            prop::LAN_BEARER_ENABLED | prop::LAN_BEARER_GROUP | prop::LAN_BEARER_INGRESS_LIMIT
35        )
36    }
37
38    /// Whether `key` holds no choice of its own and follows the default.
39    pub fn is_default(&self, key: u32) -> bool {
40        match key {
41            prop::LAN_BEARER_GROUP => self.group.is_none(),
42            prop::LAN_BEARER_INGRESS_LIMIT => self.ingress_limit.is_none(),
43            _ => false,
44        }
45    }
46
47    /// Take one property from `saved`, choice and all.
48    pub fn restore(&mut self, key: u32, saved: &Self) {
49        match key {
50            prop::LAN_BEARER_ENABLED => self.enabled = saved.enabled,
51            prop::LAN_BEARER_GROUP => self.group = saved.group,
52            prop::LAN_BEARER_INGRESS_LIMIT => self.ingress_limit = saved.ingress_limit,
53            _ => {}
54        }
55    }
56
57    /// Apply a written value. An empty group or limit returns it to the
58    /// default.
59    pub fn set(&mut self, key: u32, value: &[u8]) -> Result<(), Status> {
60        match key {
61            prop::LAN_BEARER_ENABLED => {
62                self.enabled = match value {
63                    [0] => false,
64                    [1] => true,
65                    _ => return Err(Status::INVALID_ARGUMENT),
66                }
67            }
68            prop::LAN_BEARER_GROUP => {
69                self.group = match value {
70                    [] => None,
71                    _ => Some(Group::decode(value).ok_or(Status::INVALID_ARGUMENT)?),
72                }
73            }
74            prop::LAN_BEARER_INGRESS_LIMIT => {
75                self.ingress_limit = match value {
76                    [] => None,
77                    _ => {
78                        let limit = u16::from_le_bytes(
79                            value.try_into().map_err(|_| Status::INVALID_ARGUMENT)?,
80                        );
81                        if limit == 0 {
82                            return Err(Status::INVALID_ARGUMENT);
83                        }
84                        Some(limit)
85                    }
86                }
87            }
88            _ => return Err(Status::PROP_NOT_FOUND),
89        }
90        Ok(())
91    }
92
93    /// Encode the value in effect for one configuration property; at most
94    /// [`GROUP_LEN`](umsh_ulcp::lan::GROUP_LEN) octets.
95    pub fn encode(&self, key: u32, out: &mut [u8]) -> Option<usize> {
96        let group;
97        let limit;
98        let bytes: &[u8] = match key {
99            prop::LAN_BEARER_ENABLED => &[self.enabled as u8],
100            prop::LAN_BEARER_GROUP => {
101                group = self.group().encode();
102                &group
103            }
104            prop::LAN_BEARER_INGRESS_LIMIT => {
105                limit = self.ingress_limit().to_le_bytes();
106                &limit
107            }
108            _ => return None,
109        };
110        out.get_mut(..bytes.len())?.copy_from_slice(bytes);
111        Some(bytes.len())
112    }
113}

115#[cfg(test)]
116mod tests {
117    use super::*;
118
119    #[test]
120    fn an_empty_write_returns_a_setting_to_the_default() {
121        let mut config = LanBearerConfig::default();
122        assert!(config.is_default(prop::LAN_BEARER_GROUP));
123        config
124            .set(prop::LAN_BEARER_GROUP, &[239, 1, 2, 3, 0x10, 0x27])
125            .unwrap();
126        config
127            .set(prop::LAN_BEARER_INGRESS_LIMIT, &30u16.to_le_bytes())
128            .unwrap();
129        assert_eq!(config.group().address, [239, 1, 2, 3]);
130        assert_eq!(config.ingress_limit(), 30);
131        assert!(!config.is_default(prop::LAN_BEARER_GROUP));
132
133        config.set(prop::LAN_BEARER_GROUP, &[]).unwrap();
134        config.set(prop::LAN_BEARER_INGRESS_LIMIT, &[]).unwrap();
135        assert_eq!(config, LanBearerConfig::default());
136        assert_eq!(config.group(), Group::default());
137        assert_eq!(config.ingress_limit(), DEFAULT_INGRESS_LIMIT);
138        assert_eq!(
139            config.set(prop::LAN_BEARER_ENABLED, &[]),
140            Err(Status::INVALID_ARGUMENT)
141        );
142    }
