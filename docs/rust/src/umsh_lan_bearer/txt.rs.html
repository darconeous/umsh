1//! The DNS-SD TXT schema (`lan-bearer.md` § Discovery).
2//!
3//! A participant's record says which group it is on, which node it is,
4//! and—if it has a LoRa radio—the settings that radio uses, so a client
5//! knows before sending anything which meshes the LAN reaches.
6//!
7//! Allocation-free: a radio's responder formats each value into a stack
8//! buffer as it goes, and a browser parses with whatever lookup its mDNS
9//! library offers.
10
11use core::fmt::Write as _;
12
13use heapless::String;
14use umsh_core::base58;
15
16/// The schema version this module writes, and the only one it reads.
17pub const SCHEMA_VERSION: &str = "1";
18
19/// TXT keys.
20pub mod key {
21    /// Schema version.
22    pub const V: &str = "v";
23    /// Multicast group address, dotted quad.
24    pub const GRP: &str = "grp";
25    /// Node public key, base58.
26    pub const ID: &str = "id";
27    /// LoRa center frequency, Hz.
28    pub const F: &str = "f";
29    /// LoRa bandwidth, Hz.
30    pub const BW: &str = "bw";
31    /// LoRa spreading factor.
32    pub const SF: &str = "sf";
33    /// LoRa coding rate denominator.
34    pub const CR: &str = "cr";
35    /// `1` if LoRa repeating is enabled.
36    pub const REP: &str = "rep";
37    /// `1` if MeshCore v1 forwarding is enabled.
38    pub const MC: &str = "mc";
39    /// `1` if an internet bridge link is up.
40    pub const IB: &str = "ib";
41}
42
43/// The LoRa half of an advertisement.
44#[derive(Clone, Copy, Debug, PartialEq, Eq)]
45pub struct LoraSettings {
46    pub freq_hz: u32,
47    pub bw_hz: u32,
48    pub sf: u8,
49    /// Coding rate denominator, 5 to 8.
50    pub cr: u8,
51    /// Whether the node repeats LoRa to LoRa.
52    pub repeater: bool,
53}
54
55/// Everything one participant's TXT record says.
56#[derive(Clone, Copy, Debug, PartialEq, Eq)]
57pub struct Advertisement {
58    pub group: [u8; 4],
59    pub node_key: [u8; 32],
60    /// Present for a participant with a LoRa radio.
61    pub lora: Option<LoraSettings>,
62    pub meshcore: bool,
63    pub internet_bridge: bool,
64}
65
66/// Why a TXT record could not be read as an advertisement.
67#[derive(Clone, Copy, Debug, PartialEq, Eq)]
68pub enum TxtError {
69    /// A required key is absent.
70    Missing(&'static str),
71    /// A key's value does not parse.
72    Invalid(&'static str),
73    /// A schema version this module does not read.
74    Version,
75}
76
77/// The longest value written: a 44-digit key.
78type Value = String<{ base58::ENCODED_LEN }>;
79
80impl Advertisement {
81    /// Hand each TXT key and value to `f`, in schema order. Optional
82    /// flags are written only when set.
83    pub fn for_each_property(&self, mut f: impl FnMut(&'static str, &str)) {
84        let mut value = Value::new();
85        let mut put = |key: &'static str, args: core::fmt::Arguments<'_>| {
86            value.clear();
87            // Every value fits by construction: the longest is the key.
88            let _ = value.write_fmt(args);
89            f(key, &value);
90        };
91        put(key::V, format_args!("{SCHEMA_VERSION}"));
92        let [a, b, c, d] = self.group;
93        put(key::GRP, format_args!("{a}.{b}.{c}.{d}"));
94        let id = base58::encode(&self.node_key);
95        put(
96            key::ID,
97            format_args!("{}", core::str::from_utf8(&id).unwrap_or_default()),
98        );
99        if let Some(lora) = self.lora {
100            put(key::F, format_args!("{}", lora.freq_hz));
101            put(key::BW, format_args!("{}", lora.bw_hz));
102            put(key::SF, format_args!("{}", lora.sf));
103            put(key::CR, format_args!("{}", lora.cr));
104            put(key::REP, format_args!("{}", lora.repeater as u8));
105        }
106        if self.meshcore {
107            put(key::MC, format_args!("1"));
108        }
109        if self.internet_bridge {
110            put(key::IB, format_args!("1"));
111        }
112    }
113
114    /// Read an advertisement through `get`, which returns a key's value.
115    /// Unknown keys are the caller's to ignore; a missing optional flag
116    /// reads as false.
117    pub fn parse<'a>(get: impl Fn(&str) -> Option<&'a str>) -> Result<Self, TxtError> {
118        let required = |key: &'static str| get(key).ok_or(TxtError::Missing(key));
119        if required(key::V)? != SCHEMA_VERSION {
120            return Err(TxtError::Version);
121        }
122        let group = parse_dotted_quad(required(key::GRP)?).ok_or(TxtError::Invalid(key::GRP))?;
123        let node_key = base58::decode(required(key::ID)?.as_bytes())
124            .map_err(|_| TxtError::Invalid(key::ID))?;
125        let flag = |key: &'static str| -> Result<bool, TxtError> {
126            match get(key) {
127                None | Some("0") => Ok(false),
128                Some("1") => Ok(true),
129                Some(_) => Err(TxtError::Invalid(key)),
130            }
131        };
132        let lora = match get(key::F) {
133            None => None,
134            Some(freq) => {
135                let number = |key: &'static str, value: &str| {
136                    value.parse::<u32>().map_err(|_| TxtError::Invalid(key))
137                };
138                let freq_hz = number(key::F, freq)?;
139                let bw_hz = number(key::BW, required(key::BW)?)?;
140                let sf = u8::try_from(number(key::SF, required(key::SF)?)?)
141                    .map_err(|_| TxtError::Invalid(key::SF))?;
142                let cr = u8::try_from(number(key::CR, required(key::CR)?)?)
143                    .ok()
144                    .filter(|cr| (5..=8).contains(cr))
145                    .ok_or(TxtError::Invalid(key::CR))?;
146                Some(LoraSettings {
147                    freq_hz,
148                    bw_hz,
149                    sf,
150                    cr,
151                    repeater: flag(key::REP)?,
152                })
153            }
154        };
155        Ok(Self {
156            group,
157            node_key,
158            lora,
159            meshcore: flag(key::MC)?,
160            internet_bridge: flag(key::IB)?,
161        })
162    }
163}

165fn parse_dotted_quad(text: &str) -> Option<[u8; 4]> {
166    let mut octets = [0u8; 4];
167    let mut parts = text.split('.');
168    for octet in octets.iter_mut() {
169        let part = parts.next()?;
170        // No leading zeros, signs, or empty parts: one spelling per address.
171        if part.is_empty() || (part.len() > 1 && part.starts_with('0')) {
172            return None;
173        }
174        *octet = part.parse().ok()?;
175    }
176    parts.next().is_none().then_some(octets)
177}

179#[cfg(test)]
180mod tests {
181    extern crate std;
182    use super::*;
183    use std::collections::BTreeMap;
184    use std::string::{String, ToString};
185
186    fn record(advertisement: &Advertisement) -> BTreeMap<&'static str, String> {
187        let mut map = BTreeMap::new();
188        advertisement.for_each_property(|key, value| {
189            map.insert(key, value.to_string());
190        });
191        map
192    }
193
194    fn radio() -> Advertisement {
195        Advertisement {
196            group: [239, 255, 134, 121],
197            node_key: [7; 32],
198            lora: Some(LoraSettings {
199                freq_hz: 906_875_000,
200                bw_hz: 250_000,
201                sf: 11,
202                cr: 5,
203                repeater: true,
204            }),
205            meshcore: false,
206            internet_bridge: true,
207        }
208    }
209
210    #[test]
211    fn a_radio_round_trips() {
212        let advertisement = radio();
213        let map = record(&advertisement);
214        assert_eq!(map["v"], "1");
215        assert_eq!(map["grp"], "239.255.134.121");
216        assert_eq!(map["id"].len(), 44);
217        assert_eq!(map["f"], "906875000");
218        assert_eq!(map["rep"], "1");
219        assert_eq!(map.get("mc"), None, "unset flags are not written");
220        assert_eq!(map["ib"], "1");
221        let parsed = Advertisement::parse(|key| map.get(key).map(String::as_str)).unwrap();
222        assert_eq!(parsed, advertisement);
223    }
224
225    #[test]
226    fn a_participant_without_a_radio_has_no_lora_keys() {
227        let advertisement = Advertisement {
228            lora: None,
229            internet_bridge: false,
230            ..radio()
231        };
232        let map = record(&advertisement);
233        assert_eq!(
234            map.keys().copied().collect::<std::vec::Vec<_>>(),
235            ["grp", "id", "v"]
236        );
237        let parsed = Advertisement::parse(|key| map.get(key).map(String::as_str)).unwrap();
238        assert_eq!(parsed, advertisement);
239    }
240
241    #[test]
242    fn bad_records_are_refused_by_key() {
243        let mut map = record(&radio());
244        map.insert("cr", "9".to_string());
245        assert_eq!(
246            Advertisement::parse(|key| map.get(key).map(String::as_str)),
247            Err(TxtError::Invalid("cr"))
248        );
249        map.insert("v", "2".to_string());
250        assert_eq!(
251            Advertisement::parse(|key| map.get(key).map(String::as_str)),
252            Err(TxtError::Version)
253        );
254        map.remove("v");
255        assert_eq!(
256            Advertisement::parse(|key| map.get(key).map(String::as_str)),
257            Err(TxtError::Missing("v"))
258        );
259    }
260
261    #[test]
262    fn a_group_has_one_spelling() {
263        assert_eq!(
264            parse_dotted_quad("239.255.134.121"),
265            Some([239, 255, 134, 121])
266        );
267        assert_eq!(parse_dotted_quad("239.255.042.42"), None);
268        assert_eq!(parse_dotted_quad("239.255.42"), None);
269        assert_eq!(parse_dotted_quad("239.255.134.121.1"), None);
270        assert_eq!(parse_dotted_quad("239.255.42.256"), None);
271    }
