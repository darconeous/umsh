211/// Parse a node URI locally and return only validated public identity fields.
212#[uniffi::export]
213pub fn inspect_node_uri(uri: String) -> Result<NodeUriPreviewRecord, MobileError> {
214    let reference = UriRef::from_str(&uri).map_err(|_| MobileError::InvalidUri)?;
215    let node = match umsh_uri::parse_umsh_uri(reference).map_err(|_| MobileError::InvalidUri)? {
216        UmshUri::Node(node) => node,
217        _ => return Err(MobileError::InvalidUri),
218    };
219    let identity = public_identity_record(&node.public_key);
220    let (decoded, payload) = match node.identity_data {
221        None => (None, None),
222        Some(data) => match umsh_uri::decode_base58_bytes(data).ok().and_then(|bytes| {
223            node_identity_record(&node.public_key, &bytes)
224                .ok()
225                .map(|record| (record, bytes))
226        }) {
227            None => (None, None),
228            Some((record, bytes)) => {
229                // Tampered bundles are shown as invalid but never persisted.
230                let payload =
231                    (record.signature != IdentitySignatureState::Invalid).then_some(bytes);
232                (Some(record), payload)
233            }
234        },
235    };
236    Ok(NodeUriPreviewRecord {
237        canonical_address: identity.canonical_address,
238        hint: identity.hint,
239        has_identity_data: node.identity_data.is_some(),
240        identity: decoded,
241        identity_payload: payload,
242    })
243}

245/// Decode a persisted advertised-identity payload for display.
246///
247/// `address` is the canonical Base58 address of the node the payload claims
248/// to describe; the signature state is recomputed against it on every call.
249#[uniffi::export]
250pub fn decode_node_identity(
251    address: String,
252    payload: Vec<u8>,
253) -> Result<NodeIdentityRecord, MobileError> {
254    let key = PublicKey(umsh_core::base58::decode(address.as_bytes())?);
255    node_identity_record(&key, &payload)
256}

258fn node_identity_record(
259    key: &PublicKey,
260    payload: &[u8],
261) -> Result<NodeIdentityRecord, MobileError> {
262    let identity = umsh_node::NodeIdentityPayload::from_bytes(payload)
263        .map_err(|_| MobileError::InvalidIdentityData)?;
264
265    let signature = match identity.signature {
266        None => IdentitySignatureState::Unsigned,
267        Some(signature) => {
268            // The signed range is everything before the trailing signature.
269            let signed = &payload[..payload.len() - 64];
270            if umsh_crypto::verify_ed25519_signature(key, signed, &signature) {
271                IdentitySignatureState::Valid
272            } else {
273                IdentitySignatureState::Invalid
274            }
275        }
276    };
277
278    let role_label = match identity.role {
279        umsh_node::NodeRole::Unspecified => "Unspecified".to_owned(),
280        umsh_node::NodeRole::Repeater => "Repeater".to_owned(),
281        umsh_node::NodeRole::Chat => "Chat".to_owned(),
282        umsh_node::NodeRole::Tracker => "Tracker".to_owned(),
283        umsh_node::NodeRole::Sensor => "Sensor".to_owned(),
284        umsh_node::NodeRole::Bridge => "Bridge".to_owned(),
285        umsh_node::NodeRole::ChatRoom => "Chat room".to_owned(),
286        umsh_node::NodeRole::TemporarySession => "Temporary session".to_owned(),
287        umsh_node::NodeRole::Unknown(code) => format!("Unknown ({code})"),
288    };
289
290    let capabilities = [
291        (umsh_node::NodeCapabilities::REPEATER, "Repeater"),
292        (umsh_node::NodeCapabilities::MOBILE, "Mobile"),
293        (umsh_node::NodeCapabilities::TEXT_MESSAGES, "Text messages"),
294        (umsh_node::NodeCapabilities::TELEMETRY, "Telemetry"),
295        (umsh_node::NodeCapabilities::CHAT_ROOM, "Chat room"),
296        (umsh_node::NodeCapabilities::COAP, "CoAP"),
297    ]
298    .into_iter()
299    .filter(|(bit, _)| identity.capabilities.contains(*bit))
300    .map(|(_, label)| label.to_owned())
301    .collect();
302
303    let location = identity.location.filter(|loc| !loc.is_unspecified());
304    let (latitude, longitude, location_precision) = match location {
305        None => (None, None, None),
306        Some(location) => {
307            let (lon, lat) = location.center();
308            (
309                Some(f64::from(lat)),
310                Some(f64::from(lon)),
311                Some(location.precision()),
312            )
313        }
314    };
315
316    Ok(NodeIdentityRecord {
317        role_code: identity.role.as_byte(),
318        role_label,
319        capabilities,
320        name: identity.name,
321        latitude,
322        longitude,
323        location_precision,
324        altitude_m: identity.altitude_m,
325        timestamp: identity.timestamp,
326        signature,
327    })
328}

330/// Inspect a pasted peer identity in any user-facing interchange form.
331///
332/// Accepted input is a node URI, the canonical fixed-width Base58 public
333/// address, or exactly 32 public-key bytes written as hexadecimal (with an
334/// optional `0x` prefix). The result always returns the canonical Base58 form.
335#[uniffi::export]
336pub fn inspect_peer_identity(input: String) -> Result<NodeUriPreviewRecord, MobileError> {
337    let input = input.trim();
338    if input.starts_with("umsh:") {
339        return inspect_node_uri(input.to_owned());
340    }
341
342    let hex_input = input
343        .strip_prefix("0x")
344        .or_else(|| input.strip_prefix("0X"))
345        .unwrap_or(input);
346    let hex = hex_input
347        .chars()
348        .filter(|character| !character.is_ascii_whitespace() && !matches!(character, ':' | '-'))
349        .collect::<String>();
350    if hex.len() == 64 && hex.bytes().all(|byte| byte.is_ascii_hexdigit()) {
351        let mut public_key = [0u8; 32];
352        for (index, pair) in hex.as_bytes().chunks_exact(2).enumerate() {
353            let pair =
354                core::str::from_utf8(pair).map_err(|_| MobileError::InvalidAddressCharacter)?;
355            public_key[index] =
356                u8::from_str_radix(pair, 16).map_err(|_| MobileError::InvalidAddressCharacter)?;
357        }
358        let identity = public_identity_record(&PublicKey(public_key));
359        return Ok(NodeUriPreviewRecord {
360            canonical_address: identity.canonical_address,
361            hint: identity.hint,
362            has_identity_data: false,
363            identity: None,
364            identity_payload: None,
365        });
366    }
367
368    let identity = inspect_public_identity(input.to_owned())?;
369    Ok(NodeUriPreviewRecord {
370        canonical_address: identity.canonical_address,
371        hint: identity.hint,
372        has_identity_data: false,
373        identity: None,
374        identity_payload: None,
375    })
376}

378/// Render the canonical shareable node URI for a public identity address.
379///
380/// The result is the plain `umsh:n:` form without an appended identity
381/// bundle, suitable for copying and for QR presentation of the bare key.
382#[uniffi::export]
383pub fn node_uri_for_address(address: String) -> Result<String, MobileError> {
384    let key = PublicKey(umsh_core::base58::decode(address.as_bytes())?);
385    let mut buf = [0u8; 64];
386    let len = umsh_uri::format_node_uri(&key, &mut buf).map_err(|_| MobileError::InvalidUri)?;
387    Ok(core::str::from_utf8(&buf[..len])
388        .expect("node URI is ASCII")
389        .to_owned())
390}

392/// Render the shareable node URI carrying a signed identity bundle:
393/// `umsh:n:<address>:<base58 bundle>`.
394#[uniffi::export]
395pub fn node_uri_with_identity(
396    address: String,
397    identity_payload: Vec<u8>,
398) -> Result<String, MobileError> {
399    let base = node_uri_for_address(address)?;
400    Ok(format!(
401        "{base}:{}",
402        umsh_uri::encode_base58_bytes(&identity_payload)
403    ))
404}

469/// Identity secret retained by the Rust engine after one controlled unlock
470/// transfer. Swift receives only public identity records from this object.
471#[derive(uniffi::Object)]
472pub struct MobileIdentity {
473    identity: Mutex<Option<SoftwareIdentity>>,
474    public_identity: PublicIdentityRecord,
475}
476
477#[uniffi::export]
478impl MobileIdentity {
479    #[uniffi::constructor]
480    pub fn unlock(mut secret_key: Vec<u8>) -> Result<Arc<Self>, MobileError> {
481        let result = (|| {
482            let mut bytes: [u8; 32] = secret_key
483                .as_slice()
484                .try_into()
485                .map_err(|_| MobileError::InvalidSecretKeyLength)?;
486            let identity = SoftwareIdentity::from_secret_bytes(&bytes);
487            let public_identity = public_identity_record(identity.public_key());
488            bytes.zeroize();
489            Ok(Arc::new(Self {
490                identity: Mutex::new(Some(identity)),
491                public_identity,
492            }))
493        })();
494        secret_key.zeroize();
495        result
496    }
497
498    pub fn public_identity(&self) -> PublicIdentityRecord {
499        self.public_identity.clone()
500    }
501}
502
503impl MobileIdentity {
504    fn take_for_session(&self) -> Result<SoftwareIdentity, MobileMeshError> {
505        self.identity
506            .lock()
507            .map_err(|_| MobileMeshError::SessionUnavailable)?
508            .take()
509            .ok_or(MobileMeshError::SessionUnavailable)
510    }
511}
512
513fn public_identity_record(key: &PublicKey) -> PublicIdentityRecord {
514    let canonical_address = umsh_core::base58::encode(&key.0)
515        .into_iter()
516        .map(char::from)
517        .collect();
518
519    PublicIdentityRecord {
520        canonical_address,
521        hint: render_node_hint_bytes(NodeHint::from_public_key(&key).0),
522    }
523}

621        assert!(preview.has_identity_data);
622        assert_eq!(preview.identity_payload.as_deref(), Some(bundle));
623        let record = preview.identity.unwrap();
624        assert_eq!(record.signature, IdentitySignatureState::Valid);
625        assert_eq!(record.role_label, "Chat");
626        assert_eq!(record.name.as_deref(), Some("Basecamp"));
627        assert_eq!(
628            record.capabilities,
629            vec!["Mobile".to_owned(), "Text messages".to_owned()]
630        );
631        assert_eq!(record.altitude_m, Some(72));
632        assert_eq!(record.timestamp, Some(1_760_000_000));
633        assert_eq!(record.location_precision, Some(4));
634        // A 4-byte grid cell is ~610 x 305 m; the center must land nearby.
635        assert!((record.latitude.unwrap() - 44.05).abs() < 0.01);
636        assert!((record.longitude.unwrap() + 123.09).abs() < 0.01);
637
638        // Tampering with the signed range must flag the bundle and withhold
639        // the persistable payload.
640        let mut tampered = bundle.to_vec();
641        tampered[0] ^= 0x01;
642        let uri = format!(
643            "umsh:n:{address}:{}",
644            umsh_uri::encode_base58_bytes(&tampered)
645        );
646        let preview = inspect_node_uri(uri).unwrap();
647        assert_eq!(
648            preview.identity.unwrap().signature,
649            IdentitySignatureState::Invalid
650        );
651        assert!(preview.identity_payload.is_none());
652
653        // An unsigned bundle stays displayable and persistable, but is
654        // explicitly marked unauthenticated.
655        let unsigned_len = {
656            let unsigned = umsh_node::NodeIdentityPayload {
657                signature: None,
658                ..payload.clone()
659            };
660            unsigned.encode(&mut buf).unwrap()
661        };
662        let uri = format!(
663            "umsh:n:{address}:{}",
664            umsh_uri::encode_base58_bytes(&buf[..unsigned_len])
665        );
666        let preview = inspect_node_uri(uri).unwrap();
667        assert_eq!(
668            preview.identity.unwrap().signature,
669            IdentitySignatureState::Unsigned
670        );
671        assert!(preview.identity_payload.is_some());
672    }
