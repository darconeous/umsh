218/// The writes that state a whole device configuration, ordered, for a
219/// device being configured across the mesh.
220///
221/// A phone holding a device open writes a configuration through
222/// `MobileUlcpSession::configure_device`, which owns the ordering — the
223/// PHY goes down first and comes back up last — and closes with a save.
224/// An administrator has no session to hand a record to, only the record
225/// its read produced, so it asks for the same writes here and sends them
226/// with [`MobileMeshSession::begin_management_set_many`]. The reduction is
227/// literally the same code, which is the point: the two paths cannot
228/// drift into configuring a device differently.
229///
230/// `reported` is the device as a completed read found it. Its
231/// capabilities decide which fields must be present, and the properties
232/// it would not report are left out — writing one of those fails, and a
233/// device fails the write it is on rather than the ones after it.
234#[uniffi::export]
235pub fn ulcp_device_config_writes(
236    configuration: crate::ulcp::UlcpDeviceConfigRecord,
237    reported: UlcpSyncRecord,
238) -> Result<Vec<MobileMeshPropertyWriteRecord>, MobileMeshError> {
239    let values = crate::ulcp::device_config_writes(configuration, &reported)
240        .map_err(|_| MobileMeshError::InvalidRequest)?;
241    Ok(values
242        .into_iter()
243        .map(|(property_id, value)| MobileMeshPropertyWriteRecord { property_id, value })
244        .collect())
245}

401/// Platform-side listener invoked when `poll_update` has new data waiting.
402///
403/// Called on the worker thread; implementations must only schedule a drain
404/// on their own executor and return. Notifications are coalesced: at most
405/// one call fires per pending-to-drained cycle, so a burst of protocol
406/// activity costs one crossing, and the platform needs no polling cadence.
407#[uniffi::export(with_foreign)]
408pub trait MobileMeshWakeListener: Send + Sync {
409    fn on_update_pending(&self);
410}
411
412/// Coalescing wake flag shared between the worker's producer channels and
413/// `poll_update`. `notify` fires the listener only on the false-to-true
414/// transition; `drained` re-arms it.
415struct WakeSignal {
416    pending: AtomicBool,
417    listener: Mutex<Option<Arc<dyn MobileMeshWakeListener>>>,
418}
419
420impl WakeSignal {
421    fn new() -> Self {
422        Self {
423            pending: AtomicBool::new(false),
424            listener: Mutex::new(None),
425        }
426    }
427
428    fn notify(&self) {
429        if self.pending.swap(true, Ordering::AcqRel) {
430            return;
431        }
432        let listener = self
433            .listener
434            .lock()
435            .ok()
436            .and_then(|slot| slot.as_ref().cloned());
437        if let Some(listener) = listener {
438            listener.on_update_pending();
439        }
440    }
441
442    fn drained(&self) {
443        self.pending.store(false, Ordering::Release);
444    }
445
446    fn set_listener(&self, listener: Option<Arc<dyn MobileMeshWakeListener>>) {
447        let already_pending = {
448            let Ok(mut slot) = self.listener.lock() else {
449                return;
450            };
451            *slot = listener.clone();
452            self.pending.load(Ordering::Acquire)
453        };
454        // Data enqueued before registration must not wait for the next
455        // protocol event to surface.
456        if already_pending && let Some(listener) = listener {
457            listener.on_update_pending();
458        }
459    }
460}

462/// A producer channel endpoint that arms the wake signal on every enqueue.
463struct NotifyingSender<T> {
464    tx: std_mpsc::Sender<T>,
465    wake: Arc<WakeSignal>,
466}
467
468impl<T> Clone for NotifyingSender<T> {
469    fn clone(&self) -> Self {
470        Self {
471            tx: self.tx.clone(),
472            wake: self.wake.clone(),
473        }
474    }
475}
476
477impl<T> NotifyingSender<T> {
478    fn send(&self, value: T) -> Result<(), std_mpsc::SendError<T>> {
479        self.tx.send(value)?;
480        self.wake.notify();
481        Ok(())
482    }
483}

763struct BridgeTransmitCompletions {
764    next_id: AtomicU64,
765    failure_generation: AtomicU64,
766    /// A link-wide failure was declared and its `FailOutboundTransmissions`
767    /// command has not yet been processed by the worker. While set, no new
768    /// transmission may reach the platform: the MAC's in-progress drain loop
769    /// would otherwise keep dispatching the frames queued behind the one the
770    /// failure caught mid-flight, because each later `transmit` call samples
771    /// the generation only after the bump. The worker clears the flag when
772    /// it processes the queued command and cancels the affected tickets.
773    poisoned: AtomicBool,
774    pending: Mutex<BTreeMap<u64, oneshot::Sender<bool>>>,
775}
776
777impl BridgeTransmitCompletions {
778    fn new() -> Self {
779        Self {
780            next_id: AtomicU64::new(1),
781            failure_generation: AtomicU64::new(0),
782            poisoned: AtomicBool::new(false),
783            pending: Mutex::new(BTreeMap::new()),
784        }
785    }
786
787    fn generation(&self) -> u64 {
788        self.failure_generation.load(Ordering::SeqCst)
789    }
790
791    fn allocate(
792        &self,
793        generation: u64,
794        completion: oneshot::Sender<bool>,
795    ) -> Result<Option<u64>, BridgeRadioError> {
796        let id = self.next_id.fetch_add(1, Ordering::Relaxed).max(1);
797        let mut pending = self.pending.lock().map_err(|_| BridgeRadioError::Closed)?;
798        if self.poisoned.load(Ordering::SeqCst)
799            || generation != self.failure_generation.load(Ordering::SeqCst)
800        {
801            return Ok(None);
802        }
803        pending.insert(id, completion);
804        Ok(Some(id))
805    }
806
807    /// Declare a link-wide failure: refuse new platform dispatches until the
808    /// worker processes the corresponding cancellation command.
809    fn poison(&self) {
810        self.poisoned.store(true, Ordering::SeqCst);
811        self.failure_generation.fetch_add(1, Ordering::SeqCst);
812    }
813
814    fn clear_poison(&self) {
815        self.poisoned.store(false, Ordering::SeqCst);
816    }
817
818    fn complete(&self, id: u64, transmitted: bool) -> bool {
819        self.pending
820            .lock()
821            .ok()
822            .and_then(|mut pending| pending.remove(&id))
823            .is_some_and(|completion| completion.send(transmitted).is_ok())
824    }
825
826    fn fail_all(&self) {
827        self.failure_generation.fetch_add(1, Ordering::SeqCst);
828        let completions = self
829            .pending
830            .lock()
831            .map(|mut pending| core::mem::take(&mut *pending))
832            .unwrap_or_default();
833        for completion in completions.into_values() {
834            let _ = completion.send(false);
835        }
836    }
837}

839struct BridgeRadio {
840    inbound: mpsc::UnboundedReceiver<InboundFrame>,
841    outbound: NotifyingSender<MobileMeshOutboundFrameRecord>,
842    completions: Arc<BridgeTransmitCompletions>,
843}
844
845impl Radio for BridgeRadio {
846    type Error = BridgeRadioError;
847
848    async fn transmit(
849        &mut self,
850        data: &[u8],
851        options: TxOptions,
852    ) -> Result<(), TxError<Self::Error>> {
853        if data.len() > MAX_FRAME_SIZE {
854            return Err(TxError::Io(BridgeRadioError::FrameTooLarge));
855        }
856        // The MAC skips CAD only for immediate acks (channel-access.md
857        // § Immediate ACK Transmission); every other policy asks the
858        // device to listen before talking.
859        let nocca = matches!(options.cad, umsh_hal::CadPolicy::Skip);
860        let (completion_tx, completion_rx) = oneshot::channel();
861        let generation = self.completions.generation();
862        let Some(id) = self
863            .completions
864            .allocate(generation, completion_tx)
865            .map_err(TxError::Io)?
866        else {
867            // A link-wide failure raced this send before it reached the
868            // platform. Its queued cancellation owns the ticket outcome.
869            return Ok(());
870        };
871        if self
872            .outbound
873            .send(MobileMeshOutboundFrameRecord {
874                id,
875                data: data.to_vec(),
876                nocca,
877            })
878            .is_err()
879        {
880            let _ = self.completions.complete(id, false);
881            return Err(TxError::Io(BridgeRadioError::Closed));
882        }
883
884        // Awaiting here is deliberate: Radio::transmit completes only after
885        // the frame has actually left the radio PHY. Returning at
886        // bridge-queue acceptance starts MAC ACK timers too early and causes
887        // fragmented sends to retransmit frames that are still waiting in
888        // the device queue. This is an async wait, not a thread block, so
889        // the worker keeps servicing commands and timers while the frame is
890        // in flight; the MAC itself stays serialized behind its own borrow.
891        match completion_rx.await {
892            Ok(true) => Ok(()),
893            // The public completion API poisons the bridge and queues
894            // FailOutboundTransmissions before releasing this wait. Return
895            // success here solely to keep an ordinary rejected frame from
896            // terminating the long-lived MAC driver; the queued command
897            // cancels its ACK ticket immediately.
898            Ok(false) => Ok(()),
899            Err(_) => Err(TxError::Io(BridgeRadioError::Closed)),
900        }
901    }
902
903    fn poll_receive(
904        &mut self,
905        cx: &mut Context<'_>,
906        buf: &mut [u8],
907    ) -> Poll<Result<RxInfo, Self::Error>> {
908        match self.inbound.poll_recv(cx) {
909            Poll::Ready(Some(frame)) => {
910                if frame.record.data.len() > buf.len() {
911                    return Poll::Ready(Err(BridgeRadioError::FrameTooLarge));
912                }
913                let len = frame.record.data.len();
914                buf[..len].copy_from_slice(&frame.record.data);
915                Poll::Ready(Ok(RxInfo {
916                    len,
917                    rssi: frame.record.rssi_dbm.unwrap_or(0),
918                    snr: Snr::from_centibels(frame.record.snr_cb.unwrap_or(0)),
919                    lqi: frame.record.lqi.and_then(core::num::NonZeroU8::new),
920                    origin: RxOrigin::Air,
921                }))
922            }
923            Poll::Ready(None) => Poll::Ready(Err(BridgeRadioError::Closed)),
924            Poll::Pending => Poll::Pending,
925        }
926    }
927
928    fn max_frame_size(&self) -> usize {
929        MAX_FRAME_SIZE
930    }
931    fn t_frame_ms(&self) -> u32 {
932        DEFAULT_FRAME_TIME_MS
933    }
934}

1783    /// Report the actual physical radio result for an outbound
1784    /// frame. This is intentionally distinct from accepting the frame into the

1786    pub fn complete_outbound_frame(
1787        &self,
1788        frame_id: u64,
1789        transmitted: bool,
1790    ) -> Result<(), MobileMeshError> {

1798                .map_err(|_| MobileMeshError::SessionUnavailable)?;

1801            .complete(frame_id, transmitted)
1802            .then_some(())
1803            .ok_or(MobileMeshError::SessionUnavailable)
1804    }

1958    pub fn fail_outbound_transmissions(&self) -> Result<(), MobileMeshError> {
1959        // Poison before anything else: from this instant until the worker
1960        // processes the command below (the sole clearer), every frame the
1961        // MAC's in-progress drain loop tries to hand to the platform is
1962        // suppressed instead of dispatched. Without this, releasing the
1963        // blocked transmit lets the drain advance to the next queued frame,
1964        // which samples the post-bump generation and goes out as if healthy.
1965        self.transmit_completions.poison();

1970        // is defused by the poison above.

1972        Ok(())
1973    }

1986    pub fn poll_update(&self) -> MobileMeshSessionUpdateRecord {
1987        // Re-arm before draining: anything enqueued mid-drain triggers a
1988        // fresh notification rather than being silently swallowed.
1989        self.wake.drained();
1990        let mut outbound_frames = Vec::new();
1991        if let Ok(receiver) = self.outbound.lock() {
1992            outbound_frames.extend(receiver.try_iter());
1993        }
1994        let mut ping_events = Vec::new();
1995        if let Ok(receiver) = self.events.lock() {
1996            ping_events.extend(receiver.try_iter());
1997        }
1998        let mut management_events = Vec::new();
1999        if let Ok(receiver) = self.management.lock() {
2000            management_events.extend(receiver.try_iter());
2001        }
2002        let mut advertisement_events = Vec::new();
2003        if let Ok(receiver) = self.advertisements.lock() {
2004            advertisement_events.extend(receiver.try_iter());
2005        }
2006        let mut peer_heard_events = Vec::new();
2007        if let Ok(receiver) = self.peer_heard.lock() {
2008            peer_heard_events.extend(receiver.try_iter());
2009        }
2010        let mut chat_mutations = Vec::new();
2011        let mut chat_deliveries = Vec::new();
2012        let mut chat_archive_lookups = Vec::new();
2013        let mut chat_sender_resolutions = Vec::new();
2014        let mut chat_diagnostics = Vec::new();
2015        let mut chat_batch_id = None;
2016        if let Ok(mut pending) = self.pending_chat_events.lock() {
2017            if pending.is_none()
2018                && let Ok(receiver) = self.chat_events.lock()
2019            {
2020                let events = receiver.try_iter().collect::<Vec<_>>();
2021                if !events.is_empty()
2022                    && let Ok(mut next) = self.next_chat_batch_id.lock()
2023                {
2024                    let id = *next;
2025                    *next = next.wrapping_add(1).max(1);
2026                    *pending = Some(PendingChatEventBatch { id, events });
2027                }
2028            }
2029            if let Some(batch) = pending.as_ref() {
2030                chat_batch_id = Some(batch.id);
2031                for event in batch.events.iter().cloned() {
2032                    match event {
2033                        MobileChatWorkerEvent::Mutation(record) => chat_mutations.push(record),
2034                        MobileChatWorkerEvent::Delivery(record) => chat_deliveries.push(record),
2035                        MobileChatWorkerEvent::ArchiveLookup(record) => {
2036                            chat_archive_lookups.push(record);
2037                        }
2038                        MobileChatWorkerEvent::SenderResolution(record) => {
2039                            chat_sender_resolutions.push(record);
2040                        }
2041                        MobileChatWorkerEvent::Diagnostic(record) => chat_diagnostics.push(record),
2042                    }
2043                }
2044            }
2045        }
2046        MobileMeshSessionUpdateRecord {
2047            outbound_frames,
2048            ping_events,
2049            management_events,
2050            advertisement_events,
2051            peer_heard_events,
2052            chat_batch_id,
2053            chat_mutations,
2054            chat_deliveries,
2055            chat_archive_lookups,
2056            chat_sender_resolutions,
2057            chat_diagnostics,
2058        }
2059    }

2362/// A read of a named set of properties, one batch at a time.
2363///
2364/// The caller says what to ask for, which is what keeps a screenful of
2365/// settings from costing a whole-device read. Everything out of an
2366/// administrator's reach is dropped before a single frame goes on the
2367/// air.
2368struct FetchCrawl {
2369    /// Whether the device answers multi-property requests. A hint from
2370    /// the caller, corrected by the device the moment it declines one.
2371    multi: bool,
2372    /// What the outstanding request asked for, in order.
2373    asked: Vec<u32>,
2374    /// What is still to be asked for.
2375    pending: VecDeque<u32>,
2376    /// Every answer so far, refusals included: a property the device
2377    /// would not report is a different thing from one nobody asked for,
2378    /// and only the caller can tell what that means for its screen.
2379    answers: Vec<MobileMeshManagementAnswerRecord>,
2380}
2381
2382impl FetchCrawl {
2383    fn new(properties: Vec<u32>, multi_hint: bool) -> Self {
2384        let mut pending: Vec<u32> = properties
2385            .into_iter()
2386            .filter(|&key| ids::admin_reachable(key))
2387            .collect();
2388        pending.dedup();
2389        Self {
2390            multi: multi_hint,
2391            asked: Vec::new(),
2392            pending: pending.into(),
2393            answers: Vec::new(),
2394        }
2395    }
2396
2397    /// The next request, or `None` when there is nothing left to ask.
2398    fn next_request(&mut self) -> Result<Option<Vec<u8>>, MobileMeshError> {
2399        let batch = if self.multi { SYNC_BATCH } else { 1 };
2400        self.asked = self
2401            .pending
2402            .drain(..batch.min(self.pending.len()))
2403            .collect();
2404        match self.asked.as_slice() {
2405            [] => Ok(None),
2406            [key] => encode_management(|buf| frame::prop_get(buf, 0, *key)).map(Some),
2407            keys => encode_management(|buf| frame::prop_multi_get(buf, 0, keys)).map(Some),
2408        }
2409    }
2410
2411    /// Take in one answer.
2412    fn receive(&mut self, reply: &[u8]) -> Result<(), MobileMeshError> {
2413        if let [key] = self.asked.as_slice() {
2414            let key = *key;
2415            match umsh_ulcp::reply::property(key, reply) {
2416                Ok(answer) => self.answers.push(answer_record(key, answer)),
2417                Err(_) => return Err(MobileMeshError::InvalidRequest),
2418            }
2419            return Ok(());
2420        }
2421
2422        let asked = core::mem::take(&mut self.asked);
2423        let Ok(entries) = umsh_ulcp::reply::entries(&asked, reply) else {
2424            // The device declined the command rather than the properties,
2425            // which is what one without `CAP_CMD_MULTI` does. Ask again one
2426            // at a time; the hint said otherwise, but the device is the
2427            // authority on itself.
2428            self.multi = false;
2429            for key in asked.into_iter().rev() {
2430                self.pending.push_front(key);
2431            }
2432            return Ok(());
2433        };
2434        let mut answered = 0usize;
2435        for (key, answer) in entries.flatten() {
2436            answered += 1;
2437            self.answers.push(answer_record(key, answer));
2438        }
2439        // A device stops before its answer overflows rather than
2440        // truncating one, so whatever it did not reach is simply asked for
2441        // again. An answer that reached nothing would ask forever, so that
2442        // batch is broken up instead.
2443        if answered == 0 {
2444            self.multi = false;
2445        }
2446        for key in asked.into_iter().skip(answered).rev() {
2447            self.pending.push_front(key);
2448        }
2449        Ok(())
2450    }
2451}

2498    /// Read a finished exchange. `None` means another exchange was begun
2499    /// and the operation continues.
2500    fn settle(
2501        &mut self,
2502        outcome: umsh_node_mgmt::Outcome,
2503        now_ms: u64,
2504    ) -> Option<MobileMeshManagementEventRecord> {
2505        match outcome {
2506            umsh_node_mgmt::Outcome::Failed(umsh_node_mgmt::Failure::TimedOut) => {
2507                return Some(self.event(MobileMeshManagementOutcome::TimedOut));
2508            }
2509            umsh_node_mgmt::Outcome::Failed(_) => {
2510                return Some(self.event(MobileMeshManagementOutcome::Failed));
2511            }
2512            umsh_node_mgmt::Outcome::NoResponse => {
2513                return Some(match self.plan {
2514                    // Only a reset-class command is answered by nothing.
2515                    ManagementPlan::One(ReplyShape::Acknowledgment) => {
2516                        self.event(MobileMeshManagementOutcome::Acknowledged)
2517                    }
2518                    _ => self.event(MobileMeshManagementOutcome::Failed),
2519                });
2520            }
2521            // The reply is the reassembly buffer, which the manager hands
2522            // out whole; its length is not needed separately.
2523            umsh_node_mgmt::Outcome::Replied { .. } => {}
2524        }
2525
2526        match &mut self.plan {
2527            ManagementPlan::One(shape) => {
2528                let shape = shape.clone();
2529                Some(self.replied(&shape))
2530            }
2531            ManagementPlan::Fetch(crawl) => {
2532                if crawl.receive(self.manager.reply()).is_err() {
2533                    return Some(self.event(MobileMeshManagementOutcome::Failed));
2534                }
2535                match crawl.next_request() {
2536                    Ok(Some(request)) => match self.manager.begin(&request, now_ms) {
2537                        Ok(()) => {
2538                            self.reported_remaining = None;
2539                            None
2540                        }
2541                        Err(_) => Some(self.event(MobileMeshManagementOutcome::Failed)),
2542                    },
2543                    Ok(None) => Some(self.fetched()),
2544                    Err(_) => Some(self.event(MobileMeshManagementOutcome::Failed)),
2545                }
2546            }
2547        }
2548    }
2549
2550    /// Report a single exchange's reply, read against the shape that was
2551    /// asked for.
2552    fn replied(&self, shape: &ReplyShape) -> MobileMeshManagementEventRecord {
2553        let reply = self.manager.reply();
2554        let mut event = self.event(MobileMeshManagementOutcome::Replied);
2555        match shape {
2556            ReplyShape::Property(key) => match umsh_ulcp::reply::property(*key, reply) {
2557                Ok(answer) => event.answers.push(answer_record(*key, answer)),
2558                Err(_) => return self.event(MobileMeshManagementOutcome::Failed),
2559            },
2560            ReplyShape::Entries(keys) => match umsh_ulcp::reply::entries(keys, reply) {
2561                Ok(entries) => {
2562                    for (key, answer) in entries.flatten() {
2563                        event.answers.push(answer_record(key, answer));
2564                    }
2565                }
2566                // A device without `CAP_CMD_MULTI` declines the command
2567                // itself, which is an answer about the request rather than
2568                // about any one property.
2569                Err(_) => event.status_code = umsh_ulcp::reply::status_of(reply).map(|s| s.0),
2570            },
2571            // A reset the device answered anyway — `CMD_RESTORE` with no
2572            // snapshot to restore — reports like any other status.
2573            ReplyShape::Status | ReplyShape::Acknowledgment => {
2574                match umsh_ulcp::reply::status_of(reply) {
2575                    Some(status) => event.status_code = Some(status.0),
2576                    None => return self.event(MobileMeshManagementOutcome::Failed),
2577                }
2578            }
2579        }
2580        event
2581    }
2582
2583    /// Hand back everything a finished crawl collected.
2584    ///
2585    /// Undecoded on purpose: what these values mean is the caller's
2586    /// question, and it asked for a particular set of properties because
2587    /// it already knew what it wanted with them.
2588    fn fetched(&mut self) -> MobileMeshManagementEventRecord {
2589        let ManagementPlan::Fetch(crawl) = &mut self.plan else {
2590            return self.event(MobileMeshManagementOutcome::Failed);
2591        };
2592        let answers = core::mem::take(&mut crawl.answers);
2593        let mut event = self.event(MobileMeshManagementOutcome::Replied);
2594        event.answers = answers;
2595        event
2596    }
2597}

2650/// Register the target and put the first request on the air.
2651async fn start_management<M: MacBackend>(
2652    node: &LocalNode<M>,
2653    operation_id: u64,
2654    peer: PublicKey,
2655    request: ManagementRequest,
2656    now_ms: u64,
2657    token_seed: u16,
2658) -> Option<ManagementJob<M>> {
2659    let connection = node.peer(peer).await.ok()?;
2660    let mut manager = umsh_node_mgmt::NodeManager::new(connection, token_seed);
2661    // An acknowledgment is what completes a reset and what turns an
2662    // unreachable device into an early answer; a flood budget and a trace
2663    // route are what get a first request to a device no route is known
2664    // for, and teach the MAC the way back.
2665    *manager.send_options_mut() = SendOptions::default()
2666        .with_ack_requested(true)
2667        .with_flood_hops(MANAGEMENT_FLOOD_HOPS)
2668        .with_trace_route();
2669    let (plan, request) = match request {
2670        ManagementRequest::One { frame, shape } => (ManagementPlan::One(shape), frame),
2671        ManagementRequest::Fetch {
2672            property_ids,
2673            multi_hint,
2674        } => {
2675            let mut crawl = FetchCrawl::new(property_ids, multi_hint);
2676            let request = crawl.next_request().ok()??;
2677            (ManagementPlan::Fetch(crawl), request)
2678        }
2679    };
2680    manager.begin(&request, now_ms).ok()?;
2681    Some(ManagementJob {
2682        operation_id,
2683        peer,
2684        manager,
2685        plan,
2686        reported_remaining: None,
2687    })
2688}

2690/// Carry the outstanding operation as far as it goes right now, clearing
2691/// it and reporting once it ends.
2692async fn service_management<M: MacBackend>(
2693    job: &mut Option<ManagementJob<M>>,
2694    now_ms: u64,
2695    events: &NotifyingSender<MobileMeshManagementEventRecord>,
2696) {
2697    let Some(active) = job.as_mut() else {
2698        return;
2699    };
2700    loop {
2701        match active.manager.service(now_ms).await {
2702            Err(_) => {
2703                let _ = events.send(active.event(MobileMeshManagementOutcome::Failed));
2704                *job = None;
2705                return;
2706            }
2707            Ok(umsh_node_mgmt::Progress::Waiting { .. }) => {
2708                if let Some(progress) = active.progress() {
2709                    let _ = events.send(progress);
2710                }
2711                return;
2712            }
2713            Ok(umsh_node_mgmt::Progress::Done(outcome)) => match active.settle(outcome, now_ms) {
2714                Some(event) => {
2715                    let _ = events.send(event);
2716                    *job = None;
2717                    return;
2718                }
2719                None => {
2720                    // Another exchange of the same operation. A crawl says
2721                    // so at every batch boundary, which is the only sign
2722                    // of life a long read gives before it finishes.
2723                    let _ = events.send(active.event(MobileMeshManagementOutcome::Progress));
2724                    continue;
2725                }
2726            },
2727        }
2728    }
2729}

3687                            InboundTextSource::ChannelDirect { channel, peer } => (
3688                                Envelope {
3689                                    path: DeliveryPath::BlindUnicast,
3690                                    conversation: ConversationKey::ChannelDirect { channel, peer },
3691                                    sender: SenderScope::Peer(peer),
3692                                },
3693                                Some(peer),
3694                            ),
3695                        };
3696                        let _ = chat.engine.receive(
3697                            &envelope,
3698                            sender_full_key,
3699                            &text.payload,
3700                            received_at_ms,
3701                        );
3702                        let mut drain = chat.drain();
3703                        // This drain belongs to exactly one frame, so the
3704                        // records it produced are the ones that frame caused.
3705                        attach_rx_metadata(&mut drain.mutations, &text.rx);
3706                        let transmissions = drain.transmissions.clone();
3707                        publish_chat_drain(drain, &chat_events);
3708                        if !transmissions.is_empty() || !pending_chat_transmissions.is_empty() {
3709                            let sent = queue_chat_transmissions(
3710                                &node,
3711                                transmissions,
3712                                &mut pending_chat_transmissions,
3713                                &mut in_flight_chat,
3714                                &chat_pipeline_ready,
3715                                &channel_registry,
3716                                &mut chat,
3717                                received_at_ms,
3718                            )
3719                            .await;
3720                            publish_chat_drain(chat.drain(), &chat_events);
3721                            if sent > 0 && handle.service_counter_persistence().await.is_err() {
3722                                return;
3723                            }
3724                        }
3725                    }
3726                }
3727                _ = protocol_timeout_tick.tick() => {
3728                    timeout_servicer.service().await;
3729                    let now_ms = handle.now_ms().await;
3730                    if management.is_some() {
3731                        // Retries, cursor continuations, and the batches of
3732                        // a crawl all leave on this tick. Persistence is
3733                        // serviced alongside because each of those is a
3734                        // real authenticated send.
3735                        service_management(&mut management, now_ms, &management_events).await;
3736                        if handle.service_counter_persistence().await.is_err() {
3737                            return;
3738                        }
3739                    }
3740                    chat.engine.tick(now_ms);
3741                    service_chat_tickets(
3742                        &mut chat,
3743                        &mut in_flight_chat,
3744                        &mut chat_pipeline_ready,
3745                        &chat_events,
3746                        pending_chat_transmissions.len(),
3747                        now_ms,
3748                    );
3749                    let drain = chat.drain();
3750                    let transmissions = drain.transmissions.clone();
3751                    publish_chat_drain(drain, &chat_events);
3752                    if !transmissions.is_empty() || !pending_chat_transmissions.is_empty() {
3753                        let sent = queue_chat_transmissions(
3754                            &node,
3755                            transmissions,
3756                            &mut pending_chat_transmissions,
3757                            &mut in_flight_chat,
3758                            &chat_pipeline_ready,
3759                            &channel_registry,
3760                            &mut chat,
3761                            now_ms,
3762                        )
3763                        .await;
3764                        publish_chat_drain(chat.drain(), &chat_events);
3765                        if sent > 0 && handle.service_counter_persistence().await.is_err() {
3766                            return;
3767                        }
3768                    }
3769                }
3770            }
3771        }
3772    };
3773
3774    // Either loop ending (pump error, shutdown command, fatal persistence
3775    // failure) ends the session.
3776    tokio::select! {
3777        _ = pump_loop => {}
3778        _ = command_loop => {}
3779    }
3780}

3782async fn queue_chat_transmissions<M: MacBackend>(
3783    node: &LocalNode<M>,
3784    transmissions: Vec<umsh_text::engine::Transmission>,
3785    pending: &mut VecDeque<umsh_text::engine::Transmission>,
3786    in_flight: &mut Vec<InFlightChatTransmission>,
3787    pipeline_ready: &BTreeSet<[u8; 32]>,
3788    channels: &Rc<RefCell<ChannelRegistry>>,
3789    chat: &mut MobileChatState,
3790    now_ms: u64,
3791) -> usize {
3792    pending.extend(transmissions);
3793    // Keep a bounded pipeline aligned with the device's target-selected
3794    // TX queue. The durable pending queue below handles messages larger than
3795    // this window without imposing the mobile RAM choice on embedded MACs.
3796    if in_flight.len() >= MOBILE_CHAT_TRANSMIT_WINDOW {
3797        return 0;
3798    }
3799    let mut queued = 0;
3800    while let Some(transmission) = pending.pop_front() {
3801        let gate_peer = match transmission.destination {
3802            Destination::Peer(peer) => Some(peer),
3803            // Multicast is unaddressed and blind unicast carries no ACK, so
3804            // neither has a peer whose pipeline could be confirmed.
3805            Destination::Channel(_) | Destination::ChannelPeer { .. } => None,
3806        };
3807        if let Some(peer) = gate_peer {
3808            if !pipeline_ready.contains(&peer.0)
3809                && in_flight.iter().any(|entry| entry.gate_peer == Some(peer))
3810            {
3811                // First contact may require counter synchronization. Confirm
3812                // one authenticated frame before opening this peer's full
3813                // pipeline.
3814                pending.push_front(transmission);
3815                break;
3816            }
3817        }
3818        let mut payload = Vec::with_capacity(transmission.payload.len() + 1);
3819        payload.push(PayloadType::TextMessage as u8);
3820        payload.extend_from_slice(transmission.payload.as_slice());
3821        let sent = match transmission.destination {
3822            Destination::Peer(peer) => match node.peer(peer).await {
3823                Ok(connection) => {
3824                    connection
3825                        .send(&payload, &SendOptions::default().with_ack_requested(true))
3826                        .await
3827                }
3828                Err(_) => {
3829                    chat.engine.transmit_update(
3830                        transmission.transmission_id,
3831                        DeliveryState::Failed,
3832                        now_ms,
3833                    );
3834                    continue;
3835                }
3836            },
3837            Destination::Channel(channel) => {
3838                let Some(bound) = bound_channel(node, channels, &channel) else {
3839                    chat.engine.transmit_update(
3840                        transmission.transmission_id,
3841                        DeliveryState::Failed,
3842                        now_ms,
3843                    );
3844                    continue;
3845                };
3846                // Carry the full source address: a member who misses a
3847                // fragment can only ask us to resend it if our frames name
3848                // the key to address that request to.
3849                let mut options = SendOptions::default().with_full_source();
3850                // An emergency message that only channel members can read is
3851                // not an emergency message. The spec forbids encrypting chat
3852                // on `EMERGENCY` so anyone in range can act on it, whether or
3853                // not they hold the key; the full source key it already
3854                // carries is what keeps it attributable without it.
3855                if channel == crate::emergency_channel_tag() {
3856                    options = options.unencrypted();
3857                }
3858                bound.send_all(&payload, &options).await
3859            }
3860            Destination::ChannelPeer { channel, peer } => {
3861                let Some(bound) = bound_channel(node, channels, &channel) else {
3862                    chat.engine.transmit_update(
3863                        transmission.transmission_id,
3864                        DeliveryState::Failed,
3865                        now_ms,
3866                    );
3867                    continue;
3868                };
3869                // The MAC will only address a registered peer, and a channel
3870                // member is not one — nothing about being in a channel
3871                // together registers anybody. Register on the way out rather
3872                // than on sight: only the members we actually have to ask
3873                // something of spend a peer slot, and this is the only place
3874                // we ever ask.
3875                if node.peer(peer).await.is_err() {
3876                    chat.engine.transmit_update(
3877                        transmission.transmission_id,
3878                        DeliveryState::Failed,
3879                        now_ms,
3880                    );
3881                    continue;
3882                }
3883                // A repair request; the engine owns retrying it, so no ACK is
3884                // asked for here.
3885                let mut options = SendOptions::default().with_full_source();
3886                // Repairs carry the same message the multicast did, so they
3887                // are held to the same rule — and have to be, since the
3888                // receiving side refuses encrypted emergency text whatever
3889                // family it arrives in. It stays blind unicast even so: what
3890                // an unencrypted blind unicast still carries over a plain one
3891                // is the channel it names, which is what a repeater decides
3892                // to forward on.
3893                if channel == crate::emergency_channel_tag() {
3894                    options = options.unencrypted();
3895                }
3896                let r = bound.send(&peer, &payload, &options).await;
3897                r
3898            }
3899        };
3900        let ticket = match sent {
3901            Ok(ticket) => ticket,
3902            Err(_) => {
3903                // With a registered peer and an engine-bounded payload, the
3904                // expected failure here is temporary MAC queue / pending-ACK
3905                // capacity. Preserve ordering and retry after tickets advance.
3906                pending.push_front(transmission);
3907                break;
3908            }
3909        };
3910        in_flight.push(InFlightChatTransmission {
3911            transmission_id: transmission.transmission_id,
3912            gate_peer,
3913            ticket,
3914            sent_reported: false,
3915            non_ack: gate_peer.is_none(),
3916            queued_at_ms: now_ms,
3917            stall_reported: false,
3918        });
3919        queued += 1;
3920        if in_flight.len() >= MOBILE_CHAT_TRANSMIT_WINDOW {
3921            break;
3922        }
3923    }
3924    queued
3925}

3927/// Solicit one channel member's identity over the channel they were heard on.
3928///
3929/// The request is a multicast every member receives but only the filtered hint
3930/// answers. Routing follows the evidence that member's own frames left: their
3931/// trace route if one was observed, otherwise a flood budget no larger than
3932/// the distance they were last heard from.
3933async fn request_identity_over_channel<M: MacBackend>(
3934    node: &LocalNode<M>,
3935    channels: &Rc<RefCell<ChannelRegistry>>,
3936    channel: ChannelTag,
3937    hint: NodeHint,
3938    nonce: u32,
3939    route: Option<MemberRoute>,
3940) -> Result<(), MobileMeshError> {
3941    let Some(bound) = bound_channel(node, channels, &channel) else {
3942        return Err(MobileMeshError::UnknownConversation);
3943    };
3944    let options_block = umsh_node::mac_command::IdentityRequestBuilder::new()
3945        .nonce(nonce)
3946        .and_then(|builder| builder.filter_hint(&hint))
3947        .map_err(|_| MobileMeshError::SendFailed)?
3948        .build();
3949    let cmd = umsh_node::MacCommand::IdentityRequest {
3950        options: &options_block,
3951    };
3952    let mut frame = [0u8; 128];
3953    frame[0] = PayloadType::MacCommand as u8;
3954    let length = umsh_node::mac_command::encode(&cmd, &mut frame[1..])
3955        .map_err(|_| MobileMeshError::SendFailed)?
3956        + 1;
3957    // Full source so the member can answer with a targeted unicast rather
3958    // than another multicast.
3959    let mut options = SendOptions::default().with_full_source();
3960    match route.as_ref() {
3961        Some(route) if !route.route_hints.is_empty() => {
3962            let hops = route
3963                .route_hints
3964                .iter()
3965                .filter_map(|hint| <[u8; 2]>::try_from(hint.as_slice()).ok())
3966                .map(umsh_core::RouterHint)
3967                .collect::<Vec<_>>();
3968            // An over-long observed route is not a reason to fail the
3969            // request; fall back to flooding at the distance it implies.
3970            options = match options.try_with_source_route(&hops) {
3971                Ok(options) => options,
3972                Err(_) => SendOptions::default()
3973                    .with_full_source()
3974                    .with_flood_hops(flood_budget(route.hop_count)),
3975            };
3976        }
3977        Some(MemberRoute {
3978            hop_count: Some(hops),
3979            ..
3980        }) => {
3981            options = options.with_flood_hops(flood_budget(Some(*hops)));
3982        }
3983        _ => {}
3984    }
3985    bound
3986        .send_all(&frame[..length], &options)
3987        .await
3988        .map(|_| ())
3989        .map_err(|_| MobileMeshError::SendFailed)
3990}

3992/// How long one page of a Peer Repeaters listing is waited for.
3993///
3994/// A repeater builds its answer from tables it already holds, so the wait is
3995/// the mesh crossing and the responder's own channel-access window, not any
3996/// work on its part.
3997const PEER_REPEATERS_PAGE_TIMEOUT: Duration = Duration::from_secs(30);
3998
3999/// The most pages one listing is followed across.
4000///
4001/// A responder's table is bounded, so an enumeration that keeps handing back
4002/// cursors is a responder that has lost its place; the ask ends rather than
4003/// following it forever.
4004const PEER_REPEATERS_MAX_PAGES: usize = 8;
4005
4006/// The most time one listing is followed for, across all its pages.
4007///
4008/// The worker loop serves every command in turn, so a walk that kept waiting
4009/// out page timeouts back to back would hold the whole session hostage. Pages
4010/// from a live responder arrive in seconds; a walk this old is being dripped
4011/// at, and ends with what it has.
4012const PEER_REPEATERS_WALK_TIMEOUT: Duration = Duration::from_secs(60);
4013
4014/// Ask one repeater for its peer-repeater listing, following cursors until
4015/// the answer is complete.
4016///
4017/// Each page carries its own nonce, so a late page from an abandoned ask
4018/// cannot be mistaken for the one being waited on.
4019async fn collect_peer_repeaters<M: MacBackend>(
4020    node: &LocalNode<M>,
4021    handle: &M,
4022    peer: PublicKey,
4023) -> Result<Vec<MobileMeshPeerRepeaterRecord>, MobileMeshError> {
4024    let connection = node
4025        .peer(peer)
4026        .await
4027        .map_err(|_| MobileMeshError::InvalidPeer)?;
4028
4029    let pages: Rc<RefCell<Vec<Vec<u8>>>> = Rc::new(RefCell::new(Vec::new()));
4030    let _subscription = {
4031        let pages = pages.clone();
4032        node.on_mac_command(move |from, command| {
4033            if from != peer {
4034                return;
4035            }
4036            if let umsh_node::OwnedMacCommand::PeerRepeatersResponse { body } = command {
4037                pages.borrow_mut().push(body.clone());
4038            }
4039        })
4040    };
4041
4042    let mut listing = Vec::new();
4043    let mut cursor: Option<Vec<u8>> = None;
4044    let walk_deadline = tokio::time::Instant::now() + PEER_REPEATERS_WALK_TIMEOUT;
4045    for _ in 0..PEER_REPEATERS_MAX_PAGES {
4046        let mut nonce_bytes = [0u8; 2];
4047        handle.fill_random(&mut nonce_bytes).await;
4048        let nonce = u16::from_be_bytes(nonce_bytes);
4049        pages.borrow_mut().clear();
4050        let sent = connection
4051            .request_peer_repeaters(nonce, cursor.as_deref(), &SendOptions::default())
4052            .await;
4053        if sent.is_err() {
4054            if listing.is_empty() {
4055                return Err(MobileMeshError::SendFailed);
4056            }
4057            // A follow-up ask that cannot leave ends the walk the same way
4058            // an unanswered one does: with the pages already in hand.
4059            break;
4060        }
4061
4062        let deadline =
4063            (tokio::time::Instant::now() + PEER_REPEATERS_PAGE_TIMEOUT).min(walk_deadline);
4064        let page = loop {
4065            let matched = pages.borrow_mut().iter().position(|body| {
4066                umsh_node::mac_command::PeerRepeatersResponseView::new(body).nonce() == Some(nonce)
4067            });
4068            if let Some(index) = matched {
4069                break Some(pages.borrow_mut().remove(index));
4070            }
4071            if tokio::time::Instant::now() >= deadline {
4072                break None;
4073            }
4074            // The worker's pump runs as a sibling future, so yielding here is
4075            // what lets the answer arrive at all.
4076            tokio::time::sleep(Duration::from_millis(20)).await;
4077        };
4078        let Some(page) = page else {
4079            // A listing that stopped part way is still what the repeater
4080            // said; the caller gets it rather than nothing.
4081            break;
4082        };
4083
4084        let view = umsh_node::mac_command::PeerRepeatersResponseView::new(&page);
4085        listing.extend(view.entries().map(peer_repeater_record));
4086        match view.cursor() {
4087            Some(next) => cursor = Some(next.to_vec()),
4088            None => break,
4089        }
4090        if tokio::time::Instant::now() >= walk_deadline {
4091            // No answer to the next ask would be waited for, so it is not
4092            // worth the airtime.
4093            break;
4094        }
4095    }
4096    Ok(listing)
4097}

4129fn service_chat_tickets(
4130    chat: &mut MobileChatState,
4131    in_flight: &mut Vec<InFlightChatTransmission>,
4132    pipeline_ready: &mut BTreeSet<[u8; 32]>,
4133    events: &NotifyingSender<MobileChatWorkerEvent>,
4134    pending_depth: usize,
4135    now_ms: u64,
4136) {
4137    let occupied = in_flight.len();
4138    let mut index = 0;
4139    while index < in_flight.len() {
4140        let entry = &mut in_flight[index];
4141        // Checked before the state transitions below, so an entry that is
4142        // about to retire this pass is not accused on its way out.
4143        if !entry.stall_reported
4144            && !entry.ticket.was_transmitted()
4145            && now_ms.saturating_sub(entry.queued_at_ms) >= CHAT_TRANSMISSION_STALL_MS
4146        {
4147            entry.stall_reported = true;
4148            let waited = now_ms.saturating_sub(entry.queued_at_ms) / 1000;
4149            let transmission_id = entry.transmission_id;
4150            let _ = events.send(MobileChatWorkerEvent::Diagnostic(format!(
4151                "transmission {transmission_id} has not left the radio after {waited}s \
4152                 ({occupied}/{MOBILE_CHAT_TRANSMIT_WINDOW} window slots used, \
4153                 {pending_depth} more waiting)"
4154            )));
4155        }
4156        let entry = &mut in_flight[index];
4157        if entry.ticket.was_transmitted() && !entry.sent_reported {
4158            chat.engine
4159                .transmit_update(entry.transmission_id, DeliveryState::Sent, now_ms);
4160            entry.sent_reported = true;
4161        }
4162        if entry.non_ack && entry.sent_reported {
4163            // Nothing further can happen to this one: no acknowledgement is
4164            // coming, so transmission is where it ends. Retiring it here is
4165            // what keeps channel sends from accumulating in flight forever.
4166            in_flight.swap_remove(index);
4167        } else if entry.ticket.was_acked() {
4168            if let Some(peer) = entry.gate_peer {
4169                pipeline_ready.insert(peer.0);
4170            }
4171            chat.engine
4172                .transmit_update(entry.transmission_id, DeliveryState::Acked, now_ms);
4173            in_flight.swap_remove(index);
4174        } else if entry.ticket.has_failed() {
4175            chat.engine
4176                .transmit_update(entry.transmission_id, DeliveryState::Failed, now_ms);
4177            in_flight.swap_remove(index);
4178        } else {
4179            index += 1;
4180        }
4181    }
4182}

6819    /// A batch id is issued exactly when the batch has events in it, and never
6820    /// otherwise. The platform reads the id as its whole signal to apply and
6821    /// acknowledge, so a batch made of only one kind of event — a lone sender
6822    /// resolution, say — must still be announced. One batch left
6823    /// unacknowledged holds the slot for the rest of the session, and every
6824    /// delivery receipt behind it never arrives: messages transmit fine and
6825    /// stay on "Sending" forever, in every conversation at once.
6826    fn assert_batch_id_matches_events(update: &MobileMeshSessionUpdateRecord) {
6827        let has_events = !update.chat_mutations.is_empty()
6828            || !update.chat_deliveries.is_empty()
6829            || !update.chat_archive_lookups.is_empty()
6830            || !update.chat_sender_resolutions.is_empty()
6831            || !update.chat_diagnostics.is_empty();
6832        assert_eq!(
6833            update.chat_batch_id.is_some(),
6834            has_events,
6835            "batch id {:?} disagrees with the batch's contents",
6836            update.chat_batch_id
6837        );
6838    }
