1//! Owned mobile facade records for the sans-I/O text engine.
2//!
3//! Conversations are addressed by an opaque string that discriminates the two
4//! kinds the platform can hold: a peer's base58 address for a direct
5//! conversation, and `ch:<hex tag>` for a channel's group conversation. Room
6//! profiles are still outside this facade, and are dropped rather than
7//! exported in a form Swift would have to interpret.

81/// Physical-layer metadata for a record produced by a live received frame.
82///
83/// The engine is transport-agnostic and carries none of this, so the facade
84/// attaches it alongside the mutation the frame produced. Absent on records
85/// that came from a timer, a compose, or a repair drain.
86#[derive(Clone, Debug, PartialEq, Eq, uniffi::Record)]
87pub struct MobileChatRxMetadataRecord {
88    pub rssi_dbm: Option<i16>,
89    pub snr_centibels: Option<i16>,
90    pub lqi: Option<u8>,
91    /// Radio links the frame crossed to get here, counting the final one into
92    /// this device: a frame heard directly from its sender is one hop. The
93    /// same count a ping reply reports, and absent for the same reason — a
94    /// frame source-routed without a trace route crossed hops nobody
95    /// recorded.
96    pub hop_count: Option<u8>,
97    /// Intermediate-router hints in trace-route order: each forwarding
98    /// repeater prepends its own hint, so the list starts nearest us and ends
99    /// nearest the sender. That is return-path order — usable directly as a
100    /// source route back — and the reverse of the path the frame travelled.
101    pub route_hints: Vec<Vec<u8>>,
102    pub source_authenticated: bool,
103}

296/// The channels this session can hold a conversation in, and the keys to
297/// reach them.
298///
299/// Channel membership already lives in the MAC, which resolves an inbound
300/// frame to a key by authenticating it. This is the chat layer's own view:
301/// it maps between the tag a conversation is keyed by and the key a send
302/// needs, in both directions.
303#[derive(Default)]
304pub(crate) struct ChannelRegistry {
305    entries: BTreeMap<ChannelTag, ChannelKey>,
306}
307
308impl ChannelRegistry {
309    pub fn register(&mut self, tag: ChannelTag, key: ChannelKey) {
310        self.entries.insert(tag, key);
311    }
312
313    pub fn remove(&mut self, tag: &ChannelTag) {
314        self.entries.remove(tag);
315    }
316
317    pub fn key(&self, tag: &ChannelTag) -> Option<ChannelKey> {
318        self.entries.get(tag).copied()
319    }
320
321    pub fn contains(&self, tag: &ChannelTag) -> bool {
322        self.entries.contains_key(tag)
323    }
324}

326pub(crate) struct MobileChatState {
327    /// Keep the reducer off the worker future's stack. The mobile MAC/host is
328    /// already a large bounded value, and combining both inline can exhaust a
329    /// debug-build thread stack.
330    pub engine: Box<ChatEngine>,
331    pub session_id: u64,
332    next_batch_id: u64,
333    pub pending_batches: BTreeMap<u64, PendingChatBatch>,
334    /// Shared with the worker so a batch rejection — which rebuilds the
335    /// reducer — cannot lose the channels the platform registered.
336    pub channels: Rc<RefCell<ChannelRegistry>>,
337    /// Full keys learned for claimed member hints, per channel. A hint is only
338    /// 3 bytes and two members could in principle claim the same one, so this
339    /// records the first key seen for a hint and leaves it there.
340    resolved_members: BTreeMap<(ChannelTag, [u8; 3]), PublicKey>,
341    /// Hints already reported to the platform, so one resolution is announced
342    /// once rather than on every frame.
343    announced_members: BTreeSet<(ChannelTag, [u8; 3])>,
344}

346impl MobileChatState {
347    pub fn new(local_key: PublicKey, channels: Rc<RefCell<ChannelRegistry>>) -> Self {
348        Self {
349            engine: Box::new(ChatEngine::new(
350                DirectChannelProfile,
351                local_key,
352                EngineConfig::default(),
353                rand::rng().next_u64(),
354            )),
355            session_id: rand::rng().next_u64().max(1),
356            next_batch_id: 1,
357            pending_batches: BTreeMap::new(),
358            channels,
359            resolved_members: BTreeMap::new(),
360            announced_members: BTreeSet::new(),
361        }
362    }

364    pub fn restore(
365        &mut self,
366        checkpoints: &[MobileChatCheckpointRecord],
367        now_ms: u64,
368    ) -> Vec<String> {
369        let mut diagnostics = Vec::new();
370        let mut restored = Vec::new();
371        for record in checkpoints {
372            match self.checkpoint_from_record(record) {
373                Some(checkpoint) => restored.push(checkpoint),
374                None => diagnostics.push(format!(
375                    "chat checkpoint for unknown conversation {}",
376                    record.conversation_address
377                )),
378            }
379        }
380        self.engine.restore(&restored, now_ms);
381        let _ = self.drain();
382        diagnostics
383    }

385    /// Note a channel member's full key, learned from a frame that carried
386    /// both it and the claimed hint. Returns a record the first time a hint
387    /// resolves, so the platform can upgrade rows it stored anonymously.
388    pub fn resolve_member(
389        &mut self,
390        channel: ChannelTag,
391        hint: NodeHint,
392        key: PublicKey,
393    ) -> Option<MobileChatSenderResolutionRecord> {
394        self.resolved_members
395            .entry((channel, hint.0))
396            .or_insert(key);
397        if !self.announced_members.insert((channel, hint.0)) {
398            return None;
399        }
400        Some(MobileChatSenderResolutionRecord {
401            conversation_address: channel_address(channel),
402            sender_hint: hint.0.to_vec(),
403            sender_address: address(key),
404        })
405    }

483    /// Resolve the platform's persisted identity of a regarding target to an
484    /// engine reference, on the same terms as [`Self::compose_ref`].
485    fn regarding_ref(&self, target: &MobileChatRegardingRef) -> Option<RegardingRef> {
486        if target.session_id == self.session_id {
487            return Some(RegardingRef::Handle(MessageHandle(target.handle)));
488        }
489        let sender_hint = match &target.sender_hint {
490            None => None,
491            Some(bytes) => Some(NodeHint(<[u8; 3]>::try_from(bytes.as_slice()).ok()?)),
492        };
493        Some(RegardingRef::Wire {
494            message_id: target.wire_id?,
495            direction: match target.direction? {
496                MobileChatDirection::Inbound => Direction::Inbound,
497                MobileChatDirection::Outbound => Direction::Outbound,
498            },
499            sender_hint,
500            epoch: target.epoch,
501        })
502    }

504    /// Resolve the platform's persisted identity of an original message to
505    /// an engine compose reference. Same facade session: the engine handle
506    /// is still live. Earlier session: fall back to the persisted wire
507    /// identity, which the engine validates against stream continuity.
508    fn compose_ref(&self, original: &MobileChatOriginalRef) -> Option<ComposeRef> {
509        if original.session_id == self.session_id {
510            return Some(ComposeRef::Handle(MessageHandle(original.handle)));
511        }
512        match (original.wire_id, original.epoch) {
513            (Some(message_id), Some(epoch)) => Some(ComposeRef::Wire { message_id, epoch }),
514            _ => None,
515        }
516    }

518    fn compose_batch(
519        &mut self,
520        conversation: ConversationKey,
521        client_token: u32,
522        intent: ComposeIntent<'_>,
523        now_ms: u64,
524    ) -> Result<ComposedChatBatch, ()> {
525        self.engine
526            .compose(conversation, client_token, intent, now_ms)
527            .map_err(|_| ())?;
528        let mut drain = self.drain();
529        let checkpoint = drain.checkpoint.ok_or(())?;
530        let fragment_count = u8::try_from(drain.archives.len()).map_err(|_| ())?;
531        for mutation in &mut drain.mutations {
532            if mutation.kind == MobileChatMutationKind::Insert
533                && mutation.direction == Some(MobileChatDirection::Outbound)
534            {
535                mutation.fragment_count = Some(fragment_count.max(1));
536            }
537        }
538        let batch_id = self.next_batch_id;
539        self.next_batch_id = self.next_batch_id.wrapping_add(1).max(1);
540        self.pending_batches.insert(
541            batch_id,
542            PendingChatBatch {
543                transmissions: drain.transmissions,
544            },
545        );
546        Ok(ComposedChatBatch {
547            record: MobileChatComposeBatchRecord {
548                batch_id,
549                checkpoint,
550                archive_deletes: drain.archive_deletes,
551                archives: drain.archives,
552                mutations: drain.mutations,
553            },
554            deliveries: drain.deliveries,
555            diagnostics: drain.diagnostics,
556        })
557    }

559    pub fn drain(&mut self) -> ChatDrain {
560        let mut drained = ChatDrain::new();
561        while let Some(output) = self.engine.poll_output() {
562            match output {
563                Output::Transmit(transmission) => {
564                    if let Some(archive) = transmission.archive {
565                        if let Some(record) =
566                            self.archive_record(archive, transmission.payload.as_slice())
567                        {
568                            drained.archives.push(record);
569                        }
570                    }
571                    drained.transmissions.push(transmission);
572                }
573                Output::StoreCheckpoint {
574                    conversation,
575                    next_id,
576                    epoch,
577                } => {
578                    drained.checkpoint =
579                        self.conversation_address(conversation)
580                            .map(|conversation_address| MobileChatCheckpointRecord {
581                                conversation_address,
582                                next_id,
583                                epoch,
584                            });
585                }
586                Output::LookupOutbound {
587                    request_id,
588                    conversation,
589                    sequence,
590                } => {
591                    if let Some(conversation_address) = self.conversation_address(conversation) {
592                        drained.lookups.push(MobileChatArchiveLookupRecord {
593                            request_id,
594                            conversation_address,
595                            message_id: sequence.message_id,
596                            fragment_index: sequence.fragment.map(|fragment| fragment.index),
597                        });
598                    }
599                }
600                Output::StoreArchive { key, payload } => {
601                    if let Some(record) = self.archive_record(key, payload.as_slice()) {
602                        drained.archives.push(record);
603                    }
604                }
605                Output::DeleteArchive {
606                    conversation,
607                    message_id,
608                } => {
609                    if let Some(conversation_address) = self.conversation_address(conversation) {
610                        drained.archive_deletes.push(MobileChatArchiveDeleteRecord {
611                            conversation_address,
612                            message_id,
613                        });
614                    }
615                }
616                Output::StoreMessage(mutation) => {
617                    if let Some(record) = self.mutation_record(mutation) {
618                        drained.mutations.push(record);
619                    }
620                }
621                Output::Event(Event::DeliveryStateChanged {
622                    handle,
623                    fragment,
624                    state,
625                }) => drained.deliveries.push(MobileChatDeliveryRecord {
626                    session_id: self.session_id,
627                    handle: handle.0,
628                    fragment_index: fragment,
629                    state: match state {
630                        DeliveryState::Sent => MobileChatDeliveryState::Sent,
631                        DeliveryState::Acked => MobileChatDeliveryState::Acknowledged,
632                        DeliveryState::Failed => MobileChatDeliveryState::Failed,
633                    },
634                }),
635                Output::Event(event) => drained.diagnostics.push(format!("{event:?}")),
636                Output::Diagnostic(diagnostic) => {
637                    drained.diagnostics.push(format!("{diagnostic:?}"));
638                }
639            }
640        }
641        drained
642    }

672            fragment_count: None,
673            finalized: None,
674            presence: MobileChatPresence::Present,
675            received_late: false,
676            notify: false,
677            rx: None,
678        };
679        match mutation.kind {
680            MutationKind::Insert {
681                conversation,
682                sender,
683                direction,
684                message_type,
685                wire_id,
686                epoch,
687                client_token,
688                sender_handle,
689                regarding,
690                bg_color,
691                text_color,
692                body,
693                status,
694                presence,
695                late,
696                notify,
697            } => {
698                record.conversation_address = self.conversation_address(conversation);
699                record.sender_address = self.sender_address(conversation, sender);
700                record.sender_hint = claimed_hint(sender);
701                record.direction = Some(match direction {
702                    Direction::Inbound => MobileChatDirection::Inbound,
703                    Direction::Outbound => MobileChatDirection::Outbound,
704                });
705                record.message_type = Some(message_type.to_byte());
706                record.wire_id = wire_id;
707                record.epoch = Some(epoch);
708                record.client_token = client_token;
709                record.sender_handle =
710                    sender_handle.map(|value| self.engine.body(&value).to_owned());
711                if let Some(regarding) = regarding {
712                    apply_regarding(&mut record, regarding);
713                }
714                record.background_color = bg_color.map(|color| color.to_vec());
715                record.text_color = text_color.map(|color| color.to_vec());
716                record.body = Some(self.engine.body(&body).to_owned());
717                record.presence = mobile_presence(presence);
718                record.received_late = late;
719                record.notify = notify;
720                apply_completion(&mut record, status);
721            }
722            MutationKind::UpdateBody {
723                body,
724                status,
725                late,
726                notify,
727            } => {
728                record.kind = MobileChatMutationKind::UpdateBody;
729                record.body = Some(self.engine.body(&body).to_owned());
730                record.received_late = late;
731                record.notify = notify;
732                apply_completion(&mut record, status);
733            }
734            MutationKind::Edit {
735                conversation,
736                original,
737                body,
738            } => {
739                record.kind = MobileChatMutationKind::Edit;
740                record.conversation_address = self.conversation_address(conversation);
741                apply_original(&mut record, original);
742                record.body = Some(self.engine.body(&body).to_owned());
743            }
744            MutationKind::Delete {
745                conversation,
746                original,
747            } => {
748                record.kind = MobileChatMutationKind::Delete;
749                record.conversation_address = self.conversation_address(conversation);
750                apply_original(&mut record, original);
751            }
752        }
753        Some(record)
754    }

756    /// The platform-facing address of a conversation: a peer's base58 address
757    /// for a direct one, a prefixed tag for a channel. Rooms have no address
758    /// in this facade.
759    fn conversation_address(&self, conversation: ConversationKey) -> Option<String> {
760        match conversation {
761            ConversationKey::Direct { peer } => Some(address(peer)),
762            ConversationKey::ChannelGroup { channel } => Some(channel_address(channel)),
763            ConversationKey::ChannelDirect { channel, peer } => Some(format!(
764                "{CHANNEL_DIRECT_ADDRESS_PREFIX}{}:{}",
765                hex(&channel.0),
766                address(peer)
767            )),
768            ConversationKey::Room { .. } => None,
769        }
770    }
771
772    /// The sender's full address when one is known. A multicast member claims
773    /// only a hint, so this is whatever key that hint has resolved to.
774    fn sender_address(&self, conversation: ConversationKey, sender: SenderScope) -> Option<String> {
775        match sender {
776            SenderScope::Peer(peer) => Some(address(peer)),
777            SenderScope::Local => None,
778            SenderScope::ClaimedMember(hint) => {
779                let ConversationKey::ChannelGroup { channel } = conversation else {
780                    return None;
781                };
782                self.resolved_members
783                    .get(&(channel, hint.0))
784                    .map(|peer| address(*peer))
785            }
786        }
787    }

810    /// Resolve a stored address back to a conversation. A channel address only
811    /// resolves while its key is registered — the platform registers channels
812    /// before restoring chat, so an unresolvable one means the channel was
813    /// left.
814    pub fn parse_conversation_address(&self, value: &str) -> Option<ConversationKey> {
815        if let Some(rest) = value.strip_prefix(CHANNEL_DIRECT_ADDRESS_PREFIX) {
816            let (tag, peer) = rest.split_once(':')?;
817            let channel = parse_channel_tag(tag)?;
818            if !self.channels.borrow().contains(&channel) {
819                return None;
820            }
821            return Some(ConversationKey::ChannelDirect {
822                channel,
823                peer: decode_address(peer)?,
824            });
825        }
826        if let Some(tag) = value.strip_prefix(CHANNEL_ADDRESS_PREFIX) {
827            let channel = parse_channel_tag(tag)?;
828            if !self.channels.borrow().contains(&channel) {
829                return None;
830            }
831            return Some(ConversationKey::ChannelGroup { channel });
832        }
833        Some(ConversationKey::Direct {
834            peer: decode_address(value)?,
835        })
836    }

839fn mobile_presence(presence: Presence) -> MobileChatPresence {
840    match presence {
841        Presence::Present => MobileChatPresence::Present,
842        Presence::GapPending => MobileChatPresence::GapPending,
843        Presence::Unavailable => MobileChatPresence::Unavailable,
844    }

847fn apply_completion(record: &mut MobileChatMutationRecord, status: CompletionStatus) {
848    match status {
849        CompletionStatus::Complete => record.complete = Some(true),
850        CompletionStatus::Partial {
851            present,
852            count,
853            finalized,
854        } => {
855            record.complete = Some(false);
856            record.present_fragments = Some(present);
857            record.fragment_count = Some(count);
858            record.finalized = Some(finalized);
859        }
860    }

863/// Export an edit/delete target: a live handle when resolved, otherwise the
864/// wire reference for the platform to match against its persisted rows.
865/// Room-scoped reference forms are outside this facade.
866fn apply_original(record: &mut MobileChatMutationRecord, reference: ResolvedRef) {
867    match reference {
868        ResolvedRef::Handle(MessageHandle(handle)) => {
869            record.original_handle = Some(handle);
870        }
871        ResolvedRef::Unresolved(WireRef::SenderScoped { sender, message_id }) => {
872            record.original_wire_id = Some(message_id);
873            record.original_direction = Some(wire_ref_direction(sender));
874            record.original_sender_hint = claimed_hint(sender);
875        }
876        ResolvedRef::Unresolved(WireRef::RoomCanonical { .. }) => {}
877    }
878}
879
880/// Export a reply/emote target, on the same terms as [`apply_original`].
881/// Reactions overwhelmingly target messages received long before this
882/// process started, so the unresolved form is the common case here rather
883/// than the exception.
884fn apply_regarding(record: &mut MobileChatMutationRecord, reference: ResolvedRef) {
885    match reference {
886        ResolvedRef::Handle(MessageHandle(handle)) => {
887            record.regarding_handle = Some(handle);
888        }
889        ResolvedRef::Unresolved(WireRef::SenderScoped { sender, message_id }) => {
890            record.regarding_wire_id = Some(message_id);
891            record.regarding_direction = Some(wire_ref_direction(sender));
892            record.regarding_sender_hint = claimed_hint(sender);
893        }
894        ResolvedRef::Unresolved(WireRef::RoomCanonical { .. }) => {}
895    }
896}
897
898fn wire_ref_direction(sender: SenderScope) -> MobileChatDirection {
899    match sender {
900        SenderScope::Local => MobileChatDirection::Outbound,
901        // A channel member's message is inbound like any other peer's; the
902        // hint is what tells the platform whose stream it was.
903        SenderScope::Peer(_) | SenderScope::ClaimedMember(_) => MobileChatDirection::Inbound,
904    }
905}
906
907fn claimed_hint(sender: SenderScope) -> Option<Vec<u8>> {
908    match sender {
909        SenderScope::ClaimedMember(hint) => Some(hint.0.to_vec()),
910        SenderScope::Local | SenderScope::Peer(_) => None,
911    }
912}

914/// The conversation address of a channel's group conversation.
915pub(crate) fn channel_address(channel: ChannelTag) -> String {
916    format!("{CHANNEL_ADDRESS_PREFIX}{}", hex(&channel.0))
917}
918
919fn parse_channel_tag(value: &str) -> Option<ChannelTag> {
920    if value.len() != 32 {
921        return None;
922    }
923    let mut bytes = [0u8; 16];
924    for (index, byte) in bytes.iter_mut().enumerate() {
925        *byte = u8::from_str_radix(value.get(index * 2..index * 2 + 2)?, 16).ok()?;
926    }
927    Some(ChannelTag(bytes))
928}
929
930fn hex(bytes: &[u8]) -> String {
931    bytes.iter().map(|byte| format!("{byte:02x}")).collect()
932}
933
934fn address(key: PublicKey) -> String {
935    umsh_core::base58::encode(&key.0)
936        .into_iter()
937        .map(char::from)
938        .collect()
939}
940
941fn decode_address(value: &str) -> Option<PublicKey> {
942    umsh_core::base58::decode(value.as_bytes())
943        .ok()
944        .map(PublicKey)
945}
