413/// The two relationships a phone can have with a radio.
414///
415/// They are different things with different lifecycles, and Swift should
416/// model them as different objects: "my radio" is exactly one, tethered,
417/// and re-provisioned on every attach; "radios I administer" is any
418/// number, configured but never claimed. One list must not serve both.
419#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, uniffi::Enum)]
420pub enum UlcpAttachMode {

422    /// the radio filters, queues and acknowledges on its behalf.
423    #[default]
424    Tethered,
425    /// This phone is administering the radio without claiming it. A
426    /// phone commissioning ten repeaters must not write its host key on
427    /// any of them.
428    Administrative,
429}

463    /// Begin post-attach synchronization for a new transport generation.
464    pub fn begin(
465        &self,
466        selected_host_key: Option<Vec<u8>>,
467    ) -> Result<UlcpSessionUpdateRecord, MobileError> {
468        let selected_host_key = selected_host_key
469            .map(|key| {
470                key.try_into()
471                    .map_err(|_| MobileError::InvalidPublicKeyLength)
472            })
473            .transpose()?;
474        let mut state = self.inner.lock().expect("ULCP session mutex poisoned");
475        let generation = state.generation.wrapping_add(1);
476        *state = UlcpSessionState {
477            generation,
478            mode: self.mode,
479            stage: SessionStage::Initial,
480            selected_host_key,
481            ..UlcpSessionState::default()
482        };
483
484        let mut outbound = Vec::new();
485        for property in [
486            prop::LAST_STATUS,
487            prop::PROTOCOL_VERSION,
488            prop::CAPS,
489            prop::DEV_KEY,
490            prop::DEV_NAME,
491            prop::BATTERY,
492            prop::HOST_KEY,
493        ] {
494            outbound.push(state.get_property(property)?);
495        }
496        Ok(state.update(outbound))
497    }

499    /// Replace an unclaimed or other-host configuration with this phone's key.
500    pub fn claim(&self, host_key: Vec<u8>) -> Result<UlcpSessionUpdateRecord, MobileError> {
501        // Commissioning is not tethering: an administrative session
502        // configures the radio's own domain and never writes a host key.
503        if self.mode == UlcpAttachMode::Administrative {
504            return Err(MobileError::AdministrativeSession);
505        }
506        let host_key: [u8; 32] = host_key
507            .try_into()
508            .map_err(|_| MobileError::InvalidPublicKeyLength)?;
509        let mut state = self.inner.lock().expect("ULCP session mutex poisoned");
510        if state.stage != SessionStage::AwaitingHost
511            || !matches!(
512                state.ownership(),
513                UlcpHostOwnership::Unclaimed | UlcpHostOwnership::OtherHost
514            )
515        {
516            return Err(MobileError::InvalidUlcpFrame);
517        }
518        state.selected_host_key = Some(host_key);
519        state.stage = SessionStage::Claiming;
520        state.expected.clear();
521        let tid = state.allocate_tid();
522        state.expected.insert(tid, ExpectedResponse::Claim);
523        let frame = ulcp_prop_set(tid, prop::HOST_KEY, host_key.to_vec())?;
524        Ok(state.update(vec![frame]))
525    }

527    /// Erase ALL mutable state on the radio (saved provisioning, device
528    /// identity, BLE bonds, pairing PIN, every persisted journal) and
529    /// reboot it. The radio does not reply — the reset drops the link —
530    /// so this is fire-and-forget: send the frame, then treat the ensuing
531    /// disconnect as completion. Permitted from any stage so a misbehaving
532    /// radio can always be wiped; unlike `claim`/`configure` it makes no
533    /// stage or ownership demands.
534    pub fn factory_reset(&self) -> Result<UlcpSessionUpdateRecord, MobileError> {
535        let mut state = self.inner.lock().expect("ULCP session mutex poisoned");
536        let tid = state.allocate_tid();
537        // Deliberately no ExpectedResponse: the device wipes storage and
538        // reboots without answering, so `update` reports
539        // waiting_for_responses = false and the caller does not block.
540        let frame = ulcp_factory_reset(tid)?;
541        Ok(state.update(vec![frame]))
542    }

544    /// Start or stop the radio's locate alert (`PROP_ALERT`) so a
545    /// misplaced radio can be found.
546    ///
547    /// Not part of `configure_device`, and never saved: this is live
548    /// behavior rather than configuration, and it deliberately survives
549    /// the phone walking out of BLE range — which is precisely when a
550    /// search needs it. What ends it is this call, a button press at the
551    /// radio, or the radio's own deadline; the latter two arrive as an
552    /// unsolicited `PROP_ALERT` carried on the session snapshot.
553    ///
554    /// Re-sending `Locate` while an alert is running restarts that
555    /// deadline, which is how a longer search keeps the alert alive.
556    pub fn set_alert(&self, state: UlcpAlertState) -> Result<UlcpSessionUpdateRecord, MobileError> {
557        let mut session = self.inner.lock().expect("ULCP session mutex poisoned");
558        if session.stage != SessionStage::Attached {
559            return Err(MobileError::InvalidUlcpFrame);
560        }
561        if !session.has_capability(cap::ALERT)? {
562            return Err(MobileError::UnsupportedCapability);
563        }
564        let mut value = [0u8; pui::MAX_LEN];
565        let len = pui::encode(state.to_wire().code(), &mut value)
566            .map_err(|_| MobileError::InvalidUlcpFrame)?;
567        let tid = session.allocate_tid();
568        session
569            .expected
570            .insert(tid, ExpectedResponse::Property(prop::ALERT));
571        let frame = ulcp_prop_set(tid, prop::ALERT, value[..len].to_vec())?;
572        Ok(session.update(vec![frame]))
573    }

575    /// Apply, verify, and persist a complete radio-settings snapshot.
576    pub fn configure(
577        &self,
578        settings: UlcpRadioSettingsRecord,
579    ) -> Result<UlcpSessionUpdateRecord, MobileError> {
580        let mut state = self.inner.lock().expect("ULCP session mutex poisoned");
581        if state.stage != SessionStage::Attached {

593    /// Apply, verify, and persist a complete configuration of the device's
594    /// own domain: its radio, the role it advertises, and whether and how
595    /// it forwards for the mesh on its own.
596    ///
597    /// This is what commissioning writes. It touches nothing in the host
598    /// domain — no host key, no filters, no queues — so it is equally
599    /// valid from an administrative session on someone else's radio and
600    /// from a tethered session on this phone's own.
601    pub fn configure_device(
602        &self,
603        configuration: UlcpDeviceConfigRecord,
604    ) -> Result<UlcpSessionUpdateRecord, MobileError> {
605        let mut state = self.inner.lock().expect("ULCP session mutex poisoned");
606        if state.stage != SessionStage::Attached {
607            return Err(MobileError::InvalidUlcpFrame);
608        }
609        validate_radio_settings(&configuration.radio, &state)?;
610        let device_values = validate_device_settings(&configuration, &state)?;
611
612        state.expected.clear();
613        state.configuration_queue =
614            state.writable(configuration_values(configuration.radio, device_values));
615        let mut outbound = Vec::new();
616        state.start_configuration(&mut outbound)?;
617        Ok(state.update(outbound))
618    }

621    /// snapshot. The existing snapshot remains usable while the bounded
622    /// refresh is in flight; authoritative provisioning is published when
623    /// the full capability-gated read completes.
624    pub fn refresh(&self) -> Result<UlcpSessionUpdateRecord, MobileError> {
625        let mut state = self.inner.lock().expect("ULCP session mutex poisoned");
626        if state.stage != SessionStage::Attached || !state.expected.is_empty() {
627            return Err(MobileError::InvalidUlcpFrame);
628        }
629        let capabilities = state
630            .responses
631            .get(&prop::CAPS)
632            .ok_or(MobileError::InvalidUlcpFrame)?
633            .value
634            .clone();
635        state.inspection_queue = ulcp_refresh_properties(capabilities)?.into();
636        let mut outbound = Vec::new();
637        state.start_refresh(&mut outbound)?;
638        Ok(state.update(outbound))
639    }

642    /// (`PROP_DEV_PEERS`), then persist with a chained `CMD_SAVE` when the

663        let frame = ulcp_prop_insert(tid, prop::DEV_PEERS, &public_key)?;

691    /// Queue one complete raw UMSH frame on `STR_PHY_RAW`.
692    ///
693    /// The platform adapter supplies only opaque bytes from `MobileMeshSession`;
694    /// Rust owns the ULCP command, stream identifier, metadata, TID, and
695    /// confirmation matching. `nocca` sets `TX_FLAG_NOCCA` so the device
696    /// transmits without its pre-transmit channel-activity check — used for
697    /// immediate MAC acks (see [`MobileMeshOutboundFrameRecord::nocca`]).
698    pub fn transmit_raw(
699        &self,
700        data: Vec<u8>,
701        nocca: bool,
702    ) -> Result<UlcpSessionUpdateRecord, MobileError> {
703        let mut state = self.inner.lock().expect("ULCP session mutex poisoned");
704        let raw_pipeline_active = state
705            .expected
706            .values()
707            .all(|expected| matches!(expected, ExpectedResponse::RawTransmit));
708        if state.stage != SessionStage::Attached || !raw_pipeline_active || data.is_empty() {
709            return Err(MobileError::InvalidUlcpFrame);
710        }
711        let mut available_tid = None;
712        for _ in 0..usize::from(frame::TID_MAX) {
713            let candidate = state.allocate_tid();
714            if !state.expected.contains_key(&candidate) {
715                available_tid = Some(candidate);
716                break;
717            }
718        }
719        let tid = available_tid.ok_or(MobileError::InvalidUlcpFrame)?;
720        state.expected.insert(tid, ExpectedResponse::RawTransmit);
721        let mut metadata = [0u8; umsh_ulcp::TxMeta::WIRE_LEN];
722        let flags = if nocca {
723            umsh_ulcp::meta::TX_FLAG_NOCCA
724        } else {
725            0
726        };
727        umsh_ulcp::TxMeta {
728            flags,
729            ..umsh_ulcp::TxMeta::default()
730        }
731        .encode(&mut metadata)
732        .map_err(|_| MobileError::InvalidUlcpFrame)?;
733        let mut frame = vec![0u8; data.len() + 16];
734        let len = umsh_ulcp::frame::str_send(
735            &mut frame,
736            tid,
737            umsh_ulcp::ids::stream::PHY_RAW,
738            &data,
739            &metadata,
740        )
741        .map_err(|_| MobileError::InvalidUlcpFrame)?;
742        frame.truncate(len);
743        let mut update = state.update(vec![frame]);
744        update.raw_transmit_started_transaction_id = Some(tid);
745        Ok(update)
746    }

748    /// Consume one complete ULCP frame and advance the session reducer.
749    pub fn consume(&self, frame: Vec<u8>) -> Result<UlcpSessionUpdateRecord, MobileError> {
750        let parsed = Frame::parse(&frame).map_err(|_| MobileError::InvalidUlcpFrame)?;
751        if parsed.command() == Some(Cmd::StrRecv) {
752            if parsed.header.tid() != frame::TID_UNSOLICITED {
753                return Err(MobileError::InvalidUlcpFrame);
754            }
755            let payload =
756                StreamPayload::parse(parsed.payload).map_err(|_| MobileError::InvalidUlcpFrame)?;
757            if payload.stream != umsh_ulcp::ids::stream::PHY_RAW {
758                return Err(MobileError::InvalidUlcpFrame);
759            }
760            let metadata = BufferedRxMeta::decode(payload.metadata)
761                .map_err(|_| MobileError::InvalidUlcpFrame)?;
762            let mut state = self.inner.lock().expect("ULCP session mutex poisoned");
763            if state.stage == SessionStage::Idle {
764                return Err(MobileError::InvalidUlcpFrame);
765            }
766            return Ok(state.update_with_received(vec![UlcpReceivedFrameRecord {
767                data: payload.data.to_vec(),
768                rssi_dbm: metadata.rx.rssi_dbm,
769                lqi: metadata.rx.lqi.map(core::num::NonZeroU8::get),
770                snr_cb: metadata.rx.snr_cb,
771                was_buffered: metadata.flags & RX_FLAG_BUFFERED != 0,
772                was_acknowledged: metadata.flags & RX_FLAG_ACKED != 0,
773                age_seconds: metadata.age_s,
774            }]));
775        }
776        let response = inspect_ulcp_property_frame(frame)?;
777        let mut state = self.inner.lock().expect("ULCP session mutex poisoned");
778        let mut outbound = Vec::new();
779        let mut raw_transmit_result = None;
780        let mut operation_error = None;
781
782        if response.transaction_id == frame::TID_UNSOLICITED {
783            if response.command == Cmd::PropIs as u8 {
784                state
785                    .responses
786                    .insert(response.property_id, response.clone());
787            }
788            state.apply_property(&response)?;
789            state.refresh_attached_snapshot(Some(response.property_id))?;
790            return Ok(state.update(outbound));
791        }
792
793        let expected = state
794            .expected
795            .remove(&response.transaction_id)
796            .ok_or(MobileError::InvalidUlcpFrame)?;
797        match expected {
798            ExpectedResponse::Property(property) => {
799                if response.property_id == prop::LAST_STATUS && property != prop::LAST_STATUS {
800                    // A capability-gated property the device declines is one
801                    // unknown setting, not an unusable device: drop any value
802                    // cached from an earlier read so the reduction reports it
803                    // as unreadable, and carry on with the rest of the queue.
804                    let expected_property = matches!(
805                        state.stage,
806                        SessionStage::Inspection | SessionStage::Refreshing
807                    );
808                    if expected_property {
809                        state.responses.remove(&property);
810                    } else {
811                        operation_error = Some(ulcp_operation_error(
812                            format!("read property {property}"),
813                            response.value.as_slice(),
814                        )?);
815                        let optional_initial_property = state.stage == SessionStage::Initial
816                            && matches!(
817                                property,
818                                prop::DEV_KEY | prop::DEV_NAME | prop::BATTERY | prop::HOST_KEY
819                            );
820                        state.stage_failure_pending |= !optional_initial_property;
821                        if state.stage == SessionStage::Initial && property == prop::HOST_KEY {
822                            state.host_key_unsupported = true;
823                        }
824                    }
825                } else {
826                    if response.property_id != property || response.command != Cmd::PropIs as u8 {
827                        return Err(MobileError::InvalidUlcpFrame);
828                    }
829                    state.responses.insert(property, response.clone());
830                    state.apply_property(&response)?;
831                }
832            }
833            ExpectedResponse::Claim => {
834                if response.property_id == prop::LAST_STATUS {
835                    operation_error = Some(ulcp_operation_error(
836                        "claim host identity".to_owned(),
837                        response.value.as_slice(),
838                    )?);
839                    state.stage_failure_pending = true;
840                } else {
841                    if response.property_id != prop::HOST_KEY
842                        || response.command != Cmd::PropIs as u8
843                    {
844                        return Err(MobileError::InvalidUlcpFrame);
845                    }
846                    // Whatever the device reports is its host key, even if
847                    // it is not the one just written — a claim that did not
848                    // take means this radio belongs to someone else, which
849                    // `ownership()` reads off this value and reports as
850                    // `OtherHost`. That is an answer, not a broken session.
851                    state.radio_host_key = Some(response.value.clone());
852                    state.responses.insert(prop::HOST_KEY, response);
853                    if state.has_capability(cap::SAVE)? {
854                        state.stage = SessionStage::Saving;
855                        let tid = state.allocate_tid();
856                        state.expected.insert(tid, ExpectedResponse::Save);
857                        outbound.push(ulcp_save(tid)?);
858                    } else {
859                        state.start_inspection(&mut outbound)?;
860                    }
861                }
862            }
863            ExpectedResponse::Save => {
864                if response.property_id != prop::LAST_STATUS
865                    || response.command != Cmd::PropIs as u8
866                {
867                    return Err(MobileError::InvalidUlcpFrame);
868                }
869                if inspect_ulcp_status(response.value.clone())? != 0 {
870                    operation_error = Some(ulcp_operation_error(
871                        "save claimed host identity".to_owned(),
872                        response.value.as_slice(),
873                    )?);
874                    state.stage_failure_pending = true;
875                } else {
876                    state.start_inspection(&mut outbound)?;
877                }
878            }
879            ExpectedResponse::ConfigurationProperty(property) => {
880                if response.property_id == prop::LAST_STATUS {
881                    operation_error = Some(ulcp_operation_error(
882                        format!("set property {property}"),
883                        response.value.as_slice(),
884                    )?);
885                    state.stage_failure_pending = true;
886                    // The status frame describes the write, not the property.
887                    // Filing it under the property would leave the snapshot
888                    // reducing a status code as that property's value.
889                    state.responses.remove(&property);
890                } else if response.property_id != property || response.command != Cmd::PropIs as u8
891                {
892                    return Err(MobileError::InvalidUlcpFrame);
893                } else {
894                    // A `CMD_PROP_IS` is the device's authoritative value,
895                    // whatever was written. It reports what the device holds
896                    // — clamped to hardware, reduced to what it supports,
897                    // changed for a reason this host has no view of — and a
898                    // value differing from the write is that report, not a
899                    // fault. The snapshot published to the UI is what the
900                    // device says, never what was asked for. Failure is a
901                    // `PROP_LAST_STATUS`, handled above.
902                    state.responses.insert(property, response.clone());
903                    state.apply_property(&response)?;
904                }
905            }
906            ExpectedResponse::SaveConfiguration => {
907                if response.property_id != prop::LAST_STATUS
908                    || response.command != Cmd::PropIs as u8
909                {
910                    return Err(MobileError::InvalidUlcpFrame);
911                }
912                if inspect_ulcp_status(response.value.clone())? != 0 {
913                    operation_error = Some(ulcp_operation_error(
914                        "save radio configuration".to_owned(),
915                        response.value.as_slice(),
916                    )?);
917                    state.stage_failure_pending = true;
918                } else {
919                    state.finish_configuration()?;
920                }
921            }
922            ExpectedResponse::RawTransmit => {
923                if response.property_id != prop::LAST_STATUS
924                    || response.command != Cmd::PropIs as u8
925                {
926                    return Err(MobileError::InvalidUlcpFrame);
927                }
928                let status_code = inspect_ulcp_status(response.value)?;
929                let status = umsh_ulcp::Status(status_code);
930                raw_transmit_result = Some(UlcpRawTransmitResultRecord {
931                    transaction_id: response.transaction_id,
932                    status_code,
933                    status_name: format!("{status:?}"),
934                    disposition: if status == umsh_ulcp::Status::OK {
935                        UlcpRawTransmitDisposition::Sent
936                    } else if status == umsh_ulcp::Status::BUSY
937                        || status == umsh_ulcp::Status::CCA_FAILURE
938                    {
939                        // Both are transient channel-contention refusals: the
940                        // frame never left the radio, so retry with backoff.
941                        UlcpRawTransmitDisposition::Retry
942                    } else {
943                        UlcpRawTransmitDisposition::Rejected
944                    },
945                });
946            }
947            ExpectedResponse::DevPeerInsert(item) => {
948                if response.property_id == prop::LAST_STATUS {
949                    let error = ulcp_operation_error(
950                        "insert device peer".to_owned(),
951                        response.value.as_slice(),
952                    )?;
953                    // ALREADY is the device saying the key is stored; keep
954                    // the cache truthful even though the operation "failed".
955                    if error.status_code == umsh_ulcp::Status::ALREADY.0 {
956                        state.patch_dev_peers(&item, true);
957                        state.refresh_attached_snapshot(None)?;
958                    }
959                    operation_error = Some(error);
960                } else {
961                    if response.property_id != prop::DEV_PEERS
962                        || response.command != Cmd::PropInserted as u8
963                        || response.value != item
964                    {
965                        return Err(MobileError::InvalidUlcpFrame);
966                    }
967                    state.patch_dev_peers(&item, true);
968                    if state.has_capability(cap::SAVE)? {
969                        let tid = state.allocate_tid();
970                        state.expected.insert(tid, ExpectedResponse::SaveDevPeers);
971                        outbound.push(ulcp_save(tid)?);
972                    }
973                    state.refresh_attached_snapshot(None)?;
974                }
975            }
976            ExpectedResponse::DevPeerRemove(item) => {
977                if response.property_id == prop::LAST_STATUS {
978                    let error = ulcp_operation_error(
979                        "remove device peer".to_owned(),
980                        response.value.as_slice(),
981                    )?;
982                    // ITEM_NOT_FOUND means the key is not on the device,
983                    // which is the state the caller asked for.
984                    if error.status_code == umsh_ulcp::Status::ITEM_NOT_FOUND.0 {
985                        state.patch_dev_peers(&item, false);
986                        state.refresh_attached_snapshot(None)?;
987                    }
988                    operation_error = Some(error);
989                } else {
990                    if response.property_id != prop::DEV_PEERS
991                        || response.command != Cmd::PropRemoved as u8
992                        || response.value != item
993                    {
994                        return Err(MobileError::InvalidUlcpFrame);
995                    }
996                    state.patch_dev_peers(&item, false);
997                    if state.has_capability(cap::SAVE)? {
998                        let tid = state.allocate_tid();
999                        state.expected.insert(tid, ExpectedResponse::SaveDevPeers);
1000                        outbound.push(ulcp_save(tid)?);
1001                    }
1002                    state.refresh_attached_snapshot(None)?;
1003                }
1004            }
1005            ExpectedResponse::SaveDevPeers => {
1006                if response.property_id != prop::LAST_STATUS
1007                    || response.command != Cmd::PropIs as u8
1008                {
1009                    return Err(MobileError::InvalidUlcpFrame);
1010                }
1011                if inspect_ulcp_status(response.value.clone())? != 0 {
1012                    // The live mutation stuck; only persistence failed. The
1013                    // session stays attached and the caller sees the same
1014                    // `saved` warning path a failed configuration save uses.
1015                    operation_error = Some(ulcp_operation_error(
1016                        "save device peers".to_owned(),
1017                        response.value.as_slice(),
1018                    )?);
1019                }
1020            }
1021        }
1022
1023        if state.expected.is_empty() {
1024            if state.stage_failure_pending {
1025                state.stage_failure_pending = false;
1026                state.recover_from_operation_failure(&mut outbound)?;
1027            } else {
1028                state.advance_completed_stage(&mut outbound)?;
1029            }
1030        }
1031        Ok(state.update_with(outbound, Vec::new(), raw_transmit_result, operation_error))
1032    }

1167    fn apply_property(&mut self, response: &UlcpPropertyFrameRecord) -> Result<(), MobileError> {
1168        if response.command != Cmd::PropIs as u8 {
1169            // Insert/remove notifications are valid protocol frames, but none
1170            // of the mobile snapshot fields are multi-value payloads.
1171            return Ok(());
1172        }
1173        match response.property_id {
1174            prop::DEV_KEY => {
1175                if response.value.is_empty() {
1176                    self.device_key = None;
1177                } else if response.value.len() == items::PUBLIC_KEY_LEN {
1178                    self.device_key = Some(response.value.clone());
1179                } else {
1180                    return Err(MobileError::InvalidUlcpFrame);
1181                }
1182            }
1183            prop::DEV_NAME => {
1184                let name = core::str::from_utf8(&response.value)
1185                    .map_err(|_| MobileError::InvalidUlcpFrame)?;
1186                self.device_name = (!name.is_empty()).then(|| name.to_owned());
1187            }
1188            prop::BATTERY => {
1189                self.battery = Some(inspect_ulcp_battery(response.value.clone())?);
1190            }
1191            prop::ALERT => {
1192                // Arrives both as the answer to a write and unsolicited,
1193                // when the radio ends the alert itself.
1194                self.alert = Some(inspect_ulcp_alert(response.value.clone())?);
1195            }
1196            prop::HOST_KEY => {
1197                if !response.value.is_empty() && response.value.len() != items::PUBLIC_KEY_LEN {
1198                    return Err(MobileError::InvalidUlcpFrame);
1199                }
1200                self.radio_host_key = Some(response.value.clone());
1201            }
1202            _ => {}
1203        }
1204        Ok(())
1205    }

1207    /// Whether synchronization may proceed straight to inspection without
1208    /// pausing for the user to decide about host ownership.
1209    ///
1210    /// A tethered session must pause: it is about to become the radio's
1211    /// host, and taking a radio from another phone is a decision only the
1212    /// user can make. An administrative session never claims anything, so
1213    /// there is no decision to pause for — whose radio this is stays worth
1214    /// reporting, but only as information.
1215    fn attaches_without_host_decision(&self) -> bool {
1216        self.mode == UlcpAttachMode::Administrative
1217            || matches!(
1218                self.ownership(),
1219                UlcpHostOwnership::Ours | UlcpHostOwnership::Unsupported
1220            )
1221    }

1224    /// flight, and the device actually has a device identity domain.
1225    fn begin_dev_peer_operation(&mut self) -> Result<(), MobileError> {

1227            return Err(MobileError::InvalidUlcpFrame);

1229        if !self.has_capability(cap::DEV_IDENTITY)? {
1230            return Err(MobileError::InvalidUlcpFrame);

1235    /// Patch the cached `PROP_DEV_PEERS` table after a confirmed mutation,
1236    /// keeping it lossless without a round-trip re-read.
1237    fn patch_dev_peers(&mut self, key: &[u8], present: bool) {
1238        let entry =
1239            self.responses
1240                .entry(prop::DEV_PEERS)
1241                .or_insert_with(|| UlcpPropertyFrameRecord {
1242                    transaction_id: frame::TID_UNSOLICITED,
1243                    command: Cmd::PropIs as u8,
1244                    property_id: prop::DEV_PEERS,
1245                    value: Vec::new(),
1246                });
1247        let mut value = Vec::with_capacity(entry.value.len() + key.len());
1248        let mut found = false;
1249        for chunk in entry.value.chunks(items::PUBLIC_KEY_LEN) {
1250            if chunk == key {
1251                found = true;
1252                if !present {
1253                    continue;
1254                }
1255            }
1256            value.extend_from_slice(chunk);
1257        }
1258        if present && !found {
1259            value.extend_from_slice(key);
1260        }
1261        entry.value = value;
1262    }

1264    fn has_capability(&self, capability: u32) -> Result<bool, MobileError> {
1265        let capabilities = self
1266            .responses
1267            .get(&prop::CAPS)
1268            .ok_or(MobileError::InvalidUlcpFrame)?;

1272    /// Drop the writes the device has already refused to answer for.
1273    ///
1274    /// A capability-gated property that would not read is one the device
1275    /// does not implement, so writing it fails — and one rejected write
1276    /// abandons the whole configuration pass. The caller still states a
1277    /// complete configuration; what cannot land is left out here, where the
1278    /// device's own answers are known, rather than in the form.
1279    fn writable(&self, values: Vec<(u32, Vec<u8>)>) -> VecDeque<(u32, Vec<u8>)> {
1280        let Some(unreadable) = self
1281            .provisioning
1282            .as_ref()
1283            .map(|sync| sync.unreadable_properties.as_slice())
1284            .filter(|unreadable| !unreadable.is_empty())
1285        else {
1286            return values.into();
1287        };
1288        let dropped = |property: u32| {
1289            unreadable.contains(&property)
1290                || WHOLE_WRITE_GROUPS.iter().any(|group| {
1291                    group.contains(&property) && group.iter().any(|part| unreadable.contains(part))
1292                })
1293        };
1294        values
1295            .into_iter()
1296            .filter(|(property, _)| !dropped(*property))
1297            .collect()
1298    }

1300    fn advance_completed_stage(&mut self, outbound: &mut Vec<Vec<u8>>) -> Result<(), MobileError> {
1301        match self.stage {
1302            SessionStage::Initial => {
1303                let version = self
1304                    .responses
1305                    .get(&prop::PROTOCOL_VERSION)
1306                    .ok_or(MobileError::InvalidUlcpFrame)?;
1307                if version.value != [PROTOCOL_MAJOR_VERSION, PROTOCOL_MINOR_VERSION] {
1308                    return Err(MobileError::InvalidUlcpFrame);
1309                }
1310                let capabilities = self
1311                    .responses
1312                    .get(&prop::CAPS)
1313                    .ok_or(MobileError::InvalidUlcpFrame)?;
1314                self.inspection_queue =
1315                    ulcp_inspection_properties(capabilities.value.clone())?.into();
1316                let advertises_host_filter = self.has_capability(cap::HOST_FILTER)?;
1317                if advertises_host_filter == self.host_key_unsupported {
1318                    return Err(MobileError::InvalidUlcpFrame);
1319                }
1320                if self.attaches_without_host_decision() {
1321                    self.start_inspection(outbound)?;
1322                } else {
1323                    self.stage = SessionStage::AwaitingHost;
1324                }
1325            }
1326            SessionStage::Inspection => self.start_inspection(outbound)?,
1327            SessionStage::Refreshing => self.start_refresh(outbound)?,
1328            SessionStage::Configuring => {
1329                if !self.configuration_queue.is_empty() {
1330                    self.start_configuration(outbound)?;
1331                } else if self.has_capability(cap::SAVE)? {
1332                    self.stage = SessionStage::SavingConfiguration;
1333                    let tid = self.allocate_tid();
1334                    self.expected
1335                        .insert(tid, ExpectedResponse::SaveConfiguration);
1336                    outbound.push(ulcp_save(tid)?);
1337                } else {
1338                    self.finish_configuration()?;
1339                }
1340            }
1341            SessionStage::Claiming
1342            | SessionStage::Saving
1343            | SessionStage::AwaitingHost
1344            | SessionStage::Attached
1345            | SessionStage::SavingConfiguration
1346            | SessionStage::Idle => {}
1347        }
1348        Ok(())
1349    }

1351    /// Abort only the failed operation stage. A correlated CRP status error
1352    /// never invalidates GATT framing and therefore never resets the session.
1353    fn recover_from_operation_failure(
1354        &mut self,
1355        outbound: &mut Vec<Vec<u8>>,
1356    ) -> Result<(), MobileError> {
1357        self.configuration_queue.clear();
1358        self.inspection_queue.clear();
1359        match self.stage {
1360            SessionStage::Claiming => self.stage = SessionStage::AwaitingHost,
1361            SessionStage::Saving => {
1362                // The host-key write succeeded even if persistence did not.
1363                // Continue attaching while reporting that SAVE failed.
1364                self.start_inspection(outbound)?;
1365            }
1366            SessionStage::Refreshing
1367            | SessionStage::Configuring
1368            | SessionStage::SavingConfiguration => {
1369                // Retain the last authoritative snapshot. Property echoes that
1370                // completed before the failed operation remain available for
1371                // the next explicit refresh.
1372                self.stage = SessionStage::Attached;
1373            }
1374            SessionStage::Inspection if self.provisioning.is_some() => {
1375                self.stage = SessionStage::Attached;
1376            }
1377            SessionStage::Initial | SessionStage::Inspection => {
1378                // The transport is healthy but the initial snapshot is not
1379                // trustworthy enough to attach. Stay connected and report the
1380                // operation error; reconnect/refresh may retry synchronization.
1381                self.stage = SessionStage::Initial;
1382            }
1383            SessionStage::Attached | SessionStage::AwaitingHost | SessionStage::Idle => {}
1384        }
1385        Ok(())
1386    }

1388    fn finish_configuration(&mut self) -> Result<(), MobileError> {
1389        let responses = self.responses.values().cloned().collect();
1390        self.provisioning = Some(inspect_ulcp_sync(responses)?);
1391        self.stage = SessionStage::Attached;

1395    fn start_configuration(&mut self, outbound: &mut Vec<Vec<u8>>) -> Result<(), MobileError> {
1396        self.stage = SessionStage::Configuring;
1397        for _ in 0..usize::from(frame::TID_MAX) {
1398            let Some((property, value)) = self.configuration_queue.pop_front() else {
1399                break;
1400            };
1401            let tid = self.allocate_tid();
1402            self.expected
1403                .insert(tid, ExpectedResponse::ConfigurationProperty(property));
1404            outbound.push(ulcp_prop_set(tid, property, value)?);
1405        }
1406        Ok(())
1407    }

1409    fn start_inspection(&mut self, outbound: &mut Vec<Vec<u8>>) -> Result<(), MobileError> {
1410        self.stage = SessionStage::Inspection;
1411        if self.inspection_queue.is_empty() {
1412            let responses = self.responses.values().cloned().collect();
1413            self.provisioning = Some(inspect_ulcp_sync(responses)?);
1414            self.stage = SessionStage::Attached;
1415            return Ok(());
1416        }
1417        for _ in 0..usize::from(frame::TID_MAX) {
1418            let Some(property) = self.inspection_queue.pop_front() else {
1419                break;
1420            };
1421            outbound.push(self.get_property(property)?);
1422        }
1423        Ok(())
1424    }
1425
1426    fn start_refresh(&mut self, outbound: &mut Vec<Vec<u8>>) -> Result<(), MobileError> {
1427        self.stage = SessionStage::Refreshing;
1428        if self.inspection_queue.is_empty() {
1429            let responses = self.responses.values().cloned().collect();
1430            self.provisioning = Some(inspect_ulcp_sync(responses)?);
1431            self.stage = SessionStage::Attached;
1432            return Ok(());
1433        }
1434        for _ in 0..usize::from(frame::TID_MAX) {
1435            let Some(property) = self.inspection_queue.pop_front() else {
1436                break;
1437            };
1438            outbound.push(self.get_property(property)?);
1439        }
1440        Ok(())
1441    }

1443    /// Recompute the attached provisioning snapshot after device state
1444    /// changed under an established session.
1445    ///
1446    /// `changed_property` is the property the triggering notification
1447    /// carried, or `None` for a change this session made itself.
1448    ///
1449    /// Re-opening the host decision is deliberately limited to a
1450    /// `PROP_HOST_KEY` change. Another phone claiming the radio out from
1451    /// under an attached session is a question only the user can answer,
1452    /// so that case still returns to the host prompt. Every *other*
1453    /// published value is news, not a decision: a session attached to a
1454    /// radio owned by another identity (a tethered claim that did not
1455    /// take, which attaches deliberately — see the `Claim` arm) would
1456    /// otherwise be thrown back to the prompt by any unsolicited update at
1457    /// all. `PROP_BATTERY` makes that concrete, being the one notification
1458    /// that arrives on its own schedule for the life of the session.
1459    fn refresh_attached_snapshot(
1460        &mut self,
1461        changed_property: Option<u32>,
1462    ) -> Result<(), MobileError> {
1463        if self.stage != SessionStage::Attached {
1464            return Ok(());
1465        }
1466        let responses = self.responses.values().cloned().collect();
1467        self.provisioning = Some(inspect_ulcp_sync(responses)?);
1468        if changed_property == Some(prop::HOST_KEY) && !self.attaches_without_host_decision() {
1469            self.stage = SessionStage::AwaitingHost;
1470        }
1471        Ok(())
1472    }
1473}

1554/// Validate and reduce the property responses from the read-only post-attach
1555/// inspection.
1556///
1557/// The four properties every ULCP device must answer — the interface type
1558/// and the live PHY triple — are required: without them there is no radio
1559/// to describe. Everything else is capability-gated and merely *expected*,
1560/// so a device that refuses one, or answers it with something undecodable,
1561/// yields a snapshot with that setting absent and named in
1562/// `unreadable_properties` rather than no snapshot at all.
1563#[uniffi::export]
1564pub fn inspect_ulcp_sync(
1565    responses: Vec<UlcpPropertyFrameRecord>,
1566) -> Result<UlcpSyncRecord, MobileError> {
1567    let value = |key| property_value(&responses, key);
1568    let capabilities = decode_capabilities(value(prop::CAPS)?)?;
1569    validate_capability_dependencies(&capabilities)?;
1570    let has = |capability| capabilities.contains(&capability);
1571
1572    let interface = decode_exact_pui(value(prop::INTERFACE_TYPE)?)?;
1573    if interface != INTERFACE_TYPE {
1574        return Err(MobileError::InvalidUlcpFrame);
1575    }
1576    let phy_enabled = decode_bool(value(prop::PHY_ENABLED)?)?;
1577    let frequency_khz = decode_u32(value(prop::PHY_FREQ)?)?;
1578    let transmit_power_dbm = decode_i8(value(prop::PHY_TX_POWER)?)?;
1579
1580    let mut expected = ExpectedProperties {
1581        responses: &responses,
1582        unreadable: Vec::new(),
1583    };
1584    let lora = has(cap::PHY_LORA);
1585    let bandwidth_hz = expected.read(lora, prop::PHY_LORA_BW, decode_u32);
1586    let spreading_factor = expected.read(lora, prop::PHY_LORA_SF, decode_u8);
1587    let coding_rate_denom = expected.read(lora, prop::PHY_LORA_CR, decode_u8);
1588    let duty = has(cap::PHY_DUTY_LIMIT);
1589    let duty_cycle_now = expected.read(duty, prop::PHY_DUTY_NOW, decode_u16);
1590    let duty_cycle_limit = expected.read(duty, prop::PHY_DUTY_LIMIT, decode_u16);
1591    let saved = expected.read(has(cap::SAVE), prop::SAVED, decode_saved);
1592    let queue = has(cap::HOST_RX_QUEUE);
1593    let queued_frames = expected.read(queue, prop::HOST_RX_QUEUE_COUNT, decode_u16);
1594    let dropped_frames = expected.read(queue, prop::HOST_RX_QUEUE_DROPPED, decode_u32);
1595    let filter_count = expected.read(
1596        has(cap::HOST_FILTER),
1597        prop::HOST_RX_FILTERS,
1598        decode_filter_count,
1599    );
1600    let host_keys = has(cap::HOST_KEYS);
1601    let host_channel_count = expected.read(
1602        host_keys,
1603        prop::HOST_CHANNEL_KEYS,
1604        decode_fixed_count::<{ items::CHANNEL_ID_LEN }>,
1605    );
1606    let host_peer_count = expected.read(
1607        host_keys,
1608        prop::HOST_PEER_KEYS,
1609        decode_fixed_count::<{ items::PUBLIC_KEY_LEN }>,
1610    );
1611    let auto_ack = expected.read(has(cap::HOST_AUTO_ACK), prop::HOST_AUTO_ACK, decode_bool);
1612    let dev_identity = has(cap::DEV_IDENTITY);
1613    let dev_peer_keys = expected.read(
1614        dev_identity,
1615        prop::DEV_PEERS,
1616        decode_fixed_list::<{ items::PUBLIC_KEY_LEN }>,
1617    );
1618
1619    // Every part of the policy is read before any of it is required, so one
1620    // unreadable property does not hide the others behind it.
1621    let forwards = has(cap::REPEATER);
1622    let repeater_enabled = expected.read(forwards, prop::MAC_REPEATER_ENABLED, decode_bool);
1623    let regions = expected.read(forwards, prop::MAC_REPEATER_REGIONS, decode_region_list);
1624    let default_region = expected.read(
1625        forwards,
1626        prop::MAC_REPEATER_DEFAULT_REGION,
1627        decode_optional_region,
1628    );
1629    let min_rssi_dbm = expected.read(forwards, prop::MAC_REPEATER_MIN_RSSI, |value| {
1630        decode_optional(value, decode_i16)
1631    });
1632    let min_snr_db = expected.read(forwards, prop::MAC_REPEATER_MIN_SNR, |value| {
1633        decode_optional(value, decode_i8)
1634    });
1635    let repeater = (|| {
1636        Some(UlcpRepeaterSettingsRecord {
1637            enabled: repeater_enabled?,
1638            regions: regions?,
1639            default_region: default_region?,
1640            min_rssi_dbm: min_rssi_dbm?,
1641            min_snr_db: min_snr_db?,
1642        })
1643    })();
1644
1645    // An empty PROP_IDENT_ROLE is the device saying it derives its own
1646    // role, which is the same `None` a device without CAP_IDENT reports.
1647    let ident = has(cap::IDENT);
1648    let ident_role = expected
1649        .read(ident, prop::IDENT_ROLE, |value| {
1650            decode_optional(value, decode_u8)
1651        })
1652        .flatten();
1653    let ident_mobile = expected.read(ident, prop::IDENT_MOBILE, decode_bool);
1654    let dev_discoverable = expected.read(dev_identity, prop::DEV_DISCOVERABLE, decode_bool);
1655
1656    let mut unreadable_properties = expected.unreadable;
1657    unreadable_properties.sort_unstable();
1658
1659    Ok(UlcpSyncRecord {
1660        capability_count: capabilities
1661            .len()
1662            .try_into()
1663            .map_err(|_| MobileError::InvalidUlcpFrame)?,
1664        has_host_filtering: has(cap::HOST_FILTER),
1665        supports_offline_queue: has(cap::HOST_RX_QUEUE),
1666        supports_delegated_ack: has(cap::HOST_AUTO_ACK),
1667        supports_device_name: has(cap::DEV_NAME),
1668        supports_lora: has(cap::PHY_LORA),
1669        supports_duty_cycle_limit: has(cap::PHY_DUTY_LIMIT),
1670        supports_repeater: has(cap::REPEATER),
1671        supports_ident: has(cap::IDENT),
1672        supports_device_identity: has(cap::DEV_IDENTITY),
1673        phy_enabled,
1674        frequency_khz,
1675        transmit_power_dbm,
1676        bandwidth_hz,
1677        spreading_factor,
1678        coding_rate_denom,
1679        duty_cycle_now,
1680        duty_cycle_limit,
1681        saved,
1682        queued_frames,
1683        dropped_frames,
1684        filter_count,
1685        host_channel_count,
1686        host_peer_count,
1687        auto_ack,
1688        repeater,
1689        dev_peer_keys,
1690        ident_role,
1691        ident_mobile,
1692        dev_discoverable,
1693        unreadable_properties,
1694    })
1695}

1697/// Properties only ever written as a set. Writing part of a modem profile
1698/// or part of a forwarding policy leaves the device running a configuration
1699/// nobody asked for, so one unreadable member withdraws the whole group.
1700/// These are the same groupings the reduction reports as a unit.
1701const WHOLE_WRITE_GROUPS: [&[u32]; 2] = [
1702    &[prop::PHY_LORA_BW, prop::PHY_LORA_SF, prop::PHY_LORA_CR],
1703    &[
1704        prop::MAC_REPEATER_ENABLED,
1705        prop::MAC_REPEATER_REGIONS,
1706        prop::MAC_REPEATER_DEFAULT_REGION,
1707        prop::MAC_REPEATER_MIN_RSSI,
1708        prop::MAC_REPEATER_MIN_SNR,
1709    ],
1710];
1711
1712/// The capability-gated half of an inspection: properties the device is
1713/// expected to answer, each of which it may nevertheless decline.
1714struct ExpectedProperties<'a> {
1715    responses: &'a [UlcpPropertyFrameRecord],
1716    unreadable: Vec<u32>,
1717}
1718
1719impl ExpectedProperties<'_> {
1720    /// Decode `key` when the device advertises the capability that gates it.
1721    ///
1722    /// A missing or undecodable value is recorded and reported as `None`
1723    /// rather than failing the whole reduction: the setting is unknown, which
1724    /// is a fact about one property and not about the device as a whole.
1725    fn read<T>(
1726        &mut self,
1727        gated_on: bool,
1728        key: u32,
1729        decode: impl FnOnce(&[u8]) -> Result<T, MobileError>,
1730    ) -> Option<T> {
1731        if !gated_on {
1732            return None;
1733        }
1734        match property_value(self.responses, key).and_then(decode) {
1735            Ok(value) => Some(value),
1736            Err(_) => {
1737                self.unreadable.push(key);
1738                None
1739            }
1740        }
1741    }
1742}
1743
1744fn property_value(responses: &[UlcpPropertyFrameRecord], key: u32) -> Result<&[u8], MobileError> {
1745    let mut matching = responses
1746        .iter()
1747        .filter(|response| response.property_id == key);
1748    let response = matching.next().ok_or(MobileError::InvalidUlcpFrame)?;
1749    if matching.next().is_some() || response.command != Cmd::PropIs as u8 {
1750        return Err(MobileError::InvalidUlcpFrame);
1751    }
1752    Ok(&response.value)
1753}
1754
1755fn decode_capabilities(value: &[u8]) -> Result<Vec<u32>, MobileError> {
1756    let mut capabilities = Vec::new();
1757    let mut rest = value;
1758    while !rest.is_empty() {
1759        let (capability, used) = pui::decode(rest).map_err(|_| MobileError::InvalidUlcpFrame)?;
1760        if capabilities.contains(&capability) {
1761            return Err(MobileError::InvalidUlcpFrame);
1762        }
1763        capabilities.push(capability);
1764        rest = &rest[used..];
1765    }
1766    Ok(capabilities)
1767}

1769fn validate_capability_dependencies(capabilities: &[u32]) -> Result<(), MobileError> {
1770    let has = |capability| capabilities.contains(&capability);
1771    if has(cap::HOST_RX_QUEUE) && !has(cap::HOST_FILTER)
1772        || has(cap::HOST_KEYS) && !has(cap::HOST_FILTER)
1773        || has(cap::HOST_AUTO_ACK) && (!has(cap::HOST_KEYS) || !has(cap::HOST_RX_QUEUE))
1774        // A device with no identity of its own has nothing to forward for
1775        // and nothing to advertise.
1776        || has(cap::REPEATER) && !has(cap::DEV_IDENTITY)
1777        || has(cap::IDENT) && !has(cap::DEV_IDENTITY)
1778    {
1779        return Err(MobileError::InvalidUlcpFrame);
1780    }
1781    Ok(())
1782}

1869/// Order one configuration pass: everything that changes live PHY
1870/// behavior happens with the radio down, and the radio comes back up only
1871/// once the complete new profile is in place.
1872///
1873/// `device_values` are the device-domain writes, which ride between the
1874/// two PHY_ENABLED writes for the same reason the PHY parameters do — a
1875/// repeater must not start forwarding under half of its new policy.
1876fn configuration_values(
1877    settings: UlcpRadioSettingsRecord,
1878    device_values: Vec<(u32, Vec<u8>)>,
1879) -> Vec<(u32, Vec<u8>)> {
1880    let mut values = Vec::new();
1881    if !settings.phy_enabled {
1882        values.push((prop::PHY_ENABLED, vec![0]));
1883    }
1884    if let Some(name) = settings.device_name {
1885        values.push((prop::DEV_NAME, name.into_bytes()));
1886    }
1887    values.extend([
1888        (
1889            prop::PHY_FREQ,
1890            settings.frequency_khz.to_le_bytes().to_vec(),
1891        ),
1892        (prop::PHY_TX_POWER, vec![settings.transmit_power_dbm as u8]),
1893    ]);
1894    if let (Some(bandwidth), Some(sf), Some(cr)) = (
1895        settings.bandwidth_hz,
1896        settings.spreading_factor,
1897        settings.coding_rate_denom,
1898    ) {
1899        values.extend([
1900            (prop::PHY_LORA_BW, bandwidth.to_le_bytes().to_vec()),
1901            (prop::PHY_LORA_SF, vec![sf]),
1902            (prop::PHY_LORA_CR, vec![cr]),
1903        ]);
1904    }
1905    if let Some(limit) = settings.duty_cycle_limit {
1906        values.push((prop::PHY_DUTY_LIMIT, limit.to_le_bytes().to_vec()));
1907    }
1908    values.extend(device_values);
1909    if settings.phy_enabled {
1910        values.push((prop::PHY_ENABLED, vec![1]));
1911    }
1912    values
1913}

1915/// Check the device-domain half of a commissioning record against what
1916/// the device says it can do, and reduce it to property writes.
1917///
1918/// Capability-gated fields must be present exactly when the capability
1919/// is: the record states a whole desired configuration, so a field the
1920/// device cannot honor is a caller mistake rather than something to
1921/// silently drop.
1922fn validate_device_settings(
1923    configuration: &UlcpDeviceConfigRecord,
1924    state: &UlcpSessionState,
1925) -> Result<Vec<(u32, Vec<u8>)>, MobileError> {
1926    let mut values = Vec::new();
1927
1928    let supports_ident = state.has_capability(cap::IDENT)?;
1929    if configuration.ident_mobile.is_some() != supports_ident
1930        || (configuration.ident_role.is_some() && !supports_ident)
1931    {
1932        return Err(MobileError::InvalidUlcpFrame);
1933    }
1934    if supports_ident {
1935        // An empty PROP_IDENT_ROLE hands the choice back to the device.
1936        values.push((
1937            prop::IDENT_ROLE,
1938            configuration
1939                .ident_role
1940                .map(|role| vec![role])
1941                .unwrap_or_default(),
1942        ));
1943        values.push((
1944            prop::IDENT_MOBILE,
1945            vec![configuration.ident_mobile.unwrap_or(false) as u8],
1946        ));
1947    }
1948
1949    let supports_dev_identity = state.has_capability(cap::DEV_IDENTITY)?;
1950    if configuration.dev_discoverable.is_some() != supports_dev_identity {
1951        return Err(MobileError::InvalidUlcpFrame);
1952    }
1953    if let Some(discoverable) = configuration.dev_discoverable {
1954        values.push((prop::DEV_DISCOVERABLE, vec![discoverable as u8]));
1955    }
1956
1957    let supports_repeater = state.has_capability(cap::REPEATER)?;
1958    if configuration.repeater.is_some() != supports_repeater {
1959        return Err(MobileError::InvalidUlcpFrame);
1960    }
1961    if let Some(repeater) = &configuration.repeater {
1962        let mut regions = Vec::with_capacity(repeater.regions.len() * items::REGION_CODE_LEN);
1963        for region in &repeater.regions {
1964            if region.len() != items::REGION_CODE_LEN {
1965                return Err(MobileError::InvalidUlcpFrame);
1966            }
1967            regions.extend_from_slice(region);
1968        }
1969        if let Some(default_region) = &repeater.default_region {
1970            if default_region.len() != items::REGION_CODE_LEN {
1971                return Err(MobileError::InvalidUlcpFrame);
1972            }
1973        }
1974        // Enabling last means the forwarding policy is already whole by
1975        // the time the device starts acting on it. The device does not
1976        // cross-check the default region against the forwarding list —
1977        // that is a SHOULD the presenting UI is better placed to warn on.
1978        values.extend([
1979            (prop::MAC_REPEATER_REGIONS, regions),
1980            (
1981                prop::MAC_REPEATER_DEFAULT_REGION,
1982                repeater.default_region.clone().unwrap_or_default(),
1983            ),
1984            (
1985                prop::MAC_REPEATER_MIN_RSSI,
1986                repeater
1987                    .min_rssi_dbm
1988                    .map(|rssi| rssi.to_le_bytes().to_vec())
1989                    .unwrap_or_default(),
1990            ),
1991            (
1992                prop::MAC_REPEATER_MIN_SNR,
1993                repeater
1994                    .min_snr_db
1995                    .map(|snr| vec![snr as u8])
1996                    .unwrap_or_default(),
1997            ),
1998            (prop::MAC_REPEATER_ENABLED, vec![repeater.enabled as u8]),
1999        ]);
2000    }
2001    Ok(values)
2002}

2074/// Split a ULCP frame into ATT values using the negotiated maximum write
2075/// length. The returned values include the one-octet SAR header.
2076#[uniffi::export]
2077pub fn ulcp_gatt_segments(
2078    frame: Vec<u8>,
2079    maximum_value_length: u16,
2080) -> Result<Vec<GattSegmentRecord>, MobileError> {
2081    let segment_payload = usize::from(maximum_value_length)
2082        .checked_sub(1)
2083        .filter(|length| *length > 0)
2084        .ok_or(MobileError::InvalidGattSegment)?;
2085    if frame.len() > MAX_FRAME {
2086        return Err(MobileError::InvalidUlcpFrame);
2087    }
2088
2089    Ok(gatt::segments(&frame, segment_payload)
2090        .map(|segment| {
2091            let mut value = vec![0; segment.payload().len() + 1];
2092            let length = segment
2093                .write_to(&mut value)
2094                .expect("sized from the segment payload");
2095            value.truncate(length);
2096            GattSegmentRecord { value }
2097        })
2098        .collect())
2099}

2411    /// Answer every bounded read batch until the session stops asking.
2412    fn drive_reads(
2413        session: &MobileUlcpSession,
2414        requests: Vec<Vec<u8>>,
2415        value: impl Fn(u32) -> (u32, Vec<u8>),
2416    ) -> UlcpSessionUpdateRecord {
2417        let mut pending = requests;
2418        let mut last = None;
2419        while !pending.is_empty() {
2420            let update = answer_requests(session, pending, &value);
2421            pending = update.outbound_frames.clone();
2422            last = Some(update);
2423        }
2424        last.expect("at least one batch")
2425    }

2444    /// Consume a configuration batch, returning the writes it made as a
2445    /// property map, the order they were issued in, and the `CMD_SAVE`
2446    /// transaction that closed it.
2447    fn drive_configuration(
2448        session: &MobileUlcpSession,
2449        first_batch: Vec<Vec<u8>>,
2450    ) -> (HashMap<u32, Vec<u8>>, Vec<u32>, u8) {
2451        let mut pending = VecDeque::from(first_batch);
2452        let mut written = HashMap::new();
2453        let mut order = Vec::new();
2454        loop {
2455            let request = pending.pop_front().expect("configuration ends in a save");
2456            let parsed = Frame::parse(&request).unwrap();
2457            if parsed.command() == Some(Cmd::Save) {
2458                return (written, order, parsed.header.tid());
2459            }
2460            assert_eq!(parsed.command(), Some(Cmd::PropSet));
2461            let payload = PropPayload::parse(parsed.payload).unwrap();
2462            order.push(payload.key);
2463            written.insert(payload.key, payload.value.to_vec());
2464            let update = session
2465                .consume(property_response(
2466                    parsed.header.tid(),
2467                    payload.key,
2468                    payload.value,
2469                ))
2470                .unwrap_or_else(|error| {
2471                    panic!("write of property {} failed: {error:?}", payload.key)
2472                });
2473            pending.extend(update.outbound_frames);
2474        }
2475    }

4020        // The user takes the radio; the device refuses the write and keeps

4025            .unwrap()

4047            "the claim did not take"
4048        );

4050        let pushed = session
4051            .consume(property_response(
4052                frame::TID_UNSOLICITED,
