1//! The LAN virtual radio: a soft device whose PHY is the LAN bearer's
2//! multicast group.
3//!
4//! What the device transmits is sent to the group, where every radio with
5//! its bearer enabled hears it; what any participant sends to the group is
6//! received unmeasured, since nothing here measured it. The device lives
7//! for as long as its owner runs it, whether or not a host is attached:
8//! detached, it goes on receiving, acknowledging for its host, and
9//! queueing, exactly as a companion radio does.
10//!
11//! The group is the device's own `PROP_LAN_BEARER_GROUP`, joined while
12//! its PHY is on, and `PROP_LAN_BEARER_STATE` says how that went: a host
13//! moves the radio to another group by writing the property, as it would
14//! on a radio that forwards.
15//!
16//! The host's link is the caller's. [`run`] takes the host's side as
17//! [`HostEvent`]s, each attach bringing the sink its replies go to, so one
18//! driver serves an in-memory pipe, a local socket ([`serve`]), and
19//! anything else that carries HDLC-Lite.
20

42/// The shape of a virtual radio, reporting itself as `dev_version`.
43///
44/// Spare, like a bridge's host interface: no hardware to describe and no
45/// node behind it. The PHY settings it reports are the default profile's,
46/// fixed; the frames it sends are transmitted later by whichever LoRa
47/// radios forward them, so the frame time a host's MAC derives from these
48/// is an estimate either way. The duty limit is left open—sending to the
49/// group spends no airtime—and the radios' ingress limits are the control
50/// that means something.
51///
52/// The configuration borrows a duty ledger for the life of the process,
53/// so a caller builds one per device it runs, not one per attach.

55    SessionConfig {
56        dev_version,
57        dev_model: None,
58        default_device_name: "Virtual radio",

60        tx_preamble_symbols: 32,

63        max_tx_power_dbm: 22,

69            bw_hz: profiles::DEFAULT.bw_hz,
70            sf: profiles::DEFAULT.sf,

72            tx_power_dbm: profiles::DEFAULT_TX_POWER_DBM,

106/// Exclusive use of one virtual radio's state directory, for as long as
107/// it is held.
108///
109/// Two devices sharing an identity would send under one key with two
110/// frame counters, and each would see the other's frames as replays. The
111/// lock is the operating system's, so it holds across processes—a tool
112/// and its twin, or an app and its extension—and is released when its
113/// holder exits, however it exits.
114pub struct StateLock {
115    _file: File,
116}
117
118impl StateLock {
119    /// Take the lock on `dir`, or `None` if another holder has it.
120    pub fn try_acquire(dir: &Path) -> io::Result<Option<Self>> {
121        let file = File::create(dir.join("lock"))?;
122        match file.try_lock() {
123            Ok(()) => Ok(Some(Self { _file: file })),
124            Err(TryLockError::WouldBlock) => Ok(None),
125            Err(TryLockError::Error(error)) => Err(error),
126        }
127    }
128}

130/// Where an attached host's bytes go: one HDLC-Lite frame per call.
131pub type WireSink = Box<dyn FnMut(Vec<u8>) + Send>;
132
133/// What a host does on its side of the link.
134pub enum HostEvent {
135    /// A link came up, taking the device's replies at its sink. A host
136    /// that attaches while another is attached replaces it: the earlier
137    /// sink is dropped and hears nothing more, not even what was owed to
138    /// its session.
139    Attach(WireSink),
140    /// Bytes from the host, in HDLC-Lite.
141    Bytes(Vec<u8>),
142    /// The link went down.
143    Detach,
144}
145
146/// How long to wait before trying the group again when it could not be
147/// joined, as when the host has no network yet.
148const REJOIN_DELAY: Duration = Duration::from_secs(2);
149
150/// The device's membership in its group.
151#[derive(Default)]
152struct Membership {
153    /// The group last tried, or `None` while the PHY is off.
154    group: Option<SocketAddrV4>,
155    socket: Option<LanSocket>,
156    /// When to try `group` again, after it could not be joined.
157    retry_at: Option<Instant>,
158}
159
160impl Membership {
161    /// Join the group the device asks for, if it is not already joined,
162    /// and report the outcome as the bearer's state.
163    fn follow<P: SoftPlatform>(&mut self, device: &mut SoftDevice<P>, base: LanConfig) {
164        let session = device.session();
165        let wanted = session.settings().enabled.then(|| {
166            let group = session.lan_bearer_config().group();
167            SocketAddrV4::new(Ipv4Addr::from(group.address), group.port)
168        });
169        let retry_due = self.group.is_some() && self.socket.is_none() && self.retry_at.is_none();
170        if wanted == self.group && !retry_due {
171            return;
172        }
173        self.group = wanted;
174        self.socket = None;
175        self.retry_at = None;
176        let state = match wanted {
177            None => State::Disabled,
178            Some(group) => match LanSocket::bind(LanConfig { group, ..base }) {
179                Ok(socket) => {
180                    self.socket = Some(socket);
181                    State::Active
182                }
183                Err(_) => {
184                    self.retry_at = Some(Instant::now() + REJOIN_DELAY);
185                    State::NoNetwork
186                }
187            },
188        };
189        device.set_lan_bearer_state(state);
190    }
191
192    async fn recv(&self, buf: &mut [u8]) -> io::Result<(usize, SocketAddr)> {
193        match &self.socket {
194            Some(socket) => socket.recv(buf).await,
195            None => std::future::pending().await,
196        }
197    }
198}

200/// Run `device` until the sender of `host` is dropped, joining its group
201/// on the interface and with the loopback `base` names, and telling
202/// `queued` of each frame queued while no host is attached.
203///
204/// The device starts detached; a [`HostEvent::Attach`] attaches it.
205/// Bytes that arrive while detached belong to no session and are
206/// dropped.
207pub async fn run<P: SoftPlatform>(
208    mut device: SoftDevice<P>,
209    base: LanConfig,
210    mut host: mpsc::UnboundedReceiver<HostEvent>,
211    mut queued: impl FnMut(QueuedNotice),
212) {
213    let started = Instant::now();
214    let now_ms = || started.elapsed().as_millis() as u64;
215    let mut sink: Option<WireSink> = None;
216    let mut membership = Membership::default();
217    let mut datagram = [0u8; MAX_FRAME + 1];
218    loop {
219        // A host's write may have moved the group or switched the PHY.
220        membership.follow(&mut device, base);
221        // Whatever the device has to say goes out first: the attach's
222        // own notice, then each reply in turn.
223        while let Some(wire) = device.take_outbound() {
224            if let Some(sink) = &mut sink {
225                sink(wire);
226            }
227        }
228        for frame in device.take_transmitted() {
229            // A send that fails, or finds no group, is a frame lost on
230            // the medium, which the protocol above already tolerates.
231            if let Some(socket) = &membership.socket {
232                let _ = socket.send(&frame).await;
233            }
234        }
235        tokio::select! {
236            event = host.recv() => match event {
237                None => break,
238                Some(HostEvent::Attach(next)) => {
239                    if sink.replace(next).is_some() {
240                        device.detach();
241                    }
242                    device.attach();
243                }
244                Some(HostEvent::Detach) => {
245                    if sink.take().is_some() {
246                        device.detach();
247                    }
248                }
249                // A garbled frame costs one frame: the decoder
250                // resynchronizes on the next flag.
251                Some(HostEvent::Bytes(bytes)) if sink.is_some() => {
252                    let _ = device.ingest(&bytes, now_ms());
253                }
254                Some(HostEvent::Bytes(_)) => {}
255            },
256            () = tokio::time::sleep_until(membership.retry_at.unwrap_or_else(Instant::now)),
257                if membership.retry_at.is_some() =>
258            {
259                membership.retry_at = None;
260            }
261            received = membership.recv(&mut datagram) => {
262                if let Ok((len, _)) = received
263                    && umsh_lan_bearer::classify(&datagram[..len]) == Datagram::Umsh
264                {
265                    device.inject_radio_rx(&datagram[..len], &RadioRxInfo::default(), now_ms());
266                    if let Some(notice) = device.session_mut().take_queued_notice() {
267                        queued(notice);
268                    }
269                }
270            }
271        }
272    }
273    if sink.is_some() {
274        device.detach();
275    }
276}

278/// Make the local socket a host attaches through at `path`, replacing
279/// whatever a previous holder of the state left there. The caller holds
280/// the state's [`StateLock`], so no live device owns that file.
281#[cfg(unix)]
282pub fn bind_local_socket(path: &Path) -> io::Result<UnixListener> {
283    use std::os::unix::fs::PermissionsExt;
284    match std::fs::remove_file(path) {
285        Err(error) if error.kind() != io::ErrorKind::NotFound => return Err(error),
286        _ => {}
287    }
288    let listener = UnixListener::bind(path)?;
289    std::fs::set_permissions(path, std::fs::Permissions::from_mode(0o600))?;
290    Ok(listener)
291}

293/// How much of a local socket to take in one read. ULCP frames are
294/// small; this only bounds a burst.
295#[cfg(unix)]
296const READ_CHUNK: usize = 4096;
297
298/// Attach whoever connects to `listener`, until the receiver of `host`
299/// is dropped.
300///
301/// A connection is a session: connecting attaches, closing detaches, and
302/// a new connection replaces the current one, whose socket is closed.
303#[cfg(unix)]
304pub async fn serve(listener: UnixListener, host: mpsc::UnboundedSender<HostEvent>) {
305    use tokio::io::{AsyncReadExt, AsyncWriteExt};
306
307    let mut current: Option<OwnedReadHalf> = None;
308    let mut buf = vec![0u8; READ_CHUNK];
309    loop {
310        let event = tokio::select! {
311            accepted = listener.accept() => {
312                // A failed accept costs that one connection.
313                let Ok((stream, _)) = accepted else { continue };
314                let (read, mut write) = stream.into_split();
315                let (wire_tx, mut wire) = mpsc::unbounded_channel::<Vec<u8>>();
316                // The writer ends when the device drops this sink, which
317                // closes the connection.
318                tokio::spawn(async move {
319                    while let Some(bytes) = wire.recv().await {
320                        if write.write_all(&bytes).await.is_err() {
321                            break;
322                        }
323                    }
324                });
325                current = Some(read);
326                HostEvent::Attach(Box::new(move |bytes| {
327                    let _ = wire_tx.send(bytes);
328                }))
329            }
330            read = async {
331                match &mut current {
332                    Some(read) => read.read(&mut buf).await,
333                    None => std::future::pending().await,
334                }
335            } => match read {
336                Ok(0) | Err(_) => {
337                    current = None;
338                    HostEvent::Detach
339                }
340                Ok(len) => HostEvent::Bytes(buf[..len].to_vec()),
341            },
342        };
343        if host.send(event).is_err() {
344            break;
345        }
346    }
347}

367    fn encoded(request: &[u8]) -> Vec<u8> {
368        let mut wire = vec![0; hdlc::max_encoded_len(request.len())];
369        let len = hdlc::encode_frame(request, &mut wire).unwrap();
370        wire.truncate(len);
371        wire
372    }

378    /// An attach whose replies arrive on the returned receiver.
379    fn attach() -> (HostEvent, mpsc::UnboundedReceiver<Vec<u8>>) {
380        let (tx, rx) = mpsc::unbounded_channel();
381        let event = HostEvent::Attach(Box::new(move |wire| {
382            let _ = tx.send(wire);
383        }));
384        (event, rx)
385    }

387    async fn next(out: &mut mpsc::UnboundedReceiver<Vec<u8>>) -> Option<Vec<u8>> {
388        tokio::time::timeout(Duration::from_secs(1), out.recv())
389            .await
390            .ok()
391            .flatten()
392    }

394    /// The frames in `wire`, decoded.
395    fn frames(wire: &[u8]) -> Vec<Vec<u8>> {
396        let mut decoder = hdlc::Decoder::<512>::new();
397        wire.iter()
398            .filter_map(|&byte| decoder.push(byte).map(|frame| frame.unwrap().to_vec()))
399            .collect()
400    }

402    /// The last byte of the next frame the host hears.
403    async fn next_value(out: &mut mpsc::UnboundedReceiver<Vec<u8>>) -> u8 {
404        let wire = next(out).await.expect("the device answers");
405        *frames(&wire)[0].last().unwrap()
406    }

433    /// The device answers only while attached, announces each attach with
434    /// a session reset, and a second attach takes the link from the
435    /// first.
436    #[tokio::test(flavor = "current_thread")]
437    async fn bytes_reach_the_device_only_while_attached() {
438        let dir = scratch("attach");
439        let (host, task) = running(&dir);
440
441        let mut request = [0u8; 16];
442        let len = frame::prop_get(&mut request, 1, prop::PROTOCOL_VERSION).unwrap();
443        host.send(HostEvent::Bytes(encoded(&request[..len])))
444            .unwrap();

447        host.send(HostEvent::Bytes(encoded(&request[..len])))

513    /// Over a local socket, connecting attaches and a second connection
514    /// replaces the first, whose socket closes.
515    #[cfg(unix)]
516    #[tokio::test(flavor = "current_thread")]
517    async fn a_local_socket_connection_is_a_session() {
518        use tokio::io::{AsyncReadExt, AsyncWriteExt};
519        use tokio::net::UnixStream;
520
521        let dir = scratch("socket");
522        let path = dir.join("ulcp");
523        // A stale file from an earlier holder is replaced.
524        std::fs::write(&path, b"").unwrap();
525        let listener = bind_local_socket(&path).unwrap();
526        let (host, task) = running(&dir);
527        let server = tokio::spawn(serve(listener, host));
528
529        let read_some = async |stream: &mut UnixStream| {
530            let mut buf = [0u8; 512];
531            match tokio::time::timeout(Duration::from_secs(1), stream.read(&mut buf)).await {
532                Ok(Ok(len)) => buf[..len].to_vec(),
533                _ => panic!("the device is silent"),
534            }
535        };
536        let mut first = UnixStream::connect(&path).await.unwrap();
537        // The session reset.
538        assert!(!read_some(&mut first).await.is_empty());
539        let mut request = [0u8; 16];
540        let len = frame::prop_get(&mut request, 1, prop::PROTOCOL_VERSION).unwrap();
541        first.write_all(&encoded(&request[..len])).await.unwrap();
542        assert!(!read_some(&mut first).await.is_empty());
543
544        let mut second = UnixStream::connect(&path).await.unwrap();
545        assert!(!read_some(&mut second).await.is_empty());
546        // Closed: end of stream.
547        assert!(read_some(&mut first).await.is_empty());
548
549        server.abort();
550        let _ = server.await;
551        task.await.unwrap();
552        let _ = std::fs::remove_dir_all(&dir);
553    }
