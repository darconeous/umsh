1//! The soft device core: the production device [`Session`] behind an
2//! HDLC-Lite byte stream, with what a board would supply—storage,
3//! entropy, a clock, peripherals—supplied by a [`SoftPlatform`].
4//!
5//! The core serves every effect that is the same on any host: putting a
6//! frame on the medium, saving and restoring configuration, provisioning
7//! and keeping the device identity, rebooting, and factory reset. What is
8//! a property of the host—where state persists, where randomness comes
9//! from, what the clock says, and any peripheral the configuration
10//! advertises—comes from the platform.
11//!
12//! The medium is the caller's. Transmissions accumulate until
13//! [`SoftDevice::take_transmitted`] drains them, and receptions go in
14//! through [`SoftDevice::inject_radio_rx`]. A deterministic simulation
15//! discards or replays them; a LAN virtual radio sends them to its group.
16//!
17//! A device with a settings interface of its own—a simulated
18//! communicator—opens it with [`SoftDevice::open_device_interface`] and
19//! serves it through [`SoftDevice::device_request`], as the firmware
20//! driver serves its in-process requests. Its answers and everything the
21//! device publishes wait in [`SoftDevice::take_device_frame`].
22

41/// What a host supplies to a [`SoftDevice`].
42pub trait SoftPlatform {
43    /// The configuration snapshot saved by `CMD_SAVE`, if any.
44    fn load_snapshot(&mut self) -> Option<Vec<u8>>;
45    /// Persist a snapshot, or erase it with `None`.
46    fn store_snapshot(&mut self, snapshot: Option<&[u8]>) -> Result<(), ()>;
47    /// The device identity's secret key, if one is stored.
48    fn load_identity(&mut self) -> Option<[u8; 32]>;
49    /// Persist the device identity's secret key, or erase it with `None`.
50    fn store_identity(&mut self, secret: Option<&[u8; 32]>) -> Result<(), ()>;
51    /// A fresh identity secret.
52    fn generate_secret(&mut self) -> [u8; 32];
53    /// Whether a device with no stored identity makes one as it comes up,
54    /// as firmware does, rather than waiting for a host to ask.
55    fn generates_identity_at_boot(&self) -> bool {
56        false
57    }
58
59    /// The wall clock, as a Unix second, or `None` if unknown.
60    fn wall_clock(&mut self) -> Option<u32>;
61    /// A host set the wall clock.
62    fn set_wall_clock(&mut self, epoch: Option<u32>) {
63        let _ = epoch;
64    }
65
66    /// The channel keys the device lists hold after a reset with none
67    /// kept, on a device whose own store keeps them outside the snapshot,
68    /// as a communicator's does (see [`Session::hold_lists`]). `None`, the
69    /// default, leaves them in the snapshot.
70    fn held_lists(&self) -> Option<&'static [[u8; 32]]> {
71        None
72    }
73    /// The record of the device lists the platform keeps, if any.
74    fn load_lists(&mut self) -> Option<Vec<u8>> {
75        None
76    }
77    /// Keep the record of the device lists, or erase it with `None`.
78    fn store_lists(&mut self, record: Option<&[u8]>) -> Result<(), ()> {
79        let _ = record;
80        Err(())
81    }
82
83    /// The session was just constructed, before any saved state is
84    /// restored into it: report what the platform's hardware is.
85    fn on_boot(&mut self, session: &mut DeviceSession) {
86        let _ = session;
87    }
88    /// Erase whatever else the platform keeps, for a factory reset.
89    fn erase(&mut self) {}
90    /// Serve an effect the core does not. A platform must serve every
91    /// effect its [`SessionConfig`] can produce.
92    fn peripheral(
93        &mut self,
94        effect: Effect,
95        session: &mut DeviceSession,
96        emit: &mut dyn FnMut(&[u8]),
97    ) {
98        let _ = (effect, session, emit);
99    }
100    /// A host frame was served: bring the platform's view up to date with
101    /// what the session now holds, as a board's drivers react to the
102    /// device-domain mirror.
103    fn settle(&mut self, session: &mut DeviceSession, emit: &mut dyn FnMut(&[u8])) {
104        let _ = (session, emit);
105    }
106}

108/// The device session behind an HDLC-Lite link, on a [`SoftPlatform`].
109pub struct SoftDevice<P: SoftPlatform> {
110    session: DeviceSession,
111    config: SessionConfig,
112    platform: P,
113    decoder: hdlc::Decoder<WIRE_CAPACITY>,
114    outbound: VecDeque<Vec<u8>>,
115    /// Whether the device has a settings interface of its own. Survives a
116    /// reboot: it describes the device, not the session.
117    device_interface: bool,
118    /// Raw frames for that interface, oldest first: its answers, and every
119    /// publication.
120    device_out: VecDeque<Vec<u8>>,
121    air: Vec<Vec<u8>>,
122    /// Every `CMD_ANNOUNCE` taken, oldest first. A soft device has no
123    /// node to put one on the air, so the record is the whole of what can
124    /// be observed.
125    announcements: Vec<AnnounceRequest>,
126    now_ms: u64,
127    /// The caller's clock reading when this device last came up.
128    /// `Clock::now_ms` is milliseconds since boot, and a reboot restarts
129    /// it, so a device's uptime is measured from here.
130    boot_ms: u64,
131}

133impl<P: SoftPlatform> SoftDevice<P> {
134    /// Bring the device up as from power-on, restoring whatever the
135    /// platform persisted.
136    pub fn new(config: SessionConfig, platform: P) -> Self {
137        let mut device = Self {
138            session: DeviceSession::new(
139                config,
140                Status::RESET_POWER_ON,
141                CryptoEngine::new(SoftwareAes, SoftwareSha256),
142            ),
143            config,
144            platform,
145            decoder: hdlc::Decoder::new(),
146            outbound: VecDeque::new(),
147            device_interface: false,
148            device_out: VecDeque::new(),
149            air: Vec::new(),
150            announcements: Vec::new(),
151            now_ms: 0,
152            boot_ms: 0,
153        };
154        device.boot();
155        device
156    }

158    /// Construct a fresh session and restore the platform's state into
159    /// it, as a board does on the way up.
160    fn boot(&mut self) {
161        self.session = DeviceSession::new(
162            self.config,
163            Status::RESET_POWER_ON,
164            CryptoEngine::new(SoftwareAes, SoftwareSha256),
165        );
166        self.platform.on_boot(&mut self.session);
167        if let Some(defaults) = self.platform.held_lists() {
168            let stored = self.platform.load_lists();
169            self.session.hold_lists(defaults, stored.as_deref());
170        }
171        if self.device_interface {
172            self.session.open_device_interface();
173        }
174        // The identity before the snapshot: a restore keeps the PHY off
175        // unless the snapshot was taken under the live identity. A device
176        // configured without one neither keeps nor makes one.
177        let secret = if !self.config.dev_identity {
178            None
179        } else {
180            match self.platform.load_identity() {
181                Some(secret) => Some(secret),
182                None if self.platform.generates_identity_at_boot() => {
183                    let secret = self.platform.generate_secret();
184                    self.platform
185                        .store_identity(Some(&secret))
186                        .ok()
187                        .map(|_| secret)
188                }
189                None => None,
190            }
191        };
192        if let Some(secret) = secret {
193            let public = SoftwareIdentity::from_secret_bytes(&secret).public_key().0;
194            self.session.set_boot_identity(public);
195        }
196        if let Some(saved) = self.platform.load_snapshot() {
197            let _ = self.session.restore_at_boot(&saved);
198        }
199    }

217    /// The caller's clock, rebased onto this device's boot.
218    pub fn device_ms(&self) -> u64 {
219        self.now_ms.saturating_sub(self.boot_ms)

222    /// Attach a host over a link as secure as USB: one only this host's
223    /// own processes can reach.
224    pub fn attach(&mut self) {
225        self.decoder.reset();
226        self.outbound.clear();
227        self.session.attach(true);
228        self.flush_session_reset_notice();
229    }

231    pub fn detach(&mut self) {
232        self.decoder.reset();
233        self.outbound.clear();
234        self.session.detach();
235    }

237    /// Pay whatever session-reset notice the session is owing, the way
238    /// the firmware driver does at the tail of its loop.
239    fn flush_session_reset_notice(&mut self) {
240        let mut emitted = Vec::new();
241        if self
242            .session
243            .take_session_reset_notice(&mut |frame: &[u8]| emitted.push(frame.to_vec()))
244        {
245            self.queue_emitted(emitted);
246        }
247    }

249    /// Feed bytes from the host's side of the link.
250    pub fn ingest(&mut self, bytes: &[u8], now_ms: u64) -> Result<(), String> {
251        self.now_ms = now_ms;
252        for &byte in bytes {
253            let outcome = self
254                .decoder
255                .push(byte)
256                .map(|result| result.map(<[u8]>::to_vec));
257            if let Some(outcome) = outcome {
258                let frame = outcome.map_err(|error| format!("HDLC decode error: {error:?}"))?;
259                self.handle_frame(&frame);
260            }
261        }
262        Ok(())
263    }
264
265    /// The next HDLC-encoded frame for the host, if any.
266    pub fn take_outbound(&mut self) -> Option<Vec<u8>> {
267        self.outbound.pop_front()
268    }

270    /// Give the device a settings interface of its own. From here on the
271    /// device publishes whether or not a host is attached, and every
272    /// publication reaches [`SoftDevice::take_device_frame`].
273    pub fn open_device_interface(&mut self) {
274        self.device_interface = true;
275        self.session.open_device_interface();
276    }

278    /// Serve one raw ULCP frame from the device's own interface. Its answer
279    /// waits in [`SoftDevice::take_device_frame`]; whatever the device
280    /// publishes while serving it, and whatever it changes, reaches an
281    /// attached host too. A change to a saved setting keeps an existing
282    /// snapshot in step.
283    pub fn device_request(&mut self, frame: &[u8], now_ms: u64) {
284        debug_assert!(self.device_interface, "the device has no interface");
285        self.now_ms = now_ms;
286        let writes = Writes::with(frame, |key| self.session.snapshots(key));
287        let device_ms = self.device_ms();
288        let mut emitted = Vec::new();
289        let effect = self
290            .session
291            .handle_device_frame(frame, device_ms, &mut |bytes| emitted.push(bytes.to_vec()));
292        self.execute(effect, &mut emitted);
293        self.session.end_admin_exchange();
294        let mut published = Vec::new();
295        self.platform
296            .settle(&mut self.session, &mut |frame: &[u8]| {
297                published.push(frame.to_vec())
298            });
299        for frame in emitted {
300            if is_publication(&frame) {
301                self.publish(frame);
302            } else {
303                if writes.any && self.session.is_attached() {
304                    let mut mirrors = Mirrors::new(&frame);
305                    while let Some(mirror) = mirrors.next() {
306                        self.to_host(mirror);
307                    }
308                }
309                self.device_out.push_back(frame);
310            }
311        }
312        self.queue_emitted(published);
313        self.flush_session_reset_notice();
314        if writes.saved {
315            self.keep_snapshot_in_step();
316        }
317        self.keep_lists();
318    }

320    /// The next raw frame for the device's own interface, if any.
321    pub fn take_device_frame(&mut self) -> Option<Vec<u8>> {
322        self.device_out.pop_front()
323    }

335    /// Hand the platform the device lists when one changed, as the
336    /// firmware driver does after every event.
337    fn keep_lists(&mut self) {
338        let mut buf = [0u8; HELD_LISTS_MAX];
339        if let Some(len) = self.session.lists_to_keep(&mut buf)
340            && self.platform.store_lists(Some(&buf[..len])).is_ok()
341        {
342            self.session.note_lists_kept();
343        }
344    }
345
346    /// Keep an existing snapshot in step with a change made at the device,
347    /// as the firmware driver does: never make one, and never write over a
348    /// generation boot rejected.
349    fn keep_snapshot_in_step(&mut self) {
350        if self.session.saved_status() != SavedStatus::Current {
351            return;
352        }
353        let mut buf = [0u8; SNAPSHOT_MAX];
354        if let Some(len) = self.session.encode_snapshot(&mut buf)
355            && self.platform.store_snapshot(Some(&buf[..len])).is_ok()
356        {
357            self.session.note_snapshot_saved();
358        }
359    }

372    /// Put a reception through the real device receive path.
373    pub fn inject_radio_rx(&mut self, bytes: &[u8], info: &RadioRxInfo, now_ms: u64) {
374        self.now_ms = now_ms;
375        let device_ms = self.device_ms();
376        let mut emitted = Vec::new();
377        let effect = self
378            .session
379            .on_radio_rx(bytes, info, device_ms, &mut |frame| {

382        self.execute(effect, &mut emitted);
383        self.queue_emitted(emitted);

400    fn handle_frame(&mut self, frame: &[u8]) {
401        let device_ms = self.device_ms();
402        let mut emitted = Vec::new();
403        let effect = self
404            .session
405            .handle_frame(frame, device_ms, &mut |bytes| emitted.push(bytes.to_vec()));
406        self.execute(effect, &mut emitted);
407        // What the host changed, the device's own interface hears.
408        if self.device_interface && Writes::of(frame).any {
409            for answer in emitted.iter().filter(|frame| !is_publication(frame)) {
410                let mut mirrors = Mirrors::new(answer);
411                while let Some(mirror) = mirrors.next() {
412                    self.device_out.push_back(mirror.to_vec());
413                }
414            }
415        }
416        self.platform
417            .settle(&mut self.session, &mut |frame: &[u8]| {
418                emitted.push(frame.to_vec())
419            });
420        self.queue_emitted(emitted);
421        self.flush_session_reset_notice();
422        self.keep_lists();
423    }

425    fn execute(&mut self, mut effect: Option<Effect>, emitted: &mut Vec<Vec<u8>>) {
426        while let Some(current) = effect {
427            self.execute_one(current, emitted);
428            effect = self
429                .session
430                .resume_multi(self.device_ms(), &mut |frame| emitted.push(frame.to_vec()));
431        }
432    }
433
434    fn execute_one(&mut self, effect: Effect, emitted: &mut Vec<Vec<u8>>) {
435        let mut emit = |frame: &[u8]| emitted.push(frame.to_vec());
436        match effect {
437            // No hardware to apply them to: the session's own values are
438            // the whole of the state.
439            Effect::ApplyRadio(_) | Effect::DeviceNameChanged | Effect::ApplyAlert(_) => {}
440            Effect::StartTransmit => {
441                let device_ms = self.device_ms();
442                self.air.push(self.session.tx_data().to_vec());
443                self.session
444                    .on_tx_result(TxOutcome::Sent, device_ms, &mut emit);
445            }
446            Effect::DrainQueue => {
447                let device_ms = self.device_ms();
448                while self.session.drain_step(device_ms, &mut emit) {}
449            }
450            Effect::SaveSnapshot { tid } => {
451                let mut buf = [0u8; SNAPSHOT_MAX];
452                let result = match self.session.encode_snapshot(&mut buf) {
453                    Some(len) => self.platform.store_snapshot(Some(&buf[..len])),
454                    None => Err(()),
455                };
456                self.session.respond_save(tid, result, &mut emit);
457            }
458            Effect::ClearSaved { tid } => {
459                let mut result = self
460                    .platform
461                    .store_snapshot(None)
462                    .and(self.platform.store_identity(None));
463                if result.is_ok() && self.session.holds_lists() {
464                    result = self.platform.store_lists(None);
465                }
466                self.session.respond_clear(tid, result, &mut emit);
467            }
468            Effect::FactoryReset => {
469                // The platform wipe-and-reboot. No reply: on hardware the
470                // reboot drops the link.
471                let _ = self.platform.store_snapshot(None);
472                let _ = self.platform.store_identity(None);
473                if self.session.holds_lists() {
474                    let _ = self.platform.store_lists(None);
475                }
476                self.platform.erase();
477                self.boot_ms = self.now_ms;
478                self.boot();
479            }
480            Effect::Reboot => {
481                // A power cycle and nothing else: the persisted state
482                // survives and is restored on the way up. No reply, as on
483                // hardware.
484                self.boot_ms = self.now_ms;
485                self.boot();
486            }
487            Effect::Dfu { tid, .. } => {
488                self.session
489                    .respond_dfu(tid, Err(Status::UNIMPLEMENTED), &mut emit);
490            }
491            Effect::Announce { tid, request } => {
492                self.announcements.push(request);
493                self.session.respond_announce(tid, Ok(()), &mut emit);
494            }
495            // No node, so no key to sign an identity with.
496            Effect::SignIdentity { tid } => {
497                self.session.respond_identity_blob(tid, Err(()), &mut emit);
498            }
499            Effect::ReadTime { tid } => {
500                let epoch = self.platform.wall_clock();
501                self.session.respond_time(tid, epoch, &mut emit);
502            }
503            Effect::ApplyTime { epoch } => self.platform.set_wall_clock(epoch),
504            Effect::ProvisionIdentity { tid } => {
505                let result = match self.session.identity_request() {
506                    Some(source) => {
507                        let secret = match source {
508                            IdentitySource::Install(secret) => secret,
509                            IdentitySource::Generate => self.platform.generate_secret(),
510                        };
511                        self.platform
512                            .store_identity(Some(&secret))
513                            .map(|()| SoftwareIdentity::from_secret_bytes(&secret).public_key().0)
514                    }
515                    None => Err(()),
516                };
517                self.session.respond_identity(tid, result, &mut emit);
518            }
519            other => self
520                .platform
521                .peripheral(other, &mut self.session, &mut emit),
522        }
523    }

525    /// Deliver what the session emitted outside an exchange with the
526    /// device's own interface: everything goes to the host, as it always
527    /// has, and a publication reaches the interface too.
528    fn queue_emitted(&mut self, emitted: Vec<Vec<u8>>) {
529        for frame in emitted {
530            if is_publication(&frame) {
531                self.publish(frame);
532            } else {
533                self.to_host(&frame);
534            }
535        }
536    }
537
538    /// Everyone listening hears a publication. With the interface open,
539    /// the session publishes whether or not a host is attached, so the
540    /// host is sent one only while it is there to receive it.
541    fn publish(&mut self, frame: Vec<u8>) {
542        if self.session.is_attached() || !self.device_interface {
543            self.to_host(&frame);
544        }
545        if self.device_interface {
546            self.device_out.push_back(frame);
547        }
548    }
549
550    fn to_host(&mut self, frame: &[u8]) {
551        let mut encoded = vec![0; hdlc::max_encoded_len(frame.len())];
552        let len = hdlc::encode_frame(frame, &mut encoded)
553            .expect("soft device output buffer uses HDLC worst-case size");
554        encoded.truncate(len);
555        self.outbound.push_back(encoded);
556    }
557}

559/// Whether `frame` is one everyone listening to the device receives,
560/// rather than an answer for whoever asked.
561fn is_publication(frame: &[u8]) -> bool {
562    Frame::parse(frame).is_ok_and(|parsed| parsed.is_publication())
