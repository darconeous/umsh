1//! Soft ULCP devices: the production device [`Session`] behind an
2//! HDLC-Lite byte stream, with no hardware under it.
3//!
4//! [`soft::SoftDevice`] is the core, generic over a [`soft::SoftPlatform`]
5//! that supplies storage, entropy, a clock, and peripherals. Two platforms
6//! ship here:
7//!
8//! - [`SimulatedDevice`]: a deterministic test fixture. Storage is RAM,
9//!   identities come from a counter, and the battery, receiver, Wi-Fi
10//!   station, and I2C bus are canned. Every protocol decision is still
11//!   made by the real session. Transmissions accumulate until
12//!   [`SimulatedDevice::take_transmitted`] drains them, so the caller
13//!   decides what "the air" is—the web debugger discards it, a bridge
14//!   copies it to real segments.
15//! - [`host::HostPlatform`]: a real device with no radio of its own, such
16//!   as a LAN virtual radio. State persists in files, identities come
17//!   from the operating system's generator, and the clock is the system's.
18//! - [`lan`]: that device with the LAN bearer's group as its PHY, driven
19//!   for as long as its owner runs it.
20//!
21//! The [`SessionConfig`] is the caller's: it names the device and
22//! declares its capability surface. A simulated device has no node
23//! behind it, so its configuration must leave
24//! [`SessionConfig::mac_node`] unset—a backhaul would connect the host
25//! to nothing.
26

107/// A small register-file peripheral at one address: a write sets the
108/// register pointer and stores whatever follows, a read returns from the
109/// pointer onward, and both advance it. Enough to make a register read
110/// (`write pointer, read N`) mean something.
111#[derive(Debug, Default)]
112pub struct SimulatedI2c {
113    registers: [u8; 8],
114    pointer: u8,
115    /// Transactions performed, failures included. A test that checks a
116    /// retransmission was answered from a retained response rather than
117    /// executed again needs this: an unchanged register file would not
118    /// tell an idempotent second write from a suppressed one.
119    executed: u32,
120}

122impl SimulatedI2c {
123    /// The peripheral's 7-bit address.
124    pub const ADDRESS: u8 = 0x50;

126    /// Perform one transaction, returning the concatenated read data.
127    pub fn transact(&mut self, addr: u8, request: TransferRequest<'_>) -> Result<Vec<u8>, Status> {
128        self.executed += 1;
129        if addr != Self::ADDRESS {
130            return Err(Status::NO_DEVICE);
131        }
132        let mut out = Vec::new();
133        for op in request.ops() {
134            match op.map_err(|error| error.status())? {
135                umsh_ulcp::i2c::Op::Write(data) => {
136                    self.pointer = data[0];
137                    for byte in &data[1..] {
138                        self.registers[usize::from(self.pointer % 8)] = *byte;
139                        self.pointer = self.pointer.wrapping_add(1);
140                    }
141                }
142                umsh_ulcp::i2c::Op::Read(len) => {
143                    for _ in 0..len {
144                        out.push(self.registers[usize::from(self.pointer % 8)]);
145                        self.pointer = self.pointer.wrapping_add(1);
146                    }
147                }
148            }
149        }
150        Ok(out)
151    }
152
153    /// The addresses in a range that acknowledge.
154    pub fn scan(&mut self, request: ScanRequest) -> Vec<u8> {
155        self.executed += 1;
156        if request.covers(Self::ADDRESS) {
157            vec![Self::ADDRESS]
158        } else {
159            Vec::new()
160        }
161    }
162
163    /// Transactions performed so far, scans included.
164    pub fn executed(&self) -> u32 {
165        self.executed
166    }
167
168    /// The register file as it stands.
169    pub fn registers(&self) -> &[u8; 8] {
170        &self.registers
171    }
172}

206/// The simulated Wi-Fi station and IP stack.
207///
208/// Where the driver would be: the network table with its credentials, the
209/// results of a scan, and whatever the link and the stack have made of
210/// the configuration the session holds. Nothing here is protocol state —
211/// the session owns that—which is exactly the division the effects
212/// exist to draw.
213#[derive(Default)]
214struct SimulatedNetwork {
215    /// Stored networks in item form, credential included, keyed by SSID.
216    networks: Vec<Vec<u8>>,
217    /// Whether the station is currently on the selected network.
218    associated: bool,
219    /// Whether a scan has run. A device that has never looked reports an
220    /// empty table rather than the results of a scan nobody asked for.
221    scanned: bool,
222}

224impl SimulatedNetwork {
225    /// Where a stored network with this SSID sits, if one does.
226    fn position(&self, ssid: &[u8]) -> Option<usize> {
227        self.networks
228            .iter()
229            .position(|entry| Self::ssid_of(entry) == ssid)
230    }
231
232    fn ssid_of(entry: &[u8]) -> &[u8] {
233        NetworkEntry::decode(entry)
234            .map(|entry| entry.ssid)
235            .unwrap_or(&[])
236    }
237
238    /// Store an entry, replacing any held under the same name.
239    fn store(&mut self, entry: &[u8]) {
240        match self.position(Self::ssid_of(entry)) {
241            Some(index) => self.networks[index] = entry.to_vec(),
242            None => self.networks.push(entry.to_vec()),
243        }
244    }
245}

367    /// Append a sensor without changing any existing index or name.
368    pub fn append_temperature_sensor(&mut self, name: &str, value: u16) -> Result<usize, Status> {
369        if !umsh_ulcp::temperature::valid_name(name) {
370            return Err(Status::INVALID_ARGUMENT);
371        }
372        let pending_bytes = self
373            .temperature_append_on_sample
374            .as_ref()
375            .map_or(0, |(name, _)| name.len() + 1);
376        let pending_count = usize::from(self.temperature_append_on_sample.is_some());
377        let bytes: usize = self
378            .temperatures
379            .iter()
380            .map(|(name, _)| 1 + name.len())
381            .sum();
382        if self.temperatures.len() + pending_count >= umsh_ulcp_device::MAX_TEMPERATURE_SENSORS
383            || bytes + pending_bytes + 1 + name.len() > umsh_ulcp_device::TEMPERATURE_NAMES_MAX
384        {
385            return Err(Status::NOMEM);
386        }
387        let index = self.temperatures.len();
388        self.temperatures.push((name.into(), value));
389        Ok(index)
390    }
391
392    /// Change availability or temperature while preserving the inventory.
393    pub fn set_temperature(&mut self, index: usize, value: u16) -> Result<(), Status> {
394        self.temperatures
395            .get_mut(index)
396            .ok_or(Status::INVALID_ARGUMENT)?
397            .1 = value;
398        Ok(())
399    }

420    /// What the simulated access points offer, by name.
421    ///
422    /// The office network is a WPA2/WPA3 transition deployment, the guest
423    /// network is Enhanced Open beside a plain open BSS, and the nameless
424    /// one is ordinary WPA2—three shapes a picker has to render
425    /// differently.
426    fn simulated_modes(ssid: &[u8]) -> u16 {
427        match ssid {
428            b"Field Office" => SecurityMode::Wpa2.bit() | SecurityMode::Wpa3.bit(),
429            b"Depot Guest" => SecurityMode::Open.bit() | SecurityMode::Owe.bit(),
430            _ => SecurityMode::Wpa2.bit(),
431        }

434    /// One of the tables the platform owns rather than the session,
435    /// encoded as a length-prefixed item list.
436    fn network_table(&self, session: &DeviceSession, key: u32) -> Vec<u8> {
437        let mut items = Vec::new();
438        let mut push = |item: &[u8]| {
439            let mut buf = [0u8; 1 + umsh_ulcp::wifi::NETWORK_ENTRY_MAX_LEN];
440            if let Ok(len) = items::encode_prefixed_item(item, &mut buf) {
441                items.extend_from_slice(&buf[..len]);
442            }
443        };
444        match key {
445            // Reported form: the credential stops here and goes no
446            // further, which is the entire point of the table living
447            // with the driver.
448            prop::WIFI_NETWORKS => {
449                for entry in &self.network.networks {
450                    let mut reported = [0u8; umsh_ulcp::wifi::NETWORK_ENTRY_MAX_LEN];
451                    let encoded = NetworkEntry::decode(entry)
452                        .and_then(|entry| entry.encode_reported(&mut reported));
453                    if let Ok(len) = encoded {
454                        push(&reported[..len]);
455                    }
456                }
457            }
458            // Whatever the last scan heard, strongest first, which is
459            // how a device reports its retained table.
460            prop::WIFI_SCAN_RESULTS if self.network.scanned => {
461                for &(frequency_mhz, rssi_dbm, bssid, ssid) in SIMULATED_ACCESS_POINTS {
462                    let mut buf = [0u8; umsh_ulcp::wifi::SCAN_RESULT_MAX_LEN];
463                    let result = ScanResult {
464                        modes: Self::simulated_modes(ssid),
465                        frequency_mhz,
466                        rssi_dbm,
467                        bssid,
468                        ssid,
469                    };
470                    if let Ok(len) = result.encode(&mut buf) {
471                        push(&buf[..len]);
472                    }
473                }
474            }
475            prop::IPV4_ADDRESS if self.network.associated => {
476                let mut buf = [0u8; umsh_ulcp::ip::V4_ADDRESS_LEN];
477                let address = V4Address {
478                    address: [192, 168, 8, 42],
479                    prefix: 24,
480                    gateway: [192, 168, 8, 1],
481                };
482                if let Ok(len) = address.encode(&mut buf) {
483                    push(&buf[..len]);
484                }
485            }
486            prop::IPV6_ADDRESSES if self.network.associated => {
487                let mut address = [0u8; 16];
488                address[..2].copy_from_slice(&[0x20, 0x01]);
489                address[2..4].copy_from_slice(&[0x0d, 0xb8]);
490                address[15] = 0x2a;
491                let mut router = [0u8; 16];
492                router[0] = 0xfe;
493                router[1] = 0x80;
494                router[15] = 0x01;
495                for item in [
496                    V6Item::Address {
497                        address,
498                        prefix: 64,
499                    },
500                    V6Item::Router { address: router },
501                ] {
502                    let mut buf = [0u8; umsh_ulcp::ip::V6_ADDRESS_ITEM_LEN];
503                    if let Ok(len) = item.encode(&mut buf) {
504                        push(&buf[..len]);
505                    }
506                }
507            }
508            // What the stack resolves through: the device's own list
509            // where it has one, and the network's offer where it does
510            // not.
511            prop::IP_RESOLVERS if self.network.associated => {
512                let configured: Vec<Vec<u8>> =
513                    session.dns_resolvers().iter().map(<[u8]>::to_vec).collect();
514                match configured.is_empty() {
515                    true => push(&[192, 168, 8, 1]),
516                    false => configured.iter().for_each(|resolver| push(resolver)),
517                }
518            }
519            _ => {}
520        }
521        items
522    }

524    /// Bring the simulated link and stack to what the session now holds.
525    ///
526    /// Where a driver would react to the device-domain mirror. Called
527    /// after every frame, so a write that switches the station off or
528    /// picks a different network is followed by the state that write
529    /// produced.
530    fn settle_network(&mut self, session: &mut DeviceSession, mut emit: &mut dyn FnMut(&[u8])) {
531        if !session.wifi_station() {
532            return;
533        }
534        session.set_wifi_mac(SIMULATED_WIFI_MAC);
535        let selected = session.selected_network().to_vec();
536        let associated = session.wifi_enabled()
537            && !selected.is_empty()
538            && session
539                .known_networks()
540                .get(&selected)
541                .is_some_and(|entry| {
542                    // The office AP advertises WPA2/WPA3 transition mode.
543                    // Upgrading its offered mode set never changes the saved
544                    // minimum or allows a weaker authentication method.
545                    [SecurityMode::Wpa3, SecurityMode::Wpa2]
546                        .into_iter()
547                        .any(|mode| entry.security.permits(mode, entry.raw_key))
548                        || entry.security.permits(SecurityMode::Open, entry.raw_key)
549                });
550        self.network.associated = associated;
551        let link = match associated {
552            true => Link {
553                state: LinkState::Up,
554                reason: LinkReason::None,
555                association: Some(Association {
556                    bssid: SIMULATED_ACCESS_POINTS[0].2,
557                    frequency_mhz: SIMULATED_ACCESS_POINTS[0].0,
558                }),
559            },
560            false => Link::default(),
561        };
562        session.set_wifi_link(link, &mut emit);
563        session.set_wifi_rssi(associated.then_some(SIMULATED_ACCESS_POINTS[0].1));
564        let state = match associated {
565            true => FamilyState::Ready,
566            false => FamilyState::NoLink,
567        };
568        session.set_ip_state(prop::IPV4_STATE, state, &mut emit);
569        session.set_ip_state(prop::IPV6_STATE, state, &mut emit);
570    }

572    /// The simulated receiver's view, walked one cell east per sample.
573    ///
574    /// A receiver that has been switched off reports
575    /// [`GnssSnapshot::SEARCHING`]—zero for the facts it is sure of,
576    /// empty for the position it does not have—which is the state most
577    /// worth being able to see in a debugger.
578    fn gnss_sample(&mut self, session: &DeviceSession) -> GnssSnapshot {
579        if !session.gnss_enabled() {
580            return GnssSnapshot::SEARCHING;
581        }
582        self.fix_step = self.fix_step.wrapping_add(1);
583        let mut snapshot = GnssSnapshot::SEARCHING;
584        snapshot.fix = FixKind::ThreeD;
585        snapshot.altitude_m = Some(64);
586        snapshot.accuracy_dm = Some(GnssSnapshot::accuracy_from_hdop_centi(120));
587        snapshot.sats_used = 9;
588        snapshot.sats_in_view = Some(14);
589        let step = (self.fix_step % 16) as u8;
590        snapshot.set_location(&[0x8a, 0x1f, 0x4c, 0x00, 0xd0 | step]);
591        snapshot
592    }

646    fn peripheral(
647        &mut self,
648        effect: Effect,
649        session: &mut DeviceSession,
650        mut emit: &mut dyn FnMut(&[u8]),
651    ) {
652        match effect {
653            Effect::SampleRssi { tid } => {
654                session.respond_rssi(tid, Ok(-77), &mut emit);
655            }
656            Effect::SampleBattery { tid } => {
657                // Stable, human-recognizable simulated measurement; the
658                // configured field set decides what of it is reported.
659                session.respond_battery(
660                    tid,
661                    Ok(BatteryStatus {
662                        voltage_mv: Some(4111),
663                        level_percent: Some(87),
664                        charge_state: Some(BatteryChargeState::Charging),
665                    }),
666                    &mut emit,
667                );
668            }
669            Effect::SampleBatteryGroup {
670                tid,
671                key,
672                fields: _,
673            } => {
674                use umsh_ulcp::battery_diagnostics::{KEYS, Sample, Value, VoltageRequest};
675                let mut sample = Sample::default();
676                sample.snapshot = Ok(BatteryStatus {
677                    voltage_mv: Some(4111),
678                    level_percent: Some(87),
679                    charge_state: Some(BatteryChargeState::Charging),
680                });
681                let values = [
682                    Value::Current(100),
683                    Value::Unsigned(1305),
684                    Value::Unsigned(1500),
685                    Value::Unsigned(1500),
686                    Value::Bool(true),
687                    Value::Bool(true),
688                    Value::Bool(false),
689                    Value::Bool(true),
690                    Value::Bool(false),
691                    Value::Voltage(VoltageRequest::Millivolts(4200)),
692                    Value::Format(1),
693                    Value::Unsigned(8),
694                    Value::Unsigned(0x26),
695                ];
696                for (key, value) in KEYS.into_iter().zip(values) {
697                    sample.set(key, Ok(Some(value)));
698                }
699                session.respond_battery_group(tid, key, sample, &mut emit);
700            }
701            Effect::SampleTemperatures { tid } => {
702                self.temperature_samples += 1;
703                let values: Vec<_> = self.temperatures.iter().map(|(_, value)| *value).collect();
704                if let Some((name, value)) = self.temperature_append_on_sample.take() {
705                    self.append_temperature_sensor(&name, value)
706                        .expect("staged registration fits");
707                }
708                session.respond_temperatures(tid, Ok(&values), &mut emit);
709            }
710            Effect::ReadTemperatureNames { tid } => {
711                let names: Vec<_> = self
712                    .temperatures
713                    .iter()
714                    .map(|(name, _)| name.as_str())
715                    .collect();
716                let mut value = [0; umsh_ulcp_device::TEMPERATURE_NAMES_MAX];
717                let len = umsh_ulcp::temperature::encode_names(&names, &mut value)
718                    .expect("bounded inventory");
719                session.respond_temperature_names(tid, Ok(&value[..len]), &mut emit);
720            }
721            Effect::SampleIlluminance { tid } => {
722                // A stable simulated reading: ordinary office lighting.
723                session.respond_illuminance(tid, Some(320_000), &mut emit);
724            }
725            Effect::SetPairingPin { tid, pin } => {
726                self.pairing_pin = pin;
727                session.respond_pin_set(tid, Ok(()), &mut emit);
728            }
729            Effect::ClearBleBonds { tid } => {
730                // The full security reset a board performs: bonds, PIN,
731                // and lockout together, and the pairing window a
732                // freshly-cleared device opens.
733                self.bond_count = 0;
734                self.pairing_pin = None;
735                session.set_ble_bond_count(0, &mut emit);
736                session.set_ble_pairing(true, &mut emit);
737                session.respond_ble_bond_count(tid, Ok(()), &mut emit);
738            }
739            Effect::SetBlePairing { tid, open } => {
740                // Nothing here can pair, so the window opens and nothing
741                // walks through it. A full store is not a refusal—
742                // enrollment at capacity evicts—and the one state that
743                // would refuse an open, a pairing lockout, needs failed
744                // pairings this device cannot have.
745                session.respond_ble_pairing(tid, Ok(open), &mut emit);
746            }
747            Effect::SampleGnss { tid, key } => {
748                let sample = self.gnss_sample(session);
749                session.respond_gnss(tid, key, Ok(sample), &mut emit);
750            }
751            // A scan the whole way through, in the four frames the spec
752            // orders: the clear, the reply, an insert per access point,
753            // and the drop to zero. Instant rather than progressive,
754            // which is the shape of every stack that hands a scan back
755            // only when it finishes.
756            Effect::SetWifiScanning { tid, scanning } => {
757                if session.wifi_station() && !session.wifi_enabled() {
758                    session.respond_wifi_scanning(tid, Err(Status::INVALID_STATE), &mut emit);
759                    return;
760                }
761                if !scanning {
762                    session.respond_wifi_scanning(tid, Ok(false), &mut emit);
763                    return;
764                }
765                self.network.scanned = true;
766                session.clear_scan_results(&mut emit);
767                session.respond_wifi_scanning(tid, Ok(true), &mut emit);
768                for &(frequency_mhz, rssi_dbm, bssid, ssid) in SIMULATED_ACCESS_POINTS {
769                    session.publish_scan_result(
770                        &ScanResult {
771                            modes: Self::simulated_modes(ssid),
772                            frequency_mhz,
773                            rssi_dbm,
774                            bssid,
775                            ssid,
776                        },
777                        &mut emit,
778                    );
779                }
780                session.set_wifi_scanning(false, &mut emit);
781            }
782            Effect::SelectWifiNetwork { tid } => {
783                let ssid = session.pending_network().to_vec();
784                let result = match ssid.is_empty() || self.network.position(&ssid).is_some() {
785                    true => Ok(()),
786                    false => Err(Status::ITEM_NOT_FOUND),
787                };
788                session.respond_wifi_network(tid, result, &mut emit);
789            }
790            Effect::InsertWifiNetwork { tid } => {
791                let entry = session.pending_network_entry().to_vec();
792                // A mode the simulated hardware does not do, so that a
793                // host's step-down path has something to step down from.
794                let result = match NetworkEntry::decode(&entry).map(|entry| entry.security) {
795                    Ok(SecurityMode::Wpa3Ent192) => Err(Status::UNIMPLEMENTED),
796                    _ => {
797                        self.network.store(&entry);
798                        Ok(())
799                    }
800                };
801                session.respond_wifi_network_stored(tid, result, &mut emit);
802            }
803            Effect::RemoveWifiNetwork { tid } => {
804                let ssid = session.pending_network().to_vec();
805                let result = match self.network.position(&ssid) {
806                    Some(index) => {
807                        self.network.networks.remove(index);
808                        Ok(())
809                    }
810                    None => Err(Status::ITEM_NOT_FOUND),
811                };
812                session.respond_wifi_network_forgotten(tid, result, &mut emit);
813            }
814            Effect::ReadNetworkTable { tid, key } => {
815                let items = self.network_table(session, key);
816                session.respond_network_table(tid, key, Ok(&items), &mut emit);
817            }
818            Effect::I2cTransfer { tid } => {
819                let request = session.i2c_request();
820                let result = self.i2c.transact(request.addr, request);
821                session.respond_i2c(tid, result.as_deref().map_err(|status| *status), &mut emit);
822            }
823            Effect::I2cScan { tid } => {
824                let found = self.i2c.scan(session.i2c_scan_request());
825                session.respond_i2c(tid, Ok(&found), &mut emit);
826            }
827            // Everything else is served by the core.
828            _ => {}
829        }
830    }

909    fn decode(wire: Vec<u8>) -> Vec<u8> {
910        let mut decoder = hdlc::Decoder::<WIRE_CAPACITY>::new();
911        wire.into_iter()
912            .find_map(|byte| decoder.push(byte).map(|frame| frame.unwrap().to_vec()))
913            .unwrap()
914    }

920    fn exchange_at(sim: &mut SimulatedDevice, request: &[u8], now_ms: u64) -> Vec<Vec<u8>> {
921        let mut wire = vec![0; hdlc::max_encoded_len(request.len())];
922        let len = hdlc::encode_frame(request, &mut wire).unwrap();
923        sim.ingest(&wire[..len], now_ms).unwrap();
924
925        let mut frames = Vec::new();
926        while let Some(wire) = sim.take_outbound() {
927            frames.push(decode(wire));
928        }
929        frames
930    }
