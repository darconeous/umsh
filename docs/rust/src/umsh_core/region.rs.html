1//! Region-code encoding for the Region Code packet option.
2//!
3//! A region code is a 2-byte routing-domain tag. It is not an RF band plan:
4//! it scopes flood forwarding to a locally agreed area, and repeaters match
5//! it against their configured region list.
6//!
7//! Codes come from two sources, and the encoding keeps them disjoint so a
8//! code can be rendered without knowing which one produced it:
9//!
10//! - **Airport regions** encode a 3-letter IATA code with ARNCE/HAM-16.
11//! - **Named regions** take the first two bytes of the SHA-256 of the name,
12//!   transformed away from the letter space if they happen to land in it.
13//!
14//! Because the transform vacates every three-letter encoding, a code that
15//! decodes to three letters is always an airport code, and one that does
16//! not has no recoverable text form — [`Display`](core::fmt::Display)
17//! renders it as `0xXXXX`.
18
19use core::fmt;
20use core::str::FromStr;
21
22use hamaddr::{HamAddr, HamAddrType};
23use sha2::{Digest, Sha256};
24
25/// Lowest ARNCE index that denotes a letter (`A`).
26const LETTER_MIN: u16 = 1;
27/// Highest ARNCE index that denotes a letter (`Z`).
28const LETTER_MAX: u16 = 26;
29/// First code reserved for transformed named regions.
30const TRANSFORM_BASE: u16 = 27 * 1600;
31
32/// Longest a region's string form may be, in UTF-8 bytes.
33///
34/// Names travel on the wire in identity payloads, one option per region, and
35/// this bound is what lets a list of them fit
36/// (packet-options.md § Region Code Encoding).
37pub const REGION_NAME_MAX_LEN: usize = 24;
38
39/// A 2-byte region identifier.
40///
41/// ```
42/// # use umsh_core::RegionCode;
43/// let sjc: RegionCode = "SJC".parse().unwrap();
44/// assert_eq!(sjc.to_bytes(), [0x78, 0x53]);
45/// assert_eq!(sjc.to_string(), "SJC");
46///
47/// let valley: RegionCode = "Rogue Valley".parse().unwrap();
48/// assert_eq!(valley.to_string(), "0xDF6F");
49/// ```
50#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
51pub struct RegionCode(u16);
52
53impl RegionCode {
54    /// Wrap a raw code.
55    pub const fn from_u16(value: u16) -> Self {
56        Self(value)
57    }
58
59    /// Return the raw code.
60    pub const fn as_u16(self) -> u16 {
61        self.0
62    }
63
64    /// Wrap a code from its two wire bytes.
65    pub const fn from_bytes(bytes: [u8; 2]) -> Self {
66        Self(u16::from_be_bytes(bytes))
67    }
68
69    /// Return the code's two wire bytes.
70    pub const fn to_bytes(self) -> [u8; 2] {
71        self.0.to_be_bytes()
72    }
73
74    /// Encode a 3-letter IATA airport code.
75    ///
76    /// Case-insensitive. Anything that is not exactly three ASCII letters
77    /// is rejected, so that named regions cannot be mistaken for airports.
78    pub fn from_iata(code: &str) -> Result<Self, RegionCodeError> {
79        if code.len() != 3 || !code.bytes().all(|b| b.is_ascii_alphabetic()) {
80            return Err(RegionCodeError::NotIata);
81        }
82        let addr = HamAddr::try_from_callsign(code).map_err(|_| RegionCodeError::NotIata)?;
83        Ok(Self(addr.chunk(0)))
84    }
85
86    /// Derive the code for a named region.
87    ///
88    /// The name is hashed verbatim, so callers that want names to compare
89    /// equal across operators must agree on the exact spelling.
90    pub fn from_name(name: &str) -> Self {
91        let digest = Sha256::digest(name.as_bytes());
92        Self(transform_letter_chunk(u16::from_be_bytes([
93            digest[0], digest[1],
94        ])))
95    }
96
97    /// Return the three letters this code decodes to, if it decodes to
98    /// three letters at all.
99    ///
100    /// `Some` means the code came from [`from_iata`](Self::from_iata):
101    /// named regions are transformed out of this space by construction.
102    pub fn letters(self) -> Option<[u8; 3]> {
103        let addr = HamAddr::from_chunks([self.0, 0, 0, 0]);
104        if !matches!(addr.get_type(), HamAddrType::Callsign) {
105            return None;
106        }
107        let mut sink = Letters::default();
108        fmt::write(&mut sink, format_args!("{addr}")).ok()?;
109        if sink.len != 3 || !sink.buf.iter().all(u8::is_ascii_alphabetic) {
110            return None;
111        }
112        Some(sink.buf)
113    }
114}
115
116/// Move a three-letter encoding into the space reserved for named regions.
117///
118/// Anything that is not three letters is already outside that space and is
119/// returned unchanged.
120fn transform_letter_chunk(encoded: u16) -> u16 {
121    let a = encoded / 1600;
122    let b = (encoded / 40) % 40;
123    let c = encoded % 40;
124
125    if ![a, b, c]
126        .iter()
127        .all(|x| (LETTER_MIN..=LETTER_MAX).contains(x))
128    {
129        return encoded;
130    }
131
132    let rank = (a - 1) * 26 * 26 + (b - 1) * 26 + (c - 1);
133    TRANSFORM_BASE + rank
134}

136/// A fixed-capacity sink for the at-most-three characters a single ARNCE
137/// chunk renders to. Overlong writes leave `len` past the buffer and are
138/// rejected by the caller.
139#[derive(Default)]
140struct Letters {
141    buf: [u8; 3],
142    len: usize,
143}
144
145impl fmt::Write for Letters {
146    fn write_str(&mut self, s: &str) -> fmt::Result {
147        for byte in s.bytes() {
148            if let Some(slot) = self.buf.get_mut(self.len) {
149                *slot = byte;
150            }
151            self.len += 1;
152        }
153        Ok(())
154    }
155}

157/// Parse a region code from its textual form.
158///
159/// Three ASCII letters are an IATA code, `0x` followed by exactly four hex
160/// digits is the code it spells, and anything else is a region name. The
161/// derivation is total over every string of one to
162/// [`REGION_NAME_MAX_LEN`] bytes: a string that merely looks like a literal
163/// code — `0x12`, `0xzz` — is not one, and is hashed as the name it is
164/// (packet-options.md § Region Code Encoding).
165impl FromStr for RegionCode {
166    type Err = RegionCodeError;
167
168    fn from_str(s: &str) -> Result<Self, Self::Err> {
169        let trimmed = s.trim();
170        if trimmed.is_empty() {
171            return Err(RegionCodeError::Empty);
172        }
173        if trimmed.len() > REGION_NAME_MAX_LEN {
174            return Err(RegionCodeError::TooLong);
175        }
176        if let Some(hex) = trimmed
177            .strip_prefix("0x")
178            .or_else(|| trimmed.strip_prefix("0X"))
179            && hex.len() == 4
180            && hex.bytes().all(|b| b.is_ascii_hexdigit())
181            && let Ok(value) = u16::from_str_radix(hex, 16)
182        {
183            return Ok(Self(value));
184        }
185        Self::from_iata(trimmed).or_else(|_| Ok(Self::from_name(trimmed)))
186    }
187}
188
189impl fmt::Display for RegionCode {
190    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
191        match self.letters().as_ref().map(|l| core::str::from_utf8(l)) {
192            Some(Ok(text)) => f.write_str(text),
193            _ => write!(f, "0x{:04X}", self.0),
194        }
195    }
196}
197
198impl fmt::Debug for RegionCode {
199    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
200        write!(f, "RegionCode({self})")
201    }
202}

204impl From<RegionCode> for [u8; 2] {
205    fn from(code: RegionCode) -> Self {
206        code.to_bytes()
207    }
208}
209
210impl From<[u8; 2]> for RegionCode {
211    fn from(bytes: [u8; 2]) -> Self {
212        Self::from_bytes(bytes)
213    }
214}

216/// Why a string could not be read as a region code.
217#[derive(Debug, Clone, Copy, PartialEq, Eq)]
218pub enum RegionCodeError {
219    /// The input was empty or only whitespace.
220    Empty,
221    /// The input was longer than [`REGION_NAME_MAX_LEN`] bytes.
222    TooLong,
223    /// The input was not exactly three ASCII letters.
224    NotIata,
225}
226
227impl fmt::Display for RegionCodeError {
228    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
229        match self {
230            Self::Empty => f.write_str("empty region code"),
231            Self::TooLong => write!(f, "region name longer than {REGION_NAME_MAX_LEN} bytes"),
232            Self::NotIata => f.write_str("expected a three-letter IATA code"),
233        }
234    }
235}

275    #[test]
276    fn transforms_a_named_region_that_lands_on_three_letters() {
277        // SHA-256("Southern Oregon") begins 0x6AF2, which decodes to `QDR`.
278        assert_eq!(transform_letter_chunk(0x6AF2), 0xD35F);
279        assert_eq!(RegionCode::from_name("Southern Oregon").as_u16(), 0xD35F);
280    }

282    #[test]
283    fn leaves_a_named_region_outside_the_letter_space_alone() {
284        assert_eq!(transform_letter_chunk(0xDF6F), 0xDF6F);
285        assert_eq!(transform_letter_chunk(0x31D9), 0x31D9);
286    }

288    #[test]
289    fn no_named_region_can_collide_with_an_airport_region() {
290        // The transform is what guarantees this, so assert the property
291        // over the whole 16-bit space rather than trusting the examples.
292        for raw in 0..=u16::MAX {
293            let transformed = RegionCode::from_u16(transform_letter_chunk(raw));
294            assert_eq!(
295                transformed.letters(),
296                None,
297                "0x{raw:04X} transformed to a three-letter code"
298            );
299        }
300    }

302    #[test]
303    fn every_airport_region_round_trips_through_its_text_form() {
304        for a in b'A'..=b'Z' {
305            for b in b'A'..=b'Z' {
306                for c in b'A'..=b'Z' {
307                    let text = core::str::from_utf8(&[a, b, c]).unwrap().to_string();
308                    let code = RegionCode::from_iata(&text).unwrap();
309                    assert_eq!(code.letters(), Some([a, b, c]));
310                    assert_eq!(code.to_string(), text);
311                    assert_eq!(text.parse::<RegionCode>().unwrap(), code);
312                }
313            }
314        }
315    }

335    #[test]
336    fn hashes_anything_that_only_looks_like_a_literal_code() {
337        // Only `0x` plus exactly four hex digits spells a code. Everything
338        // else is a name, which keeps the derivation total: there is no such
