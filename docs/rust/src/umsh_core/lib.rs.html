3//! Core wire-format types and packet construction/parsing utilities for UMSH.
4//!
5//! > Note: This reference implementation is a work in progress and was developed
6//! > with the assistance of an LLM. It should be considered experimental.
7//!
8//! This crate stays focused on byte layout, typed header fields, and zero-copy
9//! parsing. It does not perform any cryptographic operations or I/O.
10//!
11//! The main entry points are:
12//!
13//! - [`PacketHeader::parse`] to inspect a received frame.
14//! - [`PacketBuilder`] to construct outbound frames with typestate guards.
15//! - [`feed_aad`] to stream canonical authenticated data into a MAC state.
16//! - [`options::OptionEncoder`] and [`options::OptionDecoder`] for CoAP-style
17//!   option blocks used throughout the protocol.
18//!
19//! # Example
20//!
21//! ```rust
22//! use umsh_core::{MicSize, NodeHint, PacketBuilder, PacketHeader, PublicKey};
23//!
24//! let mut buf = [0u8; 128];
25//! let src = PublicKey([0x11; 32]);
26//! let dst = NodeHint([0xAA, 0xBB, 0xCC]);
27//!
28//! let packet = PacketBuilder::new(&mut buf)
29//!     .unicast(dst)
30//!     .source_full(&src)
31//!     .frame_counter(7)
32//!     .encrypted()
33//!     .mic_size(MicSize::Mic16)
34//!     .payload(b"hello")
35//!     .build()
36//!     .unwrap();
37//!
38//! let header = PacketHeader::parse(packet.as_bytes()).unwrap();
39//! assert_eq!(header.dst, Some(dst));
40//! assert_eq!(header.body_range.len(), 5);
41//! ```

80        let mut buf = [0u8; 32];
81        let mut enc = OptionEncoder::new(&mut buf);
82        enc.put(1, &[0x78, 0x53]).unwrap();
83        enc.put(2, &[]).unwrap();
84        enc.end_marker().unwrap();
85        let len = enc.finish();
86
87        let mut decoder = OptionDecoder::new(&buf[..len]);
88        assert_eq!(decoder.next().unwrap().unwrap(), (1, &[0x78, 0x53][..]));
89        assert_eq!(decoder.next().unwrap().unwrap(), (2, &[][..]));
90        assert!(decoder.next().is_none());
91    }

114    #[test]
115    fn empty_payload_emits_no_end_of_options_marker() {
116        // A beacon is a broadcast with an empty payload. Nothing follows the
117        // options block, so the `0xFF` marker must not be emitted—it would
118        // be a wasted byte on every beacon.
119        let mut buf = [0u8; 64];
120        let beacon = PacketBuilder::new(&mut buf)
121            .broadcast()
122            .source_hint(NodeHint([0x8B, 0x60, 0xF5]))
123            .flood_hops(5)
124            .payload(&[])
125            .build()
126            .unwrap();
127        assert_eq!(beacon, &[0xC1, 0x50, 0x8B, 0x60, 0xF5]);
128        let header = PacketHeader::parse(beacon).unwrap();
129        assert!(header.is_beacon());
130
131        // The marker returns as soon as there is a payload to delimit.
132        let mut buf = [0u8; 64];
133        let with_payload = PacketBuilder::new(&mut buf)
134            .broadcast()
135            .source_hint(NodeHint([0x8B, 0x60, 0xF5]))
136            .payload(b"hi")
137            .build()
138            .unwrap();
139        assert_eq!(with_payload, &[0xC0, 0x8B, 0x60, 0xF5, 0xFF, b'h', b'i']);
140
141        // The same holds for a secured type whose body is empty.
142        let mut buf = [0u8; 128];
143        let unicast = PacketBuilder::new(&mut buf)
144            .unicast(NodeHint([0xC3, 0xD4, 0x25]))
145            .source_hint(NodeHint([0xA1, 0xB2, 0x03]))
146            .frame_counter(7)
147            .encrypted()
148            .mic_size(MicSize::Mic8)
149            .payload(&[])
150            .build()
151            .unwrap();
152        // FCF | DST(3) | SRC(3) | SECINFO(5) | no options, no marker | MIC(8)
153        let bytes = unicast.as_bytes();
154        assert_eq!(
155            bytes,
156            &[
157                0xD0, 0xC3, 0xD4, 0x25, 0xA1, 0xB2, 0x03, 0xA0, 0x00, 0x00, 0x00, 0x07, 0x00, 0x00,
158                0x00, 0x00, 0x00, 0x00, 0x00, 0x00,
159            ],
160        );
161        let header = PacketHeader::parse(bytes).unwrap();
162        assert!(header.options_range.is_empty());
163        assert!(header.body_range.is_empty());
164        assert_eq!(header.mic_range.len(), 8);
165    }

259    #[test]
260    fn aad_excludes_dynamic_options() {
261        // New layout for unicast with hint source, encrypted, MIC8, no fhops:
262        // FCF | DST(3) | SRC(3) | SECINFO(5) | OPTIONS(5) | 0xFF | payload(3) | MIC(8)
263        // Total = 1 + 3 + 3 + 5 + 4 + 1 + 3 + 8 = 28 bytes
264        // Options: trace route (2, len 0) + region code (11, len 2)—both dynamic, excluded from AAD
265        let mut bytes = [0u8; 64];
266        bytes[0] = Fcf::new(PacketType::Unicast, false, false).0;
267        bytes[1..4].copy_from_slice(&[0xC3, 0xD4, 0x25]); // DST
268        bytes[4..7].copy_from_slice(&[0xA1, 0xB2, 0x03]); // SRC hint
269        bytes[7] = Scf::new(true, MicSize::Mic8, false).0; // SCF
270        bytes[8..12].copy_from_slice(&42u32.to_be_bytes()); // frame counter
271        bytes[12] = 0x20; // trace route: delta=2, len=0
272        bytes[13] = 0x92; // region code: delta=9, len=2
273        bytes[14] = 0x78;
274        bytes[15] = 0x53;
275        bytes[16] = 0xFF; // end marker
276        bytes[17..20].copy_from_slice(b"hey"); // payload
277        bytes[20..28].fill(0x11); // MIC
278        let header = PacketHeader::parse(&bytes[..28]).unwrap();
279        let mut aad = [0u8; 18];
280        let mut aad_len = 0usize;
281        feed_aad(&header, &bytes[..28], |chunk| {
282            let next_len = aad_len + chunk.len();
283            aad[aad_len..next_len].copy_from_slice(chunk);
284            aad_len = next_len;
285        });
286        assert_eq!(
287            &aad[..aad_len],
288            &[
289                bytes[0],
290                0xC3,
291                0xD4,
292                0x25,
293                0xA1,
294                0xB2,
295                0x03,
296                Scf::new(true, MicSize::Mic8, false).0,
297                0x00,
298                0x00,
299                0x00,
300                0x2A,
301            ]
302        );
303    }
