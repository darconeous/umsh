1// SenseCAP Solar Node P1 / P1-Pro bringup firmware — Phase 1.
2//
3// This is the *stripped* first-boot binary (see
4// docs/firmware-plan-sensecap-solar-node-p1-pro.md, Phase 1). Its job is
5// to prove the board comes up safely and to identify the LEDs / buttons
6// whose colors and polarities the hardware reconstruction left open. It
7// deliberately does NOT bring up the OLED (there is none), the SX1262
8// radio (Phase 3), or the MAC / CLI (Phase 4 parity).
9//
10// What it does:
11//   1. Initialize embassy-nrf, arm the 8 s watchdog.
12//   2. Read + clear any panic message left by the previous boot.
13//   3. Drive the board into a safe state:
14//        - RADIO_RXEN (P0.05) LOW  — safety contract: RXEN low before any
15//          radio work and until lora-phy owns it (no radio here yet).
16//        - GNSS_ENABLE (P1.05) LOW — hold the L76K powered down.
17//        - GNSS_RESET candidate (P1.03) is NEVER driven — left untouched.
18//        - Battery-divider gate (P0.14) is NOT touched in Phase 1; the
19//          gate polarity is verified in Phase 2 before we drive it.
20//   4. Bring up USB-CDC with the CdcAcmRescue escape hatch + panic replay.
21//   5. Blink LED_B (P0.19, the blue "breathing" LED) as a heartbeat, and
22//      run an interactive GPIO/button identification session over CDC.
23//
24// Confirmed on hardware 2026-07-23:
25//   - LED_A (P0.15) = white, active-high; LED_B (P0.19) = blue, active-high.
26//   - USER_BUTTON (P1.01) and the "PWR" button (P1.07) are both active-low
27//     on internal pull-ups (LOW pressed, HIGH released). PWR is a soft
28//     momentary button — pressing it does not cut the MCU rail.
29//
30// Identification session (single-char commands, echoed back as reports):
31//   '1'  toggle LED_A (P0.15) — the white user LED
32//   '?'  help + current LED_A / button states
33//   USER_BUTTON (P1.01) and PWR (P1.07) edges auto-report.
34//
35// The LED_B heartbeat runs in its own task (umsh_ux_tracker::led::LedEngine,
36// a 20 ms pulse every 4 s), decoupled from the CDC RX loop so terminal
37// activity can never affect its cadence. The watchdog is petted from that
38// same heartbeat loop, exactly as on the other nRF52840 boards.
39

96    /// Write a line (with trailing CRLF) to the CDC sender in <=64-byte
97    /// USB packets. Best-effort — drops on a closed endpoint.
98    async fn write_line(tx: &mut SolarSender, s: &str) {
99        for chunk in s.as_bytes().chunks(64) {
100            let _ = tx.write_packet(chunk).await;
101        }
102        let _ = tx.write_packet(b"\r\n").await;
103    }

105    const HELP: &str = "\
106SenseCAP Solar Node bringup (Phase 1) — GPIO identification\r
107  1  toggle LED_A (P0.15, white user LED)\r
108  ?  this help + current pin states\r
109LED_B (P0.19, blue) blinks as the heartbeat.\r
110USER_BUTTON (P1.01) and PWR (P1.07) edges auto-report.";

112    /// Interactive GPIO/button identification over CDC. Owns LED_A, both
113    /// buttons, and the CDC RX/TX endpoints. LED_B belongs to the
114    /// independent heartbeat task.
115    #[embassy_executor::task]
116    async fn ident_task(
117        mut tx: SolarSender,
118        mut rx: SolarRescue,
119        mut led_a: Output<'static>,
120        mut btn1: Input<'static>,
121        mut btn2: Input<'static>,
122        prev_panic_buf: &'static [u8; 256],
123        prev_panic_len: usize,
124    ) {
125        // Wait for the host to open the CDC port before writing the banner —
126        // otherwise the writes silently vanish into a closed IN endpoint.
127        rx.wait_connection().await;
128
129        let sha = env!("GIT_SHORT_SHA");
130        write_line(&mut tx, "").await;
131        write_line(&mut tx, "UMSH SenseCAP Solar Node bringup (Phase 1)").await;
132        write_line(&mut tx, sha).await;
133        if prev_panic_len > 0 {
134            write_line(&mut tx, "[PREV PANIC]:").await;
135            if let Ok(s) = core::str::from_utf8(&prev_panic_buf[..prev_panic_len]) {
136                write_line(&mut tx, s).await;
137            }
138        }
139        write_line(&mut tx, HELP).await;
140
141        let mut a_level = false; // LED_A assumed active-high (confirmed)
142        led_a.set_low();
143
144        let mut pkt = [0u8; 64];
145        loop {
146            // Re-arm on every iteration so a host re-attach is transparent and
147            // a disconnected port doesn't busy-loop on read_packet returning 0.
148            rx.wait_connection().await;
149
150            match select3(
151                rx.read_packet(&mut pkt),
152                btn1.wait_for_any_edge(),
153                btn2.wait_for_any_edge(),
154            )
155            .await
156            {
157                // CDC command bytes.
158                Either3::First(res) => {
159                    let n = match res {
160                        Ok(0) | Err(_) => continue, // disconnect → re-arm at top
161                        Ok(n) => n,
162                    };
163                    for &b in &pkt[..n] {
164                        match b {
165                            b'1' => {
166                                a_level = !a_level;
167                                if a_level {
168                                    led_a.set_high();
169                                } else {
170                                    led_a.set_low();
171                                }
172                                report(&mut tx, "LED_A (P0.15)", a_level).await;
173                            }
174                            b'?' => {
175                                write_line(&mut tx, HELP).await;
176                                report(&mut tx, "LED_A (P0.15)", a_level).await;
177                                report_button(&mut tx, "USER_BUTTON (P1.01)", &btn1).await;
178                                report_button(&mut tx, "PWR         (P1.07)", &btn2).await;
179                            }
180                            b'\r' | b'\n' => {}
181                            _ => {} // ignore other bytes
182                        }
183                    }
184                }
185
186                // Button edges: report the settled level. Pull-up inputs, so
187                // LOW == pressed (active-low, confirmed).
188                Either3::Second(()) => {
189                    report_button(&mut tx, "USER_BUTTON (P1.01)", &btn1).await;
190                }
191                Either3::Third(()) => {
192                    report_button(&mut tx, "PWR         (P1.07)", &btn2).await;
193                }
194            }
195        }
196    }

198    async fn report(tx: &mut SolarSender, name: &str, level: bool) {
199        write_line(tx, name).await;
200        write_line(tx, if level { "  -> HIGH" } else { "  -> LOW" }).await;
201    }
202
203    async fn report_button(tx: &mut SolarSender, name: &str, btn: &Input<'static>) {
204        write_line(tx, name).await;
205        // Pull-up input: released == HIGH, pressed == LOW (active-low, confirmed).
206        write_line(
207            tx,
208            if btn.is_low() {
209                "  = LOW  (pressed)"
210            } else {
211                "  = HIGH (released)"
212            },
213        )
214        .await;
215    }

217    // ─── Heartbeat (LED_B) + watchdog ────────────────────────────────────────
218
219    /// Blinks LED_B (P0.19, blue "breathing" LED) via the shared LedEngine
220    /// and pets the watchdog on the same schedule. Isolated from the CDC RX
221    /// loop so terminal traffic can't perturb its cadence. Matches the
222    /// heartbeat structure used on the other nRF52840 boards.
223    async fn heartbeat(mut led_b: Output<'static>, mut wdt: WatchdogHandle) -> ! {
224        let mut engine = LedEngine::new(LedTimings::default(), Instant::now().as_millis());
225        loop {
226            wdt.pet();
227            let decision = engine.tick(Instant::now().as_millis());
228            if decision.on {
229                led_b.set_high();
230            } else {
231                led_b.set_low();
232            }
233            Timer::at(Instant::from_millis(decision.next_deadline_ms)).await;
234        }
235    }

239    #[embassy_executor::main]
240    async fn main(spawner: Spawner) {
241        let p = embassy_nrf::init(umsh_bsp_nrf52840::clocks::default_config());
242
243        let mut wdt_config = WdtConfig::default();
244        wdt_config.timeout_ticks = 32768 * 8; // 8 s
245        let (_wdt, [wdt_handle]) =
246            Watchdog::try_new::<_, 1>(p.WDT, wdt_config).unwrap_or_else(|_| panic!("wdt"));
247
248        // ── Board safe state ──────────────────────────────────────────────────
249        // Held live for the lifetime of `main` (which never returns). RXEN low
250        // per the safety contract; GNSS held powered down. The GNSS reset
251        // candidate (P1.03) and battery-divider gate (P0.14) are deliberately
252        // left untouched in Phase 1.
253        let _radio_rxen = Output::new(p.P0_05, Level::Low, OutputDrive::Standard);
254        let _gnss_enable = Output::new(p.P1_05, Level::Low, OutputDrive::Standard);
255
256        // ── Previous-boot panic message ───────────────────────────────────────
257        static PREV_PANIC_BUF: StaticCell<[u8; 256]> = StaticCell::new();
258        let mut prev_panic_tmp = [0u8; 256];
259        let prev_panic_len = {
260            let mut slot = PanicSlot::new(super::panic::panic_region());
261            if let Some(msg) = slot.read() {
262                let n = msg.len().min(prev_panic_tmp.len());
263                prev_panic_tmp[..n].copy_from_slice(&msg[..n]);
264                slot.clear();
265                n
266            } else {
267                0
268            }
269        };
270        let prev_panic_buf: &'static [u8; 256] = PREV_PANIC_BUF.init(prev_panic_tmp);
271
272        // ── User LEDs + buttons ───────────────────────────────────────────────
273        let led_a = Output::new(p.P0_15, Level::Low, OutputDrive::Standard);
274        let led_b = Output::new(p.P0_19, Level::Low, OutputDrive::Standard);
275        let btn1 = Input::new(p.P1_01, Pull::Up);
276        let btn2 = Input::new(p.P1_07, Pull::Up);
277
278        // ── USB-CDC stack ─────────────────────────────────────────────────────
279        let driver = Driver::new(p.USBD, Irqs, HardwareVbusDetect::new(Irqs));
280
281        // USB IDs per the plan: VID 0x2886 / PID 0x0059 (Seeed XIAO family).
282        let mut config = Config::new(0x2886, 0x0059);
283        config.manufacturer = Some("UMSH");
284        config.product = Some("SenseCAP Solar Node Bringup");
285        config.serial_number = Some("companion-cli-sensecap-solar");
286        config.max_power = 100;
287        config.max_packet_size_0 = 64;
288
289        static CONFIG_DESC: StaticCell<[u8; 256]> = StaticCell::new();
290        static BOS_DESC: StaticCell<[u8; 256]> = StaticCell::new();
291        static MSOS_DESC: StaticCell<[u8; 0]> = StaticCell::new();
292        static CONTROL_BUF: StaticCell<[u8; 64]> = StaticCell::new();
293        static STATE: StaticCell<State> = StaticCell::new();
294
295        let mut builder = Builder::new(
296            driver,
297            config,
298            CONFIG_DESC.init([0; 256]),
299            BOS_DESC.init([0; 256]),
300            MSOS_DESC.init([0; 0]),
301            CONTROL_BUF.init([0; 64]),
302        );
303
304        let class = CdcAcmClass::new(&mut builder, STATE.init(State::new()), 64);
305        let mut usb = builder.build();
306
307        let (tx, raw_rx, ctrl) = class.split_with_control();
308        let rx = CdcAcmRescue::new(raw_rx, ctrl);
309
310        spawner
311            .spawn(
312                ident_task(tx, rx, led_a, btn1, btn2, prev_panic_buf, prev_panic_len).unwrap(),
313            );
314
315        join(usb.run(), heartbeat(led_b, wdt_handle)).await;
316    }
