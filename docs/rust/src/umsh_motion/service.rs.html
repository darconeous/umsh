1//! Shared observations and independently notified consumers. Subscription does
2//! not itself power the sensor: consumers explicitly register their demand.
3use crate::State;
4use core::cell::Cell;
5use embassy_sync::{
6    blocking_mutex::{Mutex, raw::CriticalSectionRawMutex},
7    signal::Signal,
8    watch::Watch,
9};
10
11#[derive(Clone, Copy, Debug, PartialEq, Eq)]
12pub enum Consumer {
13    Display = 1,
14    Location = 2,
15}
16
17#[derive(Clone, Copy, Debug, PartialEq, Eq)]
18pub struct Control {
19    pub generation: u32,
20    pub consumers: u8,
21    pub shutdown: bool,
22}
23
24impl Control {
25    pub fn sensing(self) -> bool {
26        !self.shutdown && self.consumers != 0
27    }
28    pub fn display(self) -> bool {
29        !self.shutdown && self.consumers & Consumer::Display as u8 != 0
30    }
31}
32
33#[derive(Clone, Copy, Debug)]
34pub struct Wake {
35    pub generation: u32,
36    pub at_ms: u64,
37}
38
39pub struct Service {
40    control: Mutex<CriticalSectionRawMutex, Cell<Control>>,
41    pub changed: Signal<CriticalSectionRawMutex, ()>,
42    pub stopped: Signal<CriticalSectionRawMutex, ()>,
43    pub observations: Watch<CriticalSectionRawMutex, State, 4>,
44    pub display_wake: Signal<CriticalSectionRawMutex, Wake>,
45}
46
47impl Default for Service {
48    fn default() -> Self {
49        Self::new()
50    }
51}
52
53impl Service {
54    pub const fn new() -> Self {
55        Self {
56            control: Mutex::new(Cell::new(Control {
57                generation: 0,
58                consumers: 0,
59                shutdown: false,
60            })),
61            changed: Signal::new(),
62            stopped: Signal::new(),
63            observations: Watch::new(),
64            display_wake: Signal::new(),
65        }
66    }
67    pub fn control(&self) -> Control {
68        self.control.lock(Cell::get)
69    }
70    pub fn request(&self, consumer: Consumer, enabled: bool) {
71        let changed = self.control.lock(|cell| {
72            let mut c = cell.get();
73            let consumers = if enabled {
74                c.consumers | consumer as u8
75            } else {
76                c.consumers & !(consumer as u8)
77            };
78            if c.consumers == consumers {
79                return false;
80            }
81            c.consumers = consumers;
82            c.generation = c.generation.wrapping_add(1);
83            cell.set(c);
84            true
85        });
86        if changed {
87            self.display_wake.reset();
88            self.changed.signal(());
89        }
90    }
91    /// Fault/reset invalidates queued wakes immediately, before async cleanup.
92    pub fn cancel(&self) {
93        self.control.lock(|cell| {
94            let mut c = cell.get();
95            c.generation = c.generation.wrapping_add(1);
96            cell.set(c);
97        });
98        self.display_wake.reset();
99    }
100    pub fn shutdown(&self) {
101        self.control.lock(|cell| {
102            let mut c = cell.get();
103            c.shutdown = true;
104            c.generation = c.generation.wrapping_add(1);
105            cell.set(c);
106        });
107        self.display_wake.reset();
108        self.changed.signal(());
109    }
110    pub fn accept(&self, wake: Wake, now_ms: u64) -> bool {
111        let c = self.control();
112        c.display()
113            && c.generation == wake.generation
114            && now_ms >= wake.at_ms
115            && now_ms - wake.at_ms <= 250
116    }
117}
