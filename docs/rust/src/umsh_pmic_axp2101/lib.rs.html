1//! X-Powers AXP2101 PMIC, async, over `embedded-hal-async` I²C.
2//!
3//! On a board like the LILYGO T-Beam Supreme the PMIC is not a telemetry
4//! peripheral, it is prerequisite hardware: the radio, GNSS, display,
5//! sensors, and SD card each sit behind a rail this chip switches. A
6//! failed I²C probe of any of them usually means the rail is still off,
7//! not that the part is missing (hardware doc §5.3). So a board brings
8//! this driver up before it probes anything else.
9//!
10//! The surface here is deliberately narrow — rails, charging, telemetry,
11//! interrupts, and the power key. The AXP2101 does considerably more
12//! (JEITA profiles, watchdogs, sleep sequencing, backup-battery
13//! charging); none of it is reachable through UMSH, so none of it is
14//! transcribed.
15//!
16//! The outputs a board does not use are the exception: those are
17//! reachable through [`UnusedOutput`], one direction only, because a
18//! channel left at the part's reset default is a channel nobody decided
19//! about.
20//!
21//! **DCDC1's enable bit is never cleared.** On the boards UMSH supports
22//! it is the MCU core supply, so no value of any type in this crate can
23//! name it, and every write that touches its register is a
24//! read-modify-write that puts it back.
25//!
26//! Register provenance and the transcription caveat: see [`reg`].
27

47/// A switchable low-dropout rail.
48///
49/// The DCDC rails are absent on purpose: DCDC1 is the MCU core supply,
50/// and the rest feed expansion interfaces no UMSH board owns.
51#[derive(Debug, Clone, Copy, PartialEq, Eq)]
52pub enum Rail {
53    Aldo1,
54    Aldo2,
55    Aldo3,
56    Aldo4,
57    Bldo1,
58    Bldo2,
59}
60
61impl Rail {
62    /// This rail's bit in [`reg::LDO_ONOFF_CTRL0`].
63    pub const fn enable_bit(self) -> u8 {
64        match self {
65            Rail::Aldo1 => 0,
66            Rail::Aldo2 => 1,
67            Rail::Aldo3 => 2,
68            Rail::Aldo4 => 3,
69            Rail::Bldo1 => 4,
70            Rail::Bldo2 => 5,
71        }
72    }
73
74    /// This rail's voltage register.
75    pub const fn voltage_reg(self) -> u8 {
76        match self {
77            Rail::Aldo1 => reg::ALDO1_VOL,
78            Rail::Aldo2 => reg::ALDO2_VOL,
79            Rail::Aldo3 => reg::ALDO3_VOL,
80            Rail::Aldo4 => reg::ALDO4_VOL,
81            Rail::Bldo1 => reg::BLDO1_VOL,
82            Rail::Bldo2 => reg::BLDO2_VOL,
83        }
84    }
85}

87/// A PMIC output UMSH never brings up, exposed only so a board can
88/// switch it off rather than inherit whatever the part's reset default
89/// left it at.
90///
91/// There is no setter for these and no voltage control: the type says
92/// "off", and enabling a converter at a voltage some previous session
93/// chose is the hazard [`Axp2101::enable_rail_at`] exists to avoid. A
94/// carrier that genuinely needs one of these up wants voltage support
95/// added deliberately, not a bool flipped here.
96///
97/// Deliberately absent:
98///
99/// - **DCDC1**, the MCU core supply on every board UMSH supports.
100///   Clearing its bit switches off the processor doing the clearing, so
101///   no value of this type can name it — the mistake is unrepresentable
102///   rather than runtime-guarded.
103/// - **CPUSLDO**, whose load is unattributed on the boards we have. An
104///   output nobody has traced is not an output to switch off blind.
105/// - The six [`Rail`]s, which have their own on-and-off API.
106#[derive(Debug, Clone, Copy, PartialEq, Eq)]
107pub enum UnusedOutput {
108    Dcdc2,
109    Dcdc3,
110    Dcdc4,
111    Dcdc5,
112    Dldo1,
113    Dldo2,
114}
115
116impl UnusedOutput {
117    /// Every output this type can name, for a board that uses none of
118    /// them. Iterate it in bring-up rather than listing variants, so a
119    /// new variant cannot be silently missed.
120    pub const ALL: [UnusedOutput; 6] = [
121        UnusedOutput::Dcdc2,
122        UnusedOutput::Dcdc3,
123        UnusedOutput::Dcdc4,
124        UnusedOutput::Dcdc5,
125        UnusedOutput::Dldo1,
126        UnusedOutput::Dldo2,
127    ];
128
129    /// This output's enable register and the bit within it.
130    pub const fn enable_bit(self) -> (u8, u8) {
131        match self {
132            UnusedOutput::Dcdc2 => (reg::DC_ONOFF_DVM_CTRL, reg::DCDC2_BIT),
133            UnusedOutput::Dcdc3 => (reg::DC_ONOFF_DVM_CTRL, reg::DCDC3_BIT),
134            UnusedOutput::Dcdc4 => (reg::DC_ONOFF_DVM_CTRL, reg::DCDC4_BIT),
135            UnusedOutput::Dcdc5 => (reg::DC_ONOFF_DVM_CTRL, reg::DCDC5_BIT),
136            UnusedOutput::Dldo1 => (reg::LDO_ONOFF_CTRL0, reg::DLDO1_BIT),
137            UnusedOutput::Dldo2 => (reg::LDO_ONOFF_CTRL1, reg::DLDO2_BIT),
138        }
139    }
140}

142/// Encode a rail voltage into its register field, rejecting values off
143/// the 100 mV grid or outside the part's range.
144pub const fn encode_rail_mv(millivolts: u16) -> Option<u8> {
145    if millivolts < reg::LDO_VOL_MIN_MV
146        || millivolts > reg::LDO_VOL_MAX_MV
147        || (millivolts - reg::LDO_VOL_MIN_MV) % reg::LDO_VOL_STEP_MV != 0
148    {
149        return None;
150    }
151    Some(((millivolts - reg::LDO_VOL_MIN_MV) / reg::LDO_VOL_STEP_MV) as u8)
152}
153
154/// Decode a rail voltage register field back to millivolts.
155pub const fn decode_rail_mv(field: u8) -> u16 {
156    (field & reg::LDO_VOL_MASK) as u16 * reg::LDO_VOL_STEP_MV + reg::LDO_VOL_MIN_MV
157}

193/// What the charger is doing, from [`reg::STATUS2`].
194#[derive(Debug, Clone, Copy, PartialEq, Eq)]
195pub enum ChargeState {
196    Trickle,
197    Precharge,
198    ConstantCurrent,
199    ConstantVoltage,
200    Done,
201    NotCharging,
202    /// A code the datasheet does not assign.
203    Unknown(u8),
204}
205
206impl ChargeState {
207    fn from_field(field: u8) -> Self {
208        match field & reg::STATUS2_CHARGE_STATE_MASK {
209            0 => ChargeState::Trickle,
210            1 => ChargeState::Precharge,
211            2 => ChargeState::ConstantCurrent,
212            3 => ChargeState::ConstantVoltage,
213            4 => ChargeState::Done,
214            5 => ChargeState::NotCharging,
215            other => ChargeState::Unknown(other),
216        }
217    }
218
219    /// Whether current is actually going into the cell.
220    pub fn is_charging(self) -> bool {
221        matches!(
222            self,
223            ChargeState::Trickle
224                | ChargeState::Precharge
225                | ChargeState::ConstantCurrent
226                | ChargeState::ConstantVoltage
227        )
228    }
229}

258/// AXP2101 interrupt sources, as a 24-bit mask across the three IRQ
259/// registers. Only the events a UMSH board acts on are named.
260#[derive(Debug, Clone, Copy, PartialEq, Eq)]
261pub struct IrqMask(pub u32);
262
263impl IrqMask {
264    pub const NONE: IrqMask = IrqMask(0);
265    /// Every bit the part defines — used to clear stale status at boot.
266    pub const ALL: IrqMask = IrqMask(0x00FF_FFFF);
267
268    // Register 0x41.
269    pub const POWER_KEY_LONG: IrqMask = IrqMask(1 << 10);
270    pub const POWER_KEY_SHORT: IrqMask = IrqMask(1 << 11);
271    pub const BATTERY_REMOVED: IrqMask = IrqMask(1 << 12);
272    pub const BATTERY_INSERTED: IrqMask = IrqMask(1 << 13);
273    pub const VBUS_REMOVED: IrqMask = IrqMask(1 << 14);
274    pub const VBUS_INSERTED: IrqMask = IrqMask(1 << 15);
275
276    // Register 0x42.
277    pub const CHARGE_STARTED: IrqMask = IrqMask(1 << 19);
278    pub const CHARGE_DONE: IrqMask = IrqMask(1 << 20);
279
280    pub const fn union(self, other: IrqMask) -> IrqMask {
281        IrqMask(self.0 | other.0)
282    }
283
284    pub const fn contains(self, other: IrqMask) -> bool {
285        self.0 & other.0 == other.0 && other.0 != 0
286    }
287
288    pub const fn is_empty(self) -> bool {
289        self.0 == 0
290    }
291
292    fn bytes(self) -> [u8; 3] {
293        [self.0 as u8, (self.0 >> 8) as u8, (self.0 >> 16) as u8]
294    }
295
296    fn from_bytes(bytes: [u8; 3]) -> Self {
297        IrqMask(bytes[0] as u32 | (bytes[1] as u32) << 8 | (bytes[2] as u32) << 16)
298    }
299}

303#[derive(Debug, Clone, Copy, PartialEq, Eq)]
304pub enum Error<E> {
305    /// The bus refused the transfer.
306    Bus(E),
307    /// `IC_TYPE` did not read back [`CHIP_ID`].
308    WrongChip(u8),
309    /// A voltage was off the 100 mV grid or outside the part's range.
310    UnsupportedVoltage(u16),
311}
312
313impl<E> From<E> for Error<E> {
314    fn from(e: E) -> Self {
315        Error::Bus(e)
316    }
317}

319type Result<T, E> = core::result::Result<T, Error<E>>;

323pub struct Axp2101<I> {
324    i2c: I,
325    addr: u8,
326}

328impl<I: I2c> Axp2101<I> {
329    /// Bind the PMIC at the standard address.
330    pub fn new(i2c: I) -> Self {
331        Self::with_address(i2c, DEFAULT_ADDRESS)
332    }
333
334    pub fn with_address(i2c: I, addr: u8) -> Self {
335        Self { i2c, addr }
336    }
337
338    pub fn release(self) -> I {
339        self.i2c
340    }
341
342    // ── Register access ──
343
344    async fn read(&mut self, register: u8) -> Result<u8, I::Error> {
345        let mut buf = [0u8; 1];
346        self.i2c
347            .write_read(self.addr, &[register], &mut buf)
348            .await?;
349        Ok(buf[0])
350    }
351
352    async fn write(&mut self, register: u8, value: u8) -> Result<(), I::Error> {
353        self.i2c.write(self.addr, &[register, value]).await?;
354        Ok(())
355    }
356
357    /// Read-modify-write, preserving every bit outside `mask`.
358    async fn update(&mut self, register: u8, mask: u8, value: u8) -> Result<(), I::Error> {
359        let current = self.read(register).await?;
360        let next = (current & !mask) | (value & mask);
361        self.write(register, next).await
362    }
363
364    async fn set_bit(&mut self, register: u8, bit: u8, on: bool) -> Result<(), I::Error> {
365        self.update(register, 1 << bit, if on { 1 << bit } else { 0 })
366            .await
367    }
368
369    async fn get_bit(&mut self, register: u8, bit: u8) -> Result<bool, I::Error> {
370        Ok(self.read(register).await? & (1 << bit) != 0)
371    }
372
373    /// Assemble a two-register ADC result: `mask` significant bits of the
374    /// high register above a full low byte.
375    async fn read_adc(&mut self, high: u8, low: u8, mask: u8) -> Result<u16, I::Error> {
376        let h = self.read(high).await?;
377        let l = self.read(low).await?;
378        Ok(((h & mask) as u16) << 8 | l as u16)
379    }
380
381    // ── Identity ──
382
383    /// Read `IC_TYPE`.
384    pub async fn chip_id(&mut self) -> Result<u8, I::Error> {
385        self.read(reg::IC_TYPE).await
386    }
387
388    /// Confirm an AXP2101 is answering. Call before touching any rail:
389    /// writing rail registers at a chip that is not this one can drive a
390    /// supply to the wrong voltage.
391    pub async fn probe(&mut self) -> Result<(), I::Error> {
392        let id = self.chip_id().await?;
393        if id == CHIP_ID {
394            Ok(())
395        } else {
396            Err(Error::WrongChip(id))
397        }
398    }
399
400    // ── Rails ──
401
402    pub async fn rail_enabled(&mut self, rail: Rail) -> Result<bool, I::Error> {
403        self.get_bit(reg::LDO_ONOFF_CTRL0, rail.enable_bit()).await
404    }
405
406    pub async fn set_rail_enabled(&mut self, rail: Rail, on: bool) -> Result<(), I::Error> {
407        self.set_bit(reg::LDO_ONOFF_CTRL0, rail.enable_bit(), on)
408            .await
409    }
410
411    pub async fn set_rail_voltage(&mut self, rail: Rail, millivolts: u16) -> Result<(), I::Error> {
412        let field = encode_rail_mv(millivolts).ok_or(Error::UnsupportedVoltage(millivolts))?;
413        self.update(rail.voltage_reg(), reg::LDO_VOL_MASK, field)
414            .await
415    }
416
417    pub async fn rail_voltage(&mut self, rail: Rail) -> Result<u16, I::Error> {
418        Ok(decode_rail_mv(self.read(rail.voltage_reg()).await?))
419    }
420
421    /// Set a rail's voltage and switch it on, in that order — bringing a
422    /// rail up at whatever the previous session left configured can
423    /// overvolt what is behind it.
424    pub async fn enable_rail_at(&mut self, rail: Rail, millivolts: u16) -> Result<(), I::Error> {
425        self.set_rail_voltage(rail, millivolts).await?;
426        self.set_rail_enabled(rail, true).await
427    }
428
429    /// Switch off an output this board does not use.
430    ///
431    /// Read-modify-write, so the neighboring bits — DCDC1's included —
432    /// come back unchanged.
433    pub async fn disable_unused(&mut self, output: UnusedOutput) -> Result<(), I::Error> {
434        let (register, bit) = output.enable_bit();
435        self.set_bit(register, bit, false).await
436    }
437
438    /// Whether an unused output is currently on, for diagnostics and for
439    /// checking what the part's reset default actually was.
440    pub async fn unused_enabled(&mut self, output: UnusedOutput) -> Result<bool, I::Error> {
441        let (register, bit) = output.enable_bit();
442        self.get_bit(register, bit).await
443    }

445    /// Drop the given rails, wait [`COLD_BOOT_SETTLE_MS`], and leave them
446    /// off.
447    ///
448    /// LILYGO's board init and MeshCore both do this on a cold boot so a
449    /// peripheral that was mid-transaction when power was cut releases
450    /// the bus before anything probes it. Restore the rails afterwards
451    /// with [`enable_rail_at`](Self::enable_rail_at).
452    pub async fn cold_boot_cycle<D: DelayNs>(
453        &mut self,
454        rails: &[Rail],
455        delay: &mut D,
456    ) -> Result<(), I::Error> {
457        for &rail in rails {
458            self.set_rail_enabled(rail, false).await?;
459        }
460        delay.delay_ms(COLD_BOOT_SETTLE_MS).await;
461        Ok(())
462    }

476    pub async fn set_charge_led(&mut self, mode: ChargeLed) -> Result<(), I::Error> {
477        let (keep, value) = match mode {
478            ChargeLed::Charger => (reg::CHGLED_CHARGER_KEEP, reg::CHGLED_CHARGER_CTRL),
479            ChargeLed::Off => (reg::CHGLED_MANUAL_KEEP, reg::CHGLED_MANUAL_CTRL),
480            ChargeLed::Blink1Hz => (
481                reg::CHGLED_MANUAL_KEEP,
482                reg::CHGLED_MANUAL_CTRL | (1 << reg::CHGLED_PATTERN_SHIFT),
483            ),
484            ChargeLed::Blink4Hz => (
485                reg::CHGLED_MANUAL_KEEP,
486                reg::CHGLED_MANUAL_CTRL | (2 << reg::CHGLED_PATTERN_SHIFT),
487            ),
488            ChargeLed::On => (
489                reg::CHGLED_MANUAL_KEEP,
490                reg::CHGLED_MANUAL_CTRL | (3 << reg::CHGLED_PATTERN_SHIFT),
491            ),
492        };
493        let current = self.read(reg::CHGLED_SET_CTRL).await?;
494        self.write(reg::CHGLED_SET_CTRL, (current & keep) | value)
495            .await
496    }

502    pub async fn charge_direction(&mut self) -> Result<ChargeDirection, I::Error> {
503        let field = (self.read(reg::STATUS2).await? >> reg::STATUS2_DIRECTION_SHIFT)
504            & reg::STATUS2_DIRECTION_MASK;
505        Ok(match field {
506            reg::DIRECTION_STANDBY => ChargeDirection::Standby,
507            reg::DIRECTION_CHARGING => ChargeDirection::Charging,

509            other => ChargeDirection::Unknown(other),
510        })
511    }

515    /// Enable the battery, VBUS, and system voltage ADC channels.
516    ///
517    /// The TS (battery thermistor) channel is not among them, because
518    /// whether it should be on is a board fact rather than a driver one:
519    /// with no thermistor populated the input floats and the charger's
520    /// thermal protection acts on noise. A board that knows its TS pin
521    /// is populated opts in with
522    /// [`set_thermistor_measurement`](Self::set_thermistor_measurement).
523    pub async fn enable_telemetry(&mut self) -> Result<(), I::Error> {
524        for bit in [reg::ADC_CH_VBAT, reg::ADC_CH_VBUS, reg::ADC_CH_VSYS] {
525            self.set_bit(reg::ADC_CHANNEL_CTRL, bit, true).await?;
526        }
527        Ok(())
528    }

530    /// Switch the TS (battery thermistor) ADC channel on or off.
531    ///
532    /// Only for a board that has confirmed its TS pin is populated. Two
533    /// things are unproven here and both want a board to answer them:
534    /// whether the part gates the charger's response to TS separately
535    /// from this ADC channel, and whether the populated part is a real
536    /// NTC or the fixed resistor often fitted in its place to keep the
537    /// charger from folding back. A reading that tracks ambient
538    /// temperature settles the second; a reading that sits still does
539    /// not.
540    pub async fn set_thermistor_measurement(&mut self, on: bool) -> Result<(), I::Error> {
541        self.set_bit(reg::ADC_CHANNEL_CTRL, reg::ADC_CH_TS, on)
542            .await
543    }

594    /// Fuel-gauge state of charge, whole percent.
595    ///
596    /// `None` with no cell attached, and also when the gauge reports a
597    /// value outside 0–100 — it returns 0xFF until it has learned the
598    /// pack, which a caller must not show as 255%.
599    pub async fn battery_percent(&mut self) -> Result<Option<u8>, I::Error> {

604        Ok(if raw <= 100 { Some(raw) } else { None })
605    }
606
607    // ── Interrupts ──
608
609    /// Clear every latched IRQ status bit.
610    ///
611    /// Do this before enabling interrupts at boot: the part latches
612    /// events from before the MCU was running, and an unhandled latched
613    /// bit holds the IRQ line asserted so no edge ever arrives.
614    pub async fn clear_all_irqs(&mut self) -> Result<(), I::Error> {
615        for register in reg::INTSTS {
616            self.write(register, 0xFF).await?;
617        }
618        Ok(())
619    }
620
621    /// Replace the enabled-interrupt set.
622    pub async fn set_irq_enabled(&mut self, mask: IrqMask) -> Result<(), I::Error> {
623        for (register, byte) in reg::INTEN.iter().zip(mask.bytes()) {
624            self.write(*register, byte).await?;
625        }
626        Ok(())
627    }
628
629    /// Read the latched interrupt status without clearing it.
630    pub async fn irq_status(&mut self) -> Result<IrqMask, I::Error> {
631        let mut bytes = [0u8; 3];
632        for (slot, register) in bytes.iter_mut().zip(reg::INTSTS) {
633            *slot = self.read(register).await?;
634        }
635        Ok(IrqMask::from_bytes(bytes))
636    }
637
638    /// Read the latched interrupt status and clear exactly the bits that
639    /// were read, so an event arriving mid-call is not lost.
640    pub async fn take_irqs(&mut self) -> Result<IrqMask, I::Error> {
641        let mut bytes = [0u8; 3];
642        for (slot, register) in bytes.iter_mut().zip(reg::INTSTS) {
643            *slot = self.read(register).await?;
644        }
645        for (register, byte) in reg::INTSTS.iter().zip(bytes) {
646            if byte != 0 {
647                self.write(*register, byte).await?;
648            }
649        }
650        Ok(IrqMask::from_bytes(bytes))
651    }
652
653    // ── Power key ──
654
655    /// How long the POWER key must be held to power the board on.
656    pub async fn set_power_on_press(&mut self, press: PowerOnPress) -> Result<(), I::Error> {
657        self.update(reg::IRQ_OFF_ON_LEVEL_CTRL, reg::PRESS_ON_MASK, press as u8)
658            .await
659    }
660
661    /// How long the POWER key must be held to force the board off. This
662    /// is the PMIC's own hard power-off, independent of any firmware
663    /// gesture.
664    pub async fn set_power_off_press(&mut self, press: PowerOffPress) -> Result<(), I::Error> {
665        self.update(
666            reg::IRQ_OFF_ON_LEVEL_CTRL,
667            reg::PRESS_OFF_MASK,
668            (press as u8) << reg::PRESS_OFF_SHIFT,
669        )
670        .await
671    }
672
673    /// Ask the PMIC to cut power.
674    ///
675    /// Returns once the request is written; the rails fall shortly after,
676    /// so a caller must not expect execution to continue meaningfully.
677    /// The board comes back with a POWER key press — no firmware runs in
678    /// between.
679    pub async fn power_off(&mut self) -> Result<(), I::Error> {
680        self.set_bit(reg::COMMON_CONFIG, reg::COMMON_CONFIG_SOFT_POWEROFF, true)
681            .await
682    }
683}
