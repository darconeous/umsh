1//! Share: a code for this node or a joined channel, with what it shares
2//! beside it. Turning moves Shows between them; Backspace goes back.
3
4use buoyant::event::{Key, KeyResponse};
5use buoyant::focus::BoundaryBehavior;
6use buoyant::view::prelude::*;
7use buoyant::view::scroll_view::{ScrollBarVisibility, ScrollDirection};
8use embedded_graphics::pixelcolor::Rgb565;
9use heapless::{String, Vec};

11use super::{STRIP, label};
12use crate::model::{ChannelAccess, ChatKind, MAX_CHATS, Shareable, State};
13use crate::nav::keys;
14use crate::qr::{Owned, QrImage};

17/// The room the code takes, with its quiet zone in modules.
18const ROOM: u32 = theme::HEIGHT - STRIP - 8;
19const QUIET: u32 = 2;
20const INSET: u32 = 8;
21const GAP: u32 = 12;
22const TRAILING: u32 = 4;
23/// The name, each line under it, and Shows.
24const NAME: u32 = 40;
25const LINE: u32 = 20;
26const ROW: u32 = 36;
27const RADIUS: u16 = 8;
28/// Shows' width: the rest of the screen beside the code.
29const COLUMN: u32 = theme::WIDTH - INSET - ROOM - GAP - TRAILING;
30/// How many things there are to show: this node and every chat.
31const SHAREABLES: usize = MAX_CHATS + 1;
32
33/// Indices `ForEach` can hold without borrowing the state.
34static INDICES: [usize; SHAREABLES] = {
35    let mut a = [0; SHAREABLES];
36    let mut i = 0;
37    while i < SHAREABLES {
38        a[i] = i;
39        i += 1;
40    }
41    a
42};
43
44pub fn view(state: &State) -> impl View<Rgb565, State> + use<> {
45    let shown = state.shared();
46    let items = state.shareables();
47    let last = items.len().saturating_sub(1);
48    let mut name: String<32> = String::new();
49    let mut lines: Vec<String<24>, 2> = Vec::new();
50    match shown {
51        Shareable::Identity => {
52            _ = name.push_str(&state.status.device_name);
53            let address = state.node.address();
54            let (first, second) = address.split_at(address.len().min(22));
55            _ = lines.push(String::try_from(first).unwrap_or_default());
56            _ = lines.push(String::try_from(second).unwrap_or_default());
57        }
58        Shareable::Channel(chat) => {
59            if let Some(chat) = state.chat(chat) {
60                _ = name.push_str(&chat.name);
61                let about = match chat.kind {
62                    ChatKind::Channel {
63                        access: ChannelAccess::Private,
64                    } => "Private channel",
65                    ChatKind::Channel {
66                        access: ChannelAccess::Clear,
67                    } => "Not encrypted",
68                    _ => "Named channel",
69                };
70                _ = lines.push(String::try_from(about).unwrap_or_default());
71            }
72        }
73    }
74    // What Shows reads at each place it turns to.
75    let values: Vec<String<32>, SHAREABLES> = items
76        .iter()
77        .map(|item| match item {
78            Shareable::Identity => String::try_from("Identity").unwrap_or_default(),
79            Shareable::Channel(chat) => state
80                .chat(*chat)
81                .map(|c| c.name.clone())
82                .unwrap_or_default(),
83        })
84        .collect();
85    let code = state
86        .node
87        .share_uri(shown)
88        .and_then(|uri| QrImage::encode(&uri, ROOM, QUIET));
89    let mono = matches!(shown, Shareable::Identity);
90    HStack::new((
91        card(code),
92        VStack::new((
93            label::<32>(&name, &font::BOLD, theme::BODY, theme::TEXT)
94                .frame()
95                .with_height(NAME),
96            lines.first().map(|line| about(line, mono)),
97            lines.get(1).map(|line| about(line, mono)),
98            // Turned in place: each place is a row as wide as the column,
99            // and the row focused scrolls into view.
100            ScrollView::new(
101                ForEach::<SHAREABLES>::new_horizontal(&INDICES[..items.len()], move |&i| {
102                    shows(COLUMN, i > 0, i < last, values[i].clone())
103                })
104                .keyed(move |&i| keys::share(items[i])),
105            )
106            .with_direction(ScrollDirection::Horizontal)
107            .with_bar_visibility(ScrollBarVisibility::Never)
108            .frame()
109            .with_width(COLUMN)
110            .with_height(ROW)
111            .padding(Edges::Top, 8),
112        ))
113        .with_spacing(0)
114        .with_alignment(HorizontalAlignment::Leading)
115        .flex_infinite_width(HorizontalAlignment::Leading),
116    ))
117    .with_spacing(GAP)
118    .with_alignment(VerticalAlignment::Top)
119    .padding(Edges::Leading, INSET)
120    .padding(Edges::Trailing, TRAILING)
121    .bound_focus(BoundaryBehavior::Stop)
122    .on_key(|key, s: &mut State| match key {
123        Key::Backspace => {
124            s.back();
125            KeyResponse::Handled
126        }
127        _ => KeyResponse::Ignored,
128    })
129}

131/// A line under the name: the address, set in monospace so its two halves
132/// line up, or what kind of channel it is.
133fn about(line: &str, mono: bool) -> impl View<Rgb565, State> + use<> {
134    label::<24>(
135        line,
136        if mono { &font::MONO } else { &font::REGULAR },
137        theme::SMALL,
138        theme::TEXT_SECONDARY,
139    )
140    .frame()
141    .with_height(LINE)
142}

144/// The code on its light card, centered in the room it has.
145/// The room is the same whatever the code's size, so what is beside it stays
146/// put as Shows turns.
147fn card(code: Option<QrImage>) -> impl View<Rgb565, State> + use<> {
148    code.map(|code| {
149        let quiet = QUIET * code.scale();
150        Image::new(Owned(code))
151            .padding(Edges::All, quiet)
152            .background(
153                Alignment::Center,
154                RoundedRectangle::new(4).foreground_color(Rgb565::new(31, 63, 31)),
155            )
156    })
157    .frame()
158    .with_width(ROOM)
159    .with_height(theme::HEIGHT - STRIP)
160}

162/// Shows at one of the places it turns to, with arrows toward the places
163/// on either side.
164fn shows(
165    width: u32,
166    before: bool,
167    after: bool,
168    value: String<32>,
169) -> impl View<Rgb565, State> + use<> {
170    Button::new(
171        |_: &mut State| {},
172        move |button: ButtonState| {
173            let focused = button.is_focused();
174            let arrow = |icon, on: bool| {
175                Image::new(icon).as_template().foreground_color(if on {
176                    theme::ACCENT
177                } else {
178                    theme::FOCUS
179                })
180            };
181            HStack::new((
182                label::<8>("Shows", &font::BOLD, theme::BODY, theme::TEXT),
183                Spacer::default(),
184                arrow(&icon::BACK, before),
185                label::<32>(&value, &font::BOLD, theme::SECONDARY, theme::TEXT),
186                arrow(&icon::CHEVRON, after),
187            ))
188            .with_spacing(6)
189            .hint_background_color(theme::FOCUS)
190            .padding(Edges::Leading, 10)
191            .padding(Edges::Trailing, 6)
192            .frame()
193            .with_height(ROW - 2)
194            .background(
195                Alignment::Center,
196                focused.then(|| {
197                    ZStack::new((
198                        RoundedRectangle::new(RADIUS).foreground_color(theme::FOCUS),
199                        RoundedRectangle::new(RADIUS)
200                            .stroked(2)
201                            .foreground_color(theme::ACCENT),
202                    ))
203                }),
204            )
205            .padding(Edges::Leading, 4)
206            .frame()
207            .with_width(width)
208        },
209    )
