1//! How a message reached this device: the rows of the iOS details sheet
2//! that a reader on a communicator needs, and who reacted to it. The wheel
3//! scrolls through them; Backspace goes back.
4
5use buoyant::event::{Key, KeyResponse};
6use buoyant::focus::BoundaryBehavior;
7use buoyant::view::prelude::*;
8use embedded_graphics::pixelcolor::Rgb565;
9use heapless::String;

11use super::conversation::header;
12use super::{label, line};
13use crate::model::{Details, MAX_REACTORS, Reactor, State};
14use crate::nav::keys;

17const ROW: u32 = 36;
18const INSET: u32 = 12;
19/// The caption over the reactions.
20const CAPTION: u32 = 30;
21const AVATAR: u32 = 26;
22/// A reactor's name, past their avatar.
23const NAME_X: u32 = INSET + AVATAR + 10;
24
25static INDICES: [usize; MAX_REACTORS] = {
26    let mut a = [0; MAX_REACTORS];
27    let mut i = 0;
28    while i < MAX_REACTORS {
29        a[i] = i;
30        i += 1;
31    }
32    a
33};

35pub fn view(details: &Details, inbound: bool) -> impl View<Rgb565, State> + use<> {
36    let details = details.clone();
37    VStack::new((
38        header("Message details", None, None),
39        super::list(
40            // One focusable block, which the wheel scrolls a row at a time.
41            Button::new(|_: &mut State| {}, move |_| rows(&details, inbound))
42                .focus_key(keys::DETAIL_LIST),
43        )
44        .with_scroll_step(ROW),
45    ))
46    .with_spacing(0)
47    .bound_focus(BoundaryBehavior::Stop)
48    .on_key(|key, s: &mut State| match key {
49        Key::Backspace => {
50            s.back();
51            KeyResponse::Handled
52        }
53        _ => KeyResponse::Ignored,
54    })
55}

57/// The rows for what is known.
58fn rows(details: &Details, inbound: bool) -> impl View<Rgb565, State> + use<> {
59    let text = |value: core::fmt::Arguments| {
60        let mut s: String<24> = String::new();
61        _ = core::fmt::write(&mut s, value);
62        s
63    };
64    // A true minus, as the mockup sets it.
65    let minus = |negative: bool| if negative { "\u{2212}" } else { "" };
66    VStack::new((
67        (!details.time.is_empty()).then(|| {
68            row(
69                if inbound { "Received" } else { "Sent" },
70                &text(format_args!("{}", details.time)),
71                false,
72            )
73        }),
74        details
75            .hops
76            .map(|hops| row("Hops", &text(format_args!("{hops}")), false)),
77        (!details.route.is_empty())
78            .then(|| row("Route", &text(format_args!("{}", details.route)), false)),
79        details.rssi.map(|rssi| {
80            let value = text(format_args!(
81                "{}{} dBm",
82                minus(rssi < 0),
83                rssi.unsigned_abs()
84            ));
85            row("Signal", &value, false)
86        }),
87        details.snr.map(|tenths| {
88            let (sign, tenths) = (minus(tenths < 0), tenths.unsigned_abs());
89            let value = text(format_args!("{sign}{}.{} dB", tenths / 10, tenths % 10));
90            row("Signal-to-noise", &value, false)
91        }),
92        inbound.then(|| {
93            let source = if details.authenticated {
94                "Authenticated"
95            } else {
96                "Not authenticated"
97            };
98            row("Source", source, false)
99        }),
100        (!details.reactors.is_empty()).then(|| {
101            label::<16>(
102                "Reactions",
103                &font::BOLD,
104                theme::SMALL,
105                theme::TEXT_SECONDARY,
106            )
107            .padding(Edges::Leading, INSET)
108            .flex_infinite_width(HorizontalAlignment::Leading)
109            .frame()
110            .with_height(CAPTION)
111            .with_alignment(Alignment::BottomLeading)
112        }),
113        {
114            let reactors = details.reactors.clone();
115            ForEach::<MAX_REACTORS>::new_vertical(&INDICES[..reactors.len()], move |&i| {
116                reactor(&reactors[i])
117            })
118            .with_spacing(0)
119        },
120    ))
121    .with_spacing(0)
122    // The whole block, so it scrolls from its first row's top.
123    .content_shape(Rectangle)
124}

126/// Someone's reaction: who, and the glyph for it.
127fn reactor(reactor: &Reactor) -> impl View<Rgb565, State> + use<> {
128    let name = |text| line(text, &font::REGULAR, theme::BODY, theme::TEXT);
129    HStack::new((
130        reactor.name.clone().map(name),
131        reactor
132            .name
133            .is_none()
134            .then(|| line("You", &font::REGULAR, theme::BODY, theme::TEXT)),
135        Spacer::default(),
136        Image::new(icon::reaction(reactor.reaction, icon::Pad::Screen)),
137    ))
138    .with_spacing(8)
139    .padding(Edges::Leading, NAME_X)
140    .padding(Edges::Trailing, 14)
141    .frame()
142    .with_height(ROW)
143    .overlay(
144        Alignment::Leading,
145        super::avatar::view(&reactor.avatar, AVATAR).padding(Edges::Leading, INSET),
146    )
147}

149/// A labeled value, with a rule under it. A `muted` label is for a value
150/// not known yet.
151pub(super) fn row(
152    name: &'static str,
153    value: &str,
154    muted: bool,
155) -> impl View<Rgb565, State> + use<> {
156    let color = if muted {
157        theme::TEXT_TERTIARY
158    } else {
159        theme::TEXT
160    };
161    HStack::new((
162        label::<16>(name, &font::REGULAR, theme::BODY, color),
163        Spacer::default(),
164        label::<48>(
165            value,
166            &font::REGULAR,
167            theme::SECONDARY,
168            theme::TEXT_SECONDARY,
169        ),
170    ))
171    .with_spacing(8)
172    .padding(Edges::Leading, INSET)
173    .padding(Edges::Trailing, 14)
174    .frame()
175    .with_height(ROW)
176    .overlay(
177        Alignment::Bottom,
178        Rectangle
179            .foreground_color(theme::LINE)
180            .frame()
181            .with_height(1)
182            .padding(Edges::Leading, INSET),
183    )
