1//! What a message wakes the panel to: the chats that got messages while it
2//! was dark, the first focused. Press or Enter opens one, typing starts a
3//! reply in it, and Backspace returns to where the panel went dark.
4
5use buoyant::event::{Key, KeyResponse};
6use buoyant::focus::BoundaryBehavior;
7use buoyant::view::prelude::*;
8use embedded_graphics::pixelcolor::Rgb565;
9use heapless::Vec;
10
11use super::label;
12use crate::model::{Age, Arrived, ChatSummary, MAX_CHATS, State};
13use crate::nav::keys;
14use crate::{font, theme};

16/// The heading's line, and the space between cards.
17const HEADING: u32 = 26;
18const GAP: u32 = 6;
19const RADIUS: u16 = 10;
20const AVATAR: u32 = 28;
21/// From a card's leading edge to its avatar and to its text.
22const AVATAR_X: u32 = 10;
23const TEXT_X: u32 = 46;
24/// Above the name, and the lines of the name and of the message.
25const TOP: u32 = 6;
26const NAME_LINE: u32 = 24;
27const MESSAGE_LINE: u32 = 20;
28
29/// Indices `ForEach` can hold without borrowing the state.
30static INDICES: [usize; MAX_CHATS] = {
31    let mut a = [0; MAX_CHATS];
32    let mut i = 0;
33    while i < MAX_CHATS {
34        a[i] = i;
35        i += 1;
36    }
37    a
38};
39
40pub fn view(state: &State) -> impl View<Rgb565, State> + use<> {
41    let cards: Vec<(Arrived, ChatSummary, Age), MAX_CHATS> = state
42        .arrived
43        .iter()
44        .filter_map(|arrived| {
45            let chat = state.chat(arrived.chat)?;
46            Some((arrived.clone(), chat.clone(), state.age(chat)))
47        })
48        .collect();
49    let ids: Vec<u32, MAX_CHATS> = cards.iter().map(|(a, ..)| a.chat).collect();
50    let count = cards.len();
51    VStack::new((
52        label::<32>(
53            "While the screen was off",
54            &font::BOLD,
55            theme::SMALL,
56            theme::TEXT_SECONDARY,
57        )
58        .padding(Edges::Leading, 10)
59        .flex_infinite_width(HorizontalAlignment::Leading)
60        .frame()
61        .with_height(HEADING),
62        super::list(
63            ForEach::<MAX_CHATS>::new_vertical(&INDICES[..count], move |&i| {
64                let (arrived, chat, age) = &cards[i];
65                card(arrived, chat, age)
66            })
67            .with_spacing(GAP)
68            .keyed(move |&i| keys::chat(ids[i])),
69        ),
70    ))
71    .with_spacing(0)
72    .bound_focus(BoundaryBehavior::Stop)
73    .on_key(|key, s: &mut State| {
74        let focused = s.focus().current().and_then(keys::chat_of);
75        match (key, focused) {
76            (Key::Backspace, _) => {
77                s.back();
78                KeyResponse::Handled
79            }
80            (Key::Enter, Some(chat)) => {
81                s.open_arrival(chat);
82                KeyResponse::Handled
83            }
84            // The chat opens with its composer focused, which takes the key.
85            (Key::Character(_), Some(chat)) => {
86                s.open_arrival(chat);
87                KeyResponse::Redirect
88            }
89            _ => KeyResponse::Ignored,
90        }
91    })
92}

94/// A chat, its age, and its newest message in up to two lines.
95fn card(arrived: &Arrived, chat: &ChatSummary, age: &Age) -> impl View<Rgb565, State> + use<> {
96    let text = arrived.text.clone();
97    let chat = chat.clone();
98    let age = age.clone();
99    let id = chat.id;
100    Button::new(
101        move |s: &mut State| s.open_arrival(id),
102        move |button: ButtonState| {
103            let focused = button.is_focused();
104            let (fill, ring, ring_width) = if focused {
105                (theme::FOCUS, theme::ACCENT, 2)
106            } else {
107                (theme::PANEL, theme::LINE, 1)
108            };
109            VStack::new((
110                HStack::new((
111                    label::<32>(&chat.name, &font::BOLD, theme::BODY, theme::TEXT),
112                    Spacer::default(),
113                    label::<6>(&age, &font::SEMIBOLD, theme::SMALL, theme::TEXT_TERTIARY),
114                ))
115                .with_spacing(8)
116                .frame()
117                .with_height(NAME_LINE),
118                Text::new(text.clone(), &font::REGULAR)
119                    .with_font_size(theme::SECONDARY)
120                    .with_line_limit(2)
121                    .foreground_color(theme::TEXT_SECONDARY)
122                    .padding(Edges::Trailing, 4)
123                    .flex_infinite_width(HorizontalAlignment::Leading),
124            ))
125            .with_spacing(0)
126            .with_alignment(HorizontalAlignment::Leading)
127            .padding(Edges::Top, TOP)
128            .padding(Edges::Bottom, MESSAGE_LINE - theme::SECONDARY)
129            .padding(Edges::Leading, TEXT_X)
130            .padding(Edges::Trailing, 10)
131            .overlay(
132                Alignment::TopLeading,
133                super::avatar::view(&chat.avatar, AVATAR)
134                    .padding(Edges::Leading, AVATAR_X)
135                    .padding(Edges::Top, 8),
136            )
137            .hint_background_color(fill)
138            .background(
139                Alignment::Center,
140                ZStack::new((
141                    RoundedRectangle::new(RADIUS).foreground_color(fill),
142                    RoundedRectangle::new(RADIUS)
143                        .stroked(ring_width)
144                        .foreground_color(ring),
145                )),
146            )
147        },
148    )
149    .padding(Edges::Horizontal, 6)
150}
