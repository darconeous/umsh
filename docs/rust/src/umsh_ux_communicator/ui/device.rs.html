18const ROW: u32 = 36;
19/// Shut down's confirmation.
20const CONFIRM_WIDTH: u32 = 220;

22/// The categories, beside the rail.
23pub fn view(state: &State) -> impl View<Rgb565, State> + use<> {
24    let channels = state
25        .chats
26        .iter()
27        .filter(|c| matches!(c.kind, ChatKind::Channel { .. }))
28        .count();
29    let confirm = matches!(
30        state.screen(),
31        Screen::Chats {
32            confirm_shut_down: true,
33            ..
34        }
35    );
36    let mut joined: String<24> = String::new();
37    _ = core::fmt::write(&mut joined, format_args!("{channels} joined"));
38    let category = |name, key, act: fn(&mut State), value: &str| {
39        let end = if value.is_empty() {
40            End::Chevron
41        } else {
42            End::value_chevron(value)
43        };
44        row(name, key, act, end, false, ROW)
45    };
46    super::list(
47        VStack::new((
48            category(
49                "Identity",
50                keys::IDENTITY,
51                State::open_share,
52                &state.status.device_name,
53            ),
54            category("Channels", keys::CHANNELS, State::open_channels, &joined),
55            category(
56                "Radio",
57                keys::RADIO,
58                State::open_radio,
59                state.profile_name(&state.node.radio()),
60            ),
61            category(
62                "Alerts",
63                keys::ALERTS,
64                |_| {},
65                if state.status.silent { "Silent" } else { "" },
66            ),
67            category("Display", keys::DISPLAY, |_| {}, ""),
68            category("Keyboard", keys::KEYBOARD, State::open_keyboard, ""),
69            category(
70                "Phone link",
71                keys::PHONE_LINK,
72                |_| {},
73                if state.status.bluetooth == Some(Bluetooth::Connected) {
74                    "Connected"
75                } else {
76                    ""
77                },
78            ),
79            category("Maintenance", keys::MAINTENANCE, |_| {}, ""),
80            row(
81                "Shut down",
82                keys::SHUT_DOWN,
83                State::ask_shut_down,
84                End::Nothing,
85                false,
86                ROW,
87            ),
88        ))
89        .with_spacing(0),
90    )
91    .with_scroll_step(ROW)
92    .bound_focus(BoundaryBehavior::Stop)
93    .on_key(|key, s: &mut State| match key {
94        Key::Backspace => {
95            s.focus_rail();
96            KeyResponse::Handled
97        }
98        _ => KeyResponse::Ignored,
99    })
100    .popover(confirm.then_some(()).as_ref(), |()| {
101        shut_down_confirmation()
102    })
103}

105/// Shut down, confirmed, or Cancel, which is focused.
106// Out of line, as `setting::row` is, against the Xtensa back end's failure.
107#[inline(never)]
108fn shut_down_confirmation() -> impl View<Rgb565, State> + use<> {
109    let text = |text: &str| String::try_from(text).unwrap_or_default();
110    card(
111        VStack::new((
112            choice(
113                text("Shut down"),
114                keys::SHUT_DOWN_CONFIRM,
115                State::shut_down,
116                theme::TEXT,
117                None,
118            ),
119            choice(
120                text("Cancel"),
121                keys::SHUT_DOWN_CANCEL,
122                State::cancel_shut_down,
123                theme::TEXT,
124                None,
125            ),
126        ))
127        .with_spacing(0),
128        CONFIRM_WIDTH,
129        HorizontalAlignment::Center,
130        State::cancel_shut_down,
131    )
132}

134/// A screen under Device.
135pub fn page(state: &State, page: &DevicePage) -> impl View<Rgb565, State> + use<> {
136    match_view!(page, {
137        DevicePage::Radio { settings } => radio(settings, state.profile_name(settings)),
138        DevicePage::Share => super::share::view(state),
139        DevicePage::Channels => super::channels::view(state),
140        DevicePage::AddChannel { text } => super::add_channel::view(state, text),
141        DevicePage::Profiles => super::profile::view(state, "Profile"),
142        DevicePage::Keyboard { settings } => super::typing::settings(settings),
143        DevicePage::TransmitPower { text } => super::typing::transmit_power(text),
144    })
145}

147/// The radio's settings, on the radio profile called `profile`.
148fn radio(settings: &RadioSettings, profile: &str) -> impl View<Rgb565, State> + use<> {
149    let mut airtime: String<24> = String::new();
150    _ = core::fmt::write(
151        &mut airtime,
152        format_args!("{}% this hour", settings.airtime),
153    );
154    let mut power: String<24> = String::new();
155    _ = core::fmt::write(&mut power, format_args!("{} dBm", settings.transmit_power));
156    VStack::new((
157        header("Radio", None, None),
158        super::list(
159            VStack::new((
160                row(
161                    "Profile",
162                    keys::PROFILE,
163                    State::open_profiles,
164                    End::value_chevron(profile),
165                    false,
166                    ROW,
167                ),
168                row(
169                    "Transmit power",
170                    keys::TRANSMIT_POWER,
171                    State::open_transmit_power,
172                    End::value_chevron(&power),
173                    false,
174                    ROW,
175                ),
176                row(
177                    "Forwarding",
178                    keys::FORWARDING,
179                    State::toggle_forwarding,
180                    End::Switch(settings.forwarding),
181                    false,
182                    ROW,
183                ),
184                row(
185                    "Share location",
186                    keys::SHARE_LOCATION,
187                    State::toggle_share_location,
188                    End::Switch(settings.share_location),
189                    false,
190                    ROW,
191                ),
192                row(
193                    "Airtime",
194                    keys::AIRTIME,
195                    |_| {},
196                    End::value(&airtime),
197                    false,
198                    ROW,
199                ),
200            ))
201            .with_spacing(0),
202        )
203        .with_scroll_step(ROW),
204    ))
205    .with_spacing(0)
206    .bound_focus(BoundaryBehavior::Stop)
207    .on_key(|key, s: &mut State| match key {
208        Key::Backspace => {
209            s.back();
210            KeyResponse::Handled
211        }
212        _ => KeyResponse::Ignored,
213    })
