1//! Add peer: one field takes a 44-character address or a `umsh:n:`
2//! invitation. The address wraps at a fixed column so its groups line up;
3//! the avatar appears from the first characters, and Message unlocks when
4//! the address is whole. Backspace goes back.
5

23/// The label above the field, and the field's place.
24const LABEL: u32 = 24;
25const INSET: u32 = 8;
26const RADIUS: u16 = 12;
27/// How many characters a line of the field holds: half an address.
28const COLUMN: u32 = 22;
29/// A character's width in the field's monospaced face, at body size.
30const ADVANCE: u32 = 10;
31const CARET: u32 = 3;
32/// The line under the field, its avatar, and the gap above it.
33const ENTRY: u32 = 44;
34const AVATAR: u32 = 34;
35const ROW: u32 = 36;
36
37pub fn view(state: &State, text: &PeerText) -> impl View<Rgb565, State> + use<> {
38    let entry = state.peer_entry();
39    let complete = matches!(entry, PeerEntry::Complete { .. });
40    VStack::new((
41        header("Add peer", None, None),
42        label::<24>(
43            "Address or invitation",
44            &font::BOLD,
45            theme::SMALL,
46            theme::TEXT_SECONDARY,
47        )
48        .padding(Edges::Leading, 10)
49        .flex_infinite_width(HorizontalAlignment::Leading)
50        .frame()
51        .with_height(LABEL),
52        field(text),
53        entry_line(&entry),
54        complete.then(|| {
55            action(
56                "Message",
57                keys::MESSAGE_PEER,
58                State::message_entry,
59                End::Chevron,
60            )
61        }),
62        (!complete).then(|| {
63            label::<8>("Message", &font::REGULAR, theme::BODY, theme::TEXT_TERTIARY)
64                .padding(Edges::Leading, 12)
65                .flex_infinite_width(HorizontalAlignment::Leading)
66                .frame()
67                .with_height(ROW)
68        }),
69    ))
70    .with_spacing(0)
71    .bound_focus(BoundaryBehavior::Stop)
72    .on_key(|key, s: &mut State| match key {
73        Key::Backspace => {
74            s.back();
75            KeyResponse::Handled
76        }
77        _ => KeyResponse::Ignored,
78    })
79}

81fn field(text: &PeerText) -> impl View<Rgb565, State> + use<> {
82    TextField::new(text, &font::MONO, State::peer_text_mut)
83        .with_font_size(theme::BODY)
84        .with_max_lines(3)
85        .with_caret_width(CARET)
86        // Enter messages the peer once the address is whole.
87        .on_submit(State::message_entry)
88        // Half an address to a line, so its groups line up.
89        .on_active(State::field_active)
90        .with_label(|status, area| field_look(status, area, COLUMN * ADVANCE + CARET))
91        .focus_key(keys::PEER_FIELD)
92        .padding(Edges::Horizontal, INSET)
93}

95/// A field typed in monospace, its text `width` wide at most.
96pub(super) fn field_look<B: EditText>(
97    status: TextFieldStatus,
98    area: TextArea<'static, B, PrerenderedFont>,
99    width: u32,
100) -> impl View<Rgb565, State> + use<B> {
101    let (fill, ring, stroke) = if status.is_active() {
102        (theme::ACTIVE_FIELD, theme::ACCENT, 2)
103    } else if status.is_focused() {
104        (theme::FIELD, theme::ACCENT, 1)
105    } else {
106        (theme::FIELD, theme::MENU_LINE, 1)
107    };
108    area.colors(AreaColors {
109        caret: Some(theme::ACCENT),
110        scroll_bar: None,
111    })
112    .foreground_color(theme::TEXT)
113    .hint_background_color(fill)
114    .frame()
115    .with_width(width)
116    .flex_infinite_width(HorizontalAlignment::Leading)
117    .padding(Edges::Horizontal, 12)
118    .padding(Edges::Vertical, 5)
119    .background(
120        Alignment::Center,
121        ZStack::new((
122            RoundedRectangle::new(RADIUS).foreground_color(fill),
123            RoundedRectangle::new(RADIUS)
124                .stroked(stroke)
125                .foreground_color(ring),
126        )),
127    )
128}

130/// Who the address belongs to as far as it goes, and what it still lacks.
131fn entry_line(entry: &PeerEntry) -> impl View<Rgb565, State> + use<> {
132    let (avatar, hint, note): (_, &str, Option<String<24>>) = match entry {
133        PeerEntry::Empty => (None, "", None),
134        PeerEntry::Invalid => (
135            None,
136            "",
137            Some(String::try_from("Not an address").unwrap_or_default()),
138        ),
139        PeerEntry::Partial {
140            avatar,
141            hint,
142            missing,
143        } => {
144            let mut note: String<24> = String::new();
145            let unit = if *missing == 1 {
146                "character"
147            } else {
148                "characters"
149            };
150            _ = core::fmt::write(&mut note, format_args!("{missing} more {unit}"));
151            (avatar.clone(), hint.as_str(), Some(note))
152        }
153        PeerEntry::Complete { avatar, hint, .. } => (Some(avatar.clone()), hint.as_str(), None),
154    };
155    HStack::new((
156        avatar.map(|avatar| super::avatar::view(&avatar, AVATAR)),
157        VStack::new((
158            (!hint.is_empty())
159                .then(|| label::<8>(hint, &font::MONO_BOLD, theme::BODY, theme::TEXT)),
160            note.map(|note| label::<24>(&note, &font::SEMIBOLD, theme::SMALL, theme::AMBER)),
161        ))
162        .with_spacing(0)
163        .with_alignment(HorizontalAlignment::Leading),
164    ))
165    .with_spacing(10)
166    .padding(Edges::Leading, 11)
167    .flex_infinite_width(HorizontalAlignment::Leading)
168    .frame()
169    .with_height(ENTRY)
170}
