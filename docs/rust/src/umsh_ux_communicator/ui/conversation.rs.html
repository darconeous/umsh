83pub fn view(state: &State, chat: ChatId) -> impl View<Rgb565, State> + use<> {
84    let summary = state.chat(chat);
85    let conversation = state.conversation(chat);
86    let messages = conversation.map_or(&[][..], |c| &c.messages);
87    let sent = conversation.map_or(0, |c| c.sent);
88    let (actions, reply, unsent) = match state.screen() {
89        Screen::Conversation {
90            actions,
91            reply,
92            unsent,
93            ..
94        } => (*actions, *reply, *unsent),
95        _ => (None, None, None),
96    };
97    // A message's actions are part of reading it.
98    let reading = actions.is_some()
99        || state
100            .screen()
101            .focus()
102            .current()
103            .is_some_and(keys::is_message);
104    let find = |id: Option<MessageId>| id.and_then(|id| messages.iter().find(|m| m.id == id));
105    // The menu offers my reaction as the one already chosen, and a
106    // member's message offers its sender.
107    let over = find(actions)
108        .map(|m| {
109            let join = m.invite.as_ref().map(|invite| {
110                let mut text = MenuText::new();
111                _ = core::fmt::write(&mut text, format_args!("Join {}", invite.name.as_str()));
112                text
113            });
114            Over::Actions(m.my_reaction(), m.sender.is_some(), join)
115        })
116        .or(unsent.map(|_| Over::Resend));
117    let quoted = find(reply).map(|message| quote(message, summary.map_or("", |s| &s.name)));
118    let kind = summary.map(|s| s.kind);
119    let channel = matches!(kind, Some(ChatKind::Channel { .. }));
120    let open = kind.is_some_and(ChatKind::is_open);
121    // Editing, the composer takes the color of what it will send: green
122    // where anyone can read it, violet under forward secrecy.
123    let tone = if open {
124        (theme::ACTIVE_FIELD_OPEN, theme::ACCENT_OPEN)
125    } else if summary.is_some_and(|s| s.forward_secret) {
126        (theme::ACTIVE_FIELD_SECRET, theme::ACCENT_SECRET)
127    } else {
128        (theme::ACTIVE_FIELD, theme::ACCENT)
129    };
130    let mut placeholder = Placeholder::new();
131    _ = core::fmt::write(
132        &mut placeholder,
133        format_args!("Message {}", summary.map_or("", |s| &s.name)),
134    );
135    VStack::new((
136        header(
137            summary.map_or("", |s| &s.name),
138            summary.map(|s| &s.avatar),
139            summary.map(|s| s.kind),
140        ),
141        history(
142            messages,
143            sent,
144            actions,
145            channel,
146            open,
147            Times {
148                now: state.wall_time,
149                offset_min: state.utc_offset_min(),
150                // A full window may have more behind it.
151                older: messages.len() == MAX_MESSAGES,
152            },
153        ),
154        state.airtime_wait.map(airtime_line),
155        quoted,
156        summary.map(|s| composer(chat, &s.draft, reading, tone, &placeholder)),
157    ))
158    .with_spacing(0)
159    .bound_focus(BoundaryBehavior::Stop)
160    .on_key(|key, s: &mut State| match key {
161        // A reply is dropped first, keeping its text; then Back.
162        Key::Backspace => {
163            if !s.drop_reply() {
164                s.back();
165            }
166            KeyResponse::Handled
167        }
168        Key::Character(_) | Key::Enter => {
169            s.focus().request(keys::COMPOSER);
170            KeyResponse::Redirect
171        }
172        _ => KeyResponse::Ignored,
173    })
174    // One popover for both: each wrapper around the conversation costs the
175    // Pager's stack a copy of its render tree.
176    .popover(over.as_ref(), |over| {
177        buoyant::match_view!(over, {
178            Over::Actions(mine, member, join) => actions_menu(*mine, *member, join.clone()),
179            Over::Resend => resend_choices(),
180        })
181    })
182}

229/// Back, the peer or channel, and how far away a peer is or that it is a
230/// channel; or a screen's title alone.
231pub(super) fn header(
232    name: &str,
233    avatar: Option<&AvatarStyle>,
234    kind: Option<ChatKind>,
235) -> impl View<Rgb565, State> + use<> {
236    let mut distance: heapless::String<16> = heapless::String::new();
237    // A channel sent in the clear says so, in red.
238    let clear = matches!(
239        kind,
240        Some(ChatKind::Channel {
241            access: ChannelAccess::Clear
242        })
243    );
244    match kind {
245        Some(ChatKind::Peer { hops: Some(hops) }) => _ = distance.push_str(&hops_text(hops)),
246        Some(ChatKind::Channel {
247            access: ChannelAccess::Clear,
248        }) => _ = distance.push_str("Not encrypted"),
249        Some(ChatKind::Channel { .. }) => _ = distance.push_str("Channel"),
250        Some(ChatKind::Peer { hops: None }) | None => {}
251    }
252    HStack::new((
253        Image::new(&icon::BACK)
254            .as_template()
255            .foreground_color(theme::TEXT_TERTIARY),
256        avatar.map(|style| super::avatar::view(style, AVATAR)),
257        label::<32>(
258            name,
259            &font::BOLD,
260            theme::BODY,
261            if clear { theme::RED } else { theme::TEXT },
262        )
263        .fixed_size(true, false),
264        label::<16>(&distance, &font::MEDIUM, theme::SMALL, theme::TEXT_TERTIARY),
265        Spacer::default(),
266    ))
267    .with_spacing(6)
268    .padding(Edges::Leading, 4)
269    .frame()
270    .with_height(HEADER)
271    .overlay(
272        Alignment::Bottom,
273        Rectangle
274            .foreground_color(theme::LINE)
275            .frame()
276            .with_height(1),
277    )
278}

280/// The messages, oldest first, ending at the composer. History shorter
281/// than the space sits on the composer. The message whose actions are open
282/// stays marked while focus is in them. A message sent brings the newest
283/// into view, however far back the history was read.
284fn history(
285    messages: &[Message],
286    sent: u32,
287    acting_on: Option<MessageId>,
288    channel: bool,
289    open: bool,
290    times: Times,
291) -> impl View<Rgb565, State> + use<> {
292    let messages = Messages::from_slice(messages).unwrap_or_default();
293    let ids: Vec<MessageId, MAX_MESSAGES> = messages.iter().map(|m| m.id).collect();
294    // Delivery shows under the newest message sent, the newest delivered,
295    // and every one unconfirmed.
296    let newest_sent = messages
297        .iter()
298        .rposition(|m| m.direction == Direction::Outbound);
299    let newest_delivered = messages
300        .iter()
301        .rposition(|m| m.direction == Direction::Outbound && m.delivery == Delivery::Delivered);
302    let count = messages.len();
303    ScrollView::new(
304        ForEach::<MAX_MESSAGES>::new_vertical(&INDICES[..count], move |&i| {
305            let delivery = Some(i) == newest_sent
306                || Some(i) == newest_delivered
307                || messages[i].delivery == Delivery::Unconfirmed;
308            item(&messages, i, delivery, acting_on, channel, open, times)
309        })
310        .keyed(move |&i| keys::message(ids[i])),
311    )
312    .with_bar_visibility(ScrollBarVisibility::Never)
313    .with_initial_pin_to_end(true)
314    .with_end_generation(sent)
315    .with_short_content_at_end(true)
316    .with_focus_margin(FOCUS_MARGIN)
317    .with_scroll_step(SCROLL_STEP)
318}

320/// A message, with its delivery below it when asked. In a channel, each run
321/// of messages from one member has the member's name above its first and
322/// their avatar beside its last.
323fn item(
324    messages: &Messages,
325    i: usize,
326    delivery: bool,
327    acting_on: Option<MessageId>,
328    channel: bool,
329    open: bool,
330    times: Times,
331) -> impl View<Rgb565, State> + use<> {
332    let message = &messages[i];
333    // A run of messages on their way again shows as the first of them.
334    let pending = message.presence == Presence::PendingRepair;
335    let collapsed = pending
336        && i.checked_sub(1)
337            .is_some_and(|p| messages[p].presence == Presence::PendingRepair);
338    let separator = (!collapsed)
339        .then(|| separator(messages, i, times))
340        .flatten();
341    let follows_same = separator.is_none()
342        && i.checked_sub(1).is_some_and(|p| {
343            messages[p].direction == message.direction && messages[p].sender == message.sender
344        });
345    let gap = if collapsed {
346        0
347    } else if i == 0 || follows_same {
348        GAP_SAME
349    } else {
350        GAP_CHANGE
351    };
352    // What stands in the place of a message that is not there.
353    let stand_in = if collapsed {
354        None
355    } else if pending {
356        Some(StandIn::Coming)
357    } else if message.presence == Presence::Unavailable {
358        Some(StandIn::Gone("Message unavailable"))
359    } else if message.deleted {
360        Some(StandIn::Gone("Message deleted"))
361    } else {
362        None
363    };
364    let shown = !collapsed && stand_in.is_none();
365    let outbound = message.direction == Direction::Outbound;
366    let alignment = if outbound {
367        HorizontalAlignment::Trailing
368    } else {
369        HorizontalAlignment::Leading
370    };
371    let sender = (!follows_same && !collapsed)
372        .then(|| message.sender.clone())
373        .flatten();
374    let ends_run = messages
375        .get(i + 1)
376        .is_none_or(|next| next.direction != message.direction || next.sender != message.sender);
377    // A channel's received bubbles stand clear of their senders' avatars.
378    let beside_avatar = channel && !outbound;
379    let avatar = message
380        .sender
381        .as_ref()
382        .filter(|_| beside_avatar && ends_run && !collapsed)
383        .map(|sender| {
384            Lazy::new(sender.avatar.clone(), member_avatar)
385                .padding(Edges::Leading, MEMBER_AVATAR_X)
386                .padding(Edges::Bottom, RING_GAP)
387        });
388    let lead = if beside_avatar { IN_X } else { SIDE };
389    // Chips over the bubble take room above it, which a sender's name
390    // beside them shares.
391    let reserve = chip_reserve(&message.reactions).saturating_sub(if sender.is_some() {
392        SENDER_LINE
393    } else {
394        0
395    });
396    let notes = match stand_in {
397        Some(stand_in) => Some(Notes::StandIn(stand_in)),
398        None if shown => {
399            let caption = caption(message, delivery);
400            let parts = message.parts;
401            (caption.is_some() || parts.is_some()).then_some(Notes::Below { caption, parts })
402        }
403        None => None,
404    };
405    let head = (separator.is_some() || sender.is_some()).then(|| Head {
406        stamp: separator,
407        sender: sender.map(|sender| sender.name),
408        beside_avatar,
409    });
410    VStack::new((
411        head.map(|head| Lazy::new(head, head_view)),
412        shown.then(|| {
413            bubble(message, acting_on == Some(message.id), open)
414                .padding(Edges::Leading, lead - SIDE)
415        }),
416        notes.map(|notes| Lazy::new((notes, outbound, beside_avatar), notes_view)),
417    ))
418    .with_spacing(0)
419    .with_alignment(alignment)
420    .flex_infinite_width(alignment)
421    // Received messages have no delivery line, so the bottom is the bubble's.
422    .overlay(Alignment::BottomLeading, avatar)
423    .padding(Edges::Top, gap + reserve)
424}

470/// A message's notes, set in from its side as its bubble is.
471///
472/// Built from its data each time it is drawn ([`Lazy`]), as a bubble's
473/// contents are.
474fn notes_view(
475    &(notes, outbound, beside_avatar): &(Notes, bool, bool),
476) -> impl View<Rgb565, ()> + use<> {
477    let lead = if beside_avatar { IN_X } else { SIDE };
478    let (stand_in, caption, parts) = match notes {
479        Notes::StandIn(stand_in) => (Some(stand_in), None, None),
480        Notes::Below { caption, parts } => (None, caption, parts),
481    };
482    VStack::new((
483        stand_in
484            .map(|stand_in| stand_in_view(stand_in, outbound).padding(Edges::Leading, lead - SIDE)),
485        caption.map(|caption| {
486            caption_line(caption, outbound)
487                .padding(Edges::Leading, lead + 4)
488                .padding(Edges::Trailing, SIDE)
489        }),
490        parts.map(|parts| part_dots(parts).padding(Edges::Leading, lead + 2)),
491    ))
492    .with_spacing(0)
493    .with_alignment(if outbound {
494        HorizontalAlignment::Trailing
495    } else {
496        HorizontalAlignment::Leading
497    })
498}

500/// A message in its bubble. Mine are in the color that says who can read
501/// them: green in an `open` channel, violet when sent under a session with
502/// forward secrecy.
503fn bubble(message: &Message, acting_on: bool, open: bool) -> impl View<Rgb565, State> + use<> {
504    let id = message.id;
505    let outbound = message.direction == Direction::Outbound;
506    let side = if outbound {
507        HorizontalAlignment::Trailing
508    } else {
509        HorizontalAlignment::Leading
510    };
511    let look = Look {
512        text: message.text.clone(),
513        reply: message.reply.clone(),
514        invite: message.invite.as_ref().map(|invite| Card {
515            name: invite.name.clone(),
516            builtin: invite.builtin,
517            tint: invite.tint,
518        }),
519        reactions: message.reactions.clone(),
520        to_come: message.parts.map(|parts| parts.lost != 0),
521        outbound,
522        open,
523        forward_secret: message.forward_secret,
524        ringed: acting_on,
525    };
526    // Pressing a message opens its actions; pressing an unconfirmed one
527    // sends it again.
528    let unconfirmed = message.delivery == Delivery::Unconfirmed;
529    Button::new(
530        move |s: &mut State| {
531            if unconfirmed {
532                s.retry(id);
533            } else {
534                s.open_actions(id);
535            }
536        },
537        move |button: ButtonState| {
538            let mut look = look.clone();
539            look.ringed |= button.is_focused();
540            // The room around the bubble, where its ring and chips stand,
541            // that the look does not hold.
542            let (corner, rest) = look.held();
543            let (outer, inner) = look.sides();
544            Lazy::new(look, bubble_look)
545                .padding(Edges::Top, RING_GAP - corner)
546                .padding(outer, RING_GAP - corner)
547                .padding(Edges::Bottom, RING_GAP - rest)
548                .padding(inner, RING_GAP - rest)
549                // Text wraps to fit; a short bubble keeps to its side.
550                .flex_frame()
551                .with_max_width(BUBBLE_MAX + 2 * RING_GAP)
552                .with_horizontal_alignment(side)
553        },
554    )
555    .padding(Edges::Horizontal, SIDE - RING_GAP)
556}

558/// What a bubble draws: its contents, its reactions, and whether it is
559/// ringed. Its colors follow from whose it is and who can read it.
560#[derive(Clone)]
561struct Look {
562    text: SharedText,
563    reply: Option<Quote>,
564    invite: Option<Card>,
565    reactions: Reactions,
566    /// Parts are still to come, and whether one will not.
567    to_come: Option<bool>,
568    outbound: bool,
569    /// In a channel anyone can read, where mine are green.
570    open: bool,
571    /// Mine, sent under a session with forward secrecy.
572    forward_secret: bool,
573    /// Focused, or its actions are open.
574    ringed: bool,
575}
576
577impl Look {
578    /// How much of the room around the bubble its look holds, which is
579    /// what focus keeps in view with it: what its ring takes while it is
580    /// ringed, out to the middle of the ring's line, and the corner its
581    /// chips hang from while it has reactions. As the room at its top and
582    /// outer side, then at its bottom and inner side.
583    fn held(&self) -> (u32, u32) {
584        let ring = if self.ringed {
585            RING_GAP - RING_LINE / 2
586        } else {
587            0
588        };
589        let chips = if self.reactions.is_empty() {
590            0
591        } else {
592            RING_GAP
593        };
594        (ring.max(chips), ring)
595    }
596
597    /// Its outer side, then its inner side: mine keep to the trailing side.
598    fn sides(&self) -> (Edges, Edges) {
599        if self.outbound {
600            (Edges::Leading, Edges::Trailing)
601        } else {
602            (Edges::Trailing, Edges::Leading)
603        }
604    }
605}

615/// A bubble, ringed while it is focused, with its reactions as chips over
616/// its outer top corner: both outside it, in the room around it, of which
617/// it holds what [`Look::held`] says.
618///
619/// Built from its data each time it is drawn ([`Lazy`]): kept in the
620/// render trees, every bubble's would outweigh the rest of the screen.
621fn bubble_look(look: &Look) -> impl View<Rgb565, ()> + use<> {
622    let gap = RING_GAP as i32;
623    let (corner, out) = if look.outbound {
624        (Alignment::TopLeading, -gap)
625    } else {
626        (Alignment::TopTrailing, gap)
627    };
628    let (held_corner, held_rest) = look.held();
629    let (outer, inner) = look.sides();
630    body_view(look)
631        .overlay(
632            Alignment::Center,
633            look.ringed.then(|| {
634                GeometryReader::new(move |size: Size| {
635                    RoundedRectangle::new(BUBBLE_RADIUS + RING_GAP as u16)
636                        .stroked(RING_LINE)
637                        .foreground_color(theme::ACCENT)
638                        .frame_sized(size.width + 2 * RING_GAP, size.height + 2 * RING_GAP)
639                        .offset(-gap, -gap)
640                })
641            }),
642        )

651        .padding(Edges::Bottom, held_rest)
652        .padding(inner, held_rest)
653}

820/// A reaction chip's diameter, the background ring that sets it off, and how
821/// far apart chips overlap.
822const CHIP: u32 = 28;
823const CHIP_HALO: u32 = 2;
824const CHIP_STEP: i32 = 21;
825/// How far a chip's center sits above the bubble's top edge.
826const CHIP_RISE: u32 = 6;
827/// The count on a chip shared by several people.
828const BADGE: u32 = 18;
829
830/// Room above a bubble for its chips, and for a count when there is one.
831fn chip_reserve(reactions: &[Reacted]) -> u32 {
832    if reactions.is_empty() {
833        return 0;
834    }
835    let counted = reactions.iter().any(|r| r.count > 1);
836    CHIP_RISE + CHIP / 2 + CHIP_HALO + if counted { 6 } else { 0 }
837}
838
839/// A bubble's reactions as round chips over its outer top corner, like a
840/// hand of cards: mine frontmost at the corner, with one dot trail falling
841/// outside the bubble. Placed from the corner outward and inward: offsets
842/// here are from the corner of the ring around the bubble.
843fn chips(reactions: &Reactions, outbound: bool, open: bool) -> impl View<Rgb565, ()> + use<> {
844    // Mine takes the color of my bubbles.
845    let (mine_fill, mine_pad) = if open {
846        (theme::OUTBOUND_OPEN, icon::Pad::MineOpen)
847    } else {
848        (theme::OUTBOUND, icon::Pad::Mine)
849    };
850    // Toward the outside of the bubble: left on a sent message.
851    let out: i32 = if outbound { -1 } else { 1 };
852    let ring = RING_GAP as i32;
853    // How far an element's outer edge reaches beyond the corner, as an
854    // offset from the side the cluster is aligned to.
855    let edge = move |beyond: i32| out * (beyond - ring);
856    let top = move |above_bubble: i32| ring + above_bubble;
857    let halo = (CHIP + 2 * CHIP_HALO) as i32;
858    // Mine at the corner, then others inward, drawn back to front.
859    let mut order: Vec<Reacted, { Reaction::ALL.len() }> = Vec::new();
860    for r in reactions.iter().filter(|r| r.mine) {
861        _ = order.push(*r);
862    }
863    for r in reactions.iter().filter(|r| !r.mine) {
864        _ = order.push(*r);
865    }
866    let count = order.len();
867    let front = order.first().map_or(theme::CHIP_THEIRS, |r| {
868        if r.mine {
869            mine_fill
870        } else {
871            theme::CHIP_THEIRS
872        }
873    });
874    // The `k`th chip drawn: the innermost first, mine at the corner last.
875    let chip = move |k: usize| {
876        let place = count.checked_sub(1 + k)?;
877        order.get(place).map(|r| {
878            let inward = CHIP_STEP * place as i32;
879            let (fill, pad) = if r.mine {
880                (mine_fill, mine_pad)
881            } else {
882                (theme::CHIP_THEIRS, icon::Pad::Theirs)
883            };
884            let number = (r.count > 1).then(|| {
885                let mut n: heapless::String<4> = heapless::String::new();
886                _ = core::fmt::write(&mut n, format_args!("{}", r.count));
887                ZStack::new((
888                    Circle.foreground_color(theme::BACKGROUND),
889                    Text::new(n, &font::BOLD)
890                        .with_font_size(theme::SMALL)
891                        .with_line_limit(1)
892                        .foreground_color(theme::TEXT),
893                ))
894                .frame_sized(BADGE, BADGE)
895                .offset(
896                    edge(BADGE as i32 / 2 + 11 - inward),
897                    top(-(CHIP_RISE as i32) - 11 - BADGE as i32 / 2),
898                )
899            });
900            ZStack::new((
901                ZStack::new((
902                    Circle.foreground_color(theme::BACKGROUND),
903                    Circle.foreground_color(fill).frame_sized(CHIP, CHIP),
904                    Image::new(icon::reaction(r.reaction, pad)),
905                ))
906                .frame_sized(halo as u32, halo as u32)
907                .offset(edge(halo / 2 - inward), top(-(CHIP_RISE as i32) - halo / 2)),
908                number,
909            ))
910            .with_alignment(if outbound {
911                Alignment::TopLeading
912            } else {
913                Alignment::TopTrailing
914            })
915        })
916    };
917    // The dot trail, from the corner chip outward and down.
918    let dot = move |distance: i32, radius: u32, color: Rgb565| {
919        let size = 2 * radius;
920        Circle
921            .foreground_color(color)
922            .frame_sized(size, size)
923            .offset(
924                edge(distance + radius as i32),
925                top(distance - CHIP_RISE as i32 - radius as i32),
926            )
927    };
928    ZStack::new((
929        dot(15, 6, theme::BACKGROUND),
930        dot(15, 4, front),
931        dot(21, 4, theme::BACKGROUND),
932        dot(21, 2, front),
933        chip(0),
934        chip(1),
935        chip(2),
936        chip(3),
937        chip(4),
938        chip(5),
939    ))
940    .with_alignment(if outbound {
941        Alignment::TopLeading
942    } else {
943        Alignment::TopTrailing
944    })
945}

1095/// When the conversation picked up again after a lull: the local day and
1096/// time of the message after it.
1097#[derive(Clone, Copy)]
1098struct Times {
1099    /// Now, in Unix seconds, when the clock is known.
1100    now: Option<u32>,
1101    offset_min: i16,
1102    /// Messages older than those shown may be kept.
1103    older: bool,
1104}
1105
1106/// How long a lull runs before the conversation says when it picked up
1107/// again: an hour, as on iOS.
1108const LULL: u32 = 60 * 60;
1109
1110/// The day and the time over the message `i`, when a lull went before it,
1111/// or when it is the first there is.
1112fn separator(messages: &Messages, i: usize, times: Times) -> Option<Stamp> {
1113    let time = messages[i].time?;
1114    let lull = match messages[..i].iter().rev().find_map(|m| m.time) {
1115        Some(before) => time.saturating_sub(before) >= LULL,
1116        None => !times.older,
1117    };
1118    if !lull {
1119        return None;
1120    }
1121    let local = |time: u32| i64::from(time) + i64::from(times.offset_min) * 60;
1122    Some(Stamp {
1123        day: local(time).div_euclid(86_400) as i32,
1124        minute: (local(time).rem_euclid(86_400) / 60) as u16,
1125        today: times.now.map(|now| local(now).div_euclid(86_400) as i32),
1126    })
1127}

1129/// When a lull ended, in local time: the day since 1970-01-01 and the
1130/// minute of that day, and today when the clock is known.
1131#[derive(Clone, Copy)]
1132struct Stamp {
1133    day: i32,
1134    minute: u16,
1135    today: Option<i32>,
1136}

1138/// A stamp's day: Today, Yesterday, a weekday within the week, or a date.
1139fn day_label(stamp: &Stamp) -> heapless::String<16> {
1140    let mut label = heapless::String::new();
1141    match stamp.today.map(|today| today - stamp.day) {
1142        Some(0) => _ = label.push_str("Today"),
1143        Some(1) => _ = label.push_str("Yesterday"),
1144        // Inside the last week a weekday places a message better than a
1145        // date does.
1146        Some(2..=6) => _ = label.push_str(WEEKDAYS[(stamp.day + 4).rem_euclid(7) as usize]),
1147        _ => {
1148            let (year, month, date) = civil(i64::from(stamp.day));
1149            _ = core::fmt::write(
1150                &mut label,
1151                format_args!("{} {date}", MONTHS[month as usize - 1]),
1152            );
1153            if stamp
1154                .today
1155                .map(|today| civil(i64::from(today)))
1156                .is_none_or(|(this_year, ..)| this_year != year)
1157            {
1158                _ = core::fmt::write(&mut label, format_args!(", {year}"));
1159            }
1160        }
1161    }
1162    label
1163}

1165const WEEKDAYS: [&str; 7] = [
1166    "Sunday",
1167    "Monday",
1168    "Tuesday",
1169    "Wednesday",
1170    "Thursday",
1171    "Friday",
1172    "Saturday",
1173];
1174
1175const MONTHS: [&str; 12] = [
1176    "Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec",
1177];

1179/// The year, month, and day of the month of `days` since 1970-01-01, in
1180/// the proleptic Gregorian calendar.
1181fn civil(days: i64) -> (i64, u32, u32) {
1182    let z = days + 719_468;
1183    let era = z.div_euclid(146_097);
1184    let doe = z.rem_euclid(146_097);
1185    let yoe = (doe - doe / 1_460 + doe / 36_524 - doe / 146_096) / 365;
1186    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
1187    let mp = (5 * doy + 2) / 153;
1188    let day = (doy - (153 * mp + 2) / 5 + 1) as u32;
1189    let month = if mp < 10 { mp + 3 } else { mp - 9 } as u32;
1190    let year = yoe + era * 400 + i64::from(month <= 2);
1191    (year, month, day)
1192}

1254) -> impl View<Rgb565, State> + use<> {

1259    .with_font_size(theme::BODY)
1260    .with_max_lines(3)
1261    .on_submit(move |s: &mut State| s.send(chat))
1262    .on_active(State::field_active)

1271fn field_look(
1272    status: TextFieldStatus,
1273    area: TextArea<'static, DraftText, PrerenderedFont>,
1274    (active_fill, active_ring): (Rgb565, Rgb565),
1275    placeholder: Option<&Placeholder>,
1276) -> impl View<Rgb565, State> + use<> {
1277    let (fill, line_width) = if status.is_active() {
1278        (active_fill, 2)
1279    } else {
1280        (theme::FIELD, 1)
1281    };
1282    let ring = if status.is_active() {
1283        active_ring
1284    } else if status.is_focused() {
1285        theme::ACCENT
1286    } else {
1287        theme::MENU_LINE
1288    };
1289    area.colors(AreaColors {
1290        caret: Some(active_ring),
1291        scroll_bar: Some((theme::MENU_LINE, theme::TEXT_SECONDARY)),
1292    })
1293    .foreground_color(theme::TEXT)
1294    .hint_background_color(fill)
1295    .overlay(
1296        Alignment::Leading,
1297        placeholder
1298            .filter(|_| !status.is_active())
1299            .map(|placeholder| {
1300                label::<48>(
1301                    placeholder,
1302                    &font::REGULAR,
1303                    theme::BODY,
1304                    theme::TEXT_TERTIARY,
1305                )
1306                .padding(Edges::Leading, 4)
1307            }),
1308    )
1309    .padding(Edges::Horizontal, 12)
1310    .padding(Edges::Vertical, 2)
1311    .background(
1312        Alignment::Center,
1313        ZStack::new((
1314            RoundedRectangle::new(FIELD_RADIUS).foreground_color(fill),
1315            RoundedRectangle::new(FIELD_RADIUS)
1316                .stroked(line_width)
1317                .foreground_color(ring),
1318        )),
1319    )
1320}

1366/// Who and what the draft replies to.
1367fn quote(message: &Message, chat_name: &str) -> impl View<Rgb565, State> + use<> {
1368    let mut title: heapless::String<48> = heapless::String::new();
1369    let whom = match (message.direction, &message.sender) {
1370        (Direction::Outbound, _) => "yourself",
1371        (Direction::Inbound, Some(sender)) => sender.name.as_str(),
1372        (Direction::Inbound, None) => chat_name,
1373    };
1374    _ = core::fmt::write(&mut title, format_args!("Replying to {whom}"));
1375    let text = message.text.clone();
1376    HStack::new((
1377        RoundedRectangle::new(2)
1378            .foreground_color(theme::ACCENT)
1379            .frame_sized(3, QUOTE - 6),
1380        VStack::new((
1381            label::<48>(&title, &font::BOLD, theme::SMALL, theme::ACCENT),
1382            super::line(
1383                text,
1384                &font::REGULAR,
1385                theme::SECONDARY,
1386                theme::TEXT_SECONDARY,
1387            ),
1388        ))
1389        .with_spacing(0)
1390        .with_alignment(HorizontalAlignment::Leading),
1391    ))
1392    .with_spacing(9)
1393    .padding(Edges::Leading, 12)
1394    .padding(Edges::Trailing, SIDE)
1395    .flex_infinite_width(HorizontalAlignment::Leading)
1396    .frame()
1397    .with_height(QUOTE)
1398}

1408/// What pressing a message gives: reactions first, then Reply and Details,
1409/// at the top of the screen beside the history. Backspace closes it, and
1410/// typing closes it and goes to the composer.
1411fn actions_menu(
1412    mine: Option<Reaction>,
1413    member: bool,
1414    join: Option<MenuText>,
1415) -> impl View<Rgb565, State> + use<> {
1416    let has_join = join.is_some();
1417    let pad = |reaction: Reaction| {
1418        Button::new(
1419            move |s: &mut State| s.react_from_actions(reaction),
1420            move |button: ButtonState| {
1421                let ring = if button.is_focused() {
1422                    Some(theme::ACCENT)
1423                } else if mine == Some(reaction) {
1424                    Some(theme::TEXT_TERTIARY)
1425                } else {
1426                    None
1427                };
1428                ZStack::new((
1429                    Circle.foreground_color(theme::REACTION_PAD),
1430                    Image::new(icon::reaction(reaction, icon::Pad::Palette)),
1431                    ring.map(|color| Circle.stroked(2).foreground_color(color)),
1432                ))
1433                .frame_sized(PAD, PAD)
1434            },
1435        )
1436        .focus_key(keys::reaction(reaction))
1437    };
1438    let [heart, up, down, laugh, exclaim, question] = Reaction::ALL;
1439    VStack::new((
1440        HStack::new((
1441            pad(heart),
1442            pad(up),
1443            pad(down),
1444            pad(laugh),
1445            pad(exclaim),
1446            pad(question),
1447        ))
1448        .with_spacing(3)
1449        .padding(Edges::Leading, 8)
1450        .flex_infinite_width(HorizontalAlignment::Leading)
1451        .frame()
1452        .with_height(PALETTE),
1453        Rectangle
1454            .foreground_color(theme::MENU_LINE)
1455            .frame()
1456            .with_height(1),
1457        // An invitation is for joining, first.
1458        join.map(|text| menu_item(text, keys::JOIN, State::join_from_actions))
1459            .padding(Edges::Top, 2),
1460        menu_item(menu_text("Reply"), keys::REPLY, State::reply_from_actions)
1461            .padding(Edges::Top, if has_join { 0 } else { 2 }),
1462        member.then(|| {
1463            menu_item(
1464                menu_text("View member"),
1465                keys::VIEW_MEMBER,
1466                State::view_member_from_actions,
1467            )
1468        }),
1469        menu_item(
1470            menu_text("Details"),
1471            keys::DETAILS,
1472            State::details_from_actions,
1473        ),
1474    ))
1475    .with_spacing(0)
1476    .padding(Edges::Bottom, 6)
1477    .frame()
1478    .with_width(MENU_WIDTH)
1479    .background(
1480        Alignment::Center,
1481        ZStack::new((
1482            RoundedRectangle::new(12)
1483                .foreground_color(Rgb565::BLACK)
1484                .offset(1, 3),
1485            RoundedRectangle::new(12).foreground_color(theme::MENU),
1486            RoundedRectangle::new(12)
1487                .stroked(1)
1488                .foreground_color(theme::MENU_LINE),
1489        )),
1490    )
1491    .padding(Edges::Top, HEADER + 6)
1492    .padding(Edges::Trailing, SIDE)
1493    .flex_infinite_width(HorizontalAlignment::Trailing)
1494    .flex_infinite_height(VerticalAlignment::Top)
1495    .on_key(|key, s: &mut State| match key {
1496        Key::Backspace => {
1497            s.close_actions();
1498            KeyResponse::Handled
1499        }
1500        Key::Character(_) | Key::Enter => {
1501            s.close_actions();
1502            s.focus().request(keys::COMPOSER);
1503            KeyResponse::Redirect
1504        }
1505        _ => KeyResponse::Ignored,
1506    })
1507}
