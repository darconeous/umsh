1//! An avatar drawn from its [`AvatarStyle`]: the tinted outline with the
2//! lettering centered over it.
3
4use buoyant::font::PrerenderedFont;
5use buoyant::view::prelude::*;
6use embedded_graphics::pixelcolor::Rgb565;
7
8use crate::avatar::{AvatarStyle, Outline, PeerShape};
9use crate::font;
10
11/// `style` at `size` pixels across.
12pub fn view<C>(style: &AvatarStyle, size: u32) -> impl View<Rgb565, C> + use<C> {
13    // A peer's hint is set in the monospaced face, two over two. A
14    // triangle's lettering is smaller and its lines closer, as on iOS, and
15    // centered on the triangle's incenter, a third of its height above its
16    // base: the triangle stands 1.16 times the avatar's size, its rounded
17    // apex above the frame, so that is 0.113 of the size below the middle.
18    let triangle = matches!(style.outline, Outline::Peer(PeerShape::Triangle));
19    let (face, text_size, below) = match style.outline {
20        Outline::Peer(PeerShape::Triangle) => {
21            (&font::MONO_BOLD, size * 27 / 80, (size * 17 + 75) / 150)
22        }
23        Outline::Peer(_) => (&font::MONO_BOLD, size * 3 / 8, 0),
24        Outline::Channel if style.lines[1].is_empty() && style.lines[0].chars().count() == 1 => {
25            (&font::BOLD, size * 3 / 4, 0)
26        }
27        Outline::Channel => (&font::BOLD, size * 2 / 5, 0),
28    };
29    let shown = face.face(text_size);
30    let pitch = if triangle {
31        i32::from(shown.size) * 15 / 16
32    } else {
33        i32::from(shown.ascent) - i32::from(shown.descent) + i32::from(shown.line_gap)
34    };
35    let two = !style.lines[1].is_empty();
36    let line = |s: &heapless::String<2>, offset: i32| {
37        Text::new(s.clone(), face)
38            .with_font_size(text_size)
39            .foreground_color(style.text)
40            .offset(0, offset)
41    };
42    ZStack::new((
43        style
44            .outline
45            .mask(size)
46            .map(|mask| Image::new(mask).as_template().foreground_color(style.fill)),
47        ZStack::new((
48            line(&style.lines[0], if two { -(pitch / 2) } else { 0 }),
49            two.then(|| line(&style.lines[1], pitch - pitch / 2)),
50        ))
51        .offset(0, below as i32 + cap_drop(face, text_size))
52        .hint_background_color(style.fill),
53    ))
54    .frame()
55    .with_height(size)
56    .with_width(size)
57}

59/// How far lettering in `face` at `size` drops to center its capitals on
60/// its line. A line holds the face's tallest ascent above the baseline, and
61/// its descent and the gap to the next line below it, so capitals centered
62/// as a line is sit high by half of what is under them less what is over.
63fn cap_drop(face: &PrerenderedFont, size: u32) -> i32 {
64    let face = face.face(size);
65    let over = face.glyph('H').map_or(0, |cap| i32::from(cap.top));
66    let under = -i32::from(face.descent) + i32::from(face.line_gap);
67    (under - over + 1) / 2
68}
