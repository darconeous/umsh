1//! An avatar drawn from its [`AvatarStyle`]: the tinted outline with the
2//! lettering centered over it.
3
4use buoyant::view::prelude::*;
5use embedded_graphics::pixelcolor::Rgb565;
6
7use crate::avatar::{AvatarStyle, Outline};
8use crate::font;
9use crate::model::State;
10
11/// `style` at `size` pixels across.
12pub fn view(style: &AvatarStyle, size: u32) -> impl View<Rgb565, State> + use<> {
13    let (face, text_size) = match style.outline {
14        // A peer's hint is set in the monospaced face, two over two.
15        Outline::Peer(_) => (&font::MONO_BOLD, size * 3 / 8),
16        Outline::Channel if style.lines[1].is_empty() && style.lines[0].chars().count() == 1 => {
17            (&font::BOLD, size * 3 / 4)
18        }
19        Outline::Channel => (&font::BOLD, size * 2 / 5),
20    };
21    let line = |s: &heapless::String<2>| {
22        Text::new(s.clone(), face)
23            .with_font_size(text_size)
24            .foreground_color(style.text)
25    };
26    let second = (!style.lines[1].is_empty()).then(|| line(&style.lines[1]));
27    ZStack::new((
28        style
29            .outline
30            .mask(size)
31            .map(|mask| Image::new(mask).as_template().foreground_color(style.fill)),
32        VStack::new((line(&style.lines[0]), second))
33            .with_spacing(0)
34            .hint_background_color(style.fill),
35    ))
36    .frame()
37    .with_height(size)
38    .with_width(size)
39}
