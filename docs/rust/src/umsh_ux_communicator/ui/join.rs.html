1//! A channel an invitation offers, before it is joined. Nothing is joined
2//! or sent until Join channel. Name is what the channel is called here;
3//! pressing it edits it. Backspace goes back.

22const WARNING: u32 = 28;
23const ROW: u32 = 36;
24const INSET: u32 = 12;
25const RADIUS: u16 = 8;

27pub fn view(invite: &Invite, name: &JoinName) -> impl View<Rgb565, State> + use<> {
28    let (kind, warning) = if invite.access.is_open() {
29        ("Public channel", "Anyone who knows the name can read it.")
30    } else {
31        (
32            "Private channel",
33            "Anyone with this invite can read and post.",
34        )
35    };
36    VStack::new((
37        header("Join channel", None, None),
38        detail_head(&invite.avatar(), &invite.name, kind),
39        HStack::new((
40            Image::new(&icon::WARNING)
41                .as_template()
42                .foreground_color(theme::AMBER),
43            label::<48>(warning, &font::MEDIUM, theme::SECONDARY, theme::AMBER),
44        ))
45        .with_spacing(6)
46        .padding(Edges::Leading, 9)
47        .flex_infinite_width(HorizontalAlignment::Leading)
48        .frame()
49        .with_height(WARNING),
50        name_row(name),
51        join_button(),
52    ))
53    .with_spacing(0)
54    .bound_focus(BoundaryBehavior::Stop)
55    .on_key(|key, s: &mut State| match key {
56        Key::Backspace => {
57            s.back();
58            KeyResponse::Handled
59        }
60        _ => KeyResponse::Ignored,
61    })
62}

64/// What the channel is to be called here, edited in place.
65fn name_row(name: &JoinName) -> impl View<Rgb565, State> + use<> {
66    let shown: String<32> = String::try_from(name.text()).unwrap_or_default();
67    TextField::new(name, &font::REGULAR, State::join_name_mut)
68        .with_font_size(theme::SECONDARY)
69        .with_max_lines(1)
70        // Enter ends the edit.
71        .on_submit(|_: &mut State| {})
72        .on_active(State::field_active)
73        .with_label(move |status, area| name_look(status, area, &shown))
74        .focus_key(keys::JOIN_NAME)
75}

77fn name_look(
78    status: TextFieldStatus,
79    area: TextArea<'static, String<32>, PrerenderedFont>,
80    shown: &str,
81) -> impl View<Rgb565, State> + use<> {
82    let editing = status.is_active();
83    let focused = status.is_focused() || editing;
84    let face = if focused { &font::BOLD } else { &font::REGULAR };
85    let fill = if focused {
86        theme::FOCUS
87    } else {
88        theme::BACKGROUND
89    };
90    HStack::new((
91        label::<8>("Name", face, theme::BODY, theme::TEXT),
92        // Edited where it stands; otherwise set at the end, as a value is.
93        editing.then(|| {
94            area.colors(AreaColors {
95                caret: Some(theme::ACCENT),
96                scroll_bar: None,
97            })
98            .foreground_color(theme::TEXT)
99        }),
100        (!editing).then(|| {
101            HStack::new((
102                Spacer::default(),
103                label::<32>(
104                    shown,
105                    &font::REGULAR,
106                    theme::SECONDARY,
107                    theme::TEXT_SECONDARY,
108                ),
109                Image::new(&icon::CHEVRON)
110                    .as_template()
111                    .foreground_color(theme::TEXT_TERTIARY),
112            ))
113            .with_spacing(8)
114        }),
115    ))
116    .with_spacing(8)
117    .hint_background_color(fill)
118    .padding(Edges::Leading, INSET - 4)
119    .padding(Edges::Trailing, 10)
120    .frame()
121    .with_height(ROW - 2)
122    .background(Alignment::Center, ring(focused))
123    .padding(Edges::Horizontal, 4)
124    .padding(Edges::Vertical, 1)
125}

127/// Joins, and opens the channel.
128fn join_button() -> impl View<Rgb565, State> + use<> {
129    Button::new(State::join, |button: ButtonState| {
130        let focused = button.is_focused();
131        label::<16>("Join channel", &font::BOLD, theme::BODY, theme::ACCENT)
132            .hint_background_color(if focused {
133                theme::FOCUS
134            } else {
135                theme::BACKGROUND
136            })
137            .flex_infinite_width(HorizontalAlignment::Center)
138            .frame()
139            .with_height(ROW - 2)
140            .background(Alignment::Center, ring(focused))
141            .padding(Edges::Horizontal, 4)
142            .padding(Edges::Vertical, 1)
143    })
144    .focus_key(keys::JOIN_CHANNEL)
145}

147/// The focused look of a row.
148fn ring(focused: bool) -> Option<impl View<Rgb565, State> + use<>> {
149    focused.then(|| {
150        ZStack::new((
151            RoundedRectangle::new(RADIUS).foreground_color(theme::FOCUS),
152            RoundedRectangle::new(RADIUS)
153                .stroked(2)
154                .foreground_color(theme::ACCENT),
155        ))
156    })
