18/// Indices `ForEach` can hold without borrowing the state.
19static INDICES: [usize; MAX_CHATS] = {
20    let mut a = [0; MAX_CHATS];
21    let mut i = 0;
22    while i < MAX_CHATS {
23        a[i] = i;
24        i += 1;
25    }
26    a
27};

29/// What a channel's row shows.
30#[derive(Clone)]
31struct Channel {
32    id: ChatId,
33    avatar: AvatarStyle,
34    name: String<32>,
35    about: &'static str,
36}
37
38pub fn view(state: &State) -> impl View<Rgb565, State> + use<> {
39    let rows: Vec<Channel, MAX_CHATS> = state
40        .chats
41        .iter()
42        .filter_map(|c| match c.kind {
43            ChatKind::Channel { access } => Some(Channel {
44                id: c.id,
45                avatar: c.avatar.clone(),
46                name: c.name.clone(),
47                about: match access {
48                    ChannelAccess::Private => "Private",
49                    ChannelAccess::Named => "Named",
50                    ChannelAccess::Clear => "Not encrypted",
51                },
52            }),
53            ChatKind::Peer { .. } => None,
54        })
55        .collect();
56    let ids: Vec<ChatId, MAX_CHATS> = rows.iter().map(|c| c.id).collect();
57    let count = rows.len();
58    VStack::new((
59        header("Channels", None, None),
60        super::list(
61            VStack::new((
62                add_row(
63                    "Add channel",
64                    State::open_add_channel,
65                    keys::ADD_CHANNEL,
66                    theme::WIDTH,
67                ),
68                ForEach::<MAX_CHATS>::new_vertical(&INDICES[..count], move |&i| row(&rows[i]))
69                    .keyed(move |&i| keys::chat(ids[i])),
70            ))
71            .with_spacing(0),
72        ),
73    ))
74    .with_spacing(0)
75    .bound_focus(BoundaryBehavior::Stop)
76    .on_key(|key, s: &mut State| match key {
77        Key::Backspace => {
78            s.back();
79            KeyResponse::Handled
80        }
81        _ => KeyResponse::Ignored,
82    })
83}
84
85fn row(channel: &Channel) -> impl View<Rgb565, State> + use<> {
86    let channel = channel.clone();
87    let id = channel.id;
88    Button::new(
89        move |s: &mut State| s.open_channel(id),
90        move |button: ButtonState| {
91            focused_row(
92                HStack::new((
93                    label::<32>(&channel.name, &font::SEMIBOLD, theme::BODY, theme::TEXT),
94                    Spacer::default(),
95                    super::line(
96                        channel.about,
97                        &font::REGULAR,
98                        theme::SECONDARY,
99                        theme::TEXT_TERTIARY,
100                    ),
101                ))
102                .with_spacing(8)
103                .padding(Edges::Leading, TEXT_X)
104                .padding(Edges::Trailing, 10)
105                .frame()
106                .with_height(ROW)
107                .overlay(
108                    Alignment::Leading,
109                    super::avatar::view(&channel.avatar, AVATAR)
110                        .frame()
111                        .with_width(TEXT_X),
112                ),
113                button.is_focused(),
114                theme::WIDTH,
115            )
116        },
117    )
118}
