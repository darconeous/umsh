13pub fn view(active: Section, unread_chats: usize) -> impl View<Rgb565, State> + use<> {
14    VStack::new((
15        item(Section::Chats, "Chats", &icon::CHATS, active, unread_chats),
16        item(Section::Peers, "Peers", &icon::PEERS, active, 0),
17        item(Section::Device, "Device", &icon::DEVICE, active, 0),
18    ))
19    .with_spacing(0)
20    .with_alignment(HorizontalAlignment::Leading)
21    .bound_focus(BoundaryBehavior::Stop)
22    .frame()
23    .with_width(RAIL)
24    .with_height(theme::HEIGHT - STRIP)
25    .with_alignment(Alignment::Leading)
26    .overlay(
27        Alignment::Trailing,
28        Rectangle
29            .foreground_color(theme::LINE)
30            .frame()
31            .with_width(1),
32    )
33    .background_color(theme::CHROME, Rectangle)
34}

36/// One section's item. Pressing it enters the section's list.
37fn item(
38    section: Section,
39    name: &'static str,
40    glyph: &'static ImageRaw<'static, Gray4>,
41    shown: Section,
42    badge: usize,
43) -> impl View<Rgb565, State> + use<> {
44    Button::new(
45        move |s: &mut State| s.enter_section(section),
46        move |button: ButtonState| {
47            item_look(name, glyph, section == shown, button.is_focused(), badge)
48        },
49    )
50    .focus_key(keys::rail(section))
51}
52
53fn item_look(
54    name: &str,
55    glyph: &'static ImageRaw<'static, Gray4>,
56    active: bool,
57    focused: bool,
58    badge: usize,
59) -> impl View<Rgb565, State> + use<> {
60    let color = if active {
61        theme::ACCENT
62    } else {
63        theme::TEXT_TERTIARY
64    };
65    let face = if active { &font::BOLD } else { &font::MEDIUM };
66    let cell = (theme::HEIGHT - STRIP) / 3;
67    VStack::new((
68        Image::new(glyph)
69            .as_template()
70            .foreground_color(color)
71            .overlay(
72                Alignment::TopTrailing,
73                (badge > 0)
74                    .then(|| badge_view(badge))
75                    .map(|b| b.offset(7, -5)),
76            ),
77        label::<8>(name, face, theme::SMALL, color),
78    ))
79    .with_spacing(2)
80    .frame()
81    .with_width(RAIL - 9)
82    .with_height(cell - 8)
83    // Focused, the item is boxed in place of its bar.
84    .background(
85        Alignment::Center,
86        focused.then(|| {
87            ZStack::new((
88                RoundedRectangle::new(7).foreground_color(theme::FOCUS),
89                RoundedRectangle::new(7)
90                    .stroked(2)
91                    .foreground_color(theme::ACCENT),
92            ))
93        }),
94    )
95    .frame()
96    .with_width(RAIL - 1)
97    .with_height(cell)
98    .overlay(
99        Alignment::Leading,
100        (active && !focused).then(|| {
101            RoundedRectangle::new(2)
102                .foreground_color(theme::ACCENT)
103                .frame_sized(4, cell - 20)
104        }),
105    )
106}

108/// The count of chats with unread messages, on the Chats glyph.
109fn badge_view(count: usize) -> impl View<Rgb565, State> + use<> {
110    let mut digits: heapless::String<4> = heapless::String::new();
111    let _ = core::fmt::write(&mut digits, format_args!("{count}"));
112    ZStack::new((
113        Circle.foreground_color(theme::RED).frame_sized(19, 19),
114        Text::new(digits, &font::BOLD)
115            .with_font_size(theme::SMALL)
116            .foreground_color(Rgb565::new(31, 63, 31))
117            .hint_background_color(theme::RED),
118    ))
119}
