1//! The status strip: device name, time, and the few indicators worth a
2//! permanent place.
3
4use buoyant::view::prelude::*;
5use embedded_graphics::pixelcolor::Rgb565;
6
7use super::{STRIP, label};
8use crate::model::{State, StatusStrip};
9use crate::{font, icon, theme};
10
11pub fn view(s: &StatusStrip) -> impl View<Rgb565, State> + use<> {
12    ZStack::new((
13        // One stack, not a stack of indicators inside it: see `root_view`.
14        HStack::new((
15            label::<24>(
16                &s.device_name,
17                &font::SEMIBOLD,
18                theme::SMALL,
19                theme::TEXT_SECONDARY,
20            ),
21            Spacer::default(),
22            s.silent.then(|| indicator(&icon::BELL_SLASH)),
23            s.phone_linked.then(|| indicator(&icon::BLUETOOTH)),
24            s.gnss.then(|| indicator(&icon::GNSS)),
25            battery(s.battery_segments),
26        ))
27        .with_spacing(3)
28        .padding(Edges::Horizontal, 8),
29        label::<8>(&s.time, &font::BOLD, theme::SMALL, theme::TEXT),
30    ))
31    .frame()
32    .with_height(STRIP)
33    .background_color(theme::CHROME, Rectangle)
34}
35
36fn indicator(
37    image: &'static embedded_graphics::image::ImageRaw<
38        'static,
39        embedded_graphics::pixelcolor::Gray4,
40    >,
41) -> impl View<Rgb565, State> + use<> {
42    Image::new(image)
43        .as_template()
44        .foreground_color(theme::TEXT_SECONDARY)
45}
46
47/// A battery outline with up to three charge segments, and its terminal.
48fn battery(segments: u8) -> impl View<Rgb565, State> + use<> {
49    let segment = move |i: u8| {
50        Rectangle
51            .foreground_color(if segments > i {
52                theme::TEXT
53            } else {
54                theme::CHROME
55            })
56            .frame_sized(5, 7)
57            .offset(4 + 6 * i32::from(i), 0)
58    };
59    ZStack::new((
60        RoundedRectangle::new(3)
61            .stroked(1)
62            .foreground_color(theme::TEXT_SECONDARY)
63            .frame_sized(25, 12),
64        segment(0),
65        segment(1),
66        segment(2),
67        Rectangle
68            .foreground_color(theme::TEXT_SECONDARY)
69            .frame_sized(2, 5)
70            .offset(26, 0),
71    ))
72    .with_alignment(Alignment::Leading)
73    .frame_sized(28, 12)
74}
