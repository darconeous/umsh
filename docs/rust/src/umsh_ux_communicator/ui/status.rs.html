12/// The strip; on first run, the battery alone, there being no name or
13/// time yet to show. Shift's and Alt's states come first among the
14/// indicators, in the keyboard's order.
15pub fn view(
16    s: &StatusStrip,
17    setup: bool,
18    (shift, alt): (Latch, Latch),
19) -> impl View<Rgb565, State> + use<> {
20    let shown = !setup;
21    ZStack::new((
22        // One stack, not a stack of indicators inside it: see `root_view`.
23        HStack::new((
24            shown.then(|| {
25                label::<24>(
26                    &s.device_name,
27                    &font::SEMIBOLD,
28                    theme::SMALL,
29                    theme::TEXT_SECONDARY,
30                )
31            }),
32            Spacer::default(),
33            modifier(alt, &icon::ALT_ONCE, &icon::ALT_LOCK, theme::ALT),
34            modifier(shift, &icon::SHIFT_ONCE, &icon::SHIFT_LOCK, theme::TEXT),
35            (shown && s.silent).then(|| indicator(&icon::BELL_SLASH)),
36            s.wifi.filter(|_| shown).map(|wifi| {
37                indicator(match wifi {
38                    Wifi::Disconnected => &icon::WIFI_LOST,
39                    Wifi::Joining => &icon::WIFI_JOINING,
40                    Wifi::Connected => &icon::WIFI,
41                })
42            }),
43            s.bluetooth.filter(|_| shown).map(bluetooth),
44            s.gnss.filter(|_| shown).map(|gnss| {
45                indicator(match gnss {
46                    Gnss::Searching => &icon::GNSS_SEARCHING,
47                    Gnss::Fixed => &icon::GNSS,
48                })
49            }),
50            s.charging.then(|| indicator(&icon::BOLT)),
51            battery(s.battery_segments, s.battery_low),
52        ))
53        .with_spacing(3)
54        .padding(Edges::Horizontal, 8),
55        shown.then(|| label::<8>(&s.time, &font::BOLD, theme::SMALL, theme::TEXT)),
56    ))
57    .frame()
58    .with_height(STRIP)
59    .background_color(theme::CHROME, Rectangle)
60}

62type Mask = embedded_graphics::image::ImageRaw<'static, embedded_graphics::pixelcolor::Gray4>;
63
64/// A modifier armed for one key, or locked.
65fn modifier(
66    latch: Latch,
67    once: &'static Mask,
68    locked: &'static Mask,
69    color: Rgb565,
70) -> Option<impl View<Rgb565, State> + use<>> {
71    let image = match latch {
72        Latch::Off => return None,
73        Latch::Once => once,
74        Latch::Locked => locked,
75    };
76    Some(Image::new(image).as_template().foreground_color(color))
77}
78
79fn indicator(image: &'static Mask) -> impl View<Rgb565, State> + use<> {
80    Image::new(image)
81        .as_template()
82        .foreground_color(theme::TEXT_SECONDARY)
83}
84
85/// The rune, flanked while a phone is connected, and in the accent color
86/// while a pairing window is open.
87fn bluetooth(state: Bluetooth) -> impl View<Rgb565, State> + use<> {
88    let (image, color) = match state {
89        Bluetooth::Waiting => (&icon::BLUETOOTH, theme::TEXT_SECONDARY),
90        Bluetooth::Connected => (&icon::BLUETOOTH_LINKED, theme::TEXT_SECONDARY),
91        Bluetooth::Pairing => (&icon::BLUETOOTH, theme::ACCENT),
92    };
93    Image::new(image).as_template().foreground_color(color)
94}

96/// A battery outline with up to three charge segments, and its terminal,
97/// amber while the charge is `low`.
98fn battery(segments: u8, low: bool) -> impl View<Rgb565, State> + use<> {
99    let outline = if low {
100        theme::AMBER
101    } else {
102        theme::TEXT_SECONDARY
103    };
104    let segment = move |i: u8| {
105        Rectangle
106            .foreground_color(if segments > i {
107                theme::TEXT
108            } else {
109                theme::CHROME
110            })
111            .frame_sized(5, 7)
112            .offset(4 + 6 * i32::from(i), 0)
113    };
114    ZStack::new((
115        RoundedRectangle::new(3)
116            .stroked(1)
117            .foreground_color(outline)
118            .frame_sized(25, 12),
119        segment(0),
120        segment(1),
121        segment(2),
122        Rectangle
123            .foreground_color(outline)
124            .frame_sized(2, 5)
125            .offset(26, 0),
126    ))
127    .with_alignment(Alignment::Leading)
128    .frame_sized(28, 12)
129}
