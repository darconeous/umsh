1//! A saved peer: who they are, their whole address, then one focusable list
2//! of actions and readouts that the wheel scrolls through. Forget is last
3//! and asks to confirm. Backspace goes back.
4
5use buoyant::event::{Key, KeyResponse};
6use buoyant::focus::BoundaryBehavior;
7use buoyant::view::prelude::*;
8use embedded_graphics::pixelcolor::Rgb565;
9use heapless::String;
10
11use super::label;
12use super::member::detail_head;
13use super::setting::{End, card, choice, row};
14use crate::model::{PeerDetail, PeerSummary, State, hops_text};
15use crate::nav::keys;
16use crate::{font, theme};
17
18const ROW: u32 = 34;
19const ADDRESS: u32 = 22;
20/// The confirmation's width.
21const CONFIRM_WIDTH: u32 = 300;
22
23pub fn view(
24    summary: &PeerSummary,
25    detail: &PeerDetail,
26    confirm: bool,
27    requested: bool,
28) -> impl View<Rgb565, State> + use<> {
29    let mut about: String<48> = String::new();
30    _ = about.push_str("Saved peer");
31    if let Some(hops) = summary.hops {
32        _ = about.push_str(" \u{B7} ");
33        _ = about.push_str(&hops_text(hops));
34    }
35    let name = summary.name.clone();
36    super::list(
37        VStack::new((
38            detail_head(&summary.avatar, &summary.name, &about),
39            label::<44>(
40                &detail.address,
41                &font::REGULAR,
42                theme::SMALL,
43                theme::TEXT_SECONDARY,
44            )
45            .padding(Edges::Leading, 10)
46            .flex_infinite_width(HorizontalAlignment::Leading)
47            .frame()
48            .with_height(ADDRESS),
49            row(
50                "Message",
51                keys::PEER_MESSAGE,
52                State::message_peer,
53                End::Chevron,
54                false,
55                ROW,
56            ),
57            row(
58                "Route",
59                keys::PEER_ROUTE,
60                |_| {},
61                detail.route.as_deref().map_or(End::Nothing, End::value),
62                false,
63                ROW,
64            ),
65            row(
66                "Last heard",
67                keys::PEER_HEARD,
68                |_| {},
69                End::value(&detail.heard),
70                false,
71                ROW,
72            ),
73            row(
74                "Forward secrecy",
75                keys::PEER_SECRECY,
76                State::toggle_forward_secrecy,
77                End::Switch(detail.forward_secrecy),
78                false,
79                ROW,
80            ),
81            row(
82                "Ping",
83                keys::PEER_PING,
84                State::ping_peer,
85                detail.ping.as_deref().map_or(End::Chevron, End::value),
86                false,
87                ROW,
88            ),
89            row(
90                "Refresh identity",
91                keys::PEER_IDENTITY,
92                State::request_peer_identity,
93                if requested {
94                    End::value("Requested")
95                } else {
96                    End::Nothing
97                },
98                false,
99                ROW,
100            ),
101            row(
102                "Forget",
103                keys::PEER_FORGET,
104                State::ask_forget,
105                End::Nothing,
106                true,
107                ROW,
108            ),
109        ))
110        .with_spacing(0),
111    )
112    .with_scroll_step(ROW)
113    .bound_focus(BoundaryBehavior::Stop)
114    .on_key(|key, s: &mut State| match key {
115        Key::Backspace => {
116            s.back();
117            KeyResponse::Handled
118        }
119        _ => KeyResponse::Ignored,
120    })
121    .popover(confirm.then_some(name).as_ref(), confirmation)
122}
123
124/// Forget, confirmed, or Cancel, which is focused.
125// Kept out of line, as `setting::row` is and for the same back-end failure,
126// here on `State::forget_peer`'s address once `root_view` took the setup
127// screens.
128#[inline(never)]
129fn confirmation(name: &String<32>) -> impl View<Rgb565, State> + use<> {
130    let mut forget: String<48> = String::new();
131    _ = core::fmt::write(&mut forget, format_args!("Forget {name}"));
132    card(
133        VStack::new((
134            choice(
135                forget,
136                keys::FORGET_CONFIRM,
137                State::forget_peer,
138                theme::RED,
139                None,
140            ),
141            choice(
142                String::try_from("Cancel").unwrap_or_default(),
143                keys::FORGET_CANCEL,
144                State::cancel_forget,
145                theme::TEXT,
146                None,
147            ),
148        ))
149        .with_spacing(0),
150        CONFIRM_WIDTH,
151        HorizontalAlignment::Center,
152        State::cancel_forget,
153    )
154}
