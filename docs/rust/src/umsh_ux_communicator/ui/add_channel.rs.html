1//! Add channel: one field takes a channel's name or a `umsh:ck:`
2//! invitation, and the line under it says which as it is typed. Join
3//! unlocks once it names a channel. Backspace goes back.
4
5use buoyant::event::{Key, KeyResponse};
6use buoyant::focus::BoundaryBehavior;
7use buoyant::view::prelude::*;
8use embedded_graphics::pixelcolor::Rgb565;
9use heapless::String;
10
11use super::add_peer::field_look;
12use super::conversation::header;
13use super::label;
14use super::member::{End, action};
15use crate::model::{ChannelEntry, State};
16use crate::nav::{ChannelText, keys};
17use crate::{font, theme};

19/// The label above the field, and the field's place.
20const LABEL: u32 = 24;
21const INSET: u32 = 8;
22/// The field's text: the screen less the field's inset and padding.
23const TEXT_WIDTH: u32 = theme::WIDTH - 2 * INSET - 24;
24const CARET: u32 = 3;
25/// The line under the field, and the row under that.
26const NOTE: u32 = 30;
27const ROW: u32 = 36;
28
29pub fn view(state: &State, text: &ChannelText) -> impl View<Rgb565, State> + use<> {
30    let entry = state.channel_entry();
31    let joinable = matches!(entry, ChannelEntry::Named | ChannelEntry::Private);
32    VStack::new((
33        header("Add channel", None, None),
34        label::<32>(
35            "Channel name or invitation",
36            &font::BOLD,
37            theme::SMALL,
38            theme::TEXT_SECONDARY,
39        )
40        .padding(Edges::Leading, 10)
41        .flex_infinite_width(HorizontalAlignment::Leading)
42        .frame()
43        .with_height(LABEL),
44        TextField::new(text, &font::MONO, State::channel_text_mut)
45            .with_font_size(theme::BODY)
46            .with_max_lines(3)
47            .with_caret_width(CARET)
48            // Enter joins once the text names a channel.
49            .on_submit(State::add_channel)
50            .on_active(State::field_active)
51            .with_label(|status, area| field_look(status, area, TEXT_WIDTH))
52            .focus_key(keys::CHANNEL_FIELD)
53            .padding(Edges::Horizontal, INSET),
54        note(entry),
55        joinable.then(|| action("Join", keys::JOIN_ENTRY, State::add_channel, End::Nothing)),
56        (!joinable).then(|| {
57            label::<8>("Join", &font::REGULAR, theme::BODY, theme::TEXT_TERTIARY)
58                .padding(Edges::Leading, 12)
59                .flex_infinite_width(HorizontalAlignment::Leading)
60                .frame()
61                .with_height(ROW)
62        }),
63    ))
64    .with_spacing(0)
65    .bound_focus(BoundaryBehavior::Stop)
66    .on_key(|key, s: &mut State| match key {
67        Key::Backspace => {
68            s.back();
69            KeyResponse::Handled
70        }
71        _ => KeyResponse::Ignored,
72    })
73}
74
75/// What the text names, and what it still lacks.
76fn note(entry: ChannelEntry) -> impl View<Rgb565, State> + use<> {
77    let mut text: String<40> = String::new();
78    let color = match entry {
79        ChannelEntry::Empty => theme::TEXT_SECONDARY,
80        ChannelEntry::Invalid => {
81            _ = text.push_str("Not a channel");
82            theme::AMBER
83        }
84        ChannelEntry::Named => {
85            _ = text.push_str("Named channel");
86            theme::TEXT_SECONDARY
87        }
88        ChannelEntry::Partial { missing } => {
89            let unit = if missing == 1 {
90                "character"
91            } else {
92                "characters"
93            };
94            _ = core::fmt::write(
95                &mut text,
96                format_args!("Private channel \u{B7} {missing} {unit} short"),
97            );
98            theme::AMBER
99        }
100        ChannelEntry::Private => {
101            _ = text.push_str("Private channel");
102            theme::TEXT_SECONDARY
103        }
104    };
105    label::<40>(&text, &font::SEMIBOLD, theme::SMALL, color)
106        .padding(Edges::Leading, 10)
107        .flex_infinite_width(HorizontalAlignment::Leading)
108        .frame()
109        .with_height(NOTE)
110}
