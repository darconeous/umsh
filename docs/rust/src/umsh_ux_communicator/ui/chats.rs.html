1//! The Chats list: two lines per chat, newest first. The last row is
2//! clipped by the screen edge when the list goes on.
3
4use buoyant::view::prelude::*;
5use buoyant::view::scroll_view::ScrollBarVisibility;
6use embedded_graphics::pixelcolor::Rgb565;
7use heapless::Vec;
8
9use super::{RAIL, label};
10use crate::model::{ChatSummary, MAX_CHATS, State};
11use crate::{font, icon, theme};
12
13const ROW: u32 = 44;
14/// The lines' heights: a line of each one's type, which together fill a row.
15const NAME_LINE: u32 = 24;
16const PREVIEW_LINE: u32 = ROW - NAME_LINE;
17const AVATAR: u32 = 34;
18/// From the list's leading edge to the text.
19const TEXT_X: u32 = 51;
20const LIST_WIDTH: u32 = theme::WIDTH - RAIL;
21
22/// Indices `ForEach` can hold without borrowing the state.
23static INDICES: [usize; MAX_CHATS] = {
24    let mut a = [0; MAX_CHATS];
25    let mut i = 0;
26    while i < MAX_CHATS {
27        a[i] = i;
28        i += 1;
29    }
30    a
31};
32
33pub fn view(chats: &Vec<ChatSummary, MAX_CHATS>) -> impl View<Rgb565, State> + use<> {
34    let rows = chats.clone();
35    let count = rows.len();
36    ScrollView::new(ForEach::<MAX_CHATS>::new_vertical(
37        &INDICES[..count],
38        move |&i| row(&rows[i]),
39    ))
40    .with_bar_visibility(ScrollBarVisibility::Never)
41}
42
43fn row(chat: &ChatSummary) -> impl View<Rgb565, State> + use<> {
44    let chat = chat.clone();
45    Button::new(
46        |_: &mut State| {},
47        move |button: ButtonState| {
48            let focused = button.is_focused();
49            let background = if focused {
50                theme::FOCUS
51            } else {
52                theme::BACKGROUND
53            };
54            // Each line has a fixed height and the avatar is an overlay, so
55            // no stack nests in another: every size a stack probes a line
56            // with is the same, and its text is measured once.
57            VStack::new((
58                name_line(&chat).frame().with_height(NAME_LINE),
59                preview_line(&chat).frame().with_height(PREVIEW_LINE),
60            ))
61            .with_spacing(0)
62            .with_alignment(HorizontalAlignment::Leading)
63            .padding(Edges::Leading, TEXT_X)
64            .flex_infinite_width(HorizontalAlignment::Leading)
65            .frame()
66            .with_height(ROW)
67            // Inside the background, so its edges blend into the row's color.
68            .overlay(
69                Alignment::Leading,
70                super::avatar::view(&chat.avatar, AVATAR)
71                    .frame()
72                    .with_width(TEXT_X),
73            )
74            // Only the focused row is filled: the others show the screen's
75            // background, and painting it again is most of what a frame
76            // would cost on the device.
77            .hint_background_color(background)
78            .background(
79                Alignment::Center,
80                focused.then(|| Rectangle.foreground_color(theme::FOCUS)),
81            )
82            .overlay(
83                Alignment::Leading,
84                focused.then(|| {
85                    Rectangle
86                        .foreground_color(theme::ACCENT)
87                        .frame_sized(4, ROW)
88                }),
89            )
90            .overlay(
91                Alignment::BottomTrailing,
92                (!focused).then(|| {
93                    Rectangle
94                        .foreground_color(theme::LINE)
95                        .frame_sized(LIST_WIDTH - TEXT_X, 1)
96                }),
97            )
98        },
99    )
100}

102/// The chat's name, then its age at the trailing edge.
103fn name_line(chat: &ChatSummary) -> impl View<Rgb565, State> + use<> {
104    let age_color = if chat.unread {
105        theme::ACCENT
106    } else {
107        theme::TEXT_TERTIARY
108    };
109    let name_face = if chat.unread {
110        &font::BOLD
111    } else {
112        &font::SEMIBOLD
113    };
114    let name_color = if chat.name.as_str() == "EMERGENCY" {
115        theme::RED
116    } else {
117        theme::TEXT
118    };
119    HStack::new((
120        label::<32>(&chat.name, name_face, theme::BODY, name_color),
121        Spacer::default(),
122        chat.muted.then(|| {
123            Image::new(&icon::BELL_SLASH)
124                .as_template()
125                .foreground_color(theme::TEXT_TERTIARY)
126        }),
127        label::<6>(&chat.age, &font::SEMIBOLD, theme::SMALL, age_color).fixed_size(true, false),
128    ))
129    .with_spacing(4)
130    .padding(Edges::Trailing, 10)
131}
132
133/// The last message, or the draft, with the unread dot at the trailing edge.
134fn preview_line(chat: &ChatSummary) -> impl View<Rgb565, State> + use<> {
135    let color = if chat.unread {
136        theme::TEXT_SECONDARY
137    } else {
138        theme::TEXT_TERTIARY
139    };
140    HStack::new((
141        chat.draft.then(|| {
142            label::<8>("Draft", &font::BOLD, theme::SECONDARY, theme::AMBER).fixed_size(true, false)
143        }),
144        label::<64>(&chat.preview, &font::REGULAR, theme::SECONDARY, color),
145        Spacer::default(),
146        chat.unread
147            .then(|| Circle.foreground_color(theme::ACCENT).frame_sized(10, 10)),
148    ))
149    .with_spacing(6)
150    .padding(Edges::Trailing, 9)
151}
