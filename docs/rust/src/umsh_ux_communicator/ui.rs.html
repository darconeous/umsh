43/// Height of the status strip.
44pub const STRIP: u32 = 22;
45/// Width of the rail beside the top-level lists.
46pub const RAIL: u32 = 66;
47/// How much of the content past the focused row a list keeps in view, so
48/// that the next row shows cut by the edge where there is one.
49const LOOK_AHEAD: u32 = 12;
50
51/// A list that scrolls with focus, with no bar: the next row showing past
52/// the focused one says there is more.
53fn list<V: ViewMarker>(content: V) -> ScrollView<V> {
54    ScrollView::new(content)
55        .with_bar_visibility(ScrollBarVisibility::Never)
56        .with_focus_margin(LOOK_AHEAD)
57}

59/// The card that says the board is shutting down.
60const SHUTTING_DOWN_WIDTH: u32 = 220;

62/// Readies an app over [`root_view`]: what can take focus, and where focus
63/// is kept.
64pub fn bind<V, F>(app: &mut App<V, State, F>)
65where
66    V: buoyant::view::ViewLayout<State>,
67    V::FocusTree: DefaultFocus,
68    V::Renderables: AnimatedJoin,
69    F: Fn(&State) -> V,
70{
71    app.set_roles(Role::Button | Role::TextEntry);
72    app.set_focus_binding(State::focus);
73    // Focus settles where the state asks, before the first input.
74    app.finalize_view();
75}

77/// The whole screen for `state`.
78///
79/// The screen's parts have fixed places, so they are laid over one another
80/// and inset rather than stacked: a stack lays each child out several times
81/// to learn how flexible it is, nested stacks multiply that, and on the
82/// device layout is the larger part of a frame.
83#[must_use]
84pub fn root_view(state: &State) -> impl View<Rgb565, State> + use<> {
85    ZStack::new((
86        status::view(&state.status, state.in_setup(), state.keyboard.latches()),
87        match_view!(state.screen(), {
88            Screen::Chats { .. } => chats_screen(state),
89            Screen::Conversation { chat, .. } => conversation::view(state, *chat),
90            Screen::Details { message, details, .. } => details::view(details, state.inbound(*message)),
91            Screen::Member {
92                member, requested, ..
93            } => member::view(
94                member,
95                *requested,
96                state
97                    .conversation
98                    .as_ref()
99                    .and_then(|c| state.chat(c.chat))
100                    .map_or("", |s| &s.name),
101            ),
102            Screen::Join { invite, name, .. } => join::view(invite, name),
103            Screen::AddPeer { text, .. } => add_peer::view(state, text),
104            Screen::Peer {
105                peer,
106                detail,
107                confirm,
108                requested,
109                ..
110            } => state
111                .peer(*peer)
112                .map(|summary| peer::view(summary, detail, *confirm, *requested)),
113            Screen::Device { page, .. } => device::page(state, page),
114            Screen::Setup { page, .. } => setup::view(state, page),
115            Screen::Wake { .. } => wake::view(state),
116        })
117        .padding(Edges::Top, STRIP),
118        // Over everything: one slot, the topmost of what could be here.
119        state.overlay().map(|overlay| overlay_view(state, overlay)),
120    ))
121    .with_alignment(Alignment::TopLeading)
122    .flex_infinite_width(HorizontalAlignment::Leading)
123    .flex_infinite_height(VerticalAlignment::Top)
124    .background_color(theme::BACKGROUND, Rectangle)
125}

127/// What lies over the screen: a banner, a glance, or the character
128/// picker, none of which takes focus; or, over even the strip, what the
129/// screen says as the board turns off.
130fn overlay_view(state: &State, overlay: Overlay) -> impl View<Rgb565, State> + use<> {
131    match_view!(overlay, {
132        Overlay::Ending(end) => ending(end),
133        Overlay::Picker(at) => typing::picker(state, at),
134        Overlay::Glance(glance) => typing::glance(glance),
135        Overlay::Banner(notice) => banner::view(state, notice),
136    })
137}

139/// What the screen says as the board turns off.
140fn ending(ending: Ending) -> impl View<Rgb565, State> + use<> {
141    match_view!(ending, {
142        Ending::ShuttingDown => shutting_down(),
143        Ending::BatteryEmpty { lit } => empty_battery(lit),
144    })
145}

147/// A card over the screen, as the board turns off as asked.
148fn shutting_down() -> impl View<Rgb565, State> + use<> {
149    VStack::new((
150        Image::new(&crate::icon::POWER)
151            .as_template()
152            .foreground_color(theme::TEXT_SECONDARY),
153        line("Shutting down", &font::SEMIBOLD, theme::BODY, theme::TEXT),
154    ))
155    .with_spacing(10)
156    .padding(Edges::Vertical, 20)
157    .frame()
158    .with_width(SHUTTING_DOWN_WIDTH)
159    .background(
160        Alignment::Center,
161        ZStack::new((
162            RoundedRectangle::new(12)
163                .foreground_color(Rgb565::new(0, 0, 0))

165            RoundedRectangle::new(12).foreground_color(theme::MENU),
166            RoundedRectangle::new(12)

168                .foreground_color(theme::MENU_LINE),

170    )
171    .flex_infinite_width(HorizontalAlignment::Center)

188/// The rail and the list beside it. Turning on the rail shows each section
189/// as it is reached; Find, while it is open, takes the Chats list's place.
190fn chats_screen(state: &State) -> impl View<Rgb565, State> + use<> {
191    let focus = state.screen().focus().current();
192    let shown = focus.and_then(keys::section).unwrap_or(state.section);
193    let find = match state.screen() {
194        Screen::Chats { find, .. } => find.as_ref().filter(|_| shown == Section::Chats),
195        _ => None,
196    };
197    let list = match (shown, find) {
198        (Section::Chats, Some(_)) => List::Find,
199        (Section::Chats, None) => List::Chats,
200        (Section::Peers, _) => List::Peers,
201        (Section::Device, _) => List::Device,
202    };
203    ZStack::new((
204        rail::view(shown, state.unread_chats()),
205        match_view!(list, {
206            List::Chats => chats::view(&state.chats, state.wall_time),
207            List::Find => find.map(|text| find::view(state, text)),
208            List::Peers => peers::view(&state.peers),
209            List::Device => device::view(state),
210        })
211        .padding(Edges::Leading, RAIL),
212    ))
213    .with_alignment(Alignment::TopLeading)
214}

216/// What shows beside the rail.
217#[derive(Clone, Copy, PartialEq, Eq)]
218enum List {
219    Chats,
220    Find,
221    Peers,
222    Device,

225/// One line of text, cut short with an ellipsis when it does not fit, and
226/// owned so the view borrows nothing.
227fn label<const N: usize>(
228    s: &str,
229    face: &'static PrerenderedFont,
230    size: u32,
231    color: Rgb565,
232) -> impl View<Rgb565, State> + use<N> {
233    let mut owned: heapless::String<N> = heapless::String::new();
234    for c in s.chars() {
235        if owned.push(c).is_err() {
236            break;
237        }
238    }
239    line(owned, face, size, color)
240}

242/// One line of `text`, which the view holds as it is given: shared text
243/// is shown without being copied.
244fn line<T: AsRef<str> + Clone, C>(
245    text: T,
246    face: &'static PrerenderedFont,
247    size: u32,
248    color: Rgb565,
249) -> impl View<Rgb565, C> + use<T, C> {
250    Text::new(text, face)
251        .with_font_size(size)
252        .with_line_limit(1)
253        .foreground_color(color)
254}
