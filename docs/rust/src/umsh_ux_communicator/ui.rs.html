1//! The screens, as Buoyant views over [`State`].
2//!
3//! Views own what they draw: Buoyant keeps the previous frame's render tree
4//! to animate from, so nothing in it may borrow the state that changed.

18/// Height of the status strip.
19pub const STRIP: u32 = 22;
20/// Width of the rail beside the top-level lists.
21pub const RAIL: u32 = 66;

23/// The whole screen for `state`.
24///
25/// The screen's parts have fixed places, so they are laid over one another
26/// and inset rather than stacked: a stack lays each child out several times
27/// to learn how flexible it is, nested stacks multiply that, and on the
28/// device layout is the larger part of a frame.
29#[must_use]
30pub fn root_view(state: &State) -> impl View<Rgb565, State> + use<> {
31    ZStack::new((
32        status::view(&state.status),
33        rail::view(state.section, state.unread_chats()).padding(Edges::Top, STRIP),
34        match state.section {
35            // Peers and Device are not built yet.
36            Section::Chats | Section::Peers | Section::Device => chats::view(&state.chats),
37        }
38        .padding(Edges::Top, STRIP)
39        .padding(Edges::Leading, RAIL),
40    ))
41    .with_alignment(Alignment::TopLeading)
42    .flex_infinite_width(HorizontalAlignment::Leading)
43    .flex_infinite_height(VerticalAlignment::Top)
44    .background_color(theme::BACKGROUND, Rectangle)
45}

47/// One line of text, cut short with an ellipsis when it does not fit, and
48/// owned so the view borrows nothing.
49fn label<const N: usize>(
50    s: &str,
51    face: &'static PrerenderedFont,
52    size: u32,
53    color: Rgb565,
54) -> impl View<Rgb565, State> + use<N> {
55    let mut owned: heapless::String<N> = heapless::String::new();
56    for c in s.chars() {
57        if owned.push(c).is_err() {
58            break;
59        }
60    }
61    Text::new(owned, face)
62        .with_font_size(size)
63        .with_line_limit(1)
64        .foreground_color(color)
65}
