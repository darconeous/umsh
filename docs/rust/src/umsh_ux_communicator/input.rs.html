1//! Input as the interface understands it, independent of how a board
2//! produces it.
3//!
4//! What an input means depends on where focus is: Backspace erases while a
5//! text field is active and is Back everywhere else, and turning the wheel
6//! moves the cursor in an active field and focus elsewhere. The screens
7//! decide that through focus; this only delivers inputs, each one to an
8//! interface whose focus has settled from the last.
9//!
10//! Keys are delivered as they are, and typed here through the
11//! [`Keyboard`](crate::keyboard::Keyboard): its modifiers, its typing aids,
12//! and the character picker.
13//!
14//! What the node has to say comes in here too: [`events`] takes in its
15//! events, after the inputs that led to them or whenever the board wakes
16//! for them.
17

28/// One user input.
29#[derive(Clone, Copy, Debug, PartialEq, Eq)]
30pub enum Input {
31    /// The wheel turned this many detents: positive is down or forward.
32    Turn(i8),
33    /// The wheel was pressed.
34    Press,
35    Enter,
36    Backspace,
37    /// A key with legend `plain`, and `alt` under Alt, was pressed.
38    Key {
39        plain: char,
40        alt: Option<char>,
41    },
42    Space,
43    /// A modifier went down, or up.
44    Modifier {
45        modifier: Modifier,
46        down: bool,
47    },
48    /// Text from a keyboard that applies its own modifiers, typed as it
49    /// is.
50    Character(char),
51}
52
53/// Applies `input`, which happened at `now` on the board's clock in
54/// milliseconds, to the running interface.
55pub fn apply<V, F>(app: &mut App<V, State, F>, input: Input, now: u64)
56where
57    V: ViewLayout<State>,
58    V::FocusTree: DefaultFocus,
59    V::Renderables: AnimatedJoin,
60    F: Fn(&State) -> V,
61{
62    // The board is turning off.
63    if app.state().ending.is_some() {
64        return;
65    }
66    app.state_mut().sync_keyboard();
67    if app.state().picker.is_some() {
68        pick(app, input, now);
69    } else {
70        deliver(app, input, now);
71    }
72    app.finalize_view();
73    app.state_mut().sync_keyboard();
74}

76/// Takes in the events the node has waiting, oldest first, at `now` on the
77/// board's clock, each through [`State::apply`]. Returns whether any calls
78/// for attention.
79///
80/// With nothing waiting, the interface is left as it was, so a board can
81/// look as often as it wakes.
82pub fn events<V, F>(app: &mut App<V, State, F>, now: u64) -> bool
83where
84    V: ViewLayout<State>,
85    V::FocusTree: DefaultFocus,
86    V::Renderables: AnimatedJoin,
87    F: Fn(&State) -> V,
88{
89    let node = app.state().node;
90    let mut attention = false;
91    let mut any = false;
92    while let Some(event) = node.next_event() {
93        any = true;
94        attention |= app.state_mut().apply(event, now);
95    }
96    if any {
97        app.finalize_view();
98    }
99    attention
100}

102fn deliver<V, F>(app: &mut App<V, State, F>, input: Input, now: u64)
103where
104    V: ViewLayout<State>,
105    V::FocusTree: DefaultFocus,
106    V::Renderables: AnimatedJoin,
107    F: Fn(&State) -> V,
108{
109    match input {
110        Input::Turn(detents) => {
111            // Turned with Alt held, the wheel sets the brightness.
112            if app.state_mut().keyboard.turn() {
113                app.state_mut().adjust_brightness(detents, now);
114                return;
115            }
116            for _ in 0..detents.unsigned_abs() {
117                if detents > 0 {
118                    app.next();
119                } else {
120                    app.previous();
121                }
122                app.finalize_view();
123            }
124        }
125        Input::Press => {
126            app.state_mut().keyboard.other();
127            app.select();
128        }
129        Input::Enter => {
130            app.state_mut().keyboard.other();
131            press(app, Key::Enter);
132        }
133        Input::Backspace => {
134            app.state_mut().keyboard.other();
135            press(app, Key::Backspace);
136        }
137        Input::Key { plain, alt } => {
138            let mut state = app.state_mut();
139            state.prepare_typing();
140            let typed = state.keyboard.letter(plain, alt);
141            press(app, Key::Character(typed));
142        }
143        Input::Space => {
144            let mut state = app.state_mut();
145            state.prepare_typing();
146            let aids = state.node.keyboard().aids();
147            match state.keyboard.space(now, aids) {
148                Spaced::Space => press(app, Key::Character(' ')),
149                Spaced::Period => {
150                    // The space just typed becomes the period's.
151                    press(app, Key::Backspace);
152                    press(app, Key::Character('.'));
153                    press(app, Key::Character(' '));
154                }
155                Spaced::Picker => {
156                    // Only into a field being edited.
157                    if state.keyboard.active() {
158                        state.picker = Some(0);
159                    }
160                }
161                Spaced::Light => state.switch_keyboard_light(now),
162            }
163        }
164        Input::Modifier { modifier, down } => {
165            app.state_mut().keyboard.modifier(modifier, down, now);
166        }
167        Input::Character(c) => {
168            app.state_mut().keyboard.other();
169            press(app, Key::Character(c));
170        }
171    }
172}

174/// An input while the character picker is open: the wheel walks it, a
175/// press or Enter types the focused character, Backspace closes it, and
176/// Space switches the keyboard light.
177fn pick<V, F>(app: &mut App<V, State, F>, input: Input, now: u64)
178where
179    V: ViewLayout<State>,
180    V::FocusTree: DefaultFocus,
181    V::Renderables: AnimatedJoin,
182    F: Fn(&State) -> V,
183{
184    let mut state = app.state_mut();
185    let Some(at) = state.picker else {
186        return;
187    };
188    match input {
189        Input::Turn(detents) => {
190            let last = PICKER.len() as i16 - 1;
191            state.picker = Some((i16::from(at) + i16::from(detents)).clamp(0, last) as u8);
192        }
193        Input::Press | Input::Enter => {
194            state.picker = None;
195            state.keyboard.other();
196            press(app, Key::Character(PICKER[usize::from(at)]));
197        }
198        Input::Backspace => {
199            state.picker = None;
200            state.keyboard.other();
201        }
202        Input::Space => {
203            state.picker = None;
204            state.keyboard.other();
205            state.switch_keyboard_light(now);
206        }
207        Input::Modifier { modifier, down } => state.keyboard.modifier(modifier, down, now),
208        // Filtering by name comes with the full picker.
209        Input::Key { .. } | Input::Character(_) => {}
210    }
211}

213/// A key pressed and released.
214fn press<V, F>(app: &mut App<V, State, F>, key: Key)
215where
216    V: ViewLayout<State>,
217    V::FocusTree: DefaultFocus,
218    V::Renderables: AnimatedJoin,
219    F: Fn(&State) -> V,
220{
221    app.key_down(key);
222    app.key_up(key);
223}

225impl State {
226    /// The focused text field, as the keyboard treats it.
227    #[must_use]
228    pub fn focused_field(&self) -> Option<Field> {
229        self.screen().focus().current().and_then(keys::field)
230    }
231
232    /// The focused field began or ended being edited.
233    pub fn field_active(&mut self, active: bool) {
234        self.keyboard.activity(active);
235    }
236
237    /// Keeps the keyboard's state with the focused field, and Shift with
238    /// where its cursor lands while it is edited.
239    pub fn sync_keyboard(&mut self) {
240        self.keyboard.focus(self.focused_field());
241        if self.keyboard.active() {
242            self.settle_capitals();
243        }
244    }
245
246    /// A key is about to type into the focused field: decides capitals as
247    /// though the field were already being edited, as the key makes it.
248    fn prepare_typing(&mut self) {
249        if self.keyboard.field().is_some() && !self.keyboard.active() {
250            self.settle_capitals();
251        }
252    }
253
254    fn settle_capitals(&mut self) {
255        if let Some((length, cursor, at_start)) = self.field_position() {
256            let aids = self.node.keyboard().aids();
257            self.keyboard.settle(length, cursor, at_start, aids);
258        }
259    }
260
261    /// The focused field's text length, its cursor, and whether the cursor
262    /// starts a sentence.
263    fn field_position(&mut self) -> Option<(usize, usize, bool)> {
264        fn of<B: EditText>(buffer: &mut EditBuffer<B>) -> (usize, usize, bool) {
265            let (text, cursor) = (buffer.text(), buffer.cursor());
266            (text.len(), cursor, starts_sentence(text, cursor))
267        }
268        match self.screen().focus().current()? {
269            keys::COMPOSER => {
270                let chat = self.conversation.as_ref()?.chat;
271                self.draft_mut(chat).map(of)
272            }
273            keys::NAME_FIELD => self.name_text_mut().map(of),
274            keys::JOIN_NAME => self.join_name_mut().map(of),
275            keys::FIND_FIELD => self.find_text_mut().map(of),
276            keys::PEER_FIELD => self.peer_text_mut().map(of),
277            keys::CHANNEL_FIELD => self.channel_text_mut().map(of),
278            keys::POWER_FIELD => self.power_text_mut().map(of),
279            _ => None,
280        }
281    }
