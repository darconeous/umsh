1//! The keyboard as the interface types with it: Shift and Alt, each off,
2//! one-shot, or locked; the two typing aids; and what each text field
3//! allows of them. See the UX book's Communicator Keyboard page.
4//!
5//! This decides what a key types. It knows nothing of the board's matrix,
6//! only of keys with a legend of their own and one under Alt, and of the
7//! text it is typed into only what it is told: which field is focused,
8//! whether it is being edited, and where its cursor is.
9
10/// What the character picker offers, while it offers only what the keys
11/// cannot type.
12pub const PICKER: [char; 16] = [
13    '#', '&', '%', '~', '(', ')', '[', ']', '{', '}', '<', '>', '|', '\\', '^', '`',
14];
15
16/// How soon a second tap locks a modifier, and a second space after a
17/// word types a period, in milliseconds.
18pub const DOUBLE_TAP_MS: u64 = 400;
19
20#[derive(Clone, Copy, Debug, PartialEq, Eq)]
21pub enum Modifier {
22    Shift,
23    Alt,
24}
25
26/// A modifier's state, apart from being held.
27#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
28pub enum Latch {
29    #[default]
30    Off,
31    /// Applies to the next key it changes.
32    Once,
33    /// Applies until tapped off.
34    Locked,
35}
36
37/// A text field, as the keyboard treats it.
38#[derive(Clone, Copy, Debug, PartialEq, Eq)]
39pub enum Field {
40    /// Running text: a message.
41    Prose,
42    /// A name: capitalized, but a period would change it.
43    Name,
44    /// Search text, codes, addresses, and invitations, typed as they are.
45    Literal,
46    /// A number, typed from the top row in symbol lock.
47    Number,
48}
49
50impl Field {
51    /// Whether Shift arms itself at the start of a sentence.
52    #[must_use]
53    pub const fn capitalizes(self) -> bool {
54        matches!(self, Self::Prose | Self::Name)
55    }
56
57    /// Whether two quick spaces after a word type a period.
58    #[must_use]
59    pub const fn periods(self) -> bool {
60        matches!(self, Self::Prose)
61    }
62
63    /// Whether the field opens in symbol lock.
64    #[must_use]
65    pub const fn symbols(self) -> bool {
66        matches!(self, Self::Number)
67    }
68}

70/// The typing aids the user leaves on. A field can only turn them off.
71#[derive(Clone, Copy, Debug, PartialEq, Eq)]
72pub struct Aids {
73    pub capitalize: bool,
74    pub period: bool,
75}

77/// What Space did.
78#[derive(Clone, Copy, Debug, PartialEq, Eq)]
79pub enum Spaced {
80    /// It types a space.
81    Space,
82    /// It ends a sentence: the space just typed becomes a period and a
83    /// space.
84    Period,
85    /// Under Alt it opens the character picker.
86    Picker,
87    /// It followed a space under Alt quickly: it switches the keyboard
88    /// light.
89    Light,
90}

92#[derive(Clone, Copy, Debug, Default)]
93struct Key {
94    latch: Latch,
95    down: bool,
96    /// Something happened while it was held, so letting go is not a tap.
97    used: bool,
98    /// When a tap armed it, for a second tap to lock it.
99    armed_at: u64,
100}
101
102impl Key {
103    /// It applies to the key typed now.
104    fn on(&self) -> bool {
105        self.down || self.latch != Latch::Off
106    }
107
108    /// It changed the key typed now.
109    fn applied(&mut self) {
110        self.used |= self.down;
111        if self.latch == Latch::Once {
112            self.latch = Latch::Off;
113        }
114    }
115}

117/// What came before a space, for the period on double space and the
118/// keyboard light.
119#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
120enum Run {
121    #[default]
122    None,
123    /// The last key typed a letter.
124    Letter,
125    /// A space just followed a letter, at this time.
126    Space(u64),
127    /// Space was pressed under Alt, at this time.
128    AltSpace(u64),
129}

131#[derive(Clone, Copy, Debug, Default)]
132pub struct Keyboard {
133    shift: Key,
134    alt: Key,
135    /// Shift was armed at the start of a sentence, not by a tap.
136    automatic: bool,
137    /// The focused text field, whose keyboard state this is.
138    field: Option<Field>,
139    /// Whether that field is being edited.
140    active: bool,
141    /// The field's length and cursor when capitals were last decided.
142    settled: Option<(usize, usize)>,
143    run: Run,
144}
145
146impl Keyboard {
147    /// Shift's and Alt's states, for the status strip.
148    #[must_use]
149    pub fn latches(&self) -> (Latch, Latch) {
150        (self.shift.latch, self.alt.latch)
151    }
152
153    /// Whether Alt is held down.
154    #[must_use]
155    pub fn alt_held(&self) -> bool {
156        self.alt.down
157    }
158
159    /// The focused text field, if any.
160    #[must_use]
161    pub fn field(&self) -> Option<Field> {
162        self.field
163    }
164
165    /// Whether the focused text field is being edited.
166    #[must_use]
167    pub fn active(&self) -> bool {
168        self.field.is_some() && self.active
169    }
170
171    /// Focus is on `field`, or on no text field. Moving between fields
172    /// starts afresh.
173    pub fn focus(&mut self, field: Option<Field>) {
174        if field != self.field {
175            self.field = field;
176            self.active = false;
177            self.clear();
178        }
179    }
180
181    /// The focused field began, or ended, being edited. The modifiers
182    /// belong to the edit, so ending it clears them.
183    pub fn activity(&mut self, active: bool) {
184        if active != self.active {
185            self.active = active;
186            if !active {
187                self.clear();
188            }
189        }
190    }
191
192    /// Back to how the focused field opens. A modifier held through this
193    /// stays held, but letting it go is no longer a tap.
194    fn clear(&mut self) {
195        for key in [&mut self.shift, &mut self.alt] {
196            *key = Key {
197                down: key.down,
198                used: key.down,
199                ..Key::default()
200            };
201        }
202        if self.field.is_some_and(Field::symbols) {
203            self.alt.latch = Latch::Locked;
204        }
205        self.automatic = false;
206        self.settled = None;
207        self.run = Run::None;
208    }
209
210    /// A modifier went down or up at `now`. Pressed and let go with
211    /// nothing in between, it was tapped.
212    pub fn modifier(&mut self, modifier: Modifier, down: bool, now: u64) {
213        self.run = Run::None;
214        let key = match modifier {
215            Modifier::Shift => &mut self.shift,
216            Modifier::Alt => &mut self.alt,
217        };
218        if down {
219            // A down repeated while held is the same hold.
220            if !key.down {
221                key.down = true;
222                key.used = false;
223            }
224        } else if core::mem::take(&mut key.down) && !key.used {
225            self.tap(modifier, now);
226        }
227    }
228
229    fn tap(&mut self, modifier: Modifier, now: u64) {
230        // Either lock is released by a tap of either, which arms nothing.
231        if self.shift.latch == Latch::Locked || self.alt.latch == Latch::Locked {
232            self.shift.latch = Latch::Off;
233            self.alt.latch = Latch::Off;
234            self.automatic = false;
235            return;
236        }
237        let automatic = core::mem::take(&mut self.automatic);
238        let (this, other) = match modifier {
239            Modifier::Shift => (&mut self.shift, &mut self.alt),
240            Modifier::Alt => (&mut self.alt, &mut self.shift),
241        };
242        // A Shift armed at the start of a sentence is not the user's: a tap
243        // of Shift disarms it, and a tap of Alt arms Alt in its place.
244        if automatic {
245            match modifier {
246                Modifier::Shift => {
247                    this.latch = Latch::Off;
248                    return;
249                }
250                Modifier::Alt => other.latch = Latch::Off,
251            }
252        }
253        // One armed, or held, then the other: the other locks.
254        if other.down || other.latch == Latch::Once {
255            this.latch = Latch::Locked;
256            other.latch = Latch::Off;
257            other.used |= other.down;
258            return;
259        }
260        this.latch = match this.latch {
261            Latch::Off => {
262                this.armed_at = now;
263                Latch::Once
264            }
265            Latch::Once if now.saturating_sub(this.armed_at) <= DOUBLE_TAP_MS => Latch::Locked,
266            _ => Latch::Off,
267        };
268    }

270    /// A key with legend `plain`, and `alt` under Alt, was pressed: what
271    /// it types.
272    pub fn letter(&mut self, plain: char, alt: Option<char>) -> char {
273        self.run = Run::None;
274        if self.alt.on()
275            && let Some(symbol) = alt
276        {
277            self.alt.applied();
278            return symbol;
279        }
280        let typed = if self.shift.on() {
281            self.shift.applied();
282            self.automatic = false;
283            plain.to_ascii_uppercase()
284        } else {
285            plain
286        };
287        if typed.is_alphabetic() {
288            self.run = Run::Letter;
289        }
290        typed
291    }

293    /// Space was pressed at `now`. In symbol lock it types a space; under
294    /// a one-shot or held Alt it opens the picker.
295    pub fn space(&mut self, now: u64, aids: Aids) -> Spaced {
296        // A second quick space after one under Alt, which needs Alt no
297        // longer: a one-shot Alt was used up by the first.
298        if let Run::AltSpace(at) = self.run
299            && now.saturating_sub(at) <= DOUBLE_TAP_MS
300        {
301            self.run = Run::None;
302            return Spaced::Light;
303        }
304        if self.alt.down || self.alt.latch == Latch::Once {
305            self.alt.applied();
306            self.run = Run::AltSpace(now);
307            return Spaced::Picker;
308        }
309        let periods = aids.period && self.field.is_some_and(Field::periods);
310        match self.run {
311            Run::Letter => {
312                self.run = Run::Space(now);
313                Spaced::Space
314            }
315            Run::Space(at) if periods && now.saturating_sub(at) <= DOUBLE_TAP_MS => {
316                self.run = Run::None;
317                Spaced::Period
318            }
319            _ => {
320                self.run = Run::None;
321                Spaced::Space
322            }
323        }
324    }
325
326    /// A key that types nothing the aids follow, such as Enter or
327    /// Backspace, was pressed. One-shot modifiers stay armed.
328    pub fn other(&mut self) {
329        self.run = Run::None;
330    }

332    /// The wheel turned. Returns whether Alt is held, for the turn to set
333    /// brightness: an Alt held for it is not tapped by letting go.
334    pub fn turn(&mut self) -> bool {
335        self.run = Run::None;
336        self.alt.used |= self.alt.down;
337        self.alt.down
338    }

340    /// The edited field's text is `length` long with the cursor at
341    /// `cursor`, which is at the start of a sentence or not. Arms Shift
342    /// there, where the field and `aids` allow, each time the cursor lands
343    /// somewhere new.
344    pub fn settle(&mut self, length: usize, cursor: usize, at_start: bool, aids: Aids) {
345        let Some(field) = self.field else {
346            return;
347        };
348        if self.settled == Some((length, cursor)) {
349            return;
350        }
351        self.settled = Some((length, cursor));
352        let free = self.shift.latch == Latch::Off && self.alt.latch == Latch::Off;
353        if at_start && aids.capitalize && field.capitalizes() && free {
354            self.shift.latch = Latch::Once;
355            self.automatic = true;
356        } else if !at_start && self.automatic {
357            self.shift.latch = Latch::Off;
358            self.automatic = false;
359        }
360    }
361}

363/// Whether `cursor` in `text` is at the start of a sentence: the start of
364/// the text, or after `.`, `!`, or `?` and a space.
365#[must_use]
366pub fn starts_sentence(text: &str, cursor: usize) -> bool {
367    let before = text.get(..cursor).unwrap_or(text);
368    if before.is_empty() {
369        return true;
370    }
371    let trimmed = before.trim_end_matches(' ');
372    trimmed.len() < before.len() && trimmed.ends_with(['.', '!', '?'])
373}

396    #[test]
397    fn a_tap_arms_for_one_key_and_a_quick_second_tap_locks() {
398        let mut k = prose();
399        tap(&mut k, Modifier::Shift, 0);
400        assert_eq!(k.latches(), (Latch::Once, Latch::Off));
401        assert_eq!(k.letter('a', Some('*')), 'A');
402        assert_eq!(k.letter('a', Some('*')), 'a');
403
404        tap(&mut k, Modifier::Shift, 1000);
405        tap(&mut k, Modifier::Shift, 1200);
406        assert_eq!(k.latches(), (Latch::Locked, Latch::Off));
407        assert_eq!(k.letter('a', Some('*')), 'A');
408        assert_eq!(k.letter('b', Some('!')), 'B');
409        tap(&mut k, Modifier::Shift, 5000);
410        assert_eq!(k.latches(), (Latch::Off, Latch::Off));
411
412        // A slow second tap disarms.
413        tap(&mut k, Modifier::Alt, 6000);
414        tap(&mut k, Modifier::Alt, 7000);
415        assert_eq!(k.latches(), (Latch::Off, Latch::Off));
416    }
