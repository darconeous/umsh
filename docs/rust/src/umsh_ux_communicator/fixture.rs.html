372            7 => condition.forward_secret = true,
373            8 => condition.reply = Some(7),

375                condition.parts = Some(Parts {
376                    count: 3,
377                    received: 0b011,
378                    lost: 0,
379                });
380            }
381            _ => {}
382        }
383    }

387/// What a reply quotes of `message`, in the chat called `chat_name`.
388fn quote_of(heap: TextHeap, message: &Message, chat_name: &str) -> Option<Quote> {
389    let name = match (message.direction, &message.sender) {
390        (Direction::Outbound, _) => None,

394    Some(Quote {
395        name,
396        text: message.text.clone(),
397    })

409/// The fixture's message, with its text kept on `heap`, when there is room.
410fn mockup_message(
411    heap: TextHeap,
412    chat: ChatId,
413    index: usize,
414    (from, text): (From, &str),
415) -> Option<Message> {
416    assert!(text.len() <= MESSAGE_LEN, "fixture message fits");
417    let count = messages(chat).len();
418    // A minute apart, the last when its chat was last active, but for the
419    // first of the states, the evening before.
420    let mut time = STARTED - quiet_for(chat) - 60 * (count - 1 - index) as u32;
421    if chat == STATES && index == 0 {
422        time -= 13 * 60 * 60;
423    }
424    let condition = condition(chat, index);
425    let reply = match condition.reply {
426        Some(quoted) => {
427            let message = mockup_message(heap, chat, quoted, *messages(chat).get(quoted)?)?;
428            let chat_name = mockup_chats(heap)
429                .into_iter()
430                .find(|row| row.id == chat)
431                .map(|row| row.name)
432                .unwrap_or_default();
433            quote_of(heap, &message, &chat_name)
434        }
435        None => None,
436    };
437    let (invite, text) = match invite(heap, chat, index) {
438        Some((invite, uri)) => (Some(invite), SharedText::new(heap, &uri)?),
439        None => (None, SharedText::new(heap, text)?),
440    };
441    Some(Message {
442        id: mockup_id(chat, index),
443        direction: match from {
444            From::Me => Direction::Outbound,
445            From::Peer | From::Member(_) => Direction::Inbound,
446        },
447        sender: match from {
448            From::Member(name) => Some(member(name)),
449            From::Me | From::Peer => None,
450        },
451        text,
452        delivery: delivery(chat, index),
453        reactions: their_reactions(chat, index)
454            .iter()
455            .map(|&(_, reaction)| Reacted {
456                reaction,
457                count: 1,
458                mine: false,
459            })
460            .collect::<Reactions>(),
461        invite,
462        reply,
463        time: Some(time),
464        presence: condition.presence,
465        edited: condition.edited,
466        deleted: condition.deleted,
467        late: condition.late,
468        forward_secret: condition.forward_secret,
469        parts: condition.parts,
470    })
471}

710/// A URI parameter's value with its percent-encoding undone; what does not
711/// decode to text stays as it was typed.
712fn decoded(value: &str) -> String<32> {
713    let mut bytes: Vec<u8, 32> = Vec::new();
714    let mut rest = value.as_bytes();
715    while let Some((&byte, tail)) = rest.split_first() {
716        let escaped = match (byte, tail) {
717            (b'%', [high, low, after @ ..]) => core::str::from_utf8(&[*high, *low])
718                .ok()
719                .and_then(|hex| u8::from_str_radix(hex, 16).ok())
720                .map(|byte| (byte, after)),
721            _ => None,
722        };
723        let (byte, tail) = escaped.unwrap_or((byte, tail));
724        if bytes.push(byte).is_err() {
725            break;
726        }
727        rest = tail;
728    }
729    core::str::from_utf8(&bytes).map_or_else(|_| cut(value), cut)
730}

732/// A channel avatar's color, from its name.
733fn tint(name: &str) -> [u8; 3] {
734    let hash = name.bytes().fold(0x811C_9DC5_u32, |h, b| {
735        (h ^ u32::from(b)).wrapping_mul(0x0100_0193)
736    });
737    let [r, g, b, _] = hash.to_be_bytes();
738    [r, g, b]
739}

741/// `name` as a URI parameter: what is not a letter, a digit, or `-._~` is
742/// percent-encoded.
743fn push_encoded(uri: &mut ShareUri, name: &str) -> Option<()> {
744    for byte in name.bytes() {
745        if byte.is_ascii_alphanumeric() || b"-._~".contains(&byte) {
746            uri.push(char::from(byte)).ok()?;
747        } else {
748            core::fmt::write(uri, format_args!("%{byte:02X}")).ok()?;
749        }
750    }
751    Some(())
752}

1481    fn inspect_peer(&self, text: &str) -> PeerEntry {
1482        use umsh_core::NodeHint;
1483        use umsh_core::base58::{ENCODED_LEN, prefix_bounds};
1484
1485        let text = text.trim();
1486        if text.is_empty() {
1487            return PeerEntry::Empty;
1488        }
1489        // An invitation carries the address, and may carry an identity after
1490        // it, whose name a node would read; the fixture does not.
1491        let (digits, name) = match text.strip_prefix(PEER_INVITATION) {
1492            Some(rest) => (
1493                rest.split_once(':').map_or(rest, |(digits, _)| digits),
1494                None,
1495            ),
1496            // An invitation still being typed.
1497            None if PEER_INVITATION.starts_with(text) => {
1498                return PeerEntry::Partial {
1499                    avatar: None,
1500                    hint: String::new(),
1501                    missing: ENCODED_LEN as u8,
1502                };
1503            }
1504            None => (text, None),
1505        };
1506        if digits.is_empty() {
1507            return PeerEntry::Partial {
1508                avatar: None,
1509                hint: String::new(),
1510                missing: ENCODED_LEN as u8,
1511            };
1512        }
1513        let Ok((low, high)) = prefix_bounds(digits.as_bytes()) else {
1514            return PeerEntry::Invalid;
1515        };
1516        let bytes = [low[0], low[1], low[2]];
1517        let mut hint: String<8> = String::new();
1518        _ = core::fmt::write(&mut hint, format_args!("{}", NodeHint(bytes)));
1519        if digits.len() == ENCODED_LEN {
1520            // The whole address settles every character of the hint.
1521            let hint: String<8> = s(&digits[..4]);
1522            let avatar = AvatarStyle::peer(bytes, &hint, PeerShape::Hexagon);
1523            return PeerEntry::Complete {
1524                avatar,
1525                hint,
1526                address: s(digits),
1527                name,
1528            };
1529        }
1530        // A peer known by nothing but their address is drawn as a hexagon.
1531        let avatar = AvatarStyle::peer(bytes, &hint, PeerShape::Hexagon);
1532        // The avatar shows once the characters settle the hint.
1533        let settled = low[..3] == high[..3];
1534        PeerEntry::Partial {
1535            avatar: settled.then_some(avatar),
1536            hint: if settled { hint } else { String::new() },
1537            missing: (ENCODED_LEN - digits.len()) as u8,
1538        }
1539    }

1552    fn inspect_channel(&self, text: &str) -> ChannelEntry {
1553        use umsh_core::base58::{ENCODED_LEN, prefix_bounds};
1554
1555        let text = text.trim();
1556        if text.is_empty() {
1557            return ChannelEntry::Empty;
1558        }
1559        let missing = |key: &str| ChannelEntry::Partial {
1560            missing: (ENCODED_LEN - key.len()) as u8,
1561        };
1562        match text.strip_prefix(CHANNEL_INVITATION) {
1563            Some(rest) => {
1564                let key = rest.split_once('?').map_or(rest, |(key, _)| key);
1565                if key.is_empty() {
1566                    missing(key)
1567                } else if key.len() > ENCODED_LEN || prefix_bounds(key.as_bytes()).is_err() {
1568                    ChannelEntry::Invalid
1569                } else if key.len() == ENCODED_LEN {
1570                    ChannelEntry::Private
1571                } else {
1572                    missing(key)
1573                }
1574            }
1575            // An invitation still being typed.
1576            None if CHANNEL_INVITATION.starts_with(text) && text.contains(':') => missing(""),
1577            // A name is any ASCII without the invitation's colon.
1578            None if text.is_ascii() && !text.contains(':') => ChannelEntry::Named,
1579            None => ChannelEntry::Invalid,
1580        }
1581    }

1584        let text = text.trim();

1587                // Joined from the start.

1649            match from {
1650                From::Me => (true, "14:27"),

1665                snr: Some(65),
1666                authenticated: true,
1667                ..Details::default()
1668            }
1669        };
1670        let index = (message & 0xFF) as usize;
1671        let theirs = if message < FIRST_KEPT {
1672            their_reactions(chat, index)
1673        } else {
1674            &[]
1675        };

1677            let (name, avatar) = match from {
1678                From::Member(name) => (name, member(name).avatar),
1679                From::Me | From::Peer => {

1688                }

1691                name: Some(SharedText::new(self.heap, name)?),
1692                avatar,

1694            });

1696        if let Some(&(_, reaction)) = self.mine.borrow().iter().find(|(id, _)| *id == message) {
1697            _ = details.reactors.push(Reactor {
1698                name: None,
1699                avatar: self.avatar(),
1700                reaction,
1701            });
1702        }
