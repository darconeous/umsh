1//! Colors and type sizes shared by every screen.
2//!
3//! The palette is the T-LoRa Pager mockup's: a near-black background, so
4//! the emissive panel spends little on what is not content.
5
6use embedded_graphics::pixelcolor::Rgb565;
7
8/// A 24-bit `0xRRGGBB` color, truncated to the panel's format.
9const fn rgb(hex: u32) -> Rgb565 {
10    Rgb565::new(
11        (hex >> 19) as u8 & 0x1f,
12        (hex >> 10) as u8 & 0x3f,
13        (hex >> 3) as u8 & 0x1f,
14    )
15}
16
