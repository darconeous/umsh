1//! The store in memory, alone or over a durable layer.
2//!
3//! It keeps two parts. Its records—chats, messages, archives, checkpoints,
4//! peers, channels, and the counters behind its identifiers—are what a
5//! power cut leaves. Its index—chats by activity, each chat's messages,
6//! wire aliases, peers by hint—is kept up to date as it goes and rebuilt
7//! from the records when it is mounted again, so a power cut can be armed
8//! and the store remounted to see what survived.
9//!
10//! Alone ([`Volatile`]), the records in memory are what survives, which
11//! serves tests and anything before durable storage. Over a layer that
12//! keeps them ([`Durable`]), the records in memory are a working copy:
13//! each one changed is handed to the layer at the end of every step of a
14//! commit, and mounting reads them all back from it.
15

40/// How much a store keeps before it evicts.
41#[derive(Clone, Copy, Debug, PartialEq, Eq)]
42pub struct Quota {
43    /// Octets of messages and archived payloads kept before the oldest
44    /// messages go.
45    pub bytes: usize,
46    /// The newest messages of each chat that eviction never takes, however
47    /// busy another chat is.
48    pub keep_per_chat: usize,
49    /// Heard peers kept besides the saved ones, most recently heard first.
50    pub heard_peers: usize,
51}
52
53impl Quota {
54    pub const DEFAULT: Self = Self {
55        bytes: 300 * 1024,
56        keep_per_chat: 50,
57        heard_peers: 32,
58    };
59}
60
61impl Default for Quota {
62    fn default() -> Self {
63        Self::DEFAULT
64    }
65}

67/// What a message costs against the quota besides its body: about what a
68/// durable back end spends on its record and its index entries.
69pub const RECORD_COST: usize = 96;

114/// How far ahead of what it has given out a durable store's kept
115/// counters reach.
116const ID_BLOCK: u32 = 64;
117
118#[derive(Clone, Debug, Default)]
119struct Disk {
120    /// The next chat, message, and peer identifiers.
121    ids: [u32; 3],
122    /// How far the kept counters reach: no identifier at or past these
123    /// has been given out.
124    reserved: [u32; 3],
125    /// Orders everything the store does.
126    clock: u64,
127    /// Each conversation's outbound checkpoint, and when it was last used.
128    checkpoints: BTreeMap<ConversationCode, (StreamCheckpoint, u64)>,
129    archives: BTreeMap<ArchiveCode, Vec<u8>>,
130    chats: BTreeMap<u32, ChatEntry>,
131    messages: BTreeMap<u32, MessageEntry>,
132    peers: BTreeMap<[u8; 32], PeerEntry>,
133    channels: BTreeMap<[u8; 16], ChannelRecord>,
134    /// The record the device keeps here of its own lists.
135    device_lists: Option<Vec<u8>>,
136}
137
138impl Disk {
139    fn tick(&mut self) -> u64 {
140        self.clock += 1;
141        self.clock
142    }
143
144    /// The next identifier of `kind`. The kept counters reach a block ahead
145    /// of what has been given out, and move on before an identifier past
146    /// them is: a store mounted after a power cut skips the rest of the
147    /// block rather than give any of it out again.
148    fn take_id(&mut self, kind: usize, dirty: &mut Dirty) -> u32 {
149        let id = self.ids[kind];
150        self.ids[kind] += 1;
151        if self.ids[kind] > self.reserved[kind] {
152            self.reserved[kind] = id + ID_BLOCK;
153            dirty.mark(RecordKey::Counters);
154        }
155        id
156    }
157}

159/// A record, by what names it.
160#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
161enum RecordKey {
162    Counters,
163    Checkpoint(ConversationCode),
164    Archive(ArchiveCode),
165    Chat(u32),
166    Message(u32),
167    Peer([u8; 32]),
168    Channel([u8; 16]),
169    DeviceLists,
170}

172/// Records changed since they were last handed to the durable layer, in
173/// the order they first changed.
174#[derive(Debug, Default)]
175struct Dirty {
176    order: Vec<RecordKey>,
177    seen: BTreeSet<RecordKey>,
178}
179
180impl Dirty {
181    fn mark(&mut self, key: RecordKey) {
182        if self.seen.insert(key) {
183            self.order.push(key);
184        }
185    }
186
187    fn take(&mut self) -> Vec<RecordKey> {
188        self.seen.clear();
189        core::mem::take(&mut self.order)
190    }
191}

195#[derive(Debug, Default, PartialEq, Eq)]
196struct Index {
197    chat_for: BTreeMap<ConversationCode, u32>,
198    by_activity: BTreeMap<u64, u32>,
199    chat_messages: BTreeMap<u32, BTreeSet<u32>>,
200    aliases: BTreeMap<AliasCode, BTreeSet<u32>>,
201    hints: BTreeMap<[u8; 3], BTreeSet<[u8; 32]>>,
202    /// Heard peers that are not saved, by when they were heard.
203    heard: BTreeMap<u64, [u8; 32]>,
204    /// Octets counted against the quota.
205    bytes: usize,
206}
207
208impl Index {
209    fn build(disk: &Disk) -> Self {
210        let mut index = Self::default();
211        for (&id, chat) in &disk.chats {
212            index.add_chat(id, chat);
213        }
214        for (&id, message) in &disk.messages {
215            index.add_message(id, message);
216        }
217        index.bytes += disk.archives.values().map(Vec::len).sum::<usize>();
218        for (key, peer) in &disk.peers {
219            index.add_peer(key, peer);
220        }
221        index
222    }
223
224    fn add_chat(&mut self, id: u32, chat: &ChatEntry) {
225        self.chat_for.insert((&chat.record.conversation).into(), id);
226        self.by_activity.insert(chat.activity, id);
227        self.chat_messages.entry(id).or_default();
228    }
229
230    fn remove_chat(&mut self, id: u32, chat: &ChatEntry) {
231        self.chat_for.remove(&(&chat.record.conversation).into());
232        self.by_activity.remove(&chat.activity);
233        self.chat_messages.remove(&id);
234    }
235
236    fn add_message(&mut self, id: u32, message: &MessageEntry) {
237        self.chat_messages
238            .entry(message.record.chat.0)
239            .or_default()
240            .insert(id);
241        if let Some(alias) = alias_code(message) {
242            self.aliases.entry(alias).or_default().insert(id);
243        }
244        self.bytes += cost(message);
245    }
246
247    fn remove_message(&mut self, id: u32, message: &MessageEntry) {
248        if let Some(messages) = self.chat_messages.get_mut(&message.record.chat.0) {
249            messages.remove(&id);
250        }
251        if let Some(alias) = alias_code(message)
252            && let Some(ids) = self.aliases.get_mut(&alias)
253        {
254            ids.remove(&id);
255            if ids.is_empty() {
256                self.aliases.remove(&alias);
257            }
258        }
259        self.bytes -= cost(message);
260    }
261
262    fn add_peer(&mut self, key: &[u8; 32], peer: &PeerEntry) {
263        self.hints.entry(hint_of(key)).or_default().insert(*key);
264        if !peer.record.saved {
265            self.heard.insert(peer.heard, *key);
266        }
267    }
268
269    fn remove_peer(&mut self, key: &[u8; 32], peer: &PeerEntry) {
270        let hint = hint_of(key);
271        if let Some(keys) = self.hints.get_mut(&hint) {
272            keys.remove(key);
273            if keys.is_empty() {
274                self.hints.remove(&hint);
275            }
276        }
277        if !peer.record.saved {
278            self.heard.remove(&peer.heard);
279        }
280    }
281}
282
283fn cost(message: &MessageEntry) -> usize {
284    RECORD_COST + message.body.len()
285}
286
287fn alias_code(message: &MessageEntry) -> Option<AliasCode> {
288    Some((
289        (&message.conversation).into(),
290        (&message.record.sender).into(),
291        message.record.wire_id?,
292    ))
293}
294
295fn hint_of(key: &[u8; 32]) -> [u8; 3] {
296    [key[0], key[1], key[2]]
297}
298
299fn reactions(reactors: &[Reactor]) -> Reactions {
300    let mut counted = Reactions::new();
301    for reactor in reactors {
302        match counted
303            .iter_mut()
304            .find(|reacted| reacted.reaction == reactor.reaction)
305        {
306            Some(reacted) => {
307                reacted.count += 1;
308                reacted.mine |= reactor.mine;
309            }
310            None => {
311                let _ = counted.push(Reacted {
312                    reaction: reactor.reaction,
313                    count: 1,
314                    mine: reactor.mine,
315                });
316            }
317        }
318    }
319    counted
320}

324/// A [`ChatStore`] in memory, alone or over a durable layer.
325pub struct MemoryStore<D: Durable = Volatile> {
326    disk: Disk,
327    index: Index,
328    quota: Quota,
329    durable: D,
330    dirty: Dirty,
331    /// Where records are encoded on their way to the durable layer.
332    scratch: Vec<u8>,
333    /// Power goes once a commit finishes this step.
334    cut: Option<CommitStep>,
335    /// Power is gone: nothing more is written until it is back.
336    off: bool,
337}
338
339impl Default for MemoryStore {
340    fn default() -> Self {
341        Self::new()
342    }
343}
344
345impl MemoryStore {
346    pub fn new() -> Self {
347        Self::with_quota(Quota::DEFAULT)
348    }
349
350    pub fn with_quota(quota: Quota) -> Self {
351        Self::over(Volatile, quota)
352    }
353}
354
355impl<D: Durable> MemoryStore<D> {
356    fn over(durable: D, quota: Quota) -> Self {
357        Self {
358            disk: Disk::default(),
359            index: Index::default(),
360            quota,
361            durable,
362            dirty: Dirty::default(),
363            scratch: Vec::new(),
364            cut: None,
365            off: false,
366        }
367    }
368
369    /// The store `durable` keeps, read back. A record that does not decode
370    /// is left out, and the rest are kept.
371    pub fn mount(durable: D, quota: Quota) -> Result<Self, StoreError> {
372        let mut store = Self::over(durable, quota);
373        store.remount()?;
374        Ok(store)
375    }
376
377    /// [`MemoryStore::mount`], or, when what `durable` keeps cannot be
378    /// read at all, an empty store over it cleared: storage that held
379    /// something else before the store was put there.
380    pub fn mount_or_clear(durable: D, quota: Quota) -> Result<Self, StoreError> {
381        let mut store = Self::over(durable, quota);
382        if store.remount().is_err() {
383            store.durable.clear()?;
384            store.remount()?;
385        }
386        Ok(store)
387    }
388
389    /// The durable layer.
390    pub fn durable(&self) -> &D {
391        &self.durable
392    }
393
394    /// The durable layer, for a test to cut its power or restore it.
395    pub fn durable_mut(&mut self) -> &mut D {
396        &mut self.durable
397    }
398
399    /// Lose power once the next commit has finished `step`: nothing after
400    /// it is written, and nothing more is until [`MemoryStore::power_on`].
401    pub fn cut_power_after(&mut self, step: CommitStep) {
402        self.cut = Some(step);
403    }
404
405    /// Mount again from what was written, as after a power cut or a
406    /// reboot.
407    pub fn power_on(&mut self) -> Result<(), StoreError> {
408        self.off = false;
409        self.cut = None;
410        self.remount()
411    }
412
413    /// Read the records back from what keeps them, and rebuild the index.
414    fn remount(&mut self) -> Result<(), StoreError> {
415        self.dirty.take();
416        if D::KEEPS {
417            let mut disk = Disk::default();
418            self.durable
419                .read(&mut |key, record| codec::restore(&mut disk, key, record))?;
420            let orphans = codec::settle(&mut disk);
421            self.disk = disk;
422            // What no longer belongs anywhere goes at the next change.
423            for orphan in orphans {
424                self.dirty.mark(orphan);
425            }
426        }
427        self.index = Index::build(&self.disk);
428        Ok(())
429    }
430
431    /// Hand the durable layer every record changed since it was last
432    /// handed any, in the order they changed.
433    fn flush(&mut self) -> Result<(), StoreError> {
434        let keys = self.dirty.take();
435        if !D::KEEPS || self.off {
436            return Ok(());
437        }
438        let mut key_bytes = heapless::Vec::<u8, { codec::KEY_MAX }>::new();
439        for key in keys {
440            codec::key(&key, &mut key_bytes);
441            self.scratch.clear();
442            let kept = codec::record(&self.disk, &key, &mut self.scratch);
443            self.durable
444                .write(&key_bytes, kept.then_some(&self.scratch[..]))?;
445        }
446        Ok(())
447    }
448
449    /// Hand over what changed. When the layer refuses, what it kept is
450    /// read back, so the store is as the layer has it and the change it
451    /// was making is undone.
452    fn settle(&mut self) -> Result<(), StoreError> {
453        let flushed = self.flush();
454        if flushed.is_err() {
455            let _ = self.remount();
456        }
457        flushed
458    }
459
460    fn writable(&self) -> Result<(), StoreError> {
461        if self.off {
462            Err(StoreError::PoweredOff)
463        } else {
464            Ok(())
465        }
466    }
467
468    /// `step` is done, and handed to the durable layer. Returns whether
469    /// power went with it.
470    fn reached(&mut self, step: CommitStep) -> Result<bool, StoreError> {
471        self.settle()?;
472        Ok(self.reached_cut(step))
473    }
474
475    fn reached_cut(&mut self, step: CommitStep) -> bool {
476        if self.cut == Some(step) {
477            self.cut = None;
478            self.off = true;
479        }
480        self.off
481    }

483    /// The index agrees with one rebuilt from storage.
484    fn check_index(&self) {
485        #[cfg(test)]
486        assert_eq!(
487            self.index,
488            Index::build(&self.disk),
489            "the index drifted from what a remount builds"
490        );
491    }

495    fn delete_archives(&mut self, conversation: ConversationCode, message_id: u8) {
496        let range = (conversation, message_id, 0)..=(conversation, message_id, u8::MAX);
497        let keys: Vec<ArchiveCode> = self
498            .disk
499            .archives
500            .range(range)
501            .map(|(key, _)| *key)
502            .collect();
503        for key in keys {
504            if let Some(payload) = self.disk.archives.remove(&key) {
505                self.index.bytes -= payload.len();
506                self.dirty.mark(RecordKey::Archive(key));
507            }
508        }
509    }
510
511    fn put_archive(&mut self, key: ArchiveCode, payload: &[u8]) {
512        self.dirty.mark(RecordKey::Archive(key));
513        if let Some(old) = self.disk.archives.insert(key, payload.to_vec()) {
514            self.index.bytes -= old.len();
515        }
516        self.index.bytes += payload.len();
517    }

521    fn open_chat(&mut self, conversation: &ConversationKey, name: Option<&str>) -> u32 {
522        if let Some(&id) = self.index.chat_for.get(&conversation.into()) {
523            return id;
524        }
525        let id = self.disk.take_id(CHAT, &mut self.dirty);
526        self.dirty.mark(RecordKey::Chat(id));
527        let activity = self.disk.tick();
528        let chat = ChatEntry {
529            record: ChatRecord {
530                id: ChatId(id),
531                conversation: *conversation,
532                name: name.map(record::name),
533                muted: false,
534                unread: 0,
535                last_activity: None,
536                has_draft: false,
537            },
538            activity,
539            draft: None,
540        };
541        self.index.add_chat(id, &chat);
542        self.disk.chats.insert(id, chat);
543        id
544    }

546    /// A message was kept in `chat` at `time`.
547    fn touch_chat(&mut self, chat: u32, time: Option<u32>, unread: bool) {
548        let activity = self.disk.tick();
549        let Some(entry) = self.disk.chats.get_mut(&chat) else {
550            return;
551        };
552        self.dirty.mark(RecordKey::Chat(chat));
553        self.index.by_activity.remove(&entry.activity);
554        entry.activity = activity;
555        self.index.by_activity.insert(activity, chat);
556        if time.is_some() {
557            entry.record.last_activity = time;
558        }
559        if unread {
560            entry.record.unread += 1;
561        }
562    }

566    /// Change a kept message, keeping the index with it.
567    fn update_message(&mut self, id: u32, change: impl FnOnce(&mut MessageEntry)) -> bool {
568        let Some(mut message) = self.disk.messages.remove(&id) else {
569            return false;
570        };
571        self.dirty.mark(RecordKey::Message(id));
572        self.index.remove_message(id, &message);
573        change(&mut message);
574        self.index.add_message(id, &message);
575        self.disk.messages.insert(id, message);
576        true
577    }

579    /// Remove a message, and its resend material with it unless a newer
580    /// message now holds its wire ID.
581    fn remove_message(&mut self, id: u32) {
582        let Some(message) = self.disk.messages.get(&id) else {
583            return;
584        };
585        if message.record.direction == Direction::Outbound
586            && let Some(alias) = alias_code(message)
587            && self.index.aliases.get(&alias).and_then(BTreeSet::last) == Some(&id)
588        {
589            self.delete_archives(alias.0, alias.2);
590        }
591        if let Some(message) = self.disk.messages.remove(&id) {
592            self.dirty.mark(RecordKey::Message(id));
593            self.index.remove_message(id, &message);
594        }
595    }

598        let message = self.disk.messages.get(&id)?;
599        let mut record = message.record.clone();
600        record.reactions = reactions(&message.reactors);

604    fn resolve(
605        &self,
606        batch: &Batch<'_>,
607        committed: &Committed,
608        handle: umsh_text::engine::sequence::MessageHandle,
609    ) -> Option<u32> {
610        committed
611            .id_of(handle)
612            .or_else(|| batch.resolve(handle))
613            .map(|id| id.0)
614            .filter(|id| self.disk.messages.contains_key(id))
615    }

617    /// The message `reference` names in `conversation`.
618    ///
619    /// A one-to-one Regarding is one octet, a wire ID with no stream, so
620    /// the engine leaves it unresolved, scoped to the peer, when the ID
621    /// could be either party's. With `either_party`, such a reference
622    /// matches the newest message with that ID in either stream. An edit
623    /// or delete never does: its original is always its own sender's.
624    fn resolve_ref(
625        &self,
626        batch: &Batch<'_>,
627        committed: &Committed,
628        reference: &ResolvedRef,
629        conversation: &ConversationKey,
630        either_party: bool,
631    ) -> Option<u32> {
632        match reference {
633            ResolvedRef::Handle(handle) => self.resolve(batch, committed, *handle),
634            ResolvedRef::Unresolved(WireRef::SenderScoped { sender, message_id }) => {
635                let alias = WireAlias {
636                    conversation: *conversation,
637                    sender: *sender,
638                    epoch: None,
639                    wire_id: *message_id,
640                };
641                let one_to_one = matches!(
642                    conversation,
643                    ConversationKey::Direct { .. } | ConversationKey::ChannelDirect { .. }
644                );
645                let ours = (either_party && one_to_one && *sender != SenderScope::Local)
646                    .then(|| {
647                        self.resolve_wire(&WireAlias {
648                            sender: SenderScope::Local,
649                            ..alias
650                        })
651                    })
652                    .flatten();
653                self.resolve_wire(&alias).max(ours).map(|id| id.0)
654            }
655            // A room numbers its messages itself, whoever sent them.
656            ResolvedRef::Unresolved(WireRef::RoomCanonical { message_id }) => {
657                let conversation = ConversationCode::from(conversation);
658                self.index
659                    .aliases
660                    .range(
661                        (conversation, SenderCode::MIN, 0)
662                            ..=(conversation, SenderCode::MAX, u8::MAX),
663                    )
664                    .filter(|((_, _, wire_id), _)| wire_id == message_id)
665                    .filter_map(|(_, ids)| ids.last().copied())
666                    .max()
667            }
668        }
669    }

671    fn apply(&mut self, batch: &Batch<'_>) -> Committed {
672        let mut committed = Committed::default();
673        for change in batch.changes() {
674            match change {
675                Change::Message { mutation, link } => {
676                    self.mutate(batch, &mut committed, mutation, link);
677                }
678                Change::Delivery {
679                    handle,
680                    fragment,
681                    state,
682                } => {
683                    if let Some(id) = self.resolve(batch, &committed, handle)
684                        && let Some(message) = self.disk.messages.get_mut(&id)
685                    {
686                        self.dirty.mark(RecordKey::Message(id));
687                        message.record.delivery.set(fragment, state);
688                        committed.touch_message(MessageId(id));
689                    }
690                }
691                Change::Secrecy {
692                    message,
693                    forward_secret,
694                } => {
695                    let id = match message {
696                        Subject::Handle(handle) => self.resolve(batch, &committed, handle),
697                        Subject::Kept(id) => Some(id.0),
698                    };
699                    if let Some(id) = id
700                        && let Some(message) = self.disk.messages.get_mut(&id)
701                        && message.record.flags.forward_secret != forward_secret
702                    {
703                        self.dirty.mark(RecordKey::Message(id));
704                        message.record.flags.forward_secret = forward_secret;
705                        committed.touch_message(MessageId(id));
706                    }
707                }
708            }
709        }
710        committed
711    }

713    fn mutate(
714        &mut self,
715        batch: &Batch<'_>,
716        committed: &mut Committed,
717        mutation: &Mutation,
718        link: Option<LinkInfo>,
719    ) {
720        let revision = mutation.revision;
721        match mutation.change {
722            chat::Change::Insert(Insert {
723                conversation,
724                sender,
725                direction,
726                message_type,
727                wire_id,
728                epoch,
729                ref sender_handle,
730                regarding,
731                bg_color,
732                text_color,
733                ref body,
734                status,
735                presence,
736                late,
737                notify,
738                ..
739            }) => {
740                let sender_handle = sender_handle.as_deref();
741                let regarding_id = regarding.and_then(|reference| {
742                    self.resolve_ref(batch, committed, &reference, &conversation, true)
743                });
744                // A slot already kept under this handle is filled in place.
745                if let Some(id) = self.resolve(batch, committed, mutation.handle) {
746                    let mut noticed = false;
747                    let mut shown = false;
748                    self.update_message(id, |message| {
749                        if revision <= message.revision {
750                            return;
751                        }
752                        let record = &mut message.record;
753                        noticed = notify && !record.flags.notify && direction == Direction::Inbound;
754                        shown =
755                            presence == Presence::Present && record.presence != Presence::Present;
756                        message.revision = revision;
757                        message.body = record::clip(body, BODY_MAX).into();
758                        record.sender = sender;
759                        record.sender_handle = sender_handle.map(record::name);
760                        record.wire_id = wire_id;
761                        record.epoch = epoch;
762                        record.message_type = message_type;
763                        record.regarding = regarding_id.map(MessageId);
764                        record.body_len = message.body.len() as u16;
765                        record.bg_color = bg_color;
766                        record.text_color = text_color;
767                        record.status = status;
768                        record.presence = presence;
769                        record.flags.late |= late;
770                        record.flags.notify |= notify;
771                        record.link = link.or(record.link);
772                        record.time = record.time.or(batch.time());
773                    });
774                    if let Some(chat) = self.disk.messages.get(&id).map(|m| m.record.chat.0)
775                        && (noticed || shown)
776                    {
777                        self.touch_chat(chat, batch.time(), noticed);
778                        committed.touch_chat(ChatId(chat));
779                    }
780                    committed.touch_message(MessageId(id));
781                    return;
782                }
783
784                // A reaction folds into what it is about.
785                if message_type == MessageType::Status && regarding.is_some() {
786                    if let Some(target) = regarding_id {
787                        let who = SenderCode::from(&sender);
788                        let reaction = Reaction::from_body(body);
789                        self.update_message(target, |message| {
790                            message.reactors.retain(|reactor| reactor.who != who);
791                            if let Some(reaction) = reaction {
792                                message.reactors.push(Reactor {
793                                    who,
794                                    mine: direction == Direction::Outbound,
795                                    reaction,
796                                });
797                            }
798                        });
799                        committed.touch_message(MessageId(target));
800                    }
801                    return;
802                }
803
804                let chat = self.open_chat(&conversation, None);
805                let id = self.disk.take_id(MESSAGE, &mut self.dirty);
806                self.dirty.mark(RecordKey::Message(id));
807                let body = record::clip(body, BODY_MAX);
808                let message = MessageEntry {
809                    record: MessageRecord {
810                        id: MessageId(id),
811                        chat: ChatId(chat),
812                        direction,
813                        sender,
814                        sender_handle: sender_handle.map(record::name),
815                        wire_id,
816                        epoch,
817                        message_type,
818                        regarding: regarding_id.map(MessageId),
819                        body_len: body.len() as u16,
820                        bg_color,
821                        text_color,
822                        status,
823                        presence,
824                        flags: MessageFlags {
825                            late,
826                            notify,
827                            ..MessageFlags::default()
828                        },
829                        delivery: Delivery::default(),
830                        link,
831                        time: batch.time(),
832                        reactions: Reactions::new(),
833                    },
834                    conversation,
835                    body: body.into(),
836                    revision,
837                    reactors: Vec::new(),
838                };
839                self.index.add_message(id, &message);
840                self.disk.messages.insert(id, message);
841                if presence == Presence::Present {
842                    let noticed = notify && direction == Direction::Inbound;
843                    self.touch_chat(chat, batch.time(), noticed);
844                }
845                committed.inserted.push((mutation.handle, MessageId(id)));
846                committed.touch_message(MessageId(id));
847                committed.touch_chat(ChatId(chat));
848            }
849            chat::Change::UpdateBody {
850                ref body,
851                status,
852                late,
853                notify,
854            } => {
855                let Some(id) = self.resolve(batch, committed, mutation.handle) else {
856                    return;
857                };
858                let mut noticed = false;
859                self.update_message(id, |message| {
860                    if revision <= message.revision {
861                        return;
862                    }
863                    let record = &mut message.record;
864                    noticed =
865                        notify && !record.flags.notify && record.direction == Direction::Inbound;
866                    message.revision = revision;
867                    message.body = record::clip(body, BODY_MAX).into();
868                    record.body_len = message.body.len() as u16;
869                    record.status = status;
870                    record.flags.late |= late;
871                    record.flags.notify |= notify;
872                });
873                if noticed && let Some(chat) = self.disk.messages.get(&id).map(|m| m.record.chat.0)
874                {
875                    self.touch_chat(chat, batch.time(), true);
876                    committed.touch_chat(ChatId(chat));
877                }
878                committed.touch_message(MessageId(id));
879            }
880            chat::Change::Edit {
881                conversation,
882                original,
883                ref body,
884            } => {
885                let Some(id) = self.resolve_ref(batch, committed, &original, &conversation, false)
886                else {
887                    return;
888                };
889                self.update_message(id, |message| {
890                    message.body = record::clip(body, BODY_MAX).into();
891                    message.record.body_len = message.body.len() as u16;
892                    message.record.flags.edited = true;
893                });
894                committed.touch_message(MessageId(id));
895            }
896            chat::Change::Delete {
897                conversation,
898                original,
899            } => {
900                let Some(id) = self.resolve_ref(batch, committed, &original, &conversation, false)
901                else {
902                    return;
903                };
904                let held = self
905                    .disk
906                    .messages
907                    .get(&id)
908                    .is_some_and(|message| message.record.presence == Presence::GapPending);
909                if held {
910                    // A slot held for a message that never came, and never
911                    // will: nothing was shown, so nothing stays.
912                    self.remove_message(id);
913                } else {
914                    // The tombstone stays as long as its wire alias does.
915                    self.update_message(id, |message| {
916                        message.body.clear();
917                        message.record.body_len = 0;
918                        message.record.flags.deleted = true;
919                        message.reactors.clear();
920                    });
921                }
922                committed.touch_message(MessageId(id));
923            }
924        }
925    }

927    /// Make room: the oldest messages go first, across every chat, except
928    /// each chat's newest.
929    fn evict(&mut self) {
930        while self.index.bytes > self.quota.bytes {
931            let keep = self.quota.keep_per_chat;
932            let victim = self.disk.messages.iter().find_map(|(&id, message)| {
933                let held = self
934                    .index
935                    .chat_messages
936                    .get(&message.record.chat.0)
937                    .map_or(0, BTreeSet::len);
938                (held > keep).then_some(id)
939            });
940            let Some(victim) = victim else {
941                break;
942            };
943            self.remove_message(victim);
944        }
945    }

949    /// Forget the heard peers beyond the quota, least recently heard first.
950    fn forget_unheard(&mut self) {
951        while self.index.heard.len() > self.quota.heard_peers {
952            let Some((_, key)) = self.index.heard.pop_first() else {
953                break;
954            };
955            if let Some(peer) = self.disk.peers.remove(&key) {
956                self.dirty.mark(RecordKey::Peer(key));
957                self.index.remove_peer(&key, &peer);
958            }
959        }
960    }

963impl<D: Durable> ChatStore for MemoryStore<D> {
964    fn commit(&mut self, batch: &Batch<'_>) -> Result<Committed, StoreError> {
965        self.writable()?;
966        for checkpoint in batch.checkpoints() {
967            let used = self.disk.tick();
968            let code = (&checkpoint.conversation).into();
969            self.dirty.mark(RecordKey::Checkpoint(code));
970            self.disk.checkpoints.insert(code, (*checkpoint, used));
971        }
972        if self.reached(CommitStep::Checkpoint)? {
973            return Err(StoreError::Io);
974        }
975        for (conversation, message_id) in batch.archive_deletes() {
976            self.delete_archives(conversation.into(), *message_id);
977        }
978        if self.reached(CommitStep::ArchiveDeletes)? {
979            return Err(StoreError::Io);
980        }
981        for (key, payload) in batch.archives() {
982            self.put_archive(archive_code(key), payload);
983        }
984        if self.reached(CommitStep::Archives)? {
985            return Err(StoreError::Io);
986        }
987        let committed = self.apply(batch);
988        self.evict();
989        self.check_index();
990        // Power lost after the transcript loses nothing: the commit is
991        // whole.
992        self.reached(CommitStep::Transcript)?;
993        Ok(committed)
994    }

1121    fn update_chat(&mut self, chat: ChatId, change: ChatChange<'_>) -> Result<ChatId, StoreError> {
1122        self.writable()?;
1123        if let ChatChange::Open { conversation, name } = change {
1124            let id = self.open_chat(&conversation, name);
1125            self.check_index();
1126            self.settle()?;
1127            return Ok(ChatId(id));
1128        }
1129        if !self.disk.chats.contains_key(&chat.0) {
1130            return Err(StoreError::NotFound);
1131        }
1132        if let ChatChange::Delete = change {
1133            let messages: Vec<u32> = self
1134                .index
1135                .chat_messages
1136                .get(&chat.0)
1137                .into_iter()
1138                .flatten()
1139                .copied()
1140                .collect();
1141            for id in messages {
1142                self.remove_message(id);
1143            }
1144            if let Some(entry) = self.disk.chats.remove(&chat.0) {
1145                self.dirty.mark(RecordKey::Chat(chat.0));
1146                self.index.remove_chat(chat.0, &entry);
1147            }
1148            self.check_index();
1149            self.settle()?;
1150            return Ok(chat);
1151        }
1152        let Some(entry) = self.disk.chats.get_mut(&chat.0) else {
1153            return Err(StoreError::NotFound);
1154        };
1155        self.dirty.mark(RecordKey::Chat(chat.0));
1156        match change {
1157            ChatChange::Rename(name) => entry.record.name = name.map(record::name),
1158            ChatChange::Mute(muted) => entry.record.muted = muted,
1159            ChatChange::Read => entry.record.unread = 0,
1160            ChatChange::Draft(draft) => {
1161                entry.draft = draft
1162                    .filter(|draft| !draft.is_empty())
1163                    .map(|draft| record::clip(draft, BODY_MAX).into());
1164                entry.record.has_draft = entry.draft.is_some();
1165            }
1166            ChatChange::Open { .. } | ChatChange::Delete => unreachable!(),
1167        }
1168        self.check_index();
1169        self.settle()?;
1170        Ok(chat)
1171    }

1210    fn update_peer(
1211        &mut self,
1212        key: &PublicKey,
1213        change: PeerChange<'_>,
1214    ) -> Result<PeerId, StoreError> {
1215        self.writable()?;
1216        let existing = self.disk.peers.remove(&key.0);
1217        if let Some(peer) = &existing {
1218            self.dirty.mark(RecordKey::Peer(key.0));
1219            self.index.remove_peer(&key.0, peer);
1220        }
1221        let mut peer = match (existing, change) {
1222            (Some(peer), PeerChange::Forget) => {
1223                self.check_index();
1224                self.settle()?;
1225                return Ok(peer.record.id);
1226            }
1227            (Some(peer), _) => peer,
1228            (None, PeerChange::Heard { .. } | PeerChange::Save { .. }) => {
1229                let id = PeerId(self.disk.take_id(PEER, &mut self.dirty));
1230                self.dirty.mark(RecordKey::Peer(key.0));
1231                PeerEntry {
1232                    record: PeerRecord {
1233                        id,
1234                        key: *key,
1235                        name: None,
1236                        identity: heapless::Vec::new(),
1237                        last_heard: None,
1238                        route: None,
1239                        forward_secrecy: false,
1240                        saved: false,
1241                    },
1242                    heard: 0,
1243                }
1244            }
1245            (None, _) => return Err(StoreError::NotFound),
1246        };
1247        match change {
1248            PeerChange::Heard { at, name, identity } => {
1249                peer.heard = self.disk.tick();
1250                peer.record.last_heard = at.or(peer.record.last_heard);
1251                // What it calls itself, unless someone here named it.
1252                if let Some(name) = name
1253                    && !(peer.record.saved && peer.record.name.is_some())
1254                {
1255                    peer.record.name = Some(record::name(name));
1256                }
1257                if let Some(identity) = identity
1258                    && identity.len() <= IDENTITY_MAX
1259                {
1260                    peer.record.identity = heapless::Vec::from_slice(identity).unwrap_or_default();
1261                }
1262            }
1263            PeerChange::Route(route) => {
1264                peer.record.route = route.map(|hints| {
1265                    heapless::Vec::from_slice(&hints[..hints.len().min(ROUTE_MAX)])
1266                        .unwrap_or_default()
1267                });
1268            }
1269            PeerChange::Save { name } => {
1270                peer.record.saved = true;
1271                if let Some(name) = name {
1272                    peer.record.name = Some(record::name(name));
1273                }
1274            }
1275            PeerChange::Unsave => {
1276                peer.record.saved = false;
1277                peer.heard = self.disk.tick();
1278            }
1279            PeerChange::Rename(name) => peer.record.name = name.map(record::name),
1280            PeerChange::ForwardSecrecy(preferred) => peer.record.forward_secrecy = preferred,
1281            PeerChange::Forget => unreachable!(),
1282        }
1283        let id = peer.record.id;
1284        self.index.add_peer(&key.0, &peer);
1285        self.disk.peers.insert(key.0, peer);
1286        self.forget_unheard();
1287        self.check_index();
1288        self.settle()?;
1289        Ok(id)
1290    }

1354    fn interrupt(&mut self) -> Result<(), StoreError> {
1355        self.writable()?;
1356        let Disk {
1357            chats, messages, ..
1358        } = &mut self.disk;
1359        for (&id, message) in messages.iter_mut() {
1360            if message.record.direction != Direction::Outbound {
1361                continue;
1362            }
1363            let multicast = chats.get(&message.record.chat.0).is_some_and(|chat| {
1364                matches!(
1365                    chat.record.conversation,
1366                    ConversationKey::ChannelGroup { .. }
1367                )
1368            });
1369            if message.record.delivery.interrupt(multicast) {
1370                self.dirty.mark(RecordKey::Message(id));
1371            }
1372        }
1373        self.settle()
1374    }

1376    fn usage(&self) -> Usage {
1377        Usage {
1378            chats: self.disk.chats.len(),
1379            messages: self.disk.messages.len(),
1380            peers: self.disk.peers.len(),
1381            channels: self.disk.channels.len(),
1382            archives: self.disk.archives.len(),
1383            checkpoints: self.disk.checkpoints.len(),
1384            bytes: self.index.bytes,
1385            quota: self.quota.bytes,
1386        }
1387    }

1389    fn erase(&mut self) -> Result<(), StoreError> {
1390        self.writable()?;
1391        self.dirty.take();
1392        if let Err(error) = self.durable.clear() {
1393            let _ = self.remount();
1394            return Err(error);
1395        }
1396        self.disk = Disk {
1397            ids: self.disk.ids,
1398            clock: self.disk.clock,
1399            ..Disk::default()
1400        };
1401        self.index = Index::build(&self.disk);
1402        // The counters are kept again at once, so that no identifier given
1403        // out before is given out again.
1404        for kind in [CHAT, MESSAGE, PEER] {
1405            self.disk.reserved[kind] = self.disk.ids[kind] + ID_BLOCK;
1406        }
1407        self.dirty.mark(RecordKey::Counters);
1408        self.settle()
1409    }
1410}

1412fn copy_text<'b>(text: &str, out: &'b mut [u8]) -> Result<&'b str, StoreError> {
1413    let out = out
1414        .get_mut(..text.len())
1415        .ok_or(StoreError::BufferTooSmall)?;
1416    out.copy_from_slice(text.as_bytes());
1417    core::str::from_utf8(out).map_err(|_| StoreError::Corrupt)
1418}

1670    #[test]
1671    fn eviction_takes_the_oldest_across_chats_but_each_keeps_its_newest() {
1672        let cost = cost_of_one();
1673        let mut store = MemoryStore::with_quota(Quota {
1674            bytes: 80 * cost,
1675            ..Quota::DEFAULT
1676        });
1677        let mut rig = Rig::new();
1678        let quiet: Vec<MessageId> = (0..10)
1679            .map(|n| {
1680                rig.say(&mut store, direct(), &alloc::format!("calm {n:03}"))
1681                    .1
1682            })
1683            .collect();
1684        let mut first_wire = None;
1685        let busy: Vec<MessageId> = (0..200)
1686            .map(|n| {
1687                let id = rig
1688                    .say(&mut store, group(), &alloc::format!("busy {n:03}"))
1689                    .1;
1690                first_wire = first_wire.or(record(&store, id).wire_id);
1691                id
1692            })
1693            .collect();
1694
1695        // The quiet chat is older than all of it, and loses nothing.
1696        for &id in &quiet {
1697            assert!(store.message(id).unwrap().is_some());
1698        }
1699        // The busy chat lost its oldest, and only as much as it had to.
1700        let usage = store.usage();
1701        assert!(usage.bytes <= usage.quota);
1702        let kept = store
1703            .messages(store.chat_for(&group()).unwrap(), None)
1704            .count();
1705        assert!((60..=70).contains(&kept), "{kept}");
1706        let oldest = busy[200 - kept];
1707        assert!(store.message(oldest).unwrap().is_some());
1708        assert!(store.message(busy[199 - kept]).unwrap().is_none());
1709
1710        // An evicted message's resend material went with it.
1711        assert_eq!(archived(&store, group(), first_wire.unwrap()), None);
1712        let newest_wire = record(&store, busy[199]).wire_id.unwrap();
1713        assert!(archived(&store, group(), newest_wire).is_some());
1714        assert!(store.checkpoint(&group()).is_some());
1715    }
