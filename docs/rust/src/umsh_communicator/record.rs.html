17/// The longest name the store keeps, in octets of UTF-8. A longer one is
18/// cut at the last whole character that fits.
19pub const NAME_MAX: usize = 32;
20
21/// A name the store keeps.
22pub type Name = String<NAME_MAX>;
23
24/// The longest body or draft, in octets: a fully reassembled message.
25pub const BODY_MAX: usize = REASSEMBLED_BODY_MAX;
26
27/// The longest advertised identity a peer record keeps.
28pub const IDENTITY_MAX: usize = 64;
29
30/// The most router hints a learned route keeps.
31pub const ROUTE_MAX: usize = 16;
32
33/// `text` as a [`Name`], cut at a character boundary.
34pub(crate) fn name(text: &str) -> Name {
35    let mut end = text.len().min(NAME_MAX);
36    while !text.is_char_boundary(end) {
37        end -= 1;
38    }
39    let mut name = Name::new();
40    let _ = name.push_str(&text[..end]);
41    name
42}
43
44/// `text` cut to at most `max` octets, at a character boundary.
45pub(crate) fn clip(text: &str, max: usize) -> &str {
46    let mut end = text.len().min(max);
47    while !text.is_char_boundary(end) {
48        end -= 1;
49    }
50    &text[..end]
51}

90/// How each fragment of an outbound message fared, as bitmaps over
91/// fragment indices. An unfragmented message is fragment 0.
92#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
93pub struct Delivery {
94    pub sent: u16,
95    pub acked: u16,
96    pub failed: u16,
97}
98
99impl Delivery {
100    /// Takes the latest word on one fragment. A failed fragment that is
101    /// sent again, or acknowledged after all, has not failed.
102    pub fn set(&mut self, fragment: Option<u8>, state: DeliveryState) {
103        let bit = 1u16 << fragment.unwrap_or(0).min(15);
104        match state {
105            DeliveryState::Sent => self.sent |= bit,
106            DeliveryState::Acked => self.acked |= bit,
107            DeliveryState::Failed => self.failed |= bit,
108        }
109        if state != DeliveryState::Failed {
110            self.failed &= !bit;
111        }
112    }
113
114    /// Whether word is still to come: nothing has failed, a fragment sent
115    /// is unacknowledged or none was sent, and, for a `multicast`, which
116    /// no acknowledgment follows, nothing was sent yet.
117    pub fn awaited(&self, multicast: bool) -> bool {
118        let delivered = self.acked != 0 && self.sent & !self.acked == 0;
119        self.failed == 0 && !delivered && !(multicast && self.sent != 0)
120    }
121
122    /// Fails what was awaited of a message whose frames went with a
123    /// session that is gone: each fragment sent but unacknowledged, or the
124    /// first when none was sent. Returns whether anything was awaited.
125    pub fn interrupt(&mut self, multicast: bool) -> bool {
126        if !self.awaited(multicast) {
127            return false;
128        }
129        let open = self.sent & !self.acked;
130        self.failed |= if open == 0 { 1 } else { open };
131        true
132    }
133
134    /// The latest word on one fragment.
135    pub fn state(&self, fragment: Option<u8>) -> Option<DeliveryState> {
136        let bit = 1u16 << fragment.unwrap_or(0).min(15);
137        if self.acked & bit != 0 {
138            Some(DeliveryState::Acked)
139        } else if self.failed & bit != 0 {
140            Some(DeliveryState::Failed)
141        } else if self.sent & bit != 0 {
142            Some(DeliveryState::Sent)
143        } else {
144            None
145        }
146    }
147}
