1//! The store kept in flash, as a `sequential-storage` map.
2//!
3//! [`FlashMapStore`] is the [`MemoryStore`] over a [`FlashMap`]: its records
4//! are worked on in memory, and each one changed is written to the map at
5//! the end of every step of a commit. Mounting reads the whole map once.
6//!
7//! The map is keyed by slot, not by record. A record keeps its slot while
8//! it lives; a record forgotten has its slot marked free, and the slot is
9//! given to the next new record. Marking a slot free is a write rather than
10//! a removal, since `sequential-storage` finds an item to remove by reading
11//! the whole partition; and since slots are reused, the free marks never
12//! outnumber the most records ever kept at once.
13//!
14//! Each item is the record's key, after its length, then the record. A free
15//! slot's item is empty.
16//!
17//! The map's calls are asynchronous, but over a blocking flash driver each
18//! finishes on its first poll. The store polls each once, and treats one
19//! that has not finished as a failed write.
20

41/// Slots whose place in flash the map remembers, so that recycling a page
42/// does not search for each of its items.
43const CACHED_SLOTS: usize = 4096;
44
45/// A store's records, kept in a `sequential-storage` map on `F`.
46pub struct FlashMap<F: NorFlash> {
47    map: Option<MapStorage<u32, F, HeapKeyPointerCache<u32>>>,
48    range: Range<u32>,
49    /// Where items are laid out on their way to and from flash.
50    buffer: &'static mut [u8],
51    /// Each record's slot, by its key.
52    slots: BTreeMap<Vec<u8>, u32>,
53    /// Slots marked free, to be given out again.
54    free: Vec<u32>,
55    /// The first slot never given out.
56    next_slot: u32,
57    /// An item, as it is written.
58    item: Vec<u8>,
59}
60
61impl<F: NorFlash> FlashMap<F> {
62    /// The map in `range` of `flash`, read and written through `buffer`.
63    ///
64    /// `buffer` holds the largest item: a message's record with its key,
65    /// about 1.8 KB. Some drivers want it word-aligned, and in memory they
66    /// can write from.
67    pub fn new(flash: F, range: Range<u32>, buffer: &'static mut [u8]) -> Self {
68        let mut map = Self {
69            map: None,
70            range,
71            buffer,
72            slots: BTreeMap::new(),
73            free: Vec::new(),
74            next_slot: 0,
75            item: Vec::new(),
76        };
77        map.open(flash);
78        map
79    }
80
81    fn open(&mut self, flash: F) {
82        let pages = (self.range.end - self.range.start) as usize / F::ERASE_SIZE;
83        self.map = Some(MapStorage::new(
84            flash,
85            MapConfig::new(self.range.clone()),
86            HeapKeyPointerCache::new(pages, CACHED_SLOTS),
87        ));
88    }
89
90    /// The flash the map is on.
91    pub fn flash_mut(&mut self) -> &mut F {
92        self.map().flash()
93    }
94
95    fn map(&mut self) -> &mut MapStorage<u32, F, HeapKeyPointerCache<u32>> {
96        self.map.as_mut().expect("the map is open")
97    }
98
99    fn store(&mut self, slot: u32, item: &[u8]) -> Result<(), StoreError> {
100        let map = self.map.as_mut().expect("the map is open");
101        now(map.store_item(self.buffer, &slot, &item))?.map_err(error)
102    }
103}

105impl<F: NorFlash> Durable for FlashMap<F> {
106    const KEEPS: bool = true;
107
108    fn write(&mut self, key: &[u8], record: Option<&[u8]>) -> Result<(), StoreError> {
109        let Some(record) = record else {
110            let Some(slot) = self.slots.remove(key) else {
111                return Ok(());
112            };
113            self.store(slot, &[])?;
114            self.free.push(slot);
115            return Ok(());
116        };
117        let slot = match self.slots.get(key) {
118            Some(&slot) => slot,
119            None => {
120                let slot = self.free.pop().unwrap_or_else(|| {
121                    self.next_slot += 1;
122                    self.next_slot - 1
123                });
124                self.slots.insert(key.to_vec(), slot);
125                slot
126            }
127        };
128        let mut item = core::mem::take(&mut self.item);
129        item.clear();
130        item.push(key.len() as u8);
131        item.extend_from_slice(key);
132        item.extend_from_slice(record);
133        let stored = self.store(slot, &item);
134        self.item = item;
135        stored
136    }
137
138    fn read(&mut self, each: &mut dyn FnMut(&[u8], &[u8])) -> Result<(), StoreError> {
139        // A fresh cache: what the old one remembers may predate a power
140        // cut.
141        let map = self.map.take().expect("the map is open");
142        let (flash, _) = map.destroy();
143        self.open(flash);
144        self.slots.clear();
145        self.free.clear();
146        self.next_slot = 0;
147
148        // Each slot's newest item comes last.
149        let mut newest: BTreeMap<u32, Vec<u8>> = BTreeMap::new();
150        let map = self.map.as_mut().expect("the map is open");
151        let mut items = now(map.fetch_all_items(self.buffer))?.map_err(error)?;
152        while let Some((slot, item)) = now(items.next::<&[u8]>(self.buffer))?.map_err(error)? {
153            newest.insert(slot, item.to_vec());
154        }
155
156        for (slot, item) in newest {
157            self.next_slot = self.next_slot.max(slot + 1);
158            let record = item.split_first().and_then(|(&len, rest)| {
159                (usize::from(len) <= rest.len()).then(|| rest.split_at(usize::from(len)))
160            });
161            match record {
162                Some((key, record)) if !key.is_empty() => {
163                    if let Some(older) = self.slots.insert(key.to_vec(), slot) {
164                        self.free.push(older);
165                    }
166                    each(key, record);
167                }
168                _ => self.free.push(slot),
169            }
170        }
171        Ok(())
172    }
173
174    fn clear(&mut self) -> Result<(), StoreError> {
175        now(self.map().erase_all())?.map_err(error)?;
176        self.slots.clear();
177        self.free.clear();
178        self.next_slot = 0;
179        Ok(())
180    }
181}
182
183fn error<E>(error: sequential_storage::Error<E>) -> StoreError {
184    match error {
185        sequential_storage::Error::FullStorage => StoreError::Full,
186        sequential_storage::Error::Corrupted { .. } => StoreError::Corrupt,
187        sequential_storage::Error::BufferTooSmall(_) => StoreError::BufferTooSmall,
188        _ => StoreError::Io,
189    }
190}
191
192/// `future`'s output, if it finishes on its first poll; a flash driver
193/// that blocks always does.
194fn now<T>(future: impl Future<Output = T>) -> Result<T, StoreError> {
195    let mut future = pin!(future);
196    match future
197        .as_mut()
198        .poll(&mut Context::from_waker(Waker::noop()))
199    {
200        Poll::Ready(output) => Ok(output),
201        Poll::Pending => Err(StoreError::Io),
202    }
203}

205/// A NOR flash in memory, for tests and simulators. Its bits only clear
206/// until their page is erased, and its power can be made to fail partway
207/// through a write.
208pub struct RamFlash {
209    bytes: Vec<u8>,
210    /// Words that may still be written, or pages erased, before power
211    /// fails.
212    budget: Option<usize>,
213    /// Words written and pages erased so far.
214    spent: usize,
215}
216
217/// What a [`RamFlash`] refuses.
218#[derive(Clone, Copy, Debug, PartialEq, Eq)]
219pub enum RamFlashError {
220    OutOfBounds,
221    NotAligned,
222    /// Its power has failed.
223    Off,
224}
225
226impl NorFlashError for RamFlashError {
227    fn kind(&self) -> NorFlashErrorKind {
228        match self {
229            Self::OutOfBounds => NorFlashErrorKind::OutOfBounds,
230            Self::NotAligned => NorFlashErrorKind::NotAligned,
231            Self::Off => NorFlashErrorKind::Other,
232        }
233    }
234}
235
236impl RamFlash {
237    /// `size` octets, erased.
238    pub fn new(size: usize) -> Self {
239        Self::from_bytes(alloc::vec![0xFF; size])
240    }
241
242    /// A flash holding `bytes`.
243    pub fn from_bytes(bytes: Vec<u8>) -> Self {
244        Self {
245            bytes,
246            budget: None,
247            spent: 0,
248        }
249    }
250
251    pub fn bytes(&self) -> &[u8] {
252        &self.bytes
253    }
254
255    /// Power fails once `words` more words are written or pages erased: the
256    /// write it fails in stops partway, and nothing more is written.
257    pub fn fail_after(&mut self, words: usize) {
258        self.budget = Some(words);
259    }
260
261    /// Power is back.
262    pub fn power_on(&mut self) {
263        self.budget = None;
264    }
265
266    /// Whether power has failed.
267    pub fn failed(&self) -> bool {
268        self.budget == Some(0)
269    }
270
271    /// Words written and pages erased so far.
272    pub fn spent(&self) -> usize {
273        self.spent
274    }
275
276    /// Spend one unit of the budget, if power holds.
277    fn spend(&mut self) -> Result<(), RamFlashError> {
278        match &mut self.budget {
279            Some(0) => return Err(RamFlashError::Off),
280            Some(left) => *left -= 1,
281            None => {}
282        }
283        self.spent += 1;
284        Ok(())
285    }
286
287    fn range(&self, offset: u32, len: usize, align: usize) -> Result<Range<usize>, RamFlashError> {
288        let start = offset as usize;
289        if !start.is_multiple_of(align) || !len.is_multiple_of(align) {
290            return Err(RamFlashError::NotAligned);
291        }
292        let end = start.checked_add(len).ok_or(RamFlashError::OutOfBounds)?;
293        if end > self.bytes.len() {
294            return Err(RamFlashError::OutOfBounds);
295        }
296        Ok(start..end)
297    }
298}
299
300impl ErrorType for RamFlash {
301    type Error = RamFlashError;
302}
303
304impl ReadNorFlash for RamFlash {
305    const READ_SIZE: usize = 4;
306
307    async fn read(&mut self, offset: u32, bytes: &mut [u8]) -> Result<(), Self::Error> {
308        let range = self.range(offset, bytes.len(), Self::READ_SIZE)?;
309        bytes.copy_from_slice(&self.bytes[range]);
310        Ok(())
311    }
312
313    fn capacity(&self) -> usize {
314        self.bytes.len()
315    }
316}
317
318impl NorFlash for RamFlash {
319    const WRITE_SIZE: usize = 4;
320    const ERASE_SIZE: usize = 4096;
321
322    async fn erase(&mut self, from: u32, to: u32) -> Result<(), Self::Error> {
323        let range = self.range(from, (to - from) as usize, Self::ERASE_SIZE)?;
324        for page in range.step_by(Self::ERASE_SIZE) {
325            self.spend()?;
326            self.bytes[page..page + Self::ERASE_SIZE].fill(0xFF);
327        }
328        Ok(())
329    }
330
331    async fn write(&mut self, offset: u32, bytes: &[u8]) -> Result<(), Self::Error> {
332        let range = self.range(offset, bytes.len(), Self::WRITE_SIZE)?;
333        for (at, word) in range
334            .step_by(Self::WRITE_SIZE)
335            .zip(bytes.chunks(Self::WRITE_SIZE))
336        {
337            self.spend()?;
338            for (cell, &bit) in self.bytes[at..at + Self::WRITE_SIZE].iter_mut().zip(word) {
339                *cell &= bit;
340            }
341        }
342        Ok(())
343    }
344}

473    /// Power failing at any word of a run of commits leaves a flash that
474    /// mounts, holds no identifier given out twice, and takes commits
475    /// again.
476    #[test]
477    fn power_fails_partway_through_a_write() {
478        // The words the whole run writes, measured on a run without a cut.
479        let mut probe = Flash.fresh();
480        let mut rig = Rig::new();
481        rig.say(&mut probe, direct(), "warm");
482        let before = probe.durable_mut().flash_mut().spent();
483        for n in 0..3 {
484            rig.say(&mut probe, group(), &alloc::format!("busy {n}"));
485        }
486        let words = probe.durable_mut().flash_mut().spent() - before;
487
488        for cut in (0..words).step_by(7) {
489            let mut store = Flash.fresh();
490            let mut rig = Rig::new();
491            let (_, first) = rig.say(&mut store, direct(), "warm");
492            store.durable_mut().flash_mut().fail_after(cut);
493            let mut kept = alloc::vec![first];
494            for n in 0..3 {
495                match rig.try_say(&mut store, group(), &alloc::format!("busy {n}")) {
496                    Some(id) => kept.push(id),
497                    None => break,
498                }
499            }
500            let mut flash = RamFlash::from_bytes(store.durable_mut().flash_mut().bytes().to_vec());
501            flash.power_on();
502            let mut mounted = mount(flash);
503            let (_, next) = Rig::new().say(&mut mounted, group(), "after");
504            assert!(kept.iter().all(|&id| id < next), "cut at {cut}");
505            assert!(mounted.message(first).unwrap().is_some(), "cut at {cut}");
506        }
507    }
