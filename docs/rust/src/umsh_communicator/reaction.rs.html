1//! Reactions: emotes about a message, folded into it.
2//!
3//! On the wire a reaction is a status message regarding its target, its
4//! body one of UMSH's emote tokens; an empty body withdraws the sender's
5//! earlier one. The store keeps the latest from each sender inside the
6//! target, and counts them by token when the message is read.
7
8use heapless::Vec;
9
10/// One of UMSH's emote tokens.
11#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
12pub enum Reaction {
13    Heart,
14    ThumbsUp,
15    ThumbsDown,
16    Laugh,
17    Exclaim,
18    Question,
19}
20
21impl Reaction {
22    /// The palette, in the order it is offered.
23    pub const ALL: [Self; 6] = [
24        Self::Heart,
25        Self::ThumbsUp,
26        Self::ThumbsDown,
27        Self::Laugh,
28        Self::Exclaim,
29        Self::Question,
30    ];
31
32    /// The emote's body on the air.
33    pub const fn token(self) -> &'static str {
34        match self {
35            Self::Heart => "<3",
36            Self::ThumbsUp => "+1",
37            Self::ThumbsDown => "-1",
38            Self::Laugh => "ha",
39            Self::Exclaim => "!",
40            Self::Question => "?",
41        }
42    }
43
44    /// The emote a reaction's body stands for.
45    ///
46    /// Reading is looser than writing, as it is on iOS: the tokens, their
47    /// longer spellings, and the emoji other clients send all count, and
48    /// a body that is none of them counts as its first character. Anything
49    /// else stands for no emote this palette has.
50    pub fn from_body(body: &str) -> Option<Self> {
51        let trimmed = body.trim();
52        Self::spelled(trimmed).or_else(|| {
53            let first = trimmed.chars().next()?;
54            let mut buf = [0u8; 4];
55            Self::spelled(first.encode_utf8(&mut buf))
56        })
57    }
58
59    fn spelled(body: &str) -> Option<Self> {
60        let mut lower = [0u8; 16];
61        let body = ascii_lowercase(body, &mut lower)?;
62        Some(match body {
63            "<3" | "\u{2665}" | "\u{2665}\u{fe0e}" | "\u{2764}" | "\u{2764}\u{fe0f}"
64            | "\u{1fa77}" => Self::Heart,
65            "+1" | "\u{1f44d}" => Self::ThumbsUp,
66            "-1" | "\u{1f44e}" => Self::ThumbsDown,
67            "ha" | "haha" | "hahaha" | "lol" | "lmao" | "\u{1f923}" => Self::Laugh,
68            "!" | "!!" | "\u{2757}" | "\u{2755}" | "\u{203c}\u{fe0f}" | "\u{203c}" => Self::Exclaim,
69            "?" | "??" | "\u{2753}" | "\u{2754}" => Self::Question,
70            _ => match strip_skin_tone(body) {
71                "\u{1f44d}" => Self::ThumbsUp,
72                "\u{1f44e}" => Self::ThumbsDown,
73                _ => return None,
74            },
75        })
76    }
77}
78
79/// `body` lowercased, when it is short enough to be a spelling at all.
80fn ascii_lowercase<'b>(body: &str, out: &'b mut [u8; 16]) -> Option<&'b str> {
81    let bytes = body.as_bytes();
82    let out = out.get_mut(..bytes.len())?;
83    out.copy_from_slice(bytes);
84    out.make_ascii_lowercase();
85    core::str::from_utf8(out).ok()
86}
87
88/// A thumb without the skin-tone modifier after it.
89fn strip_skin_tone(body: &str) -> &str {
90    let mut chars = body.char_indices();
91    match (chars.next(), chars.next(), chars.next()) {
92        (Some(_), Some((at, '\u{1f3fb}'..='\u{1f3ff}')), None) => &body[..at],
93        _ => body,
94    }
95}

97/// Everyone who reacted to a message one way.
98#[derive(Clone, Copy, Debug, PartialEq, Eq)]
99pub struct Reacted {
100    pub reaction: Reaction,
101    pub count: u16,
102    /// One of them is this node.
103    pub mine: bool,
104}
105
106/// A message's reactions, one entry per emote, in the order each emote
107/// first appeared.
108pub type Reactions = Vec<Reacted, { Reaction::ALL.len() }>;

110#[cfg(test)]
111mod tests {
112    use super::*;
113
114    #[test]
115    fn every_token_reads_as_itself() {
116        for reaction in Reaction::ALL {
117            assert_eq!(Reaction::from_body(reaction.token()), Some(reaction));
118        }
119    }
120
121    #[test]
122    fn other_spellings_read_as_the_emote_they_mean() {
123        assert_eq!(
124            Reaction::from_body("\u{2764}\u{fe0f}"),
125            Some(Reaction::Heart)
126        );
127        assert_eq!(Reaction::from_body(" LOL "), Some(Reaction::Laugh));
128        assert_eq!(
129            Reaction::from_body("\u{1f44d}\u{1f3fd}"),
130            Some(Reaction::ThumbsUp)
131        );
132        assert_eq!(Reaction::from_body("??"), Some(Reaction::Question));
133        // A sentence counts as its first character.
134        assert_eq!(Reaction::from_body("!nice"), Some(Reaction::Exclaim));
135        assert_eq!(Reaction::from_body("nice"), None);
136        assert_eq!(Reaction::from_body(""), None);
137    }
138}
