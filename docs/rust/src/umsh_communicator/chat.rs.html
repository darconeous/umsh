52/// The most of a message a reply's quote keeps, in octets: more than one

56/// How many outbound checkpoints seed the engine when the chats start.
57const RESTORED_CHECKPOINTS: usize = 32;

59/// How many times one input's outputs are settled. Answering a resend
60/// request or failing a frame can produce more, which settle in turn.
61const SETTLE_ROUNDS: usize = 4;

63/// How many of the chats' frames the device node holds at once: half its
64/// transmit queue, so a long message leaves room for what else it sends.

67/// How far a message travels when nothing narrower applies: as far as

76/// What the node needs of the board's cryptography, which it does not
77/// carry itself.
78#[derive(Clone, Copy)]
79pub struct ChannelCrypto {
80    /// The keys of `public` and `EMERGENCY`.
81    pub well_known: WellKnownChannels,
82    /// A channel's tag, as `umsh_crypto::CryptoEngine::derive_channel_tag`
83    /// derives it.
84    pub tag: fn(&ChannelKey) -> ChannelTag,
85    /// A named channel's key, as
86    /// `umsh_crypto::CryptoEngine::derive_named_channel_key` derives it, or
87    /// `None` for a name that names no channel.
88    pub named: fn(&str) -> Option<ChannelKey>,

91impl ChannelCrypto {
92    /// The channel keys a communicator's device lists hold after a reset
93    /// with none kept: `public` and `EMERGENCY` (`ulcp-device.md`, Lists
94    /// Held by the Device's Store). The board hands them to the session as
95    /// it holds the lists.
96    pub fn starting_keys(&self) -> [[u8; 32]; 2] {
97        [self.well_known.public.0, self.well_known.emergency.0]
98    }

100    /// The software crypto back end's derivations.
101    #[cfg(feature = "software-crypto")]
102    pub fn software() -> Self {
103        fn engine() -> umsh_crypto::CryptoEngine<
104            umsh_crypto::software::SoftwareAes,
105            umsh_crypto::software::SoftwareSha256,
106        > {
107            umsh_crypto::CryptoEngine::new(
108                umsh_crypto::software::SoftwareAes,
109                umsh_crypto::software::SoftwareSha256,
110            )
111        }
112        Self {
113            well_known: WellKnownChannels::derive(),
114            tag: |key| engine().derive_channel_tag(key),
115            named: |name| engine().derive_named_channel_key(name).ok(),
116        }
117    }

119    fn builtin(&self, key: &ChannelKey) -> Option<BuiltinChannel> {
120        if self.well_known.is_public(key) {
121            Some(BuiltinChannel::Public)
122        } else if self.well_known.is_emergency(key) {
123            Some(BuiltinChannel::Emergency)
124        } else {
125            None
126        }
127    }

206    /// Starts the chats as `local_key`, the device identity, carrying on
207    /// each outbound stream from where the store left it. What the store
208    /// shows on its way went with the last session, and has failed.
209    pub fn start(
210        &mut self,
211        local_key: PublicKey,
212        seeds: Seeds,
213        now_ms: u64,
214        events: &mut dyn FnMut(NodeEvent),
215    ) {
216        if self.facade.is_some() {
217            return;
218        }
219        // Left unwritten, those messages read as sending until retried.
220        _ = self.store.interrupt();
221        let mut facade = ChatFacade::new(
222            DirectChannelProfile,
223            local_key,
224            EngineConfig::default(),
225            seeds,
226        );
227        let checkpoints: Vec<_> = self
228            .store
229            .recent_checkpoints(RESTORED_CHECKPOINTS)
230            .collect();
231        facade.restore(&checkpoints, now_ms);
232        self.facade = Some(facade);
233        self.seeds = seeds;
234        let held = core::mem::take(&mut self.held);
235        self.hold(held.into_iter(), events);
236    }

238    /// The device holds exactly `tags`.
239    pub fn hold(
240        &mut self,
241        tags: impl Iterator<Item = ChannelTag>,
242        events: &mut dyn FnMut(NodeEvent),
243    ) {
244        self.held.clear();
245        self.held.extend(tags);
246        let Some(facade) = &mut self.facade else {
247            return;
248        };
249        let gone: Vec<ChannelTag> = facade
250            .channels()
251            .tags()
252            .filter(|tag| !self.held.contains(tag))
253            .collect();
254        for tag in &gone {
255            facade.channels_mut().remove(tag);
256        }
257        let well_known = self.crypto.well_known;
258        for key in [well_known.public, well_known.emergency] {
259            let tag = (self.crypto.tag)(&key);
260            if self.held.contains(&tag) {
261                self.register(tag, key, events);
262            }
263        }
264    }

283    /// Holds the channel `key` names in the facade, and keeps a chat for
284    /// it.
285    fn register(&mut self, tag: ChannelTag, key: ChannelKey, events: &mut dyn FnMut(NodeEvent)) {
286        let Some(facade) = &mut self.facade else {
287            return;
288        };
289        if facade.channels().contains(&tag) {
290            return;
291        }
292        facade.channels_mut().register(tag, key, None);
293        if !self
294            .store
295            .channels()
296            .flatten()
297            .any(|channel| channel.tag == tag)
298        {
299            let builtin = self.crypto.builtin(&key);
300            let (name, access) = match builtin {
301                Some(builtin @ BuiltinChannel::Public) => {
302                    (record::name(builtin.name()), ChannelAccess::Named)
303                }
304                Some(builtin @ BuiltinChannel::Emergency) => {
305                    (record::name(builtin.name()), ChannelAccess::Clear)
306                }
307                None => (channels::default_name(&key), ChannelAccess::Private),
308            };
309            let change = ChannelChange::Add {
310                name: name.as_str(),
311                access,
312                builtin,
313            };
314            if self.store.update_channel(&tag, change).is_err() {
315                return;
316            }
317        }
318        let conversation = ConversationKey::ChannelGroup { channel: tag };
319        if self.store.chat_for(&conversation).is_none() {
320            let open = ChatChange::Open {
321                conversation,
322                name: None,
323            };
324            if let Ok(chat) = self.store.update_chat(ChatId(0), open) {
325                events(NodeEvent::ChatChanged(chat.0));
326            }
327        }
328    }

332    /// Takes one received text frame, kept at `time`, the wall-clock
333    /// seconds when known.
334    pub fn receive(
335        &mut self,
336        inbound: InboundText,
337        now_ms: u64,
338        time: Option<u32>,
339        events: &mut dyn FnMut(NodeEvent),
340    ) {
341        // The key that authenticated the frame is one the node holds.
342        if let Some(key) = inbound.channel_key() {
343            let tag = (self.crypto.tag)(&key);
344            self.register(tag, key, events);
345        }
346        let Some(facade) = &mut self.facade else {
347            return;
348        };
349        let Some(source) = inbound.source(facade.channels()) else {
350            return;
351        };
352        if let Some(key) = source.full_key() {
353            self.hear(&key, &inbound.rx, now_ms, events);
354        }
355        let Some(facade) = &mut self.facade else {
356            return;
357        };
358        let at = inbound.received_at_ms.unwrap_or(now_ms);
359        // A frame the engine cannot take is dropped; its diagnostic drains
360        // with everything else.
361        let _ = facade.receive(source, &inbound.payload, at);
362        let link = LinkInfo {
363            rssi_dbm: inbound.rx.rssi_dbm,
364            snr_centibels: inbound.rx.snr_centibels,
365            hop_count: inbound.rx.hop_count,
366            authenticated: inbound.rx.source_authenticated,
367        };
368        self.settle(Some(link), now_ms, time, events);
369    }

371    /// Advances the engine's timers: reassembly, notification, repair.
372    pub fn tick(&mut self, now_ms: u64, time: Option<u32>, events: &mut dyn FnMut(NodeEvent)) {
373        let Some(facade) = &mut self.facade else {
374            return;
375        };
376        facade.tick(now_ms);
377        self.settle(None, now_ms, time, events);
378        self.lapse_links(now_ms, events);
379    }

381    /// Commits what the facade produced and answers what it asked. `link`
382    /// describes the frame behind the first round's outputs.
383    fn settle(
384        &mut self,
385        mut link: Option<LinkInfo>,
386        now_ms: u64,
387        time: Option<u32>,
388        events: &mut dyn FnMut(NodeEvent),
389    ) {
390        for _ in 0..SETTLE_ROUNDS {
391            let Some(facade) = &mut self.facade else {
392                return;
393            };
394            let drain = facade.drain();
395            if drain.is_empty() {
396                return;
397            }
398            self.outbox.enqueue(drain.outbound);
399            self.commit(&drain.outputs, link.take(), time, events);
400            for output in &drain.outputs {
401                match output {
402                    ChatOutput::Lookup(lookup) => {
403                        let mut payload = [0u8; MAX_FRAME];
404                        let result = match self.store.archive(&lookup.key, &mut payload) {
405                            Ok(Some(payload)) => ArchiveResult::Found { payload },
406                            _ => ArchiveResult::Unknown,
407                        };
408                        if let Some(facade) = &mut self.facade {
409                            facade.archive_result(lookup.request_id, result, now_ms);
410                        }
411                    }
412                    ChatOutput::SenderResolved(_) => events(NodeEvent::IdentitiesLearned),
413                    _ => {}
414                }
415            }
416        }
417    }

419    fn commit(
420        &mut self,
421        outputs: &[ChatOutput],
422        link: Option<LinkInfo>,
423        time: Option<u32>,
424        events: &mut dyn FnMut(NodeEvent),
425    ) {
426        let committed = {
427            let handles = &self.handles;
428            let resolve = |handle| handles.get(&handle).copied();
429            let mut batch = Batch::new(time, &resolve);
430            let mut kept = false;
431            for output in outputs {
432                kept |= batch.push_chat(output, link);
433            }
434            if !kept {
435                return;
436            }
437            // A received message the store cannot keep is lost to the
438            // transcript; there is nothing to take back.
439            let Ok(committed) = self.store.commit(&batch) else {
440                return;
441            };
442            committed
443        };
444        self.handles.extend(committed.inserted.iter().copied());
445        for output in outputs {
446            if let ChatOutput::Delivery(delivery) = output
447                && let Some(id) = self.handles.get(&delivery.handle)
448            {
449                self.retrying.remove(id);
450            }
451        }
452
453        // What arrives in full, inbound and meant to be noticed, is
454        // announced as an arrival; anything else that changed, as a change.
455        let arrived: Vec<MessageId> = outputs
456            .iter()
457            .filter_map(|output| match output {
458                ChatOutput::Mutation(mutation) => Some(mutation),
459                _ => None,
460            })
461            .filter(|mutation| match &mutation.change {
462                Change::Insert(insert) => {
463                    insert.direction == engine::Direction::Inbound
464                        && insert.presence == engine::Presence::Present
465                        && insert.notify
466                }
467                Change::UpdateBody { notify, .. } => *notify,
468                Change::Edit { .. } | Change::Delete { .. } => false,
469            })
470            .filter_map(|mutation| self.handles.get(&mutation.handle).copied())
471            .collect();
472        self.announce(&committed, &arrived, events);
473    }

475    /// Tells the interface what a commit changed. The messages in
476    /// `arrived` are announced as arrivals.
477    fn announce(
478        &self,
479        committed: &Committed,
480        arrived: &[MessageId],
481        events: &mut dyn FnMut(NodeEvent),
482    ) {
483        for &id in &committed.messages {
484            let Ok(Some(record)) = self.store.message(id) else {
485                continue;
486            };
487            let (chat, message) = (record.chat.0, id.0);
488            events(if arrived.contains(&id) {
489                NodeEvent::Arrived { chat, message }
490            } else {
491                NodeEvent::MessageChanged { chat, message }
492            });
493        }
494        for chat in &committed.chats {
495            events(NodeEvent::ChatChanged(chat.0));
496        }
497    }

510    /// Keeps `text`, in reply to `reply_to` if it is one, as the newest
511    /// message of `chat`, and queues it to go. A reply whose original
512    /// can no longer be named goes as a plain message. With
513    /// `forward_secret`, as the composer showed it, it goes to the chat's
514    /// peer under a session with forward secrecy or not at all; without,
515    /// never under one.
516    pub fn send(
517        &mut self,
518        chat: model::ChatId,
519        text: &str,
520        reply_to: Option<model::MessageId>,
521        forward_secret: bool,
522        now_ms: u64,
523        time: Option<u32>,
524        events: &mut dyn FnMut(NodeEvent),
525    ) -> Option<Message> {
526        let conversation = self.conversation(chat)?;
527        let intent = match reply_to.and_then(|id| self.regarding(id, &conversation)) {
528            Some(regarding) => ComposeIntent::Reply {
529                body: text,
530                regarding,
531                status: false,
532            },
533            None => ComposeIntent::Text {
534                body: text,
535                status: false,
536            },
537        };
538        // A peer who prefers forward secrecy has a session asked for
539        // before anything is said, when none is.
540        if let ConversationKey::Direct { peer } = conversation {
541            self.keep_secrecy(peer);
542        }
543        let committed = self.compose(conversation, intent, forward_secret, now_ms, time, events)?;
544        let &(_, id) = committed.inserted.first()?;
545        _ = self
546            .store
547            .update_chat(ChatId(chat), ChatChange::Draft(None));
548        let record = self.store.message(id).ok()??;
549        self.message_of(&record, &conversation)
550    }

552    /// Keeps `reaction` as mine to `message` in `chat`, and sends it;
553    /// `None` withdraws mine.
554    pub fn react(
555        &mut self,
556        chat: model::ChatId,
557        message: model::MessageId,
558        reaction: Option<model::Reaction>,
559        now_ms: u64,
560        time: Option<u32>,
561        events: &mut dyn FnMut(NodeEvent),
562    ) {
563        let Some(conversation) = self.conversation(chat) else {
564            return;
565        };
566        let Some(regarding) = self.regarding(message, &conversation) else {
567            return;
568        };
569        let body = reaction.map_or("", |reaction| emote(reaction).token());
570        let intent = ComposeIntent::Reply {
571            body,
572            regarding,
573            status: true,
574        };
575        let forward_secret = self.holds_secrecy(&conversation);
576        _ = self.compose(conversation, intent, forward_secret, now_ms, time, events);
577    }

579    /// Sends `message` in `chat`, which was to go under a session with
580    /// forward secrecy and went unacknowledged, again without one, as the
581    /// user chose to. Returns whether it is on its way.
582    pub fn retry_without_secrecy(
583        &mut self,
584        chat: model::ChatId,
585        message: model::MessageId,
586        now_ms: u64,
587        time: Option<u32>,
588        events: &mut dyn FnMut(NodeEvent),
589    ) -> bool {
590        let Ok(Some(record)) = self.store.message(MessageId(message)) else {
591            return false;
592        };
593        if record.chat.0 != chat || !record.flags.forward_secret {
594            return false;
595        }
596        let resolve = |_| None;
597        let mut batch = Batch::new(time, &resolve);
598        batch.secrecy(Subject::Kept(record.id), false);
599        let Ok(committed) = self.store.commit(&batch) else {
600            return false;
601        };
602        self.announce(&committed, &[], events);
603        self.retry(chat, message, now_ms, time, events)
604    }

606    /// Sends again each fragment of `message` in `chat` that went
607    /// unacknowledged, while its wire ID is still the message's in its
608    /// stream, before a restart or since. Returns whether any is on its
609    /// way.
610    pub fn retry(
611        &mut self,
612        chat: model::ChatId,
613        message: model::MessageId,
614        now_ms: u64,
615        time: Option<u32>,
616        events: &mut dyn FnMut(NodeEvent),
617    ) -> bool {
618        let Some(conversation) = self.conversation(chat) else {
619            return false;
620        };
621        let Ok(Some(record)) = self.store.message(MessageId(message)) else {
622            return false;
623        };
624        let Some(message_id) = record.wire_id else {
625            return false;
626        };
627        if record.chat.0 != chat || record.direction != engine::Direction::Outbound {
628            return false;
629        }
630        // One to go under forward secrecy asks for a session, if none is.
631        if let ConversationKey::Direct { peer } = conversation
632            && record.flags.forward_secret
633        {
634            self.keep_secrecy(peer);
635        }
636        let known = self.handle_of(record.id);
637        let Some(facade) = &mut self.facade else {
638            return false;
639        };
640        // A message sent before a restart is taken back under its wire ID.
641        let handle = match known {
642            Some(handle) => handle,
643            None => {
644                let stored = self.store.checkpoint(&conversation);
645                let Some(handle) = facade.adopt(
646                    conversation,
647                    message_id,
648                    record.epoch,
649                    stored.as_ref(),
650                    now_ms,
651                ) else {
652                    return false;
653                };
654                self.handles.insert(handle, record.id);
655                handle
656            }
657        };
658        let mut payload = [0u8; MAX_FRAME];
659        let mut queued = false;
660        // An unfragmented message is archived whole; a fragmented one by
661        // fragment, from the first.
662        let whole = ArchiveKey {
663            conversation,
664            message_id,
665            fragment: None,
666        };
667        let keys = core::iter::once(whole).chain((0..FRAGMENTS).map(|index| ArchiveKey {
668            fragment: Some(index),
669            ..whole
670        }));
671        for key in keys {
672            let bit = 1u16 << key.fragment.unwrap_or(0);
673            match self.store.archive(&key, &mut payload) {
674                Ok(Some(frame)) if record.delivery.acked & bit == 0 => {
675                    queued |= facade.retransmit(&key, handle, frame);
676                }
677                Ok(Some(_)) => {}
678                _ if key.fragment.is_some() => break,
679                _ => {}
680            }
681            if key.fragment.is_none() && queued {
682                break;
683            }
684        }
685        if queued {
686            self.retrying.insert(record.id);
687            events(NodeEvent::MessageChanged { chat, message });
688        }
689        self.settle(None, now_ms, time, events);
690        queued
691    }

693    /// Composes `intent` into `conversation`, keeps what it makes, and
694    /// queues its frames. `None` when it cannot be composed or kept.
695    fn compose(
696        &mut self,
697        conversation: ConversationKey,
698        intent: ComposeIntent<'_>,
699        forward_secret: bool,
700        now_ms: u64,
701        time: Option<u32>,
702        events: &mut dyn FnMut(NodeEvent),
703    ) -> Option<Committed> {
704        let facade = self.facade.as_mut()?;
705        self.token = self.token.wrapping_add(1);
706        // The stored checkpoint carries the conversation's sequence on when
707        // the engine no longer remembers it.
708        let stored = self.store.checkpoint(&conversation);
709        let composed = facade
710            .compose(conversation, self.token, intent, stored.as_ref(), now_ms)
711            .ok()?;
712        let kept = {
713            let handles = &self.handles;
714            let resolve = |handle| handles.get(&handle).copied();
715            let mut batch = Batch::new(time, &resolve);
716            batch.push_composed(&composed);
717            // Kept with the message, so that it is never found without.
718            if forward_secret && matches!(conversation, ConversationKey::Direct { .. }) {
719                for mutation in &composed.mutations {
720                    if matches!(mutation.change, Change::Insert(_)) {
721                        batch.secrecy(Subject::Handle(mutation.handle), true);
722                    }
723                }
724            }
725            self.store.commit(&batch)
726        };
727        let Ok(committed) = kept else {
728            self.reject(composed, now_ms, time, events);
729            return None;
730        };
731        self.handles.extend(committed.inserted.iter().copied());
732        if let Some(facade) = &mut self.facade
733            && let Ok(outbound) = facade.commit(composed)
734        {
735            self.outbox.enqueue(outbound);
736        }
737        self.announce(&committed, &[], events);
738        Some(committed)
739    }

741    /// Gives up on a composed message the store could not keep. The engine
742    /// is rebuilt from what the store holds, as a new session: everything
743    /// on its way fails first, since no word of it will be taken.
744    fn reject(
745        &mut self,
746        composed: umsh_text::chat::PendingBatch,
747        now_ms: u64,
748        time: Option<u32>,
749        events: &mut dyn FnMut(NodeEvent),
750    ) {
751        let Some(facade) = &mut self.facade else {
752            return;
753        };
754        // Their receipts are not canceled: whatever still goes out, its
755        // word is the old session's, and ignored.
756        _ = self.outbox.fail_all(facade, now_ms);
757        let checkpoints: Vec<_> = self
758            .store
759            .recent_checkpoints(RESTORED_CHECKPOINTS)
760            .collect();
761        self.seeds.session_id = self.seeds.session_id.wrapping_add(1).max(1);
762        let Ok(drain) = facade.reject(composed, &checkpoints, self.seeds, now_ms) else {
763            return;
764        };
765        self.commit(&drain.outputs, None, time, events);
766        self.handles.clear();
767        self.retrying.clear();
768    }

770    /// The next frame for the device node, or `None` while nothing may go.
771    pub fn depart(
772        &mut self,
773        now_ms: u64,
774        time: Option<u32>,
775        events: &mut dyn FnMut(NodeEvent),
776    ) -> Option<Departure> {
777        loop {
778            let facade = self.facade.as_mut()?;
779            match self.outbox.depart(facade.channels(), &self.policy)? {
780                Ok(mut departure) => {
781                    self.bind_secrecy(&mut departure);
782                    return Some(departure);
783                }
784                Err(undeliverable) => {
785                    undeliverable.fail(facade, now_ms);
786                    self.settle(None, now_ms, time, events);
787                }
788            }
789        }
790    }

792    /// Holds a frame to a peer to what was decided when its message was
793    /// sent: under a session with forward secrecy, or never under one.
794    fn bind_secrecy(&self, departure: &mut Departure) {
795        if departure.peer().is_none() {
796            return;
797        }
798        let Some(record) = departure
799            .message()
800            .and_then(|handle| self.handles.get(&handle))
801            .and_then(|&id| self.store.message(id).ok().flatten())
802        else {
803            return;
804        };
805        departure.require_forward_secrecy(record.flags.forward_secret);
806    }

808    /// Takes how the device node took `departure`. Returns whether it has
809    /// room for more.
810    pub fn departed(
811        &mut self,
812        departure: Departure,
813        sent: Sent,
814        now_ms: u64,
815        time: Option<u32>,
816        events: &mut dyn FnMut(NodeEvent),
817    ) -> bool {
818        match self.outbox.departed(departure, sent, now_ms) {
819            Departed::Sent => true,
820            Departed::Failed(undeliverable) => {
821                if let Some(facade) = &mut self.facade {
822                    undeliverable.fail(facade, now_ms);
823                }
824                self.settle(None, now_ms, time, events);
825                true
826            }
827            Departed::Busy => false,
828        }
829    }

946    /// How `message` reached this device, and who reacted to it: this
947    /// node drawn as `mine`.
948    pub fn details(
949        &self,
950        chat: model::ChatId,
951        message: model::MessageId,
952        utc_offset_min: i16,
953        mine: &AvatarStyle,
954    ) -> Option<Details> {
955        let record = self.store.message(MessageId(message)).ok()??;
956        if record.chat.0 != chat {
957            return None;
958        }
959        let mut details = Details::default();
960        if let Some(time) = record.time {
961            let local = i64::from(time) + i64::from(utc_offset_min) * 60;
962            let minutes = local.div_euclid(60);
963            _ = write!(
964                details.time,
965                "{:02}:{:02}",
966                minutes.div_euclid(60).rem_euclid(24),
967                minutes.rem_euclid(60)
968            );
969        }
970        if let Some(link) = record.link {
971            details.hops = link.hop_count;
972            details.rssi = link.rssi_dbm;
973            details.snr = link.snr_centibels;
974            details.authenticated = link.authenticated;
975        }
976        let conversation = self.conversation(chat)?;
977        let reactors: Vec<_> = self.store.reactors(MessageId(message)).collect();
978        // The latest, as many as are shown.
979        let skip = reactors.len().saturating_sub(model::MAX_REACTORS);
980        for reactor in &reactors[skip..] {
981            let shown = if reactor.mine {
982                Some((None, mine.clone()))
983            } else {
984                self.reactor(reactor.sender, &conversation)
985                    .and_then(|(name, avatar)| {
986                        Some((Some(SharedText::new(self.heap, &name)?), avatar))
987                    })
988            };
989            if let Some((name, avatar)) = shown {
990                _ = details.reactors.push(model::Reactor {
991                    name,
992                    avatar,
993                    reaction: reaction(reactor.reaction),
994                });
995            }
996        }
997        Some(details)
998    }

1000    /// Who reacted to a message in `conversation`, as its details show
1001    /// them: the peer of a direct chat as the chat shows them; a channel's
1002    /// member by the name they are known by, or else by their address.
1003    fn reactor(
1004        &self,
1005        sender: SenderScope,
1006        conversation: &ConversationKey,
1007    ) -> Option<(String<32>, AvatarStyle)> {
1008        let ConversationKey::ChannelGroup { channel } = conversation else {
1009            let chat = self.store.chat_for(conversation)?;
1010            let record = self
1011                .store
1012                .chats()
1013                .flatten()
1014                .find(|record| record.id == chat)?;
1015            let (name, avatar, _) = self.face(&record)?;
1016            return Some((name, avatar));
1017        };
1018        let (hint, key) = match sender {
1019            SenderScope::Peer(key) => (key.hint(), Some(key)),
1020            SenderScope::ClaimedMember(hint) => (
1021                hint,
1022                self.facade
1023                    .as_ref()
1024                    .and_then(|facade| facade.member_key(channel, &hint))
1025                    .or_else(|| self.only_peer_with(hint)),
1026            ),
1027            SenderScope::Local => return None,
1028        };
1029        let mut lettering = String::<8>::new();
1030        let shape = match key {
1031            Some(key) => {
1032                _ = lettering.push_str(address(&key.0).get(..4).unwrap_or_default());
1033                PeerShape::Circle
1034            }
1035            None => {
1036                _ = write!(lettering, "{hint}");
1037                PeerShape::Hexagon
1038            }
1039        };
1040        let named = key
1041            .and_then(|key| self.store.peer(&key).ok().flatten())
1042            .and_then(|kept| kept.name.as_deref().map(cut));
1043        let name = named.unwrap_or_else(|| cut(&unnamed_member(&lettering)));
1044        Some((name, AvatarStyle::peer(hint.0, &lettering, shape)))
1045    }

1107    /// How a chat is shown: its name, its avatar, and whom it is with.
1108    fn face(&self, record: &ChatRecord) -> Option<(String<32>, AvatarStyle, ChatKind)> {
1109        match record.conversation {
1110            ConversationKey::ChannelGroup { channel } => {
1111                let kept = self
1112                    .store
1113                    .channels()
1114                    .flatten()
1115                    .find(|kept| kept.tag == channel)?;
1116                // A built-in channel is shown by its own name, whatever was
1117                // kept for it.
1118                let name = record
1119                    .name
1120                    .as_deref()
1121                    .or(kept.builtin.map(BuiltinChannel::name))
1122                    .unwrap_or(kept.name.as_str());
1123                let builtin = kept.builtin.map(|builtin| match builtin {
1124                    BuiltinChannel::Public => avatar::BuiltinChannel::Public,
1125                    BuiltinChannel::Emergency => avatar::BuiltinChannel::Emergency,
1126                });
1127                let tint = [channel.0[0], channel.0[1], channel.0[2]];
1128                let access = match kept.access {
1129                    ChannelAccess::Named => model::ChannelAccess::Named,
1130                    ChannelAccess::Private => model::ChannelAccess::Private,
1131                    ChannelAccess::Clear => model::ChannelAccess::Clear,
1132                };
1133                Some((
1134                    cut(name),
1135                    AvatarStyle::channel(builtin, name, tint),
1136                    ChatKind::Channel { access },
1137                ))
1138            }
1139            ConversationKey::Direct { peer } | ConversationKey::ChannelDirect { peer, .. } => {
1140                let kept = self.store.peer(&peer).ok().flatten();
1141                let lettering = address(&peer.0);
1142                let lettering = lettering.get(..4).unwrap_or_default();
1143                let name = match record.name.as_deref() {
1144                    Some(name) => cut(name),
1145                    None => match kept.as_ref().and_then(|kept| kept.name.as_deref()) {
1146                        Some(name) => cut(name),
1147                        None => cut(lettering),
1148                    },
1149                };
1150                let hint = peer.hint().0;
1151                let hops = self.links.get(&peer.0).and_then(Link::hops);
1152                Some((
1153                    name,
1154                    AvatarStyle::peer(hint, lettering, shape(kept.as_ref())),
1155                    ChatKind::Peer { hops },
1156                ))
1157            }
1158            ConversationKey::Room { .. } => None,
1159        }
1160    }

1162    fn message_of(
1163        &self,
1164        record: &MessageRecord,
1165        conversation: &ConversationKey,
1166    ) -> Option<Message> {
1167        let (text, invite) = {
1168            let mut body = self.body.borrow_mut();
1169            let body = self.store.body(record.id, &mut body[..]).ok()??;
1170            let invite = (record.message_type == MessageType::Basic && !record.flags.deleted)
1171                .then(|| self.invite_in(body))
1172                .flatten();
1173            (SharedText::new(self.heap, body)?, invite)
1174        };
1175        let direction = match record.direction {
1176            engine::Direction::Inbound => model::Direction::Inbound,
1177            engine::Direction::Outbound => model::Direction::Outbound,
1178        };
1179        let channel = matches!(conversation, ConversationKey::ChannelGroup { .. });
1180        let delivery = match direction {
1181            model::Direction::Inbound => model::Delivery::Delivered,
1182            model::Direction::Outbound => {
1183                let delivery = record.delivery;
1184                if self.retrying.contains(&record.id) {
1185                    model::Delivery::Sending
1186                } else if delivery.failed != 0 {
1187                    model::Delivery::Unconfirmed
1188                } else if delivery.acked != 0 && delivery.sent & !delivery.acked == 0 {
1189                    model::Delivery::Delivered
1190                } else if delivery.sent != 0 && channel {
1191                    model::Delivery::Sent
1192                } else {
1193                    model::Delivery::Sending
1194                }
1195            }
1196        };
1197        let reactions = record
1198            .reactions
1199            .iter()
1200            .map(|reacted| model::Reacted {
1201                reaction: reaction(reacted.reaction),
1202                count: u8::try_from(reacted.count).unwrap_or(u8::MAX),
1203                mine: reacted.mine,
1204            })
1205            .collect();
1206        let presence = match record.presence {
1207            engine::Presence::Present => Presence::Present,
1208            engine::Presence::GapPending => Presence::PendingRepair,
1209            engine::Presence::Unavailable => Presence::Unavailable,
1210        };
1211        Some(Message {
1212            id: record.id.0,
1213            direction,
1214            sender: self.sender(record, conversation),
1215            text,
1216            delivery,
1217            reactions,
1218            invite,
1219            reply: record
1220                .regarding
1221                .filter(|_| record.message_type == MessageType::Basic)
1222                .and_then(|regarding| self.quote(regarding, conversation)),
1223            time: record.time,
1224            presence,
1225            edited: record.flags.edited,
1226            deleted: record.flags.deleted,
1227            late: record.flags.late,
1228            forward_secret: record.flags.forward_secret,
1229            parts: match record.status {
1230                engine::CompletionStatus::Complete => None,
1231                engine::CompletionStatus::Partial {
1232                    present,
1233                    count,
1234                    finalized,
1235                } => {
1236                    let all = 1u16
1237                        .checked_shl(u32::from(count))
1238                        .map_or(u16::MAX, |bit| bit - 1);
1239                    Some(model::Parts {
1240                        count,
1241                        received: present,
1242                        lost: if finalized { all & !present } else { 0 },
1243                    })
1244                }
1245            },
1246        })
1247    }

1289                    .as_ref()
1290                    .and_then(|facade| facade.member_key(channel, &hint))
1291                    .or_else(|| self.only_peer_with(hint)),
1292            )),
1293            SenderScope::Peer(key) => Some((key.hint(), Some(key))),
1294            SenderScope::Local => None,
1295        }
1296    }
1297
1298    /// Who sent an incoming channel message, as the channel shows them:
1299    /// by the name they gave, or else by their address.
1300    fn sender(&self, record: &MessageRecord, conversation: &ConversationKey) -> Option<Sender> {
1301        if record.direction != engine::Direction::Inbound {
1302            return None;
1303        }
1304        let (hint, key) = self.sender_key(record, conversation)?;
1305        let mut lettering = String::<8>::new();
1306        let shape = match key {
1307            Some(key) => {
1308                _ = lettering.push_str(address(&key.0).get(..4).unwrap_or_default());
1309                PeerShape::Circle
1310            }
1311            // Known by nothing but a hint.
1312            None => {
1313                _ = write!(lettering, "{hint}");
1314                PeerShape::Hexagon
1315            }
1316        };
1317        let mut name = String::new();
1318        match record.sender_handle.as_deref() {
1319            Some(handle) if !handle.trim().is_empty() => push_cut(&mut name, handle),
1320            _ => name = unnamed_member(&lettering),
1321        }
1322        Some(Sender {
1323            name,
1324            avatar: AvatarStyle::peer(hint.0, &lettering, shape),
1325        })
1326    }
1327}

1364    }

1368fn cut<const N: usize>(text: &str) -> String<N> {
1369    let mut out = String::new();
1370    push_cut(&mut out, text);
1371    out

1374fn push_cut<const N: usize>(out: &mut String<N>, text: &str) {
1375    for c in text.chars() {
1376        if out.push(c).is_err() {
1377            break;
1378        }
1379    }
