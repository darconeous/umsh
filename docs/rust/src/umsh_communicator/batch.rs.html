18/// The steps of a commit, in the order a store takes them. Power lost
19/// after any of them leaves the store safe:
20///
21/// | Lost after | Effect |
22/// |---|---|
23/// | [`Checkpoint`](Self::Checkpoint) | one wire ID goes unused; a resend request for it is answered Message Unavailable |
24/// | [`ArchiveDeletes`](Self::ArchiveDeletes) | an edited original no longer resends: it is answered Unavailable, never with superseded content |
25/// | [`Archives`](Self::Archives) | resend material for a message neither sent nor shown |
26/// | [`Transcript`](Self::Transcript) | complete: the transmissions are released |
27#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
28pub enum CommitStep {
29    Checkpoint,
30    ArchiveDeletes,
31    Archives,
32    Transcript,
33}
34
35impl CommitStep {
36    pub const ALL: [Self; 4] = [
37        Self::Checkpoint,
38        Self::ArchiveDeletes,
39        Self::Archives,
40        Self::Transcript,
41    ];
42}

44/// A transcript change, as a store reads it out of a [`Batch`].
45#[derive(Clone, Copy, Debug)]
46pub enum Change<'b> {
47    /// One of the engine's transcript mutations, its body and Sender
48    /// Handle already cut to what the store keeps.
49    Message {
50        mutation: &'b Mutation,
51        /// What the radio said about the frame behind it, for a mutation
52        /// a received frame produced.
53        link: Option<LinkInfo>,
54    },
55    /// Word on one fragment of an outbound message.
56    Delivery {
57        handle: MessageHandle,
58        fragment: Option<u8>,
59        state: DeliveryState,
60    },
61    /// That an outbound message goes under a session with forward
62    /// secrecy, or never does.
63    Secrecy {
64        message: Subject,
65        forward_secret: bool,
66    },
67}
68
69/// The message a change is to: one the engine knows by its handle, as one
70/// composed in the same commit is, or one the store keeps.
71#[derive(Clone, Copy, Debug, PartialEq, Eq)]
72pub enum Subject {
73    Handle(MessageHandle),
74    Kept(MessageId),
75}
76
77// Most entries are messages, so the mutation is kept inline rather than
78// boxed for the sake of the rarer deliveries.
79#[allow(clippy::large_enum_variant)]
80enum Entry {
81    Message {
82        mutation: Mutation,
83        link: Option<LinkInfo>,
84    },
85    Delivery {
86        handle: MessageHandle,
87        fragment: Option<u8>,
88        state: DeliveryState,
89    },
90    Secrecy {
91        message: Subject,
92        forward_secret: bool,
93    },
94}

96/// What one commit writes: the outbound checkpoints, then the archive
97/// deletes, then the archives, then the transcript.
98///
99/// A batch is filled from what the chat facade drains, [`Batch::push_chat`]
100/// taking each output in turn, or from the engine's own outputs through
101/// [`Batch::push`]. It keeps its own copies—bodies out of the engine's
102/// render arena, which is reused once the queue is empty, and payloads out
103/// of the outputs—so it outlives both. What it borrows is the caller's map
104/// from the engine's message handles to the store's message IDs, for
105/// mutations of messages kept by an earlier commit.
106pub struct Batch<'a> {
107    time: Option<u32>,
108    handles: &'a dyn Fn(MessageHandle) -> Option<MessageId>,
109    checkpoints: Vec<StreamCheckpoint>,
110    archive_deletes: Vec<(ConversationKey, u8)>,
111    archives: Vec<(ArchiveKey, heapless::Vec<u8, MAX_FRAME>)>,
112    entries: Vec<Entry>,
113}
114
115impl<'a> Batch<'a> {
116    /// A batch whose messages are kept at `time`, in seconds of wall-clock
117    /// time when it is known. `handles` maps the engine's handles to the
118    /// store's IDs, as the [`Committed`] of earlier commits gave them.
119    pub fn new(time: Option<u32>, handles: &'a dyn Fn(MessageHandle) -> Option<MessageId>) -> Self {
120        Self {
121            time,
122            handles,
123            checkpoints: Vec::new(),
124            archive_deletes: Vec::new(),
125            archives: Vec::new(),
126            entries: Vec::new(),
127        }
128    }
129
130    /// Take one drained output. `body` resolves the output's body
131    /// references, which is [`Engine::body`] while the output is fresh,
132    /// and `link` is what the radio said about the frame that produced it,
133    /// if a frame did. Returns whether the output was the store's.
134    ///
135    /// A transmission's payload is kept for resending when it carries an
136    /// archive key; the transmission itself is the caller's to hold until
137    /// the batch is committed.
138    ///
139    /// [`Engine::body`]: umsh_text::engine::Engine::body
140    pub fn push<'e>(
141        &mut self,
142        output: &Output,
143        body: impl Fn(&BodyRef) -> &'e str,
144        link: Option<LinkInfo>,
145    ) -> bool {
146        match output {
147            Output::StoreCheckpoint {
148                conversation,
149                next_id,
150                epoch,
151            } => self.checkpoint(StreamCheckpoint {
152                conversation: *conversation,
153                next_id: *next_id,
154                epoch: *epoch,
155            }),
156            Output::DeleteArchive {
157                conversation,
158                message_id,
159            } => self.archive_deletes.push((*conversation, *message_id)),
160            Output::StoreArchive { key, payload } => self.archives.push((*key, payload.clone())),
161            Output::Transmit(transmission) => {
162                let Some(key) = transmission.archive else {
163                    return false;
164                };
165                self.archives.push((key, transmission.payload.clone()));
166            }
167            Output::StoreMessage(mutation) => {
168                let mutation = Mutation::from_engine(*mutation, body, |_, sender| match sender {
169                    SenderScope::Peer(key) => Some(key),
170                    SenderScope::Local | SenderScope::ClaimedMember(_) => None,
171                });
172                self.message(mutation, link);
173            }
174            Output::Event(Event::DeliveryStateChanged {
175                handle,
176                fragment,
177                state,
178            }) => self.delivery(*handle, *fragment, *state),
179            Output::Event(_) | Output::LookupOutbound { .. } | Output::Diagnostic(_) => {
180                return false;
181            }
182        }
183        true
184    }
185
186    /// Take one output the chat facade drained. `link` is what the radio
187    /// said about the frame that produced it, for an output that
188    /// [describes its receipt](Mutation::describes_receipt). Returns whether
189    /// the output was the store's.
190    pub fn push_chat(&mut self, output: &ChatOutput, link: Option<LinkInfo>) -> bool {
191        match output {
192            ChatOutput::Mutation(mutation) => {
193                let link = link.filter(|_| mutation.describes_receipt());
194                self.message(mutation.clone(), link);
195            }
196            ChatOutput::Delivery(delivery) => {
197                self.delivery(delivery.handle, delivery.fragment, delivery.state);
198            }
199            ChatOutput::StoreArchive(archive) => {
200                let Ok(payload) = heapless::Vec::from_slice(&archive.payload) else {
201                    return false;
202                };
203                self.archives.push((archive.key, payload));
204            }
205            ChatOutput::DeleteArchive(delete) => {
206                self.archive_deletes
207                    .push((delete.conversation, delete.message_id));
208            }
209            ChatOutput::Lookup(_)
210            | ChatOutput::SenderResolved(_)
211            | ChatOutput::Event(_)
212            | ChatOutput::Diagnostic(_) => return false,
213        }
214        true
215    }
216
217    /// Take a composed message: its checkpoint, archive deletes, archives,
218    /// and transcript, and the deliveries among the rest of what composing
219    /// it produced. Its frames are the facade's to release once the batch
220    /// is committed.
221    pub fn push_composed(&mut self, composed: &PendingBatch) {
222        self.checkpoint(composed.checkpoint);
223        for delete in &composed.archive_deletes {
224            self.archive_deletes
225                .push((delete.conversation, delete.message_id));
226        }
227        for archive in &composed.archives {
228            if let Ok(payload) = heapless::Vec::from_slice(&archive.payload) {
229                self.archives.push((archive.key, payload));
230            }
231        }
232        for mutation in &composed.mutations {
233            self.message(mutation.clone(), None);
234        }
235        for output in &composed.outputs {
236            if let ChatOutput::Delivery(delivery) = output {
237                self.delivery(delivery.handle, delivery.fragment, delivery.state);
238            }
239        }
240    }
241
242    fn checkpoint(&mut self, checkpoint: StreamCheckpoint) {
243        match self
244            .checkpoints
245            .iter_mut()
246            .find(|kept| kept.conversation == checkpoint.conversation)
247        {
248            Some(kept) => *kept = checkpoint,
249            None => self.checkpoints.push(checkpoint),
250        }
251    }
252
253    fn message(&mut self, mut mutation: Mutation, link: Option<LinkInfo>) {
254        match &mut mutation.change {
255            chat::Change::Insert(insert) => {
256                cut(&mut insert.body, BODY_MAX);
257                if let Some(handle) = &mut insert.sender_handle {
258                    cut(handle, NAME_MAX);
259                }
260            }
261            chat::Change::UpdateBody { body, .. } | chat::Change::Edit { body, .. } => {
262                cut(body, BODY_MAX);
263            }
264            chat::Change::Delete { .. } => {}
265        }
266        self.entries.push(Entry::Message { mutation, link });
267    }
268
269    /// Take that the outbound message `message` goes under a session with
270    /// forward secrecy, or never does.
271    pub fn secrecy(&mut self, message: Subject, forward_secret: bool) {
272        self.entries.push(Entry::Secrecy {
273            message,
274            forward_secret,
275        });
276    }
277
278    fn delivery(&mut self, handle: MessageHandle, fragment: Option<u8>, state: DeliveryState) {
279        self.entries.push(Entry::Delivery {
280            handle,
281            fragment,
282            state,
283        });
284    }
285
286    /// When the batch's messages were kept.
287    pub fn time(&self) -> Option<u32> {
288        self.time
289    }
290
291    /// The store's ID for a message an earlier commit kept, by the
292    /// engine's handle for it.
293    pub fn resolve(&self, handle: MessageHandle) -> Option<MessageId> {
294        (self.handles)(handle)
295    }
296
297    pub fn checkpoints(&self) -> &[StreamCheckpoint] {
298        &self.checkpoints
299    }
300
301    /// Message IDs whose archived fragments are all to go.
302    pub fn archive_deletes(&self) -> &[(ConversationKey, u8)] {
303        &self.archive_deletes
304    }
305
306    pub fn archives(&self) -> impl Iterator<Item = (&ArchiveKey, &[u8])> + '_ {
307        self.archives
308            .iter()
309            .map(|(key, payload)| (key, payload.as_slice()))
310    }
311
312    pub fn changes(&self) -> impl Iterator<Item = Change<'_>> + '_ {
313        self.entries.iter().map(|entry| match entry {
314            Entry::Message { mutation, link } => Change::Message {
315                mutation,
316                link: *link,
317            },
318            Entry::Delivery {
319                handle,
320                fragment,
321                state,
322            } => Change::Delivery {
323                handle: *handle,
324                fragment: *fragment,
325                state: *state,
326            },
327            Entry::Secrecy {
328                message,
329                forward_secret,
330            } => Change::Secrecy {
331                message: *message,
332                forward_secret: *forward_secret,
333            },
334        })
335    }
336
337    pub fn is_empty(&self) -> bool {
338        self.checkpoints.is_empty()
339            && self.archive_deletes.is_empty()
340            && self.archives.is_empty()
341            && self.entries.is_empty()
342    }
343}

345/// What a commit did.
346#[derive(Clone, Debug, Default, PartialEq, Eq)]
347pub struct Committed {
348    /// The IDs given to messages the batch inserted, by the engine's
349    /// handles for them.
350    pub inserted: Vec<(MessageHandle, MessageId)>,
351    /// Messages inserted or changed, in the order first touched.
352    pub messages: Vec<MessageId>,
353    /// Chats made or changed, in the order first touched.
354    pub chats: Vec<ChatId>,
355}
356
357impl Committed {
358    pub(crate) fn touch_message(&mut self, id: MessageId) {
359        if !self.messages.contains(&id) {
360            self.messages.push(id);
361        }
362    }
363
364    pub(crate) fn touch_chat(&mut self, id: ChatId) {
365        if !self.chats.contains(&id) {
366            self.chats.push(id);
367        }
368    }
369
370    /// The ID this commit gave the message `handle` names.
371    pub fn id_of(&self, handle: MessageHandle) -> Option<MessageId> {
372        self.inserted
373            .iter()
374            .find(|(inserted, _)| *inserted == handle)
375            .map(|(_, id)| *id)
376    }
377}

379/// Cuts `text` to at most `max` bytes, at a character boundary.
380fn cut(text: &mut String, max: usize) {
381    let len = clip(text, max).len();
382    text.truncate(len);
