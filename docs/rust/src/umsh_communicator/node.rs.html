34/// How many events wait for the interface before they collapse into a
35/// resync.
36const EVENTS: usize = 16;
37
38/// The board's clocks, as the node reads them.
39pub trait Clock {
40    /// Milliseconds on the board's monotonic clock.
41    fn now_ms(&self) -> u64;
42    /// The time in Unix seconds, once it is known.
43    fn wall(&self) -> Option<u32>;
44    /// Local time's offset from UTC, in minutes.
45    fn utc_offset_min(&self) -> i16;
46}
47
48/// A communicator's node, over the device's own ULCP session reached
49/// through `P`, keeping its chats in `S`.
50///
51/// Every method borrows the node for the length of the call and no
52/// longer, so the interface and the board's tasks can share it on one
53/// executor.
54pub struct CommunicatorNode<P, S = MemoryStore> {
55    settings: RefCell<Settings<P>>,
56    chats: RefCell<ChatHalf<S>>,
57    events: RefCell<EventQueue<EVENTS>>,
58    clock: &'static dyn Clock,
59    /// What the chats start with, once the device identity is known.
60    seeds: Seeds,
61    /// The chats have frames for the device node.
62    work: Cell<bool>,
63    worker: RefCell<WakerRegistration>,
64}
65
66impl<P: LocalPort, S: ChatStore> CommunicatorNode<P, S> {
67    /// A node whose messages' text is kept on `heap`. Its chats start as
68    /// the device identity once the device reports it, from `seeds`, which
69    /// are random.
70    pub fn new(
71        port: P,
72        store: S,
73        crypto: ChannelCrypto,
74        heap: TextHeap,
75        clock: &'static dyn Clock,
76        seeds: Seeds,
77    ) -> Self {
78        Self {
79            settings: RefCell::new(Settings::new(port)),
80            chats: RefCell::new(ChatHalf::new(store, crypto, heap)),
81            events: RefCell::new(EventQueue::new()),
82            clock,
83            seeds,
84            work: Cell::new(false),
85            worker: RefCell::new(WakerRegistration::new()),
86        }
87    }
88
89    pub fn clock(&self) -> &'static dyn Clock {
90        self.clock
91    }
92
93    /// Whether the chats have started.
94    pub fn started(&self) -> bool {
95        self.chats.borrow().started()
96    }
97
98    /// Hands the device what is waiting and takes in what it sent. The
99    /// board calls this after the interface has taken an input, whenever
100    /// the port has something, and at [`CommunicatorNode::next_deadline`].
101    ///
102    /// The chats start here, as the device identity, as soon as the
103    /// device has reported it, and hold the device's channels from then
104    /// on.
105    pub fn pump(&self) {
106        let now_ms = self.clock.now_ms();
107        let mut events = self.events.borrow_mut();
108        let mut push = |event| events.push(event);
109        let mut settings = self.settings.borrow_mut();
110        settings.pump(now_ms, &mut push);
111        if !settings.primed() {
112            return;
113        }
114        let mut chats = self.chats.borrow_mut();
115        if !chats.started()
116            && let Some(key) = settings.key()
117        {
118            chats.start(PublicKey(*key), self.seeds, now_ms, &mut push);
119        }
120        chats.hold(settings.channels(), &mut push);
121        for tag in settings.take_refused_joins() {
122            chats.let_go(tag, &mut push);
123        }
124        chats.hold_peers(&settings.peers(), &mut push);
125        chats.set_handle(&settings.device_name());
126    }
127
128    /// Takes what the device node received for the chats.
129    pub fn arrive(&self, arrival: Arrival) {
130        self.with_chats(|chats, now_ms, time, events| match arrival {
131            Arrival::Text(inbound) => chats.receive(inbound, now_ms, time, events),
132            Arrival::Identity { key, identity, rx } => {
133                chats.identify(key, &identity, &rx, now_ms, time, events);
134            }
135            Arrival::Pong {
136                peer,
137                round_trip_ms,
138                rx,
139            } => chats.pong(peer, round_trip_ms, &rx, now_ms, events),
140            Arrival::Secrecy { peer, change } => chats.secrecy(peer, change, events),
141        });
142    }
143
144    /// Whether the chats wait on the device node for a channel's key: the
145    /// device lists a channel by its tag alone.
146    pub fn misses_channel_keys(&self) -> bool {
147        self.chats.borrow().misses_keys()
148    }
149
150    /// Takes `keys`, those of the device node's channels.
151    pub fn learn_channel_keys(&self, keys: &[ChannelKey]) {
152        self.with_chats(|chats, _, _, events| {
153            for key in keys {
154                chats.learn_key(*key, events);
155            }
156        });
157    }

159    /// What the chats ask of the device node next, besides sending frames.
160    pub fn command(&self) -> Option<Command> {
161        self.chats.borrow_mut().command()
162    }

164    /// Advances the chats' timers.
165    pub fn tick(&self) {
166        self.with_chats(|chats, now_ms, time, events| chats.tick(now_ms, time, events));
167    }
168
169    /// Takes the word on the chats' frames in flight.
170    pub fn service(&self) {
171        self.with_chats(|chats, now_ms, time, events| chats.service(now_ms, time, events));
172    }
173
174    /// The chats' next frame for the device node, or `None` while nothing
175    /// may go. Send it with [`umsh_text::node_adapter::transmit`], then
176    /// hand back how that went to [`CommunicatorNode::departed`].
177    pub fn depart(&self) -> Option<Departure> {
178        self.with_chats(|chats, now_ms, time, events| chats.depart(now_ms, time, events))
179    }
180
181    /// Takes how the device node took a frame. Returns whether it has room
182    /// for more.
183    pub fn departed(&self, departure: Departure, sent: Sent) -> bool {
184        self.with_chats(|chats, now_ms, time, events| {
185            chats.departed(departure, sent, now_ms, time, events)
186        })
187    }
188
189    /// Waits until the interface has given the chats something to send.
190    pub async fn work(&self) {
191        poll_fn(|cx| {
192            if self.work.take() {
193                return Poll::Ready(());
194            }
195            self.worker.borrow_mut().register(cx.waker());
196            Poll::Pending
197        })
198        .await;
199    }
200
201    fn wake(&self) {
202        self.work.set(true);
203        self.worker.borrow_mut().wake();
204    }
205
206    /// Runs `f` with the chats, the time, and the event queue.
207    fn with_chats<R>(
208        &self,
209        f: impl FnOnce(&mut ChatHalf<S>, u64, Option<u32>, &mut dyn FnMut(NodeEvent)) -> R,
210    ) -> R {
211        let mut events = self.events.borrow_mut();
212        f(
213            &mut self.chats.borrow_mut(),
214            self.clock.now_ms(),
215            self.clock.wall(),
216            &mut |event| events.push(event),
217        )
218    }
219
220    /// Whether events wait for the interface.
221    pub fn has_events(&self) -> bool {
222        !self.events.borrow().is_empty()
223    }
224
225    /// When [`CommunicatorNode::pump`] next has work of its own.
226    pub fn next_deadline(&self) -> Option<u64> {
227        self.settings.borrow().next_deadline()
228    }
229
230    /// Whether every setting has been read from the device once.
231    pub fn primed(&self) -> bool {
232        self.settings.borrow().primed()
233    }
234
235    /// Runs `f` with the port, such as to wait on it.
236    pub fn with_port<R>(&self, f: impl FnOnce(&P) -> R) -> R {
237        f(self.settings.borrow().port())
238    }
239
240    /// The device's lists as the store keeps them, laid out in `out`
241    /// (see [`ChatStore::device_lists`]): what the board's device session
242    /// starts from.
243    pub fn device_lists<'b>(&self, out: &'b mut [u8]) -> Result<Option<&'b [u8]>, StoreError> {
244        self.chats.borrow().store().device_lists(out)
245    }
246
247    /// The store keeps `record` as the device's lists, or forgets them
248    /// with `None`, as the board's device session asks.
249    pub fn keep_device_lists(&self, record: Option<&[u8]>) -> Result<(), StoreError> {
250        self.chats
251            .borrow_mut()
252            .store_mut()
253            .keep_device_lists(record)
254    }
255
256    /// Asks the device to hold the channel a join here holds already, and
257    /// lets it go if it cannot be asked.
258    fn joined(&self, joined: Option<Joined>) -> Option<ChatSummary> {
259        let (summary, join) = joined?;
260        if let Some((key, tag)) = join
261            && !self.settings.borrow_mut().join_channel(&key.0, tag)
262        {
263            self.with_chats(|chats, _, _, events| chats.let_go(tag, events));
264            return None;
265        }
266        Some(summary)
267    }
268
269    fn change(&self, setting: Setting, change: impl FnOnce(&mut Settings<P>) -> bool) {
270        if !change(&mut self.settings.borrow_mut()) {
271            self.events
272                .borrow_mut()
273                .push(NodeEvent::SettingRefused(setting));
274        }
275    }
276}
