14/// A communicator's durable chat state.
15///
16/// Iterators are lazy and their items owned: taking a page, filtering, and
17/// stopping early cost only what they read, and a flash back end decodes
18/// each record as it is asked for. Items are results, because a read can
19/// fail partway through. The store is always a type parameter, never a
20/// trait object.
21pub trait ChatStore {
22    /// Writes `batch` in order—checkpoint, archive deletes, archives,
23    /// transcript—and returns the IDs given to inserted messages.
24    fn commit(&mut self, batch: &Batch<'_>) -> Result<Committed, StoreError>;
25
26    /// The chat's messages, newest first, from just before `before`.
27    fn messages(
28        &self,
29        chat: ChatId,
30        before: Option<MessageId>,
31    ) -> impl Iterator<Item = Result<MessageRecord, StoreError>> + '_;
32    fn message(&self, id: MessageId) -> Result<Option<MessageRecord>, StoreError>;
33    fn body<'b>(&self, id: MessageId, out: &'b mut [u8]) -> Result<Option<&'b str>, StoreError>;
34    /// Who reacted to the message, and how: the latest from each sender,
35    /// in the order they came. Their counts are the record's `reactions`.
36    fn reactors(&self, id: MessageId) -> impl Iterator<Item = ReactorRecord> + '_;
37    fn resolve_wire(&self, alias: &WireAlias) -> Option<MessageId>;
38
39    fn archive<'b>(
40        &self,
41        key: &ArchiveKey,
42        out: &'b mut [u8],
43    ) -> Result<Option<&'b [u8]>, StoreError>;
44    fn checkpoint(&self, conversation: &ConversationKey) -> Option<StreamCheckpoint>;
45    /// The most recently used checkpoints, oldest first, for `Engine::restore`.
46    fn recent_checkpoints(&self, limit: usize) -> impl Iterator<Item = StreamCheckpoint> + '_;
47
48    /// Chats, most recently active first.
49    fn chats(&self) -> impl Iterator<Item = Result<ChatRecord, StoreError>> + '_;
50    fn chat_for(&self, conversation: &ConversationKey) -> Option<ChatId>;
51    fn draft<'b>(&self, chat: ChatId, out: &'b mut [u8]) -> Result<Option<&'b str>, StoreError>;
52    fn update_chat(&mut self, chat: ChatId, change: ChatChange<'_>) -> Result<ChatId, StoreError>;
53
54    fn peer(&self, key: &PublicKey) -> Result<Option<PeerRecord>, StoreError>;
55    fn peers_with_hint(&self, hint: NodeHint) -> impl Iterator<Item = PublicKey> + '_;
56    fn peers(
57        &self,
58        filter: PeerFilter,
59    ) -> impl Iterator<Item = Result<PeerRecord, StoreError>> + '_;
60    fn update_peer(
61        &mut self,
62        key: &PublicKey,
63        change: PeerChange<'_>,
64    ) -> Result<PeerId, StoreError>;
65
66    fn channels(&self) -> impl Iterator<Item = Result<ChannelRecord, StoreError>> + '_;
67    fn update_channel(
68        &mut self,
69        tag: &ChannelTag,
70        change: ChannelChange<'_>,
71    ) -> Result<(), StoreError>;
72
73    /// The record the device keeps here of its own lists, beside the
74    /// chats: what it persists outside its saved snapshot (spec §Lists
75    /// Held by the Device's Store). Gone with everything else on
76    /// [`ChatStore::erase`].
77    fn device_lists<'b>(&self, out: &'b mut [u8]) -> Result<Option<&'b [u8]>, StoreError>;
78    /// Keeps `record` as the device's lists, or forgets them with `None`.
79    fn keep_device_lists(&mut self, record: Option<&[u8]>) -> Result<(), StoreError>;
80
81    /// Fails what each outbound message still awaited (see
82    /// [`Delivery::interrupt`]), as a new session starts: no word on the
83    /// last one's frames will be taken.
84    ///
85    /// [`Delivery::interrupt`]: crate::Delivery::interrupt
86    fn interrupt(&mut self) -> Result<(), StoreError>;
87
88    fn usage(&self) -> Usage;
89    /// Forget everything: a factory reset. Identifiers are still never
90    /// reused.
91    fn erase(&mut self) -> Result<(), StoreError>;
92}
