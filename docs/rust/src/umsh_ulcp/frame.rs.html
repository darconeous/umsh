1//! ULCP frame grammar.
2//!
3//! A frame is a one-byte header, a one-byte command identifier, and a
4//! command-defined payload. Frame length is provided by the framing
5//! layer (see [`crate::hdlc`] for asynchronous serial links).
6
7use crate::pui;
8use crate::status::Status;
9
10/// Mask of the two most significant header bits (the `FLG` field).
11pub const HEADER_FLG_MASK: u8 = 0xC0;
12/// Required value of the `FLG` field (`0b10` in the top two bits).
13pub const HEADER_FLG_PATTERN: u8 = 0x80;
14/// Mask of the three reserved header bits, which must be zero.
15pub const HEADER_RESERVED_MASK: u8 = 0x38;
16/// Mask of the three-bit transaction identifier.
17pub const HEADER_TID_MASK: u8 = 0x07;
18
19/// TID reserved for unsolicited commands and stream traffic.
20pub const TID_UNSOLICITED: u8 = 0;
21/// Largest usable transaction identifier.
22pub const TID_MAX: u8 = 7;
23
24/// Validated frame header byte.
25#[derive(Clone, Copy, Debug, PartialEq, Eq)]
26pub struct Header(u8);
27
28impl Header {
29    /// Build a header for the given transaction identifier.
30    ///
31    /// Returns `None` if `tid` exceeds [`TID_MAX`].
32    pub const fn new(tid: u8) -> Option<Self> {
33        if tid <= TID_MAX {
34            Some(Self(HEADER_FLG_PATTERN | tid))
35        } else {
36            None
37        }
38    }
39
40    /// Validate a received header byte.
41    ///
42    /// Returns `None` when the `FLG` pattern does not match (the frame
43    /// is not a ULCP frame) or a reserved bit is set (the
44    /// frame must be ignored).
45    pub const fn from_byte(byte: u8) -> Option<Self> {
46        if byte & HEADER_FLG_MASK == HEADER_FLG_PATTERN && byte & HEADER_RESERVED_MASK == 0 {
47            Some(Self(byte))
48        } else {
49            None
50        }
51    }
52
53    pub const fn to_byte(self) -> u8 {
54        self.0
55    }
56
57    pub const fn tid(self) -> u8 {
58        self.0 & HEADER_TID_MASK
59    }
60}

62/// Command identifiers defined by the minimal and full specs.
63#[derive(Clone, Copy, Debug, PartialEq, Eq)]
64#[repr(u8)]
65pub enum Cmd {
66    /// No-operation liveness check (host to device).
67    Nop = 0,
68    /// Software reset request (host to device).
69    Reset = 1,
70    /// Get property value (host to device).
71    PropGet = 2,
72    /// Set property value (host to device).
73    PropSet = 3,
74    /// Insert an item into a multi-value property (host to device).
75    PropInsert = 4,
76    /// Remove an item from a multi-value property (host to device).
77    PropRemove = 5,
78    /// Property value notification (device to host).
79    PropIs = 6,
80    /// Item-inserted notification (device to host).
81    PropInserted = 7,
82    /// Item-removed notification (device to host).
83    PropRemoved = 8,
84    /// Send data to a stream (host to device).
85    StrSend = 9,
86    /// Data received from a stream (device to host).
87    StrRecv = 10,
88    /// Deliver queued inbound frames (host to device).
89    QueueDrain = 11,
90    /// Save state to non-volatile storage (host to device).
91    Save = 12,
92    /// Erase all saved state (host to device).
93    Clear = 13,
94    /// Restore state from the saved snapshot (host to device).
95    Restore = 14,
96    /// Factory reset (host to device): erase ALL mutable state—saved
97    /// provisioning, device identity, BLE bonds, pairing PIN, and every
98    /// other persisted journal—then reboot. Unlike `CMD_CLEAR` (which
99    /// exempts bonds and the PIN and leaves the live session running),
100    /// this returns the radio to a blank factory state and does not reply:
101    /// the reboot drops the link.
102    FactoryReset = 15,
103    /// Restart the hardware (host to device): the device reboots as if
104    /// power-cycled, keeping every piece of persisted state. Unlike
105    /// `CMD_RST` (which returns protocol state to its post-reset values
106    /// with the device still running), this drops the link, and like
107    /// `CMD_FACTORY_RESET` it does not reply. Requires `CAP_REBOOT`; a
108    /// device without it answers `STATUS_UNIMPLEMENTED` instead.
109    Reboot = 16,
110    /// Enter bootloader DFU after transmitting a successful status response.
111    /// Empty payload selects Default; otherwise one [`crate::DfuMode`] byte.
112    Dfu = 17,
113    /// Announce the device now (host to device): send an advertisement or
114    /// a beacon, as a broadcast or on one of the device's own channels.
115    /// The payload is an option list; see [`crate::announce`]. Requires
116    /// `CAP_ADVERT`.
117    Announce = 19,
118    /// Get several property values (host to device). Requires
119    /// `CAP_CMD_MULTI`.
120    PropMultiGet = 21,
121    /// Set several property values in order (host to device). Requires
122    /// `CAP_CMD_MULTI`.
123    PropMultiSet = 22,
124    /// Multiple property value notification (device to host), answering
125    /// `CMD_PROP_MULTI_GET` or `CMD_PROP_MULTI_SET`. Never unsolicited.
126    PropAre = 23,
127    /// Session reset notification (device to host): the device has
128    /// discarded session state, so every session-scoped property is back
129    /// at its documented default. Always unsolicited, and never carried
130    /// over the administrative binding. The payload is one
131    /// [`SessionResetReason`].
132    SessionReset = 24,
133    /// Perform one I2C transaction (host to device): a bus, a 7-bit
134    /// address, and an operation list with embedded-hal `transaction`
135    /// semantics. Answered by [`Cmd::I2cResult`]. See [`crate::i2c`].
136    /// Requires `CAP_I2C`.
137    I2cTransfer = 25,
138    /// The data an I2C transfer read, or the addresses a scan found
139    /// (device to host). Only ever a reply to [`Cmd::I2cTransfer`] or
140    /// [`Cmd::I2cScan`], with that command's TID; never unsolicited.
141    I2cResult = 26,
142    /// Probe a range of 7-bit addresses on one bus (host to device).
143    /// Answered by [`Cmd::I2cResult`]. Requires `CAP_I2C`.
144    I2cScan = 27,
145}

148    pub const fn from_u8(value: u8) -> Option<Self> {
149        match value {
150            0 => Some(Self::Nop),
151            1 => Some(Self::Reset),
152            2 => Some(Self::PropGet),
153            3 => Some(Self::PropSet),
154            4 => Some(Self::PropInsert),
155            5 => Some(Self::PropRemove),
156            6 => Some(Self::PropIs),
157            7 => Some(Self::PropInserted),
158            8 => Some(Self::PropRemoved),
159            9 => Some(Self::StrSend),
160            10 => Some(Self::StrRecv),
161            11 => Some(Self::QueueDrain),
162            12 => Some(Self::Save),
163            13 => Some(Self::Clear),
164            14 => Some(Self::Restore),
165            15 => Some(Self::FactoryReset),
166            16 => Some(Self::Reboot),
167            17 => Some(Self::Dfu),
168            19 => Some(Self::Announce),
169            21 => Some(Self::PropMultiGet),
170            22 => Some(Self::PropMultiSet),
171            23 => Some(Self::PropAre),
172            24 => Some(Self::SessionReset),
173            25 => Some(Self::I2cTransfer),
174            26 => Some(Self::I2cResult),
175            27 => Some(Self::I2cScan),
176            _ => None,
177        }
178    }

181/// Why the device discarded session state, carried by
182/// [`Cmd::SessionReset`].
183///
184/// Diagnostic: the host's obligation is the same for every value, so an
185/// unrecognized reason is a session reset without a name for it, never a
186/// parse failure.
187#[derive(Clone, Copy, Debug, PartialEq, Eq)]
188pub enum SessionResetReason {
189    /// A host attached.
190    Attached,
191    /// `CMD_RST`.
192    Reset,
193    /// `CMD_RESTORE` in its reset form.
194    Restored,
195    /// A reason this implementation does not have a name for.
196    Other(u32),
197}
198
199impl SessionResetReason {
200    pub const fn from_u32(value: u32) -> Self {
201        match value {
202            0 => Self::Attached,
203            1 => Self::Reset,
204            2 => Self::Restored,
205            other => Self::Other(other),
206        }
207    }
208
209    pub const fn to_u32(self) -> u32 {
210        match self {
211            Self::Attached => 0,
212            Self::Reset => 1,
213            Self::Restored => 2,
214            Self::Other(value) => value,
215        }
216    }
217}

219#[derive(Clone, Copy, Debug, PartialEq, Eq)]
220pub enum ParseError {
221    /// The input ended before the structure was complete.
222    Truncated,
223    /// The header `FLG` pattern did not match; not a ULCP frame.
224    BadFlag,
225    /// A reserved header bit was set; the frame must be ignored.
226    ReservedBits,
227    /// The command identifier had its most significant bit set; the
228    /// frame must be ignored.
229    BadCommand,
230    /// A packed unsigned integer inside the payload was malformed.
231    BadPui,
232}
233
234impl From<pui::Error> for ParseError {
235    fn from(error: pui::Error) -> Self {
236        match error {
237            pui::Error::Truncated => Self::Truncated,
238            _ => Self::BadPui,
239        }
240    }
241}

243/// A parsed frame borrowing its payload from the input.
244///
245/// `cmd` is kept as the raw identifier so receivers can distinguish an
246/// unknown-but-well-formed command (respond with
247/// `STATUS_INVALID_COMMAND`) from a malformed frame (ignore).
248#[derive(Clone, Copy, Debug, PartialEq, Eq)]
249pub struct Frame<'a> {
250    pub header: Header,
251    pub cmd: u8,
252    pub payload: &'a [u8],
253}
254
255impl<'a> Frame<'a> {
256    pub fn parse(bytes: &'a [u8]) -> Result<Self, ParseError> {
257        let [header_byte, cmd, payload @ ..] = bytes else {
258            return Err(ParseError::Truncated);
259        };
260        if header_byte & HEADER_FLG_MASK != HEADER_FLG_PATTERN {
261            return Err(ParseError::BadFlag);
262        }
263        let header = Header::from_byte(*header_byte).ok_or(ParseError::ReservedBits)?;
264        if cmd & 0x80 != 0 {
265            return Err(ParseError::BadCommand);
266        }
267        Ok(Self {
268            header,
269            cmd: *cmd,
270            payload,
271        })
272    }
273
274    /// The command, if it is one defined by this crate.
275    pub const fn command(&self) -> Option<Cmd> {
276        Cmd::from_u8(self.cmd)
277    }
278}

280#[derive(Clone, Copy, Debug, PartialEq, Eq)]
281pub enum WriteError {
282    /// The output buffer cannot hold the frame.
283    BufferTooSmall,
284    /// The transaction identifier exceeds [`TID_MAX`].
285    InvalidTid,
286    /// A value exceeds the range of its wire representation.
287    ValueTooLarge,
288}
289
290impl From<pui::Error> for WriteError {
291    fn from(error: pui::Error) -> Self {
292        match error {
293            pui::Error::BufferTooSmall => Self::BufferTooSmall,
294            _ => Self::ValueTooLarge,
295        }
296    }
297}

299/// Incremental frame builder over a caller-provided buffer.
300pub struct FrameWriter<'a> {
301    buf: &'a mut [u8],
302    len: usize,
303}
304
305impl<'a> FrameWriter<'a> {
306    /// Start a frame with the given TID and command.
307    pub fn new(buf: &'a mut [u8], tid: u8, cmd: Cmd) -> Result<Self, WriteError> {
308        let header = Header::new(tid).ok_or(WriteError::InvalidTid)?;
309        let mut writer = Self { buf, len: 0 };
310        writer.write_u8(header.to_byte())?;
311        writer.write_u8(cmd as u8)?;
312        Ok(writer)
313    }
314
315    pub fn write_u8(&mut self, byte: u8) -> Result<(), WriteError> {
316        if self.len >= self.buf.len() {
317            return Err(WriteError::BufferTooSmall);
318        }
319        self.buf[self.len] = byte;
320        self.len += 1;
321        Ok(())
322    }
323
324    pub fn write_bytes(&mut self, bytes: &[u8]) -> Result<(), WriteError> {
325        let end = self
326            .len
327            .checked_add(bytes.len())
328            .ok_or(WriteError::BufferTooSmall)?;
329        if end > self.buf.len() {
330            return Err(WriteError::BufferTooSmall);
331        }
332        self.buf[self.len..end].copy_from_slice(bytes);
333        self.len = end;
334        Ok(())
335    }
336
337    pub fn write_pui(&mut self, value: u32) -> Result<(), WriteError> {
338        let written = pui::encode(value, &mut self.buf[self.len..])?;
339        self.len += written;
340        Ok(())
341    }
342
343    pub fn write_u16_le(&mut self, value: u16) -> Result<(), WriteError> {
344        self.write_bytes(&value.to_le_bytes())
345    }
346
347    pub fn write_u32_le(&mut self, value: u32) -> Result<(), WriteError> {
348        self.write_bytes(&value.to_le_bytes())
349    }
350
351    /// Bytes still available in the buffer.
352    pub const fn remaining(&self) -> usize {
353        self.buf.len() - self.len
354    }
355
356    /// Bytes written so far.
357    pub const fn len(&self) -> usize {
358        self.len
359    }
360
361    pub const fn is_empty(&self) -> bool {
362        self.len == 0
363    }
364
365    /// Append one multi-property entry: the combined length of the key
366    /// and value, then the key, then the value.
367    ///
368    /// The entry is written whole or not at all, so a caller that runs
369    /// out of room keeps a well-formed frame of the entries that fit.
370    pub fn write_entry(&mut self, key: u32, value: &[u8]) -> Result<(), WriteError> {
371        let total = entry_len(key, value.len()).ok_or(WriteError::ValueTooLarge)?;
372        if self.remaining() < total {
373            return Err(WriteError::BufferTooSmall);
374        }
375        let body = pui::encoded_len(key) + value.len();
376        self.write_pui(body as u32)?;
377        self.write_pui(key)?;
378        self.write_bytes(value)
379    }
380
381    /// Append an entry reporting a status in the position of the property
382    /// it answers.
383    pub fn write_status_entry(&mut self, status: Status) -> Result<(), WriteError> {
384        let mut value = [0u8; pui::MAX_LEN];
385        let len = pui::encode(status.0, &mut value)?;
386        self.write_entry(crate::ids::prop::LAST_STATUS, &value[..len])
387    }
388
389    /// Finish the frame, returning its total length in the buffer.
390    pub fn finish(self) -> usize {
391        self.len
392    }
393}
394
395/// Space a multi-property entry occupies, its length prefix included.
396///
397/// Returns `None` when the combined key and value exceed what the length
398/// prefix can express.
399pub const fn entry_len(key: u32, value_len: usize) -> Option<usize> {
400    let body = pui::encoded_len(key) + value_len;
401    if body > pui::MAX_VALUE as usize {
402        return None;
403    }
404    Some(pui::encoded_len(body as u32) + body)
405}

515/// One entry of a `CMD_PROP_MULTI_SET` or `CMD_PROP_ARE` payload.
516#[derive(Clone, Copy, Debug, PartialEq, Eq)]
517pub struct MultiEntry<'a> {
518    pub key: u32,
519    pub value: &'a [u8],
520}
521
522/// Iterator over the entries of a `CMD_PROP_MULTI_SET` or `CMD_PROP_ARE`
523/// payload.
524///
525/// A malformed entry yields one error and ends the iteration: the
526/// remaining bytes cannot be located once a length is untrustworthy.
527#[derive(Clone, Copy, Debug)]
528pub struct MultiEntries<'a> {
529    rest: &'a [u8],
530}
531
532impl<'a> MultiEntries<'a> {
533    pub const fn new(payload: &'a [u8]) -> Self {
534        Self { rest: payload }
535    }
536
537    /// The bytes not yet consumed, so a caller serving entries across
538    /// several calls can record where it stopped.
539    pub const fn remainder(&self) -> &'a [u8] {
540        self.rest
541    }
542
543    fn take_entry(&mut self) -> Result<MultiEntry<'a>, ParseError> {
544        let payload = core::mem::take(&mut self.rest);
545        let (body_len, consumed) = pui::decode(payload)?;
546        let after_len = &payload[consumed..];
547        let body_len = body_len as usize;
548        if after_len.len() < body_len {
549            return Err(ParseError::Truncated);
550        }
551        let (body, tail) = after_len.split_at(body_len);
552        let (key, key_len) = pui::decode(body)?;
553        self.rest = tail;
554        Ok(MultiEntry {
555            key,
556            value: &body[key_len..],
557        })
558    }
559}
560
561impl<'a> Iterator for MultiEntries<'a> {
562    type Item = Result<MultiEntry<'a>, ParseError>;
563
564    fn next(&mut self) -> Option<Self::Item> {
565        if self.rest.is_empty() {
566            return None;
567        }
568        Some(self.take_entry())
569    }
570}

572/// Iterator over the property identifiers of a `CMD_PROP_MULTI_GET`
573/// payload.
574#[derive(Clone, Copy, Debug)]
575pub struct MultiGetKeys<'a> {
576    rest: &'a [u8],
577}
578
579impl<'a> MultiGetKeys<'a> {
580    pub const fn new(payload: &'a [u8]) -> Self {
581        Self { rest: payload }
582    }
583
584    /// The bytes not yet consumed.
585    pub const fn remainder(&self) -> &'a [u8] {
586        self.rest
587    }
588}
589
590impl Iterator for MultiGetKeys<'_> {
591    type Item = Result<u32, ParseError>;
592
593    fn next(&mut self) -> Option<Self::Item> {
594        if self.rest.is_empty() {
595            return None;
596        }
597        let payload = core::mem::take(&mut self.rest);
598        match pui::decode(payload) {
599            Ok((key, consumed)) => {
600                self.rest = &payload[consumed..];
601                Some(Ok(key))
602            }
603            Err(error) => Some(Err(error.into())),
604        }
605    }
606}

635/// Encode `CMD_DFU`. Omission and explicit Default have identical semantics.
636pub fn dfu(buf: &mut [u8], tid: u8, mode: Option<crate::DfuMode>) -> Result<usize, WriteError> {
637    let mut writer = FrameWriter::new(buf, tid, Cmd::Dfu)?;
638    if let Some(mode) = mode {
639        writer.write_u8(mode as u8)?;
640    }
641    Ok(writer.finish())
642}

670pub fn session_reset(buf: &mut [u8], reason: SessionResetReason) -> Result<usize, WriteError> {

676/// Decode the payload of a `CMD_SESSION_RESET` frame.
677pub fn parse_session_reset(payload: &[u8]) -> Result<SessionResetReason, ParseError> {
678    let (reason, _) = pui::decode(payload)?;
679    Ok(SessionResetReason::from_u32(reason))

682fn stream_payload(
683    writer: &mut FrameWriter<'_>,
684    stream: u32,
685    data: &[u8],
686    metadata: &[u8],
687) -> Result<(), WriteError> {
688    let data_len = u16::try_from(data.len()).map_err(|_| WriteError::ValueTooLarge)?;
689    writer.write_pui(stream)?;
690    writer.write_u16_le(data_len)?;
691    writer.write_bytes(data)?;
692    writer.write_bytes(metadata)
693}

695/// Encode a `CMD_STR_SEND` frame.
696pub fn str_send(
697    buf: &mut [u8],
698    tid: u8,
699    stream: u32,
700    data: &[u8],
701    metadata: &[u8],
702) -> Result<usize, WriteError> {
703    let mut writer = FrameWriter::new(buf, tid, Cmd::StrSend)?;
704    stream_payload(&mut writer, stream, data, metadata)?;
705    Ok(writer.finish())
706}

720/// Payload of `CMD_PROP_GET`, `CMD_PROP_SET`, and `CMD_PROP_IS`.
721///
722/// For `CMD_PROP_GET` the value is empty.
723#[derive(Clone, Copy, Debug, PartialEq, Eq)]
724pub struct PropPayload<'a> {
725    pub key: u32,
726    pub value: &'a [u8],
727}
728
729impl<'a> PropPayload<'a> {
730    pub fn parse(payload: &'a [u8]) -> Result<Self, ParseError> {
731        let (key, consumed) = pui::decode(payload)?;
732        Ok(Self {
733            key,
734            value: &payload[consumed..],
735        })
736    }
737}
738
739/// Payload of `CMD_STR_SEND` and `CMD_STR_RECV`.
740#[derive(Clone, Copy, Debug, PartialEq, Eq)]
741pub struct StreamPayload<'a> {
742    pub stream: u32,
743    pub data: &'a [u8],
744    /// Stream-defined trailing metadata; may be empty.
745    pub metadata: &'a [u8],
746}
747
748impl<'a> StreamPayload<'a> {
749    pub fn parse(payload: &'a [u8]) -> Result<Self, ParseError> {
750        let (stream, consumed) = pui::decode(payload)?;
751        let rest = &payload[consumed..];
752        let [len_lo, len_hi, rest @ ..] = rest else {
753            return Err(ParseError::Truncated);
754        };
755        let data_len = usize::from(u16::from_le_bytes([*len_lo, *len_hi]));
756        if rest.len() < data_len {
757            return Err(ParseError::Truncated);
758        }
759        let (data, metadata) = rest.split_at(data_len);
760        Ok(Self {
761            stream,
762            data,
763            metadata,
764        })
765    }
766}
