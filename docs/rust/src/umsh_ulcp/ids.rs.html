356/// Whether a property is reachable from a mesh administrator.
357///
358/// The out-of-reach set is small and named explicitly by the spec rather
359/// than derived, because "device domain" is not a property of the key: a
360/// host-domain table and a device-domain one differ only by which half of
361/// the device they configure. A device answers all three with
362/// `STATUS_PROP_NOT_FOUND` — an administrator learns that the property does
363/// not exist for it, not that it exists and was refused — and an
364/// administrator that reads this first spends no airtime asking.
365pub fn admin_reachable(key: u32) -> bool {
366    !matches!(
367        key,
368        // The host domain in full: the assistance the device owes to
369        // whatever host it serves is that host's business.
370        prop::HOST_KEY
371            | prop::HOST_CHANNEL_KEYS
372            | prop::HOST_PEER_KEYS
373            | prop::HOST_RX_FILTERS
374            | prop::HOST_AUTO_ACK
375            | prop::HOST_RX_QUEUE_COUNT
376            | prop::HOST_RX_QUEUE_CAPACITY
377            | prop::HOST_RX_QUEUE_DROPPED
378            // Session state: an exchange is not an attach, so there is no
379            // session for it to describe.
380            | prop::MAC_PROMISCUOUS
381            | prop::MAC_BACKHAUL
382            // A device identity cannot be installed over the mesh.
383            | prop::DEV_PRIVATE_KEY
384    )
385}

390/// Shortest accepted `PROP_ADVERT_INTERVAL` / `PROP_BEACON_INTERVAL`, in
391/// seconds.
392///
393/// An absolute floor, and one a device may only ever round *up* from:
394/// scheduling jitter delays an announcement and never brings it forward,
395/// so no configuration can put an unsolicited broadcast on the air more
396/// often than this. The duty ledger remains the airtime control; this is
397/// what keeps a mistyped interval from spending the whole budget on
398/// announcements before anything else can speak.
399pub const MIN_AUTO_ANNOUNCE_INTERVAL_S: u32 = 20 * 60;

406/// has stopped being one, and 0 says so more honestly.
