126/// `PROP_SAVED` values.
127///
128/// A snapshot that exists but cannot be read is distinguishable from no
129/// snapshot at all, and running on an older generation than the one last
130/// written is distinguishable from running on the newest.
131pub mod saved {
132    /// Nothing is saved.
133    pub const NONE: u8 = 0;
134    /// The newest saved generation is in effect.
135    pub const CURRENT: u8 = 1;
136    /// A newer generation was rejected; an older one is in effect.
137    pub const FALLBACK: u8 = 2;
138    /// A snapshot exists but no generation could be read.
139    pub const UNREADABLE: u8 = 3;
140}
