322/// How a property's octets are meant to be read.
323///
324/// Enough to render a value a person can check and to accept one typed
325/// by hand. Anything whose shape is a structure rather than a scalar—
326/// a battery snapshot, an interleaved location, a table of keys—is
327/// [`PropertyType::Bytes`], which is an honest answer: the octets are
328/// the value, and whatever decodes them knows more than this table does.
329///
330/// Numerics are little-endian, as ULCP has them throughout.
331#[derive(Clone, Copy, Debug, PartialEq, Eq)]
332pub enum PropertyType {
333    /// Optional battery scalar or tagged request, decoded by `battery_diagnostics::Value`.
334    BatteryDiagnostic,
335    /// One octet, 0 or 1.
336    Bool,
337    U8,
338    U16,
339    U32,
340    I8,
341    I16,
342    I32,
343    /// UTF-8, NUL-terminated on the wire.
344    Text,
345    /// A 32-octet Ed25519 public key or channel key.
346    Key32,
347    /// A status code in PUI form.
348    Status,
349    /// Octets whose meaning is structural, not scalar.
350    Bytes,
351}
352
353/// How to read a known property's octets.
354///
355/// A property absent from the table has no scalar reading, which is the
356/// same answer as [`PropertyType::Bytes`] but says so by omission: the
357/// caller knows it is looking at something this module has never heard
358/// of rather than something deliberately opaque.
359pub const fn property_type(key: u32) -> Option<PropertyType> {
360    if crate::battery_diagnostics::index(key).is_some() {
361        return Some(PropertyType::BatteryDiagnostic);
362    }
363    use PropertyType::{Bool, Bytes, I8, I16, I32, Key32, Status, Text, U8, U16, U32};
364    Some(match key {
365        prop::BATTERY_GAUGE_CONFIG | prop::BATTERY_GAUGE_TELEMETRY => Bytes,
366        // Tables of bus and device items, decoded by `i2c::buses` and
367        // `i2c::devices`.
368        prop::I2C_BUSES | prop::I2C_DEVICES => Bytes,
369        prop::TEMPERATURES | prop::TEMPERATURE_NAMES => Bytes,
370        prop::LAST_STATUS => Status,
371        // Major and minor, one octet each.
372        prop::PROTOCOL_VERSION => Bytes,
373        prop::DEV_VERSION | prop::DEV_MODEL | prop::DEV_NAME => Text,
374        prop::INTERFACE_TYPE => U32,
375        // A list of capability codes in PUI form.
376        prop::CAPS => Bytes,
377        prop::UPTIME => U32,
378        prop::PHY_ENABLED => Bool,
379        prop::PHY_FREQ => U32,
380        prop::PHY_TX_POWER | prop::PHY_RSSI => I8,

382        prop::PHY_LORA_SF | prop::PHY_LORA_CR => U8,

540/// A display adapter for one ULCP frame.
541///
542/// Values are summarized by length and never dumped, so callers can safely use
543/// the result in logs even for secret-bearing property writes.
544pub struct FrameDescription<'a>(pub &'a [u8]);
545
546impl fmt::Display for FrameDescription<'_> {
547    fn fmt(&self, out: &mut fmt::Formatter<'_>) -> fmt::Result {
548        let bytes = self.0;
549        let Ok(frame) = Frame::parse(bytes) else {
550            return write!(out, "malformed frame ({} bytes)", bytes.len());
551        };
552        let tid = frame.header.tid();
553        let Some(command) = frame.command() else {
554            return write!(out, "unknown command tid={tid} ({} bytes)", bytes.len());
555        };
556        match command {
557            Cmd::Dfu => write!(
558                out,
559                "Dfu tid={tid} mode={:?}",
560                crate::DfuMode::parse(frame.payload)
561            ),
562            Cmd::Nop
563            | Cmd::Reset
564            | Cmd::QueueDrain
565            | Cmd::Save
566            | Cmd::Clear
567            | Cmd::Restore
568            | Cmd::FactoryReset
569            | Cmd::Reboot => {
570                write!(out, "{command:?} tid={tid}")
571            }
572            Cmd::PropGet
573            | Cmd::PropSet
574            | Cmd::PropIs
575            | Cmd::PropInsert
576            | Cmd::PropRemove
577            | Cmd::PropInserted
578            | Cmd::PropRemoved => {
579                let Ok(payload) = PropPayload::parse(frame.payload) else {
580                    return write!(out, "{command:?} tid={tid} (malformed payload)");
581                };
582                let name = property_name(payload.key);
583                if payload.key == prop::LAST_STATUS && command == Cmd::PropIs {
584                    let status = pui::decode(payload.value)
585                        .map(|(code, _)| Status(code))
586                        .unwrap_or(Status::FAILURE);
587                    if let Some(name) = name {
588                        write!(out, "{command:?} tid={tid} {name} = {status:?}")
589                    } else {
590                        write!(
591                            out,
592                            "{command:?} tid={tid} prop {} = {status:?}",
593                            payload.key
594                        )
595                    }
596                } else if let Some(name) = name {
597                    write!(
598                        out,
599                        "{command:?} tid={tid} {name} ({} value bytes)",
600                        payload.value.len()
601                    )
602                } else {
603                    write!(
604                        out,
605                        "{command:?} tid={tid} prop {} ({} value bytes)",
606                        payload.key,
607                        payload.value.len()
608                    )
609                }
610            }
611            Cmd::PropMultiGet => {
612                let mut count = 0usize;
613                for key in MultiGetKeys::new(frame.payload) {
614                    if key.is_err() {
615                        return write!(out, "{command:?} tid={tid} (malformed payload)");
616                    }
617                    count += 1;
618                }
619                write!(out, "{command:?} tid={tid} ({count} properties)")
620            }
621            Cmd::PropMultiSet | Cmd::PropAre => {
622                let mut count = 0usize;
623                let mut value_bytes = 0usize;
624                for entry in MultiEntries::new(frame.payload) {
625                    let Ok(entry) = entry else {
626                        return write!(out, "{command:?} tid={tid} (malformed payload)");
627                    };
628                    count += 1;
629                    value_bytes += entry.value.len();
630                }
631                write!(
632                    out,
633                    "{command:?} tid={tid} ({count} entries, {value_bytes} value bytes)"
634                )
635            }
636            Cmd::Announce => match crate::announce::Announcement::parse(frame.payload) {
637                Ok(request) => write!(
638                    out,
639                    "{command:?} tid={tid} {:?} hops={} full_source={} {}",
640                    request.kind,
641                    request.flood_hops,
642                    request.full_source,
643                    if request.channel.is_some() {
644                        "multicast"
645                    } else {
646                        "broadcast"
647                    }
648                ),
649                Err(_) => write!(out, "{command:?} tid={tid} (malformed payload)"),
650            },
651            Cmd::SessionReset => match crate::frame::parse_session_reset(frame.payload) {
652                Ok(reason) => write!(out, "{command:?} tid={tid} {reason:?}"),
653                Err(_) => write!(out, "{command:?} tid={tid} (malformed payload)"),
654            },
655            Cmd::I2cTransfer => match crate::i2c::TransferRequest::parse(frame.payload) {
656                Ok(request) => {
657                    let shape = request.shape();
658                    write!(
659                        out,
660                        "{command:?} tid={tid} bus={} addr=0x{:02X} ({} ops, {} write bytes, {} read bytes)",
661                        request.bus,
662                        request.addr,
663                        shape.ops,
664                        shape.data_len - shape.read_len,
665                        shape.read_len
666                    )
667                }
668                Err(_) => write!(out, "{command:?} tid={tid} (malformed payload)"),
669            },
670            Cmd::I2cScan => match crate::i2c::ScanRequest::parse(frame.payload) {
671                Ok(request) => write!(
672                    out,
673                    "{command:?} tid={tid} bus={} 0x{:02X}..=0x{:02X}",
674                    request.bus, request.first, request.last
675                ),
676                Err(_) => write!(out, "{command:?} tid={tid} (malformed payload)"),
677            },
678            Cmd::I2cResult => {
679                write!(out, "{command:?} tid={tid} ({} bytes)", frame.payload.len())
680            }
681            Cmd::StrSend | Cmd::StrRecv => match StreamPayload::parse(frame.payload) {
682                Ok(payload) => write!(
683                    out,
684                    "{command:?} tid={tid} stream={} ({} data bytes, {} meta bytes)",
685                    payload.stream,
686                    payload.data.len(),
687                    payload.metadata.len()
688                ),
689                Err(_) => write!(out, "{command:?} tid={tid} (malformed payload)"),
690            },
691        }
692    }
693}
