23/// A soft device's own interface, on the host's clock.
24pub struct Port {
25    device: Rc<RefCell<SoftDevice<HostPlatform>>>,
26    start: Instant,
27}
28
29impl Port {
30    /// The device behind it.
31    pub fn device(&self) -> &Rc<RefCell<SoftDevice<HostPlatform>>> {
32        &self.device
33    }
34}
35
36impl LocalPort for Port {
37    fn send(&mut self, frame: &[u8]) -> Result<(), PortBusy> {
38        let now = self.start.elapsed().as_millis() as u64;
39        self.device.borrow_mut().device_request(frame, now);
40        Ok(())
41    }
42
43    fn receive(&mut self, out: &mut [u8]) -> Option<PortFrame> {
44        let frame = self.device.borrow_mut().take_device_frame()?;
45        out[..frame.len()].copy_from_slice(&frame);
46        Some(PortFrame::Frame(frame.len()))
47    }
48}

52/// The host's clocks. Local time is UTC.
53struct HostClock(Instant);
54
55impl Clock for HostClock {
56    fn now_ms(&self) -> u64 {
57        self.0.elapsed().as_millis() as u64
58    }
59
60    fn wall(&self) -> Option<u32> {
61        SystemTime::now()
62            .duration_since(UNIX_EPOCH)
63            .ok()
64            .map(|since| since.as_secs() as u32)
65    }
66
67    fn utc_offset_min(&self) -> i16 {
68        0
69    }
70}

126/// A node over a soft device keeping its state under `dir`, with its
127/// settings read and its chats mounted.
128pub fn open(dir: &Path, start: Instant) -> std::io::Result<&'static Node> {
129    let starting_keys = Box::leak(Box::new(ChannelCrypto::software().starting_keys()));
130    let platform = HostPlatform::new(dir)?.holding_lists(starting_keys);
131    let mut device = SoftDevice::new(config(), platform);
132    device.open_device_interface();
133    let flash = FileFlash::open(&dir.join("chats.flash"))?;
134    let buffer = Box::leak(vec![0u8; 2048].into_boxed_slice());
135    let store = FlashMapStore::mount(
136        FlashMap::new(flash, 0..flash::SIZE as u32, buffer),
137        Quota::DEFAULT,
138    )
139    .map_err(|error| std::io::Error::other(format!("the chats did not mount: {error:?}")))?;
140    let node: &'static Node = Box::leak(Box::new(CommunicatorNode::new(
141        Port {
142            device: Rc::new(RefCell::new(device)),
143            start,
144        },
145        store,
146        ChannelCrypto::software(),
147        &std::alloc::System,
148        Box::leak(Box::new(HostClock(start))),
149        Seeds {
150            session_id: rand::random(),
151            jitter_seed: rand::random(),
152        },
153    )));
154    node.pump();
155    while node.next_event().is_some() {}
156    Ok(node)
157}

159/// The interface's state over `node`, on the Chats screen.
160pub fn state(node: &'static Node) -> State {
161    let mut state = State::new(node);
162    state.status.device_name = node.device_name();
163    state.set_wall_time(
164        std::time::SystemTime::now()
165            .duration_since(std::time::UNIX_EPOCH)
166            .ok()
167            .map(|since| since.as_secs() as u32),
168    );
169    state.reload();
170    state.focus_list();
171    state
172}
