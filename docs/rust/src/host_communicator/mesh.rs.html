1//! The device node behind the window: a MAC of the device identity's own,
2//! on the LAN bearer, as the Pager's runs on its radio. It holds the
3//! channels and peers the device's settings hold, the chats are fed what
4//! it receives, and they send through it. Other simulators, and phones on
5//! the network, are on the same group.
6//!
7//! It runs on a single-threaded runtime that the window's loop drives
8//! between frames, so the node, the chats, and the window share a thread.
9

39type Platform = TokioPlatform<UdpMulticastRadio, TokioFileCounterStore, TokioFileKeyValueStore>;

41/// The Pager's device node, in its capacities: one identity, 32 peers, 8
42/// channels, 8 pending acknowledgments, 8 transmit slots.
43type MeshMac = Mac<Platform, 1, 32, 8, 8, 8, 255, 32, 4, 2, 0>;
44type Handle = MacHandle<'static, Platform, 1, 32, 8, 8, 8, 255, 32, 4, 2, 0>;
45
46/// What the device's settings give the node: its channels' keys, and the
47/// peers and administrators it is to know.
48#[derive(Clone, Default)]
49struct Domain {
50    channels: Vec<[u8; 32]>,
51    peers: Vec<[u8; 32]>,
52}

54/// The device node, running.
55pub struct Mesh {
56    runtime: Runtime,
57    local: LocalSet,
58    device: Rc<RefCell<SoftDevice<HostPlatform>>>,
59    /// The device-domain version the node was last given its domain at.
60    version: Cell<Option<u32>>,
61    domain: watch::Sender<Domain>,
62}

64impl Mesh {
65    /// Brings the node up as `node`'s device identity, with its frame
66    /// counters kept under `dir`.
67    pub fn start(dir: &Path, node: &'static Node) -> io::Result<Self> {
68        let device = node.with_port(|port| port.device().clone());
69        let secret = device
70            .borrow_mut()
71            .platform_mut()
72            .load_identity()
73            .ok_or_else(|| io::Error::other("the device has no identity"))?;
74        let identity = SoftwareIdentity::from_secret_bytes(&secret);
75        let key = *identity.public_key();
76
77        let runtime = tokio::runtime::Builder::new_current_thread()
78            .enable_all()
79            .build()?;
80        let radio = {
81            let _context = runtime.enter();
82            UdpMulticastRadio::bind(
83                LanConfig::lan(),
84                RxMode::Bearer(BearerId::LAN),
85                UdpMulticastRadio::SIMULATED_T_FRAME_MS,
86            )?
87        };
88        let mac: &'static AsyncRefCell<MeshMac> = Box::leak(Box::new(AsyncRefCell::new(Mac::new(
89            radio,
90            CryptoEngine::new(SoftwareAes, SoftwareSha256),
91            StdClock::new(),
92            rand::rng(),
93            TokioFileCounterStore::new(dir.join("counters"))?,
94            RepeaterConfig::default(),
95            OperatingPolicy::default(),
96        ))));
97        let handle = MacHandle::new(mac);
98        let local = LocalSet::new();
99        let (host, mesh_node) = local.block_on(&runtime, async {
100            let id = handle
101                .add_identity(identity)
102                .await
103                .map_err(|error| io::Error::other(format!("identity: {error:?}")))?;
104            handle
105                .load_persisted_counter(id)
106                .await
107                .map_err(|error| io::Error::other(format!("frame counter: {error:?}")))?;
108            // A stranger that sends its full key is known for as long as
109            // there is room, as on the board.
110            handle.set_auto_register_full_key_peers(true).await;
111            let mut host = Host::new(handle);
112            let mesh_node = host.add_node(id);
113            io::Result::Ok((host, mesh_node))
114        })?;
115
116        let inbox: &'static Inbox<NoopRawMutex> = Box::leak(Box::new(Inbox::new()));
117        // The taps live as long as the node.
118        std::mem::forget(task::listen(
119            &mesh_node,
120            inbox,
121            key,
122            WellKnownChannels::derive(),
123        ));
124        let (domain, followed) = watch::channel(Domain::default());
125        local.spawn_local(pump(host, handle));
126        local.spawn_local(follow(mesh_node.clone(), followed));
127        local.spawn_local(chats(node, mesh_node, inbox));
128        Ok(Self {
129            runtime,
130            local,
131            device,
132            version: Cell::new(None),
133            domain,
134        })
135    }
136
137    /// Lets the node and the chats run for `span`, holding the channels
138    /// and peers the device's settings now hold.
139    pub fn run_for(&self, span: Duration) {
140        {
141            let device = self.device.borrow();
142            let session = device.session();
143            let version = session.dev_domain_version();
144            if self.version.replace(Some(version)) != Some(version) {
145                self.domain.send_replace(Domain {
146                    channels: session.dev_channel_keys().collect(),
147                    peers: session.dev_peers().chain(session.dev_admins()).collect(),
148                });
149            }
150        }
151        self.runtime
152            .block_on(self.local.run_until(tokio::time::sleep(span)));
153    }
154}

156async fn pump(mut host: Host<Handle>, handle: Handle) {
157    loop {
158        if let Err(error) = host.pump_once().await {
159            eprintln!("host-communicator: the device node stopped: {error:?}");
160            return;
161        }
162        if let Err(error) = handle.service_counter_persistence().await {
163            eprintln!("host-communicator: frame counters not saved: {error:?}");
164        }
165    }
166}

168/// Joins and leaves channels, and knows and forgets peers, as the
169/// device's settings do.
170async fn follow(node: LocalNode<Handle>, mut followed: watch::Receiver<Domain>) {
171    let mut joined: Vec<[u8; 32]> = Vec::new();
172    let mut known: Vec<[u8; 32]> = Vec::new();
173    while followed.changed().await.is_ok() {
174        let domain = followed.borrow_and_update().clone();
175        for key in joined.extract_if(.., |key| !domain.channels.contains(key)) {
176            let _ = node.leave(&Channel::private(ChannelKey(key), "")).await;
177        }
178        for key in domain.channels {
179            if !joined.contains(&key)
180                && node
181                    .join(&Channel::private(ChannelKey(key), ""))
182                    .await
183                    .is_ok()
184            {
185                joined.push(key);
186            }
187        }
188        for key in known.extract_if(.., |key| !domain.peers.contains(key)) {
189            node.remove_peer(&PublicKey(key)).await;
190        }
191        for key in domain.peers {
192            if !known.contains(&key) && node.peer(PublicKey(key)).await.is_ok() {
193                known.push(key);
194            }
195        }
196    }
197}

199async fn chats(node: &'static Node, mesh: LocalNode<Handle>, inbox: &'static Inbox<NoopRawMutex>) {
200    task::run(
201        node,
202        &mesh,
203        inbox,
204        |at| {
205            let now = node.clock().now_ms();
206            tokio::time::sleep(Duration::from_millis(at.saturating_sub(now)))
207        },
208        // The window takes the node's events every frame.
209        || {},
210    )
211    .await
