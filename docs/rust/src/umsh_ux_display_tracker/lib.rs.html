3//! UX mechanism for display-tracker-class UMSH boards.
4//!
5//! This crate is the user-experience layer for boards whose physical UX
6//! is a display plus navigation controls: T-Echo, Heltec LoRa32 V3,
7//! Wio Tracker L1, T-Beam Supreme, and T-LoRa Pager in tracker mode.
8//! The Pager uses its wheel, wheel press, and Backspace; this class
9//! does not provide text entry.
10//! They differ in panel technology and input richness, but a user who
11//! learns one should already know the others, so the interaction model
12//! lives here rather than in each firmware.
13//!
14//! Like [`umsh_ux_tracker`], this crate provides only **mechanism**:
15//!
16//! - [`menu`]—the on-screen menu: a wrapping item list narrowed
17//!   per-board, with confirmation in front of the destructive entries.
18//! - [`attention`]—when to stop assuming the user is looking, and what
19//!   that means for an emissive panel (power it off) versus a persistent
20//!   one (send the menu home).
21//! - [`gate`]—what a gesture means when the screen is dark, an alert
22//!   is running, the boot splash is visible, or the panel is mid-refresh.
23//! - [`boot`]—the boot splash's two seconds of visible time, independent
24//!   of the About page that shares its artwork.
25//! - [`screen`]—what a frame looks like: the rows, what they say, and
26//!   the battery indicator ordinary frames carry. Boot and About share
27//!   a header-free logo and firmware version. Behind the `screen`
28//!   feature, since two boards in this family have no panel.
29//!
30//! Button gestures themselves come from
31//! [`umsh_ux_tracker::button::ButtonFsm`]; this crate adds only the
32//! shared timing policy, [`button_timings`]. Policy that depends on the
33//! board—which effects the menu items map to, what shutting down
34//! entails—belongs in the firmware.
35//!
36//! Rendering used to belong there too, on the theory that a 128×64 OLED
37//! and a 200×200 e-paper have too little in common to share a layout.
38//! They have more in common than they have pixels: [`screen`] takes a
39//! per-board [`screen::Layout`] and draws the rest once, so the class
40//! agrees on what a status page is rather than on how tall it is.
41//!
42//! See `docs/ux/` for the user-facing rules these modules encode and
43//! `docs/firmware-architecture.md` for the BSP / UX / App / Binary
44//! layering.

57/// The gesture timing shared by every display tracker.
58///
59/// One set of numbers across the class is the point: the same hold
60/// powers off a T-Echo and a Heltec V3, and muscle memory carries.
61///
62/// A click is at most 500 ms; a chord continues while presses are within
63/// 400 ms of each other. Holding for 1 s and releasing is Back; holding
64/// through 4 s powers off without waiting for the release, so the
65/// gesture confirms itself while the user is still committing to it.
66pub fn button_timings() -> ButtonTimings {
67    ButtonTimings {
68        max_click_hold: Duration::from_millis(500),
69        inter_click_gap: Duration::from_millis(400),
70        long_press: Duration::from_secs(1),
71        very_long_press: Some(Duration::from_secs(4)),
72    }
73}

75#[cfg(test)]
76mod tests {
77    use super::*;
78    use umsh_ux_tracker::button::{ButtonEdge, ButtonEvent, ButtonFsm};
79
80    /// The class vocabulary, end to end through the real recognizer:
81    /// single = Forward, double = Select, 1 s hold-release = Back,
82    /// 4 s hold = power off.
83    #[test]
84    fn the_shared_timings_recognize_the_class_vocabulary() {
85        let t = button_timings();
86
87        let mut fsm = ButtonFsm::new(t);
88        fsm.on_edge(ButtonEdge::Press, 0);
89        assert_eq!(fsm.on_edge(ButtonEdge::Release, 100), None);
90        assert_eq!(fsm.poll(500), Some(ButtonEvent::Single));
91
92        let mut fsm = ButtonFsm::new(t);
93        fsm.on_edge(ButtonEdge::Press, 0);
94        fsm.on_edge(ButtonEdge::Release, 100);
95        fsm.on_edge(ButtonEdge::Press, 300);
96        assert_eq!(fsm.on_edge(ButtonEdge::Release, 400), None);
97        assert_eq!(fsm.poll(800), Some(ButtonEvent::Double));
98
99        let mut fsm = ButtonFsm::new(t);
100        fsm.on_edge(ButtonEdge::Press, 0);
101        assert_eq!(
102            fsm.on_edge(ButtonEdge::Release, 1_500),
103            Some(ButtonEvent::Long)
104        );
105
106        let mut fsm = ButtonFsm::new(t);
107        fsm.on_edge(ButtonEdge::Press, 0);
108        // Fires while still held, before any release.
109        assert_eq!(fsm.poll(4_000), Some(ButtonEvent::VeryLong));
110    }
111
112    /// A hold released between Back and power-off must not power off.
113    #[test]
114    fn releasing_before_the_power_off_threshold_is_only_back() {
115        let mut fsm = ButtonFsm::new(button_timings());
116        fsm.on_edge(ButtonEdge::Press, 0);
117        assert_eq!(fsm.poll(3_999), None);
118        assert_eq!(
119            fsm.on_edge(ButtonEdge::Release, 3_999),
120            Some(ButtonEvent::Long)
121        );
122    }
