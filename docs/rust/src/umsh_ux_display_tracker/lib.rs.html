3//! UX mechanism for display-tracker-class UMSH boards.
4//!
5//! This crate is the user-experience layer for boards whose physical UX
6//! is a small display plus a button — optionally with a D-pad — and no
7//! keyboard: T-Echo, Heltec LoRa32 V3, Wio Tracker L1, T-Beam Supreme.
8//! They differ in panel technology and input richness, but a user who
9//! learns one should already know the others, so the interaction model
10//! lives here rather than in each firmware.
11//!
12//! Like [`umsh_ux_tracker`], this crate provides only **mechanism**:
13//!
14//! - [`menu`] — the on-screen menu: a wrapping item list narrowed
15//!   per-board, with confirmation in front of the destructive entries.
16//! - [`attention`] — when to stop assuming the user is looking, and what
17//!   that means for an emissive panel (power it off) versus a persistent
18//!   one (send the menu home).
19//! - [`gate`] — what a gesture means when the screen is dark, an alert
20//!   is running, or the panel is mid-refresh.
21//!
22//! Button gestures themselves come from
23//! [`umsh_ux_tracker::button::ButtonFsm`]; this crate adds only the
24//! shared timing policy, [`button_timings`]. Policy that depends on the
25//! board — which effects the menu items map to, how a frame is drawn,
26//! what shutting down entails — belongs in the firmware.
27//!
28//! Rendering is deliberately absent. Every panel in this class is an
29//! `embedded-graphics` `DrawTarget`, but their geometries are far enough
30//! apart (128×64 OLED versus 200×200 e-paper) that layout is board work.
31//!
32//! See `docs/ux/` for the user-facing rules these modules encode and
33//! `docs/firmware-architecture.md` for the BSP / UX / App / Binary
34//! layering.

43/// The gesture timing shared by every display tracker.
44///
45/// One set of numbers across the class is the point: the same hold
46/// powers off a T-Echo and a Heltec V3, and muscle memory carries.
47///
48/// A click is at most 500 ms; a chord continues while presses are within
49/// 400 ms of each other. Holding for 1 s and releasing is Back; holding
50/// through 4 s powers off without waiting for the release, so the
51/// gesture confirms itself while the user is still committing to it.
52pub fn button_timings() -> ButtonTimings {
53    ButtonTimings {
54        max_click_hold: Duration::from_millis(500),
55        inter_click_gap: Duration::from_millis(400),
56        long_press: Duration::from_secs(1),
57        very_long_press: Some(Duration::from_secs(4)),
58    }
59}

61#[cfg(test)]
62mod tests {
63    use super::*;
64    use umsh_ux_tracker::button::{ButtonEdge, ButtonEvent, ButtonFsm};
65
66    /// The class vocabulary, end to end through the real recognizer:
67    /// single = Forward, double = Select, 1 s hold-release = Back,
68    /// 4 s hold = power off.
69    #[test]
70    fn the_shared_timings_recognize_the_class_vocabulary() {
71        let t = button_timings();
72
73        let mut fsm = ButtonFsm::new(t);
74        fsm.on_edge(ButtonEdge::Press, 0);
75        assert_eq!(fsm.on_edge(ButtonEdge::Release, 100), None);
76        assert_eq!(fsm.poll(500), Some(ButtonEvent::Single));
77
78        let mut fsm = ButtonFsm::new(t);
79        fsm.on_edge(ButtonEdge::Press, 0);
80        fsm.on_edge(ButtonEdge::Release, 100);
81        fsm.on_edge(ButtonEdge::Press, 300);
82        assert_eq!(fsm.on_edge(ButtonEdge::Release, 400), None);
83        assert_eq!(fsm.poll(800), Some(ButtonEvent::Double));
84
85        let mut fsm = ButtonFsm::new(t);
86        fsm.on_edge(ButtonEdge::Press, 0);
87        assert_eq!(
88            fsm.on_edge(ButtonEdge::Release, 1_500),
89            Some(ButtonEvent::Long)
90        );
91
92        let mut fsm = ButtonFsm::new(t);
93        fsm.on_edge(ButtonEdge::Press, 0);
94        // Fires while still held, before any release.
95        assert_eq!(fsm.poll(4_000), Some(ButtonEvent::VeryLong));
96    }
97
98    /// A hold released between Back and power-off must not power off.
99    #[test]
100    fn releasing_before_the_power_off_threshold_is_only_back() {
101        let mut fsm = ButtonFsm::new(button_timings());
102        fsm.on_edge(ButtonEdge::Press, 0);
103        assert_eq!(fsm.poll(3_999), None);
104        assert_eq!(
105            fsm.on_edge(ButtonEdge::Release, 3_999),
106            Some(ButtonEvent::Long)
107        );
108    }
109}
